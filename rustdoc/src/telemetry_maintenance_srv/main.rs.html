8use anyhow::Result;
9use clap::Parser;
10use micromegas::analytics::lakehouse::lakehouse_context::LakehouseContext;
11use micromegas::analytics::lakehouse::static_tables_configurator::StaticTablesConfigurator;
12use micromegas::analytics::lakehouse::view_definition_store::PgViewDefinitionStore;
13use micromegas::analytics::lakehouse::view_factory::default_view_factory;
14use micromegas::analytics::lakehouse::view_registry::ViewRegistry;
15use micromegas::micromegas_main;
16use micromegas::servers::maintenance::daemon;
17use micromegas::servers::shutdown::wait_for_sigterm;
18use std::sync::Arc;
19
20#[derive(Parser, Debug)]
21#[clap(name = "Micromegas Telemetry Maintenance")]
22#[clap(
23    about = "Maintenance daemon for a Micromegas telemetry data lake",
24    version,
25    author
26)]
27struct Cli {
28    /// Delete lake data older than this many days (retention horizon)
29    #[clap(long, default_value = "90", env = "MICROMEGAS_RETENTION_DAYS")]
30    retention_days: i32,
31
32    #[command(flatten)]
33    common: micromegas::config::CommonServerArgs,
34}
35
36#[micromegas_main(interop_max_level = "info")]
37async fn main() -> Result<()> {
38    let args = Cli::parse();
39
40    let lakehouse = LakehouseContext::from_env().await?;
41    let data_lake = lakehouse.lake().clone();
42    let base_view_factory = default_view_factory(
43        lakehouse.runtime().clone(),
44        data_lake.clone(),
45        lakehouse.default_audience(),
46    )
47    .await?;
48    // Resolves the same `MICROMEGAS_STATIC_TABLES_URL` the FlightSQL builder uses, instead of a
49    // no-op configurator: a DDL-defined view reading a static table must build the same way in
50    // both services.
51    let session_configurator = StaticTablesConfigurator::from_env(
52        "MICROMEGAS_STATIC_TABLES_URL",
53        lakehouse.runtime().clone(),
54    )
55    .await?;
56    let view_registry = Arc::new(ViewRegistry::new(
57        Arc::new(base_view_factory),
58        Arc::new(PgViewDefinitionStore::new(lakehouse.lake().db_pool.clone())),
59        lakehouse.runtime().clone(),
60        data_lake,
61        session_configurator,
62    ));
63    let grace = args.common.grace();
64    daemon(
65        lakehouse,
66        view_registry,
67        args.retention_days,
68        wait_for_sigterm(),
69        grace,
70    )
71    .await
72}
