1//! Post-axum graceful shutdown for `object-cache-srv`: stop the prefetch
2//! producer, drain in-flight origin fetches, then close the foyer cache --
3//! see `tasks/1291_graceful_shutdown_detached_fetch_plan.md` (Design §3/§4).
4//!
5//! Axum's own HTTP drain is awaited directly in `main()`, outside this
6//! module entirely, and keeps the full shutdown grace period. `run` owns a
7//! single `tokio::time::timeout` around the three stages that come after it,
8//! bounded by whatever of that grace period axum's drain didn't use.
9
10use std::sync::Arc;
11use std::time::Duration;
12
13use micromegas::object_cache::foyer_backend::FoyerBackend;
14use micromegas::object_cache::range_cache::RangeCache;
15use micromegas::tracing::prelude::*;
16use tokio::task::JoinHandle;
17
18/// Runs, in order: abort-and-join the prefetch worker (so it can no longer
19/// spawn fetch work), wait for every outstanding detached fetch task to
20/// drain, then close `foyer` so a drained fetch's bytes survive a restart.
21/// All three are wrapped in one `tokio::time::timeout(remaining, ...)`; if it
22/// elapses, the block is dropped mid-stage and whatever it was doing is
23/// abandoned, exactly like today's runtime-teardown drop of an in-flight
24/// task. Never returns an error: an elapsed deadline is logged and swallowed
25/// here, never propagated.
26pub async fn run(
27    remaining: Duration,
28    prefetch_worker: JoinHandle<()>,
29    cache: RangeCache,
30    foyer: Arc<FoyerBackend>,
31) {
32    // `cache` itself (not a clone) is moved into the timed block for the
33    // drain; a clone is kept outside so `outstanding_fetch_tasks()` is still
34    // readable for the elapsed-deadline warning below even if the block was
35    // dropped mid-drain.
36    let cache_for_warning = cache.clone();
37    let finished = tokio::time::timeout(remaining, async move {
38        // Stage 1: stop the prefetch producer. `abort()` is synchronous;
39        // awaiting the handle afterwards observes the worker future as
40        // already dropped, so it can no longer pull an item from the channel
41        // and spawn one more fetch task. `JoinHandle::Output` is
42        // `Result<(), JoinError>`, `#[must_use]`, so it must be handled --
43        // and after an `abort()` an `Err` is the *expected* outcome
44        // (cancellation), so only a real panic is worth a warning.
45        prefetch_worker.abort();
46        match prefetch_worker.await {
47            Ok(()) => {}
48            Err(e) if e.is_panic() => warn!("prefetch worker panicked: {e:#}"),
49            Err(_) => {} // cancelled by the abort above -- expected
50        }
51
52        // Stage 2: let in-flight origin GETs land in the cache.
53        cache.wait_for_fetch_tasks_drain().await;
54        info!("origin fetch tasks drained");
55
56        // Stage 3: persist the RAM tier. `mark_shutting_down()` is set
57        // immediately before the close -- the only place it's needed, so
58        // `RamEvictionListener::on_leave` can tell this flush apart from
59        // capacity-driven thrashing (#1281).
60        foyer.mark_shutting_down();
61        match foyer.close().await {
62            Ok(()) => info!("foyer cache closed"),
63            Err(e) => warn!("foyer cache close failed: {e:#}"),
64        }
65    })
66    .await
67    .is_ok();
68
69    if !finished {
70        warn!(
71            "shutdown grace period elapsed before the post-axum sequence finished; \
72             whichever stage was still running was abandoned"
73        );
74        let outstanding = cache_for_warning.outstanding_fetch_tasks();
75        if outstanding > 0 {
76            warn!("abandoning {outstanding} in-flight origin fetch task(s) at shutdown");
77        }
78    }
79}
