192/// Validate the write-tuning knobs added for the foyer 0.22 upgrade,
193/// mirroring the fatal `anyhow!` startup guards in `object_cache_srv.rs`'s
194/// `main` for the other numeric knobs. Split out as a plain function (rather
195/// than inlined in `main`) so it is directly unit-testable from the
196/// integration-test crate.
197pub fn validate_write_tuning(flushers: usize, write_buffer_mb: usize) -> Result<()> {
198    if flushers == 0 {
199        return Err(anyhow!(
200            "MICROMEGAS_OBJECT_CACHE_FLUSHERS must be greater than 0"
201        ));
202    }
203    if write_buffer_mb == 0 {
204        return Err(anyhow!(
205            "MICROMEGAS_OBJECT_CACHE_WRITE_BUFFER_MB must be greater than 0"
206        ));
207    }
208    Ok(())
209}

212    /// Pick the listen address, falling back to the legacy
213    /// `MICROMEGAS_OBJECT_CACHE_LISTEN` value (passed in so tests never touch
214    /// process env).
215    pub fn resolve_listen_addr(&self, legacy_env: Option<String>) -> Result<SocketAddr> {
216        match (self.object_cache_listen_addr, legacy_env) {
217            (Some(_), Some(_)) => Err(anyhow!(
218                "both --object-cache-listen-addr/--listen/MICROMEGAS_OBJECT_CACHE_LISTEN_ADDR and \
219                 the deprecated MICROMEGAS_OBJECT_CACHE_LISTEN are set; remove \
220                 MICROMEGAS_OBJECT_CACHE_LISTEN"
221            )),
222            (Some(addr), None) => Ok(addr),
223            (None, Some(legacy)) => {
224                let addr = legacy.parse::<SocketAddr>().with_context(|| {
225                    format!("parsing MICROMEGAS_OBJECT_CACHE_LISTEN value {legacy:?}")
226                })?;
227                warn!(
228                    "MICROMEGAS_OBJECT_CACHE_LISTEN is deprecated, use MICROMEGAS_OBJECT_CACHE_LISTEN_ADDR"
229                );
230                Ok(addr)
231            }
232            (None, None) => Ok(DEFAULT_OBJECT_CACHE_LISTEN_ADDR),
233        }
234    }
235
236    /// Validate all numeric knobs. Fatal-at-startup config errors, kept next to
237    /// the type and unit-testable from the integration-test crate (like
238    /// `validate_write_tuning`).
239    pub fn validate(&self) -> Result<()> {
240        // `block_size` is the divisor for block-index math (`start / block_size`);
241        // a value of 0 would panic on the first range read. Reject it at the startup
242        // boundary as a fatal config error rather than letting it reach the cache.
243        if self.block_size == 0 {
244            return Err(anyhow!(
245                "MICROMEGAS_OBJECT_CACHE_BLOCK_SIZE must be greater than 0"
246            ));
247        }
248
249        if self.max_concurrent_fetches == 0 {
250            return Err(anyhow!(
251                "MICROMEGAS_OBJECT_CACHE_MAX_CONCURRENT_FETCHES must be greater than 0"
252            ));
253        }
254        // `FetchScheduler` computes `total - demand_reserved` as a plain
255        // subtraction; a misconfigured pair would panic deep inside the cache
256        // instead of at startup.
257        if self.demand_reserved_fetches >= self.max_concurrent_fetches {
258            return Err(anyhow!(
259                "MICROMEGAS_OBJECT_CACHE_DEMAND_RESERVED_FETCHES ({}) must be less than \
260                 MICROMEGAS_OBJECT_CACHE_MAX_CONCURRENT_FETCHES ({})",
261                self.demand_reserved_fetches,
262                self.max_concurrent_fetches
263            ));
264        }
265        // The byte-budget analog of the count-budget check above:
266        // `FetchScheduler::new` asserts the same relationship and panics
267        // deep inside the cache if it's violated, since a run larger than
268        // the byte budget's prefetch pool would hang forever acquiring its
269        // `bytes` permits (`acquire_many_owned` never completes past the
270        // semaphore's total). `saturating_mul`/`saturating_add` since
271        // `demand_reserved_fetches` is an unranged `usize`; a saturated
272        // floor is simply rejected below.
273        let max_run = max_run_bytes(self.block_size, self.max_coalesced_get_bytes);
274        let floor_bytes = (self.demand_reserved_fetches as u64)
275            .saturating_add(1)
276            .saturating_mul(max_run);
277        let budget_bytes = self.fetch_memory_budget_mb.saturating_mul(MIB);
278        if budget_bytes < floor_bytes {
279            let floor_mb = floor_bytes.div_ceil(MIB);
280            return Err(anyhow!(
281                "MICROMEGAS_OBJECT_CACHE_FETCH_MEMORY_BUDGET_MB ({}) must be at least {floor_mb} \
282                 MiB -- the floor implied by MICROMEGAS_OBJECT_CACHE_DEMAND_RESERVED_FETCHES ({}), \
283                 MICROMEGAS_OBJECT_CACHE_MAX_COALESCED_GET_BYTES ({}), and \
284                 MICROMEGAS_OBJECT_CACHE_BLOCK_SIZE ({}): each reserved demand slot must fit one \
285                 full-size run, or a run could hang forever acquiring its byte-budget permits",
286                self.fetch_memory_budget_mb,
287                self.demand_reserved_fetches,
288                self.max_coalesced_get_bytes,
289                self.block_size,
290            ));
291        }
292        // A zero budget would make every non-empty data request hang forever
293        // acquiring its mem_permits charge while /health and /ready still pass;
294        // fail at startup instead.
295        if self.memory_budget_mb == 0 {
296            return Err(anyhow!(
297                "MICROMEGAS_OBJECT_CACHE_MEMORY_BUDGET_MB must be greater than 0"
298            ));
299        }
300        // A large streaming read still charges a full window's worth of permits
301        // (`stream_window_bytes`, capped rather than rejected outright — see
302        // `handlers::stream_window_bytes`), and `Semaphore::acquire_many_owned`
303        // never completes (and never errors) if the requested count exceeds the
304        // semaphore's total permits. Without this floor, a deployment configured
305        // with a smaller `--memory-budget-mb` would hang every large read
306        // instead of failing fast here at startup.
307        let window_mb = permits_for_bytes(stream_window_bytes(self.block_size));
308        if self.memory_budget_mb < window_mb {
309            return Err(anyhow!(
310                "MICROMEGAS_OBJECT_CACHE_MEMORY_BUDGET_MB ({}) must be at least {window_mb} MiB \
311                 (2 * DEMAND_WINDOW_BLOCKS * block_size, the largest charge a single streaming \
312                 request can make), or every large read would hang acquiring mem_permits",
313                self.memory_budget_mb
314            ));
315        }
316        if self.prefetch_queue_capacity == 0 {
317            return Err(anyhow!(
318                "MICROMEGAS_OBJECT_CACHE_PREFETCH_QUEUE_CAPACITY must be greater than 0"
319            ));
320        }
321        if self.prefetch_worker_concurrency == 0 {
322            return Err(anyhow!(
323                "MICROMEGAS_OBJECT_CACHE_PREFETCH_WORKER_CONCURRENCY must be greater than 0"
324            ));
325        }
326        validate_write_tuning(self.flushers, self.write_buffer_mb)
327    }
