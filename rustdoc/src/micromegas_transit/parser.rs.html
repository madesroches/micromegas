9/// A reader for a custom (dynamically-sized) transit type.
10///
11/// Two higher-ranked lifetimes keep the returned `Value<'a>` (which borrows the
12/// arena and source buffer) independent of the `&'dep` borrow of the dependency
13/// map, so the caller can insert the result back into the map afterwards.
14pub type CustomReader = Arc<
15    dyn for<'a, 'dep> Fn(
16        &'a Bump,
17        &'a UserDefinedType,
18        &'a [UserDefinedType],
19        &'dep HashMap<u64, Value<'a>>,
20        &'a [u8],
21    ) -> Result<Value<'a>>,
22>;
23pub type CustomReaderMap = HashMap<String, CustomReader>;

25pub fn read_dependencies<'a>(
26    bump: &'a Bump,
27    custom_readers: &CustomReaderMap,
28    udts: &'a [UserDefinedType],
29    buffer: &'a [u8],
30) -> Result<HashMap<u64, Value<'a>>> {
31    let mut hash = HashMap::new();
32    let mut offset = 0;
33    while offset < buffer.len() {
34        let type_index = buffer[offset] as usize;
35        if type_index >= udts.len() {
36            bail!(
37                "Invalid type index parsing transit dependencies: {}",
38                type_index
39            );
40        }
41        offset += 1;
42        let udt = &udts[type_index];
43        let object_size = match udt.size {
44            0 => {
45                //dynamic size
46                let obj_size = try_read_pod_at::<u32>(buffer, offset)?;
47                offset += std::mem::size_of::<u32>();
48                obj_size as usize
49            }
50            static_size => static_size,
51        };
52        // Single guard: bounds every slice/advance derived from this object below
53        // (offset..object_end) against the actual buffer length.
54        let object_end = match offset.checked_add(object_size) {
55            Some(end) if end <= buffer.len() => end,
56            _ => bail!(
57                "corrupt block: object at offset {offset} with size {object_size} exceeds {}-byte buffer",
58                buffer.len()
59            ),
60        };
61
62        match udt.name.as_str() {
63            "StaticString" => {
64                // zero-copy: the string borrows the (whole-block) source buffer.
65                let string_id = try_read_pod_at::<u64>(buffer, offset)?;
66                let nb_utf8_bytes = object_size
67                    .checked_sub(std::mem::size_of::<usize>())
68                    .with_context(|| {
69                        format!("StaticString object_size {object_size} smaller than header")
70                    })?;
71                let str_start = offset + std::mem::size_of::<usize>();
72                let s = std::str::from_utf8(&buffer[str_start..str_start + nb_utf8_bytes])
73                    .with_context(|| "str::from_utf8")?;
74                let insert_res = hash.insert(string_id, Value::String(s));
75                if insert_res.is_some() {
76                    bail!("duplicate dependency id {string_id}");
77                }
78            }
79            "StaticStringDependency" => {
80                let mut window = &buffer[offset..object_end];
81                let string_id: u64 = try_read_consume_pod(&mut window)?;
82                let s =
83                    read_advance_string_in(bump, &mut window).with_context(|| "parsing string")?;
84                let insert_res = hash.insert(string_id, Value::String(s));
85                if insert_res.is_some() {
86                    bail!("duplicate dependency id {string_id}");
87                }
88            }
89            type_name => {
90                if let Some(reader) = custom_readers.get(type_name) {
91                    let window = &buffer[offset..object_end];
92                    if let Value::Object(obj) = (*reader)(bump, udt, udts, &hash, window)
93                        .with_context(|| "parsing custom dependency")?
94                    {
95                        let id: u64 = obj
96                            .get("id")
97                            .with_context(|| "reading id of custom dependency")?;
98                        let value = *obj
99                            .get_ref("value")
100                            .with_context(|| "reading value of custom dependency")?;
101                        let insert_res = hash.insert(id, value);
102                        if insert_res.is_some() {
103                            bail!("duplicate dependency id {id}");
104                        }
105                    } else {
106                        anyhow::bail!("custom dependency is not an object");
107                    }
108                } else {
109                    if udt.size == 0 {
110                        anyhow::bail!("invalid user-defined type {:?}", udt);
111                    }
112                    let instance =
113                        parse_pod_instance(bump, udt, udts, &hash, &buffer[offset..object_end])
114                            .with_context(|| "parse_pod_instance")?;
115                    if let Value::Object(obj) = instance {
116                        let id = obj.get::<u64>("id")?;
117                        let insert_res = hash.insert(id, Value::Object(obj));
118                        if insert_res.is_some() {
119                            bail!("duplicate dependency id {id}");
120                        }
121                    }
122                }
123            }
124        }
125        offset = object_end;
126    }
127
128    Ok(hash)
129}

131fn parse_custom_instance<'a>(
132    bump: &'a Bump,
133    custom_readers: &CustomReaderMap,
134    udt: &'a UserDefinedType,
135    udts: &'a [UserDefinedType],
136    dependencies: &HashMap<u64, Value<'a>>,
137    object_window: &'a [u8],
138) -> Result<Value<'a>> {
139    if let Some(reader) = custom_readers.get(&*udt.name) {
140        (*reader)(bump, udt, udts, dependencies, object_window)
141    } else {
142        log::warn!("unknown custom object {}", udt.name);
143        Ok(Value::Object(bump.alloc(Object {
144            type_name: udt.name.as_str(),
145            members: &[],
146        })))
147    }
148}

150// Stream metadata is untrusted: a self-referential or heavily fanned-out type
151// would otherwise overflow the stack or exhaust memory, both of which abort
152// the process. Real types nest a few levels with tens of members. The member
153// budget scales with the object's byte size so that total work and arena use
154// stay linear in the block size; the ceiling bounds a single large object.
155const MAX_UDT_NESTING_DEPTH: usize = 32;
156const MAX_UDT_NODES_PER_INSTANCE: usize = 4096;
157// Legitimate layouts visit at most ~1 member per byte (non-overlapping leaves of >= 1 byte);
158// the slack covers a few wrapper structs around tiny objects.
159const UDT_NODES_PER_BYTE: usize = 2;
160const UDT_NODES_SLACK: usize = 8;
161
162fn instance_node_budget(window_len: usize) -> usize {
163    window_len
164        .saturating_mul(UDT_NODES_PER_BYTE)
165        .saturating_add(UDT_NODES_SLACK)
166        .min(MAX_UDT_NODES_PER_INSTANCE)
167}

169struct ParseBudget {
170    depth: usize,
171    nodes_left: usize,
172    initial_nodes: usize,
173    instance_len: usize,
174}
175
176pub fn parse_pod_instance<'a>(
177    bump: &'a Bump,
178    udt: &'a UserDefinedType,
179    udts: &'a [UserDefinedType],
180    dependencies: &HashMap<u64, Value<'a>>,
181    object_window: &'a [u8],
182) -> Result<Value<'a>> {
183    let initial_nodes = instance_node_budget(object_window.len());
184    let mut budget = ParseBudget {
185        depth: 0,
186        nodes_left: initial_nodes,
187        initial_nodes,
188        instance_len: object_window.len(),
189    };
190    parse_pod_instance_bounded(bump, udt, udts, dependencies, object_window, &mut budget)
191}

193fn parse_pod_instance_bounded<'a>(
194    bump: &'a Bump,
195    udt: &'a UserDefinedType,
196    udts: &'a [UserDefinedType],
197    dependencies: &HashMap<u64, Value<'a>>,
198    object_window: &'a [u8],
199    budget: &mut ParseBudget,
200) -> Result<Value<'a>> {
201    if budget.depth >= MAX_UDT_NESTING_DEPTH {
202        bail!(
203            "corrupt metadata: udt {} nesting exceeds {MAX_UDT_NESTING_DEPTH} levels",
204            udt.name
205        );
206    }
207    let mut members =
208        bumpalo::collections::Vec::with_capacity_in(udt.members.len().min(budget.nodes_left), bump);
209    for member_meta in &udt.members {
210        if budget.nodes_left == 0 {
211            bail!(
212                "corrupt metadata: member budget of {} for {}-byte instance exhausted at udt {}",
213                budget.initial_nodes,
214                budget.instance_len,
215                udt.name
216            );
217        }
218        budget.nodes_left -= 1;
219        let name: &'a str = member_meta.name.as_str();
220        // Bounds the reference-key read and every intrinsic read below against
221        // the object window (member metadata is untrusted, same as the payload).
222        match member_meta.offset.checked_add(member_meta.size) {
223            Some(end) if end <= object_window.len() => {}
224            _ => bail!(
225                "corrupt block: member {member_meta:?} exceeds {}-byte object window",
226                object_window.len()
227            ),
228        }
229        let value = if member_meta.is_reference {
230            if member_meta.size < std::mem::size_of::<u64>() {
231                bail!(
232                    "member references have to be at least 8 bytes {:?}",
233                    member_meta
234                );
235            }
236            let key = try_read_pod_at::<u64>(object_window, member_meta.offset)?;
237            if let Some(v) = dependencies.get(&key) {
238                *v
239            } else {
240                bail!("dependency not found: member={member_meta:?} key={key}");
241            }
242        } else {
243            match member_meta.type_name.as_str() {
244                "u8" | "uint8" => {
245                    if std::mem::size_of::<u8>() != member_meta.size {
246                        bail!("type size mismatch for member {member_meta:?}");
247                    }
248                    Value::U8(try_read_pod_at::<u8>(object_window, member_meta.offset)?)
249                }
250                "u32" | "uint32" => {
251                    if std::mem::size_of::<u32>() != member_meta.size {
252                        bail!("type size mismatch for member {member_meta:?}");
253                    }
254                    Value::U32(try_read_pod_at::<u32>(object_window, member_meta.offset)?)
255                }
256                "u64" | "uint64" => {
257                    if std::mem::size_of::<u64>() != member_meta.size {
258                        bail!("type size mismatch for member {member_meta:?}");
259                    }
260                    Value::U64(try_read_pod_at::<u64>(object_window, member_meta.offset)?)
261                }
262                "i64" | "int64" => {
263                    if std::mem::size_of::<i64>() != member_meta.size {
264                        bail!("type size mismatch for member {member_meta:?}");
265                    }
266                    Value::I64(try_read_pod_at::<i64>(object_window, member_meta.offset)?)
267                }
268                "f64" => {
269                    if std::mem::size_of::<f64>() != member_meta.size {
270                        bail!("type size mismatch for member {member_meta:?}");
271                    }
272                    Value::F64(try_read_pod_at::<f64>(object_window, member_meta.offset)?)
273                }
274                non_intrinsic_member_type_name => {
275                    if let Some(index) = udts
276                        .iter()
277                        .position(|t| *t.name == non_intrinsic_member_type_name)
278                    {
279                        let member_udt = &udts[index];
280                        // member_udt.size may differ from member_meta.size, so it
281                        // needs its own guard before slicing.
282                        let udt_end = match member_meta.offset.checked_add(member_udt.size) {
283                            Some(end) if end <= object_window.len() => end,
284                            _ => bail!(
285                                "corrupt block: nested member {member_meta:?} exceeds {}-byte object window",
286                                object_window.len()
287                            ),
288                        };
289                        budget.depth += 1;
290                        let nested = parse_pod_instance_bounded(
291                            bump,
292                            member_udt,
293                            udts,
294                            dependencies,
295                            &object_window[member_meta.offset..udt_end],
296                            budget,
297                        );
298                        budget.depth -= 1;
299                        nested.with_context(|| "parse_pod_instance")?
300                    } else {
301                        bail!("unknown member type {}", non_intrinsic_member_type_name);
302                    }
303                }
304            }
305        };
306        members.push((name, value));
307    }
308
309    if udt.is_reference {
310        // reference objects need a member called 'id' which is the key to the dependency
311        if let Some(id_index) = members.iter().position(|m| m.0 == "id") {
312            return Ok(members[id_index].1);
313        }
314        bail!("reference object has no 'id' member");
315    }
316
317    Ok(Value::Object(bump.alloc(Object {
318        type_name: udt.name.as_str(),
319        members: members.into_bump_slice(),
320    })))
321}

323// parse_object_buffer calls fun for each object in the buffer until fun returns
324// `false`
325pub fn parse_object_buffer<'a, F>(
326    bump: &'a Bump,
327    custom_readers: &CustomReaderMap,
328    dependencies: &HashMap<u64, Value<'a>>,
329    udts: &'a [UserDefinedType],
330    buffer: &'a [u8],
331    mut fun: F,
332) -> Result<bool>
333where
334    F: FnMut(Value<'a>) -> Result<bool>,
335{
336    let mut offset = 0;
337    while offset < buffer.len() {
338        let type_index = buffer[offset] as usize;
339        if type_index >= udts.len() {
340            bail!("Invalid type index parsing transit objects: {}", type_index);
341        }
342        offset += 1;
343        let udt = &udts[type_index];
344        let (object_size, is_size_dynamic) = match udt.size {
345            0 => {
346                //dynamic size
347                let obj_size = try_read_pod_at::<u32>(buffer, offset)?;
348                offset += std::mem::size_of::<u32>();
349                (obj_size as usize, true)
350            }
351            static_size => (static_size, false),
352        };
353        let object_end = match offset.checked_add(object_size) {
354            Some(end) if end <= buffer.len() => end,
355            _ => bail!(
356                "corrupt block: object at offset {offset} with size {object_size} exceeds {}-byte buffer",
357                buffer.len()
358            ),
359        };
360        let instance = if is_size_dynamic {
361            parse_custom_instance(
362                bump,
363                custom_readers,
364                udt,
365                udts,
366                dependencies,
367                &buffer[offset..object_end],
368            )
369            .with_context(|| "parse_custom_instance")?
370        } else {
371            parse_pod_instance(bump, udt, udts, dependencies, &buffer[offset..object_end])
372                .with_context(|| "parse_pod_instance")?
373        };
374        if !fun(instance)? {
375            return Ok(false);
376        }
377        offset = object_end;
378    }
379    Ok(true)
