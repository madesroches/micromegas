8/// The authenticated write audience a request ingests under (AbAC Stage 5, #1373).
9/// `None` means the credential carries no audience -- data stays unstamped, exactly as
10/// before this stage.
11///
12/// Per `rust/CLAUDE.md`'s Rust-API stance, every process-insert call site must state its
13/// audience explicitly -- there is no `Default` impl, so a call site that has no audience to
14/// pass must say so with [`WriteAudience::none`], and the compiler enumerates every site that
15/// needs updating when this type's callers change.
16#[derive(Debug, Clone, PartialEq, Eq)]
17pub struct WriteAudience(Option<Arc<str>>);
18
19impl WriteAudience {
20    /// Builds a `WriteAudience` from an optional audience label.
21    ///
22    /// Rejects a malformed label: `ingestion_api_keys.audience` is already `CHECK`-constrained
23    /// to `[A-Za-z0-9_-]{1,255}`, so this is defence in depth against a future producer of
24    /// `bound_audience` that doesn't go through that column. The HTTP-edge caller
25    /// (`micromegas::servers::write_audience::resolve_write_audience`) no longer treats this
26    /// `Err` as a rejection -- it warns and degrades to [`WriteAudience::none`] instead, since
27    /// that caller has no `Result` to propagate the failure through. The charset check
28    /// duplicates `micromegas_auth::policy::is_valid_audience` (`policy.rs`) rather than
29    /// depending on `micromegas-auth` from `micromegas-ingestion` -- the same crate-boundary
30    /// trade-off `read_scope.rs`'s `is_well_formed_audience` already makes for
31    /// `micromegas-analytics`. Keep the three copies in step if the charset ever changes.
32    pub fn new(audience: Option<&str>) -> anyhow::Result<Self> {
33        match audience {
34            None => Ok(Self(None)),
35            Some(aud) => {
36                if is_valid_audience(aud) {
37                    Ok(Self(Some(Arc::from(aud))))
38                } else {
39                    Err(anyhow::anyhow!(
40                        "invalid write audience {aud:?}: must match [A-Za-z0-9_-]{{1,255}}"
41                    ))
42                }
43            }
44        }
45    }
46
47    /// No audience at all -- the credential carries none, or none is available (e.g. no auth
48    /// provider configured). Data stamped with this stays unstamped.
49    pub fn none() -> Self {
50        Self(None)
51    }
52
53    /// Returns the audience label, or `None` if this is an unstamped write.
54    pub fn as_str(&self) -> Option<&str> {
55        self.0.as_deref()
56    }
57}
58
59/// `true` if `aud` is a valid audience name: `[A-Za-z0-9_-]{1,255}`, checked in bytes. See
60/// `WriteAudience::new`'s doc comment for why this is a local copy rather than a dependency on
61/// `micromegas-auth`.
62fn is_valid_audience(aud: &str) -> bool {
63    !aud.is_empty()
64        && aud.len() <= 255
65        && aud
66            .bytes()
67            .all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'-')
68}
