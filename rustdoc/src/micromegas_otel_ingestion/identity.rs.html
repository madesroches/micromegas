1//! Synthesizing micromegas Process / Stream / Block identity from OTLP `Resource` attributes.
2//!
3//! OTel has no `Process` object; just a `Resource` carrying `repeated KeyValue attributes`.
4//! We hash an OS-honest tuple of identifying attributes to a UUIDv5. Long-term stability of
5//! `process_id` values across upgrades is not a design goal — re-deriving existing ids is
6//! always acceptable, so the formula can be extended in-place when needed.
7
8use crate::proto::{AnyValue, KeyValue, Resource, any_value};
9use micromegas_tracing::prelude::*;
10use std::sync::Once;
11use uuid::{Uuid, uuid};
12
13/// `AnyValue.string_value_strindex` / `KeyValue.key_strindex` reference a
14/// `ProfilesDictionary.string_table` that exists **only** for the Profiling signal. Per the
15/// OTLP spec, receivers of logs/metrics/traces MUST treat these as absent. Warn once per
16/// process so a misconfigured profiling producer is noticeable without flooding the logs.
17fn warn_unexpected_strindex() {
18    static WARNED: Once = Once::new();
19    WARNED.call_once(|| {
20        warn!(
21            "ignoring profiling-only string-interning field on a non-profiling OTLP signal; \
22             treating it as absent (OTLP spec)"
23        );
24    });
25}

37/// ASCII unit separator — used between concatenated string fields in identity formulas
38/// to prevent tuple-boundary collisions like `("abc", "")` vs `("ab", "c")`.
39///
40/// `pub(crate)`, not private: `block.rs` prepends an audience-tagged prefix ahead of its own
41/// hash input using this exact separator, rather than a `\x1F` string literal, so the two
42/// modules can never drift on what the separator character is.
43pub(crate) const SEPARATOR: char = '\x1F';
44const SEPARATOR_STR: &str = "\x1F";
45
46/// Identity inputs beyond the OTLP payload itself.
47///
48/// `Default` (both fields absent) is a no-op: the deployment default's un-salted namespace, or
49/// any call site with no audience threaded through it, sees zero id churn.
50#[derive(Debug, Clone, Copy, Default)]
51pub struct IdentityContext<'a> {
52    /// Authenticated write audience. Folded into `process_id` and `block_id` so two audiences
53    /// posting identical resources/payloads never collapse onto one process or dedup against
54    /// each other. `None` means the resolved write audience is the deployment default, which
55    /// keeps the un-salted namespace.
56    pub audience: Option<&'a str>,
57    /// Webhook-only: canonicalized incoming header bytes.
58    pub extra_hash_input: &'a [u8],
59}
60
61/// OTel signal label used in stream-id derivation.
62#[derive(Debug, Clone, Copy)]
63pub enum SignalKey {
64    Logs,
65    Metrics,
66    Traces,
67}
68
69impl SignalKey {
70    pub fn as_str(&self) -> &'static str {
71        match self {
72            Self::Logs => "logs",
73            Self::Metrics => "metrics",
74            Self::Traces => "traces",
75        }
76    }
77}

79/// Convenience accessor — fetch one resource attribute by key, returning `None` when absent.
80///
81/// Matches on `kv.key` only; `kv.key_strindex` (profiling-only) is intentionally ignored, so an
82/// interned key (empty `key`) is treated as absent — the OTLP spec behavior for non-profiling
83/// signals.
84pub fn attr<'a>(attrs: &'a [KeyValue], key: &str) -> Option<&'a AnyValue> {
85    attrs
86        .iter()
87        .find(|kv| kv.key == key)
88        .and_then(|kv| kv.value.as_ref())
89}

91/// Returns the attribute's value rendered as a stable string.
92///
93/// - `string_value` → as-is
94/// - `int_value`    → decimal (some SDKs emit timestamps as int nanos)
95/// - `bool_value` / `double_value` / `bytes_value` / `array_value` / `kvlist_value` →
96///   their `Debug` form is unstable, so we just stringify with `format!`. In practice
97///   the resource attributes that feed identity are always strings or ints, but having
98///   a fallback prevents identity drift if an SDK emits something exotic.
99pub fn attr_to_string(v: &AnyValue) -> String {
100    match v.value.as_ref() {
101        Some(any_value::Value::StringValue(s)) => s.clone(),
102        Some(any_value::Value::IntValue(i)) => i.to_string(),
103        Some(any_value::Value::BoolValue(b)) => b.to_string(),
104        Some(any_value::Value::DoubleValue(d)) => d.to_string(),
105        Some(any_value::Value::BytesValue(b)) => format!("{b:?}"),
106        // Profiling-only string-table reference: no dictionary exists for this signal, so the
107        // index is meaningless. Treat as absent (per OTLP spec). NOTE: this value feeds the
108        // load-bearing UUIDv5 identity hash — never stringify the index back into it.
109        Some(any_value::Value::StringValueStrindex(_)) => {
110            warn_unexpected_strindex();
111            String::new()
112        }
113        Some(any_value::Value::ArrayValue(_)) | Some(any_value::Value::KvlistValue(_)) => {
114            // Structured values shouldn't appear in identity-bearing fields. If one does,
115            // hashing the Debug form is at least deterministic for a given prost version.
116            format!("{:?}", v.value)
117        }
118        None => String::new(),
119    }
120}

122/// Lower-case + trim. Applied to free-form string fields where the SDK may render
123/// the same logical value with different casing.
124fn norm(s: &str) -> String {
125    s.trim().to_ascii_lowercase()
126}
127
128/// Reads `attr` and returns the lower-cased + trimmed string value (or empty).
129///
130/// Public so `cloudwatch_metrics::is_cloudwatch_metric_stream_resource` can share the exact
131/// same emptiness semantics `is_degenerate_resource` uses for its own fields, rather than
132/// reimplementing trim+lowercase locally and risking the two checks drifting apart.
133pub fn attr_norm(attrs: &[KeyValue], key: &str) -> String {
134    attr(attrs, key)
135        .map(|v| norm(&attr_to_string(v)))
136        .unwrap_or_default()
137}
138
139/// Reads `attr` as-is (no case folding) — used for opaque values like `process.start_time`.
140fn attr_raw(attrs: &[KeyValue], key: &str) -> String {
141    attr(attrs, key).map(attr_to_string).unwrap_or_default()
142}

144/// Resolves the process owner's username.
145///
146/// `process.owner` is the OTel semantic-conventions attribute emitted by process resource
147/// detectors. When it is absent we fall back, in decreasing semantic closeness, to the
148/// process-scoped effective (`process.user.name`) and real (`process.real_user.name`) user
149/// names, then to the generic `user.name` for producers that only set that.
150pub fn process_owner_string(attrs: &[KeyValue]) -> String {
151    for key in [
152        "process.owner",
153        "process.user.name",
154        "process.real_user.name",
155        "user.name",
156    ] {
157        let s = attr_raw(attrs, key);
158        if !s.is_empty() {
159            return s;
160        }
161    }
162    String::new()
163}

165/// Resolves the OTel process-creation timestamp.
166///
167/// `process.creation.time` is the stable OTel semantic-conventions attribute and is
168/// what real SDKs emit; `process.start_time` is accepted as a fallback for any
169/// non-standard producer that still uses the older name.
170pub fn process_start_string(attrs: &[KeyValue]) -> String {
171    let s = attr_raw(attrs, "process.creation.time");
172    if !s.is_empty() {
173        return s;
174    }
175    attr_raw(attrs, "process.start_time")
176}

178/// Returns true when none of the four identifying fields are populated. Caller may want
179/// to log a warning so a degenerate-resource scenario doesn't silently collapse multiple
180/// processes onto one `process_id`.
181pub fn is_degenerate_resource(attrs: &[KeyValue]) -> bool {
182    attr_norm(attrs, "host.id").is_empty()
183        && attr_norm(attrs, "host.name").is_empty()
184        && attr_raw(attrs, "process.pid").is_empty()
185        && attr_norm(attrs, "service.instance.id").is_empty()
186}

188/// Derives `process_id` from a resource by hashing the identifying tuple under `NS_OTEL_PROCESS_V1`.
189///
190/// All fields pass through `attr_norm` (lower-case + trim) except `process.pid` and
191/// `process.creation.time` which are `attr_raw`. Field order (separated by `\x1F`):
192///
193///   host.id · host.name · process.pid · process.creation.time ·
194///   service.namespace · service.name · service.instance.id · process.owner ·
195///   os.type · os.version · os.name · os.description · os.build_id ·
196///   host.arch · host.type ·
197///   host.image.id · host.image.name · host.image.version ·
198///   host.cpu.model.id · host.cpu.model.name · host.cpu.family ·
199///   host.cpu.vendor.id · host.cpu.stepping · host.cpu.cache.l2.size ·
200///   service.version ·
201///   telemetry.sdk.name · telemetry.sdk.language · telemetry.sdk.version ·
202///   process.runtime.name · process.runtime.version · process.runtime.description
203///
204/// Fields are appended in-place under the same namespace UUID rather than bumping to `_V2` —
205/// the same pattern used when `process.owner` was added. Long-term stability of `process_id`
206/// values is not a design goal; re-deriving existing ids is always acceptable.
207///
208/// `ctx.audience` domain-separates the hash, **only when `Some`**: a present audience derives a
209/// per-audience namespace UUID (`NS_OTEL_PROCESS_V1` salted with the audience) and hashes the
210/// joined field key under *that* namespace, rather than appending the audience string onto the
211/// joined key with the same `\x1F` separator that joins the 31 fields above. Concatenating into
212/// the shared-separator string would let a crafted resource-attribute value ending in
213/// `\x1F<victim-audience>` reproduce the exact byte string a real stamped request would hash,
214/// defeating the audience scoping this exists to provide -- OTLP attribute values are arbitrary
215/// UTF-8, and `norm` only trims and lower-cases, it does not escape `\x1F`.
216///
217/// `ctx.audience: None` means "the resolved write audience is the deployment default"
218/// (`WriteAudience::id_namespace`, `micromegas-ingestion`) -- the write path always resolves a
219/// real audience, and the deployment default is the one that keeps this un-salted namespace,
220/// `Uuid::new_v5(&NS_OTEL_PROCESS_V1, key)`. This is what stops two audiences sending identical
221/// resource attributes (the same containerized app in two tenants, a degenerate resource, a
222/// CloudWatch namespace) from silently colliding on one `process_id`, including via a crafted
223/// attribute value -- the mapping audience → namespace stays injective, with every non-default
224/// audience salted and the default alone un-salted.
225pub fn process_id_from_resource(resource: Option<&Resource>, ctx: IdentityContext) -> Uuid {
226    let attrs = resource.map(|r| r.attributes.as_slice()).unwrap_or(&[]);
227
228    let fields = [
229        attr_norm(attrs, "host.id"),
230        attr_norm(attrs, "host.name"),
231        attr_raw(attrs, "process.pid"),
232        process_start_string(attrs),
233        attr_norm(attrs, "service.namespace"),
234        attr_norm(attrs, "service.name"),
235        attr_norm(attrs, "service.instance.id"),
236        norm(&process_owner_string(attrs)),
237        // OS identity
238        attr_norm(attrs, "os.type"),
239        attr_norm(attrs, "os.version"),
240        attr_norm(attrs, "os.name"),
241        attr_norm(attrs, "os.description"),
242        attr_norm(attrs, "os.build_id"),
243        // Host hardware
244        attr_norm(attrs, "host.arch"),
245        attr_norm(attrs, "host.type"),
246        attr_norm(attrs, "host.image.id"),
247        attr_norm(attrs, "host.image.name"),
248        attr_norm(attrs, "host.image.version"),
249        attr_norm(attrs, "host.cpu.model.id"),
250        attr_norm(attrs, "host.cpu.model.name"),
251        attr_norm(attrs, "host.cpu.family"),
252        attr_norm(attrs, "host.cpu.vendor.id"),
253        attr_norm(attrs, "host.cpu.stepping"),
254        attr_norm(attrs, "host.cpu.cache.l2.size"),
255        // Service / SDK
256        attr_norm(attrs, "service.version"),
257        attr_norm(attrs, "telemetry.sdk.name"),
258        attr_norm(attrs, "telemetry.sdk.language"),
259        attr_norm(attrs, "telemetry.sdk.version"),
260        // Runtime
261        attr_norm(attrs, "process.runtime.name"),
262        attr_norm(attrs, "process.runtime.version"),
263        attr_norm(attrs, "process.runtime.description"),
264    ];
265
266    let key = fields.join(SEPARATOR_STR);
267    match ctx.audience {
268        Some(audience) => {
269            let audience_ns = Uuid::new_v5(&NS_OTEL_PROCESS_V1, audience.as_bytes());
270            Uuid::new_v5(&audience_ns, key.as_bytes())
271        }
272        None => Uuid::new_v5(&NS_OTEL_PROCESS_V1, key.as_bytes()),
273    }
274}

276/// Derives `stream_id` from `(process_id, signal)`. Max three streams per process.
277pub fn stream_id_from_process_signal(process_id: Uuid, signal: SignalKey) -> Uuid {
278    let key = format!("{process_id}{}{}", SEPARATOR, signal.as_str());
279    Uuid::new_v5(&NS_OTEL_STREAM_V1, key.as_bytes())
280}

282/// Derives `block_id` by hashing whatever bytes the caller hands it. `Uuid::new_v5` SHA-1s its
283/// input internally, so we don't pre-hash.
284///
285/// The signature is a plain `&[u8]`, but callers routinely concatenate up to three inputs ahead
286/// of calling this: the webhook path's canonicalized header set (`extra_hash_input`), and an
287/// `"aud{SEPARATOR}{audience}{SEPARATOR}"` prefix (`block.rs`) so two audiences posting
288/// byte-identical payloads never dedup against each other. See `block.rs`'s `split_logs` /
289/// `split_metrics` / `split_traces` for the actual hash-input assembly.
290pub fn block_id_from_payload(payload: &[u8]) -> Uuid {
291    Uuid::new_v5(&NS_OTEL_BLOCK_V1, payload)
292}
