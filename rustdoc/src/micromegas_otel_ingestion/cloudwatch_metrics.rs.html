1//! CloudWatch Metric Streams resource rewrite.
2//!
3//! Metrics delivered via `POST /ingestion/otlp/v1/metrics/firehose` from a CloudWatch Metric
4//! Stream carry a `Resource` with no `service.*`/`host.*`/`process.*` identity at all — only
5//! `cloud.account.id`, `cloud.provider`, `cloud.region`, and `aws.exporter.arn`. Every stream
6//! from every AWS account/region therefore hashes to the same degenerate `process_id`
7//! (`identity::is_degenerate_resource` already flags this shape, but nothing corrects it).
8//!
9//! This module detects that fingerprint and partitions each matching `ResourceMetrics` into
10//! one synthetic `ResourceMetrics` per CloudWatch **namespace** (`AWS/RDS`, `AWS/ECS`,
11//! `ECS/ContainerInsights`, `AWS/S3`, …), read off the `Namespace` attribute CloudWatch
12//! attaches to every datapoint — giving `exe` a bounded, meaningful value while folding the
13//! exporter ARN into `service.instance.id` so different accounts/regions never collide onto
14//! the same process. Mirrors what `cloudwatch_logs.rs` does for the Logs Firehose route, but
15//! unlike that route this producer's resource carries no per-stream identity of its own to
16//! reuse — the identity has to be synthesized from datapoint attributes instead.
17//!
18//! Pure, unit-testable, no HTTP/framework dependency. Called from
19//! `handler::ingest_firehose_metrics` only — the shared `/v1/metrics` OTLP endpoint is left
20//! untouched (see the design plan for the rationale).
21
22use crate::identity;
23use crate::proto::{
24    AnyValue, ExportMetricsServiceRequest, KeyValue, Metric, ResourceMetrics, ScopeMetrics,
25    any_value,
26};
27use std::collections::BTreeMap;

29/// `service.name` fallback for a metric with no usable `Namespace` datapoint attribute, so
30/// `exe` is never empty on this route. Public (rather than private) so
31/// `tests/cloudwatch_metrics_tests.rs` can assert against it directly, matching the
32/// `cloudwatch_logs.rs` precedent (`cloudwatch_logs.rs:21-22`).
33pub const UNKNOWN_NAMESPACE: &str = "AWS/Unknown";
34
35/// True when `attrs` looks like a CloudWatch Metric Streams resource: a non-empty
36/// `aws.exporter.arn` marker, no `service.name`/`service.namespace`, and otherwise degenerate
37/// (`identity::is_degenerate_resource`).
38///
39/// All three conjuncts are gated on emptiness via `identity::attr_norm` (trim + lowercase),
40/// not `Option::is_some()`/`is_none()`, to match the same-strength check
41/// `is_degenerate_resource` already applies to its own fields — a present-but-empty
42/// `StringValue("")` must not be treated as "has a value" for any of the three: an empty
43/// `aws.exporter.arn` is not a real marker (and would otherwise still leave the resource
44/// degenerate), and a present-but-empty `service.name`/`service.namespace` must not be
45/// treated as "has a service name" either.
46///
47/// Public so `tests/cloudwatch_metrics_tests.rs` can call it directly.
48pub fn is_cloudwatch_metric_stream_resource(attrs: &[KeyValue]) -> bool {
49    !identity::attr_norm(attrs, "aws.exporter.arn").is_empty()
50        && identity::attr_norm(attrs, "service.name").is_empty()
51        && identity::attr_norm(attrs, "service.namespace").is_empty()
52        && identity::is_degenerate_resource(attrs)
53}

55/// Reads the `Namespace` attribute off a `Metric`'s **first** data point, across whichever
56/// OTel data type it holds (`Sum`/`Gauge`/`Histogram`/`ExponentialHistogram`/`Summary`).
57///
58/// Sampling only the first data point is sufficient because the namespace is constant per
59/// `Metric` — CloudWatch encodes it directly in `Metric.name`
60/// (`amazonaws.com/<Namespace>/<MetricName>`), so every data point under one `Metric` carries
61/// the same `Namespace` value in practice.
62///
63/// Returns `None` — routing the metric to the `UNKNOWN_NAMESPACE` fallback bucket rather than
64/// dropping it — when the metric has no first data point, no `Namespace` attribute, or an
65/// empty/whitespace-only value. The trimmed value flows untouched into both the bucket key and
66/// `service.name`, so it must match what `process_id_from_resource` will fold through
67/// `attr_norm` (trim + lowercase) downstream — an untrimmed `" AWS/RDS"` and a trimmed
68/// `"AWS/RDS"` would otherwise land in two different `BTreeMap` buckets (two blocks, two
69/// conflicting `exe` values) while still hashing to the same `process_id`.
70///
71/// Public so `tests/cloudwatch_metrics_tests.rs` can call it directly.
72pub fn metric_namespace(metric: &Metric) -> Option<String> {
73    use crate::proto::metric::Data;
74    let attrs: &[KeyValue] = match metric.data.as_ref()? {
75        Data::Sum(s) => s.data_points.first()?.attributes.as_slice(),
76        Data::Gauge(g) => g.data_points.first()?.attributes.as_slice(),
77        Data::Histogram(h) => h.data_points.first()?.attributes.as_slice(),
78        Data::ExponentialHistogram(h) => h.data_points.first()?.attributes.as_slice(),
79        Data::Summary(s) => s.data_points.first()?.attributes.as_slice(),
80    };
81    identity::attr(attrs, "Namespace")
82        .map(identity::attr_to_string)
83        .map(|s| s.trim().to_string())
84        .filter(|s| !s.is_empty())
85}

87/// Replace-if-present `KeyValue` setter: overwrites `key`'s value in place when already
88/// present, pushing only when absent — never produces a duplicate key (unlike a plain
89/// `push`), matching the shape of `cloudwatch_logs.rs`'s `kv` helper.
90fn set_attr(attrs: &mut Vec<KeyValue>, key: &str, value: &str) {
91    let new_value = Some(AnyValue {
92        value: Some(any_value::Value::StringValue(value.to_string())),
93    });
94    if let Some(existing) = attrs.iter_mut().find(|kv| kv.key == key) {
95        existing.value = new_value;
96    } else {
97        attrs.push(KeyValue {
98            key: key.to_string(),
99            key_strindex: 0,
100            value: new_value,
101        });
102    }
103}
104
105/// Sets an existing entry's value to `""` in place when `key` is already present; a no-op
106/// otherwise (never pushes) — used to clear `service.namespace` without adding a spurious
107/// empty attribute to resources that never carried one.
108fn clear_attr_if_present(attrs: &mut [KeyValue], key: &str) {
109    if let Some(existing) = attrs.iter_mut().find(|kv| kv.key == key) {
110        existing.value = Some(AnyValue {
111            value: Some(any_value::Value::StringValue(String::new())),
112        });
113    }
114}

116/// Case-insensitive bucketing key for a namespace as returned by `metric_namespace`. Namespaces
117/// are already trimmed by `metric_namespace`; this additionally lower-cases so that CloudWatch
118/// namespaces differing only in case (e.g. `MyApp/Prod` vs `myapp/prod`) land in the same
119/// bucket — matching `identity::attr_norm`'s trim+lowercase semantics, which is what
120/// `process_id_from_resource` folds `service.name` through downstream. Without this, two case
121/// variants would produce two buckets (two `ResourceMetrics`, two `exe` values) that still hash
122/// to the same `process_id`, and `register_otel_process`'s `ON CONFLICT DO NOTHING` would let
123/// whichever `exe` lands first silently win.
124fn namespace_bucket_key(namespace: &Option<String>) -> Option<String> {
125    namespace.as_ref().map(|s| s.to_ascii_lowercase())
126}

128/// Partitions one CloudWatch-shaped `ResourceMetrics` into one synthetic `ResourceMetrics` per
129/// namespace found across its metrics (`metric_namespace`'s `None` result buckets under
130/// `UNKNOWN_NAMESPACE`). A namespace appearing in more than one original `ScopeMetrics` still
131/// ends up as a single output resource's `scope_metrics` list — `BTreeMap` for deterministic
132/// iteration order, load-bearing for reproducible `block_id`s across repeated rewrites of the
133/// same input.
134///
135/// Bucketing is keyed on `namespace_bucket_key` (case-insensitive) so that case variants of the
136/// same namespace merge into one bucket; the first-seen raw (case-preserving) namespace string
137/// is retained alongside the bucket and used as `service.name`, so the output `exe` is stable
138/// regardless of which case variant happens to sort first.
139///
140/// Every other field on the synthetic `ResourceMetrics`/`Resource` is carried over from the
141/// original unchanged — only `attributes` differs.
142fn partition_resource_metrics(rm: ResourceMetrics) -> Vec<ResourceMetrics> {
143    let original_attrs = rm
144        .resource
145        .as_ref()
146        .map(|r| r.attributes.clone())
147        .unwrap_or_default();
148    let dropped_attributes_count = rm
149        .resource
150        .as_ref()
151        .map(|r| r.dropped_attributes_count)
152        .unwrap_or(0);
153    let entity_refs = rm
154        .resource
155        .as_ref()
156        .map(|r| r.entity_refs.clone())
157        .unwrap_or_default();
158    let schema_url = rm.schema_url.clone();
159
160    // Each bucket carries the first-seen raw namespace string alongside its accumulated
161    // metrics/scope_metrics, so case variants collapse onto one bucket keyed by
162    // `namespace_bucket_key` while still emitting a single, stable `service.name`.
163    let mut buckets: BTreeMap<Option<String>, (Option<String>, Vec<ScopeMetrics>)> =
164        BTreeMap::new();
165    for scope in rm.scope_metrics {
166        let scope_scope = scope.scope.clone();
167        let scope_schema_url = scope.schema_url.clone();
168        let mut per_namespace: BTreeMap<Option<String>, (Option<String>, Vec<Metric>)> =
169            BTreeMap::new();
170        for metric in scope.metrics {
171            let namespace = metric_namespace(&metric);
172            let key = namespace_bucket_key(&namespace);
173            per_namespace
174                .entry(key)
175                .or_insert_with(|| (namespace.clone(), Vec::new()))
176                .1
177                .push(metric);
178        }
179        for (key, (raw_namespace, metrics)) in per_namespace {
180            buckets
181                .entry(key)
182                .or_insert_with(|| (raw_namespace, Vec::new()))
183                .1
184                .push(ScopeMetrics {
185                    scope: scope_scope.clone(),
186                    metrics,
187                    schema_url: scope_schema_url.clone(),
188                });
189        }
190    }
191
192    let arn = identity::attr(&original_attrs, "aws.exporter.arn")
193        .map(identity::attr_to_string)
194        .unwrap_or_default();
195
196    buckets
197        .into_iter()
198        .map(|(_key, (namespace, scope_metrics))| {
199            let mut resource_attrs = original_attrs.clone();
200            // cloud.account.id, cloud.provider, cloud.region, aws.exporter.arn carried over
201            // (via original_attrs.clone() above); only these three are added/changed:
202            set_attr(&mut resource_attrs, "service.instance.id", &arn);
203            set_attr(
204                &mut resource_attrs,
205                "service.name",
206                namespace.as_deref().unwrap_or(UNKNOWN_NAMESPACE),
207            );
208            // So exe = service.name, not "{svc_ns}/{service.name}" — matters even though the
209            // fingerprint already requires attr_norm(service.namespace) to be empty: attr_norm
210            // trims+lowercases before comparing, so a present-but-whitespace-only value
211            // satisfies the fingerprint's emptiness gate but is not itself empty, and
212            // ProcessFromResource::build reads service.namespace with raw attr_to_string (no
213            // trim).
214            clear_attr_if_present(&mut resource_attrs, "service.namespace");
215            ResourceMetrics {
216                resource: Some(crate::proto::Resource {
217                    attributes: resource_attrs,
218                    dropped_attributes_count,
219                    entity_refs: entity_refs.clone(),
220                }),
221                scope_metrics,
222                schema_url: schema_url.clone(),
223            }
224        })
225        .collect()
226}

228/// Rewrites every `ResourceMetrics` in `req` that matches
229/// `is_cloudwatch_metric_stream_resource` into its per-namespace partition (see
230/// `partition_resource_metrics`); non-matching entries pass through completely untouched —
231/// this rewrite is purely additive for this one AWS-specific shape, not a change to the
232/// shared OTLP/metrics path.
233///
234/// Public entry point, called from `handler::ingest_firehose_metrics` only (not the shared
235/// `ingest_parsed_metrics`/`ingest_metrics` used by the plain `/v1/metrics` OTLP endpoint).
236pub fn rewrite_cloudwatch_metric_streams(
237    req: ExportMetricsServiceRequest,
238) -> ExportMetricsServiceRequest {
239    let mut out = Vec::with_capacity(req.resource_metrics.len());
240    for rm in req.resource_metrics {
241        let is_match = rm
242            .resource
243            .as_ref()
244            .map(|r| is_cloudwatch_metric_stream_resource(&r.attributes))
245            .unwrap_or(false);
246        if is_match {
247            out.extend(partition_resource_metrics(rm));
248        } else {
249            out.push(rm);
250        }
251    }
252    ExportMetricsServiceRequest {
253        resource_metrics: out,
254    }
255}
