1//! Errors surfaced by the OTLP ingestion adapter.
2//!
3//! Variants map onto HTTP status codes at the handler boundary:
4//!  - `Parse`        → 400 (malformed protobuf, malformed gzip)
5//!  - `Database`     → 503 (transient — client should retry per OTLP/HTTP spec)
6//!  - `Storage`      → 503 (transient)
7//!  - `Denied`       → 403 (a conflicting process or stream re-registration under a
8//!    different audience)
9//!
10//! 415 (Content-Type / Content-Encoding) and 413 (body limit) are enforced upstream
11//! in the axum layer stack before the request reaches the OtelError surface.
12
13use micromegas_ingestion::web_ingestion_service::IngestionServiceError;
14use thiserror::Error;
15
16/// OTLP signal name (used purely for diagnostic messages).
17#[derive(Debug, Clone, Copy)]
18pub enum Signal {
19    Logs,
20    Metrics,
21    Traces,
22}
23
24impl Signal {
25    pub fn as_str(&self) -> &'static str {
26        match self {
27            Signal::Logs => "logs",
28            Signal::Metrics => "metrics",
29            Signal::Traces => "traces",
30        }
31    }
32}
33
34impl std::fmt::Display for Signal {
35    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
36        f.write_str(self.as_str())
37    }
38}
39
40#[derive(Debug, Error)]
41pub enum OtelError {
42    /// Malformed protobuf, malformed gzip, or unrepresentable input.
43    /// Maps to 400.
44    #[error("OTLP parse error ({signal}): {message}")]
45    Parse { signal: Signal, message: String },
46
47    /// PostgreSQL transient failure. Maps to 503 + Retry-After.
48    #[error("OTLP database error ({signal}): {message}")]
49    Database { signal: Signal, message: String },
50
51    /// Object-store transient failure. Maps to 503 + Retry-After.
52    #[error("OTLP storage error ({signal}): {message}")]
53    Storage { signal: Signal, message: String },
54
55    /// A conflicting re-registration under a different audience. Maps to 403 -- never
56    /// retryable, since retrying with the same credential produces the same denial.
57    #[error("OTLP write denied ({signal}): {message}")]
58    Denied { signal: Signal, message: String },
59}
60
61impl OtelError {
62    pub fn signal(&self) -> Signal {
63        match self {
64            Self::Parse { signal, .. }
65            | Self::Database { signal, .. }
66            | Self::Storage { signal, .. }
67            | Self::Denied { signal, .. } => *signal,
68        }
69    }
70
71    /// gRPC canonical `Code` for the embedded `google.rpc.Status` proto on error responses.
72    /// Despite the name, this travels over OTLP/HTTP — the spec just reuses `google.rpc.Status`
73    /// (and its gRPC code enum) as the error body format.
74    pub fn grpc_code(&self) -> i32 {
75        match self {
76            // INVALID_ARGUMENT = 3
77            Self::Parse { .. } => 3,
78            // UNAVAILABLE = 14
79            Self::Database { .. } | Self::Storage { .. } => 14,
80            // PERMISSION_DENIED = 7
81            Self::Denied { .. } => 7,
82        }
83    }
84
85    /// HTTP status code for the response.
86    pub fn http_status(&self) -> u16 {
87        match self {
88            Self::Parse { .. } => 400,
89            Self::Database { .. } | Self::Storage { .. } => 503,
90            Self::Denied { .. } => 403,
91        }
92    }
93
94    /// True when the OTLP/HTTP spec marks this status retryable.
95    pub fn is_retryable(&self) -> bool {
96        matches!(self, Self::Database { .. } | Self::Storage { .. })
97    }
98
99    /// Client-facing message for the `google.rpc.Status` body. Strips internal
100    /// detail (raw sqlx errors, object-store messages) that the full `Display`
101    /// form would otherwise leak — those still get logged server-side via
102    /// `error!`. `Parse` keeps its detail so clients can debug malformed
103    /// payloads, since prost / decoder messages don't reference server state.
104    ///
105    /// Deliberately source-neutral (no "OTLP" prefix): this surface is also
106    /// reused by the webhook route, whose clients never spoke OTLP and would be
107    /// confused by OTLP-branded wording. The `Display` form keeps "OTLP" for
108    /// server-side logs.
109    pub fn public_message(&self) -> String {
110        match self {
111            Self::Parse { signal, message } => format!("parse error ({signal}): {message}"),
112            Self::Database { signal, .. } => format!("database error ({signal})"),
113            Self::Storage { signal, .. } => format!("storage error ({signal})"),
114            // Sanitized: no internal detail (audience labels, process/stream ids) leaks to the
115            // client. Covers both the process- and stream-side conflict guards.
116            Self::Denied { .. } => {
117                "write denied: already registered under a different audience".to_string()
118            }
119        }
120    }
121}
122
123impl OtelError {
124    /// Wraps an `IngestionServiceError` with the OTLP signal of the request that
125    /// triggered it. Forces the caller to supply the signal at the conversion
126    /// site so the resulting label can't be mismatched against the route.
127    pub fn from_ingestion(err: IngestionServiceError, signal: Signal) -> Self {
128        match err {
129            IngestionServiceError::ParseError(m) => OtelError::Parse { signal, message: m },
130            IngestionServiceError::DatabaseError(m) => OtelError::Database { signal, message: m },
131            IngestionServiceError::StorageError(m) => OtelError::Storage { signal, message: m },
132            // Reachable: `register_otel_process` runs the same conflict guard as the native
133            // `insert_process` path (see `web_ingestion_service.rs`'s doc comment), and rejects
134            // a same-`process_id`, different-audience OTLP registration with this variant --
135            // closing cross-path squatting where a credential pre-registers (via
136            // `insert_process`) the `process_id` a victim's OTLP producer would later derive.
137            IngestionServiceError::AudienceConflict {
138                process_id,
139                existing,
140                incoming,
141            } => OtelError::Denied {
142                signal,
143                message: format!(
144                    "process_id {process_id} was registered under audience {existing:?}, this \
145                     request carries {incoming:?}"
146                ),
147            },
148            // The stream-side mirror of the `AudienceConflict` arm above: `register_otel_stream`
149            // rejects a same-`stream_id`, different-audience re-registration the same way.
150            IngestionServiceError::StreamAudienceConflict {
151                stream_id,
152                existing,
153                incoming,
154            } => OtelError::Denied {
155                signal,
156                message: format!(
157                    "stream_id {stream_id} was registered under audience {existing:?}, this \
158                     request carries {incoming:?}"
159                ),
160            },
161        }
162    }
163
164    /// Prepends a diagnostic prefix (e.g. `firehose record[i] message[j]`) to the
165    /// error's message, preserving its variant (and therefore HTTP status / retryability).
166    /// Used to localize which record/message failed within a multi-message batch.
167    pub fn with_context(self, prefix: impl std::fmt::Display) -> Self {
168        match self {
169            Self::Parse { signal, message } => Self::Parse {
170                signal,
171                message: format!("{prefix}: {message}"),
172            },
173            Self::Database { signal, message } => Self::Database {
174                signal,
175                message: format!("{prefix}: {message}"),
176            },
177            Self::Storage { signal, message } => Self::Storage {
178                signal,
179                message: format!("{prefix}: {message}"),
180            },
181            Self::Denied { signal, message } => Self::Denied {
182                signal,
183                message: format!("{prefix}: {message}"),
184            },
185        }
186    }
