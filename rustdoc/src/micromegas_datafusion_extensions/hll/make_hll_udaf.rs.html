21fn is_supported(data_type: &DataType) -> bool {
22    match data_type {
23        DataType::Dictionary(_, value_type) => is_supported(value_type),
24        DataType::Utf8
25        | DataType::LargeUtf8
26        | DataType::Utf8View
27        | DataType::Binary
28        | DataType::LargeBinary
29        | DataType::Int8
30        | DataType::Int16
31        | DataType::Int32
32        | DataType::Int64
33        | DataType::UInt8
34        | DataType::UInt16
35        | DataType::UInt32
36        | DataType::UInt64
37        | DataType::Float32
38        | DataType::Float64 => true,
39        _ => false,
40    }
41}

43// Each value is wrapped in an explicit hash strategy because the default `Hash` impls for
44// strings and slices add terminators/length prefixes that other DataSketches implementations
45// do not, which would make sketches unmergeable with theirs.
46fn add_values(union: &mut HllUnion, array: &ArrayRef) -> Result<()> {
47    macro_rules! add_primitive {
48        ($arr_ty:ty, $wrap:expr) => {{
49            let arr = array.as_any().downcast_ref::<$arr_ty>().ok_or_else(|| {
50                DataFusionError::Internal(format!("{NAME}: unexpected array type"))
51            })?;
52            for v in arr.iter().flatten() {
53                union.update_value($wrap(v));
54            }
55        }};
56    }
57    macro_rules! add_str {
58        ($iter:expr) => {{
59            for v in $iter.flatten() {
60                if !v.is_empty() {
61                    union.update_value(raw_bytes::from_str(v));
62                }
63            }
64        }};
65    }
66    macro_rules! add_bytes {
67        ($iter:expr) => {{
68            for v in $iter.flatten() {
69                if !v.is_empty() {
70                    union.update_value(raw_bytes::from_slice(v));
71                }
72            }
73        }};
74    }
75
76    match array.data_type() {
77        DataType::Dictionary(_, value_type) => {
78            let unpacked = cast(array, value_type)?;
79            return add_values(union, &unpacked);
80        }
81        DataType::Null => {}
82        DataType::Utf8 => add_str!(array.as_string::<i32>().iter()),
83        DataType::LargeUtf8 => add_str!(array.as_string::<i64>().iter()),
84        DataType::Utf8View => {
85            let arr = array
86                .as_any()
87                .downcast_ref::<StringViewArray>()
88                .ok_or_else(|| {
89                    DataFusionError::Internal(format!("{NAME}: unexpected array type"))
90                })?;
91            add_str!(arr.iter());
92        }
93        DataType::Binary => {
94            let arr = array
95                .as_any()
96                .downcast_ref::<BinaryArray>()
97                .ok_or_else(|| {
98                    DataFusionError::Internal(format!("{NAME}: unexpected array type"))
99                })?;
100            add_bytes!(arr.iter());
101        }
102        DataType::LargeBinary => {
103            let arr = array
104                .as_any()
105                .downcast_ref::<LargeBinaryArray>()
106                .ok_or_else(|| {
107                    DataFusionError::Internal(format!("{NAME}: unexpected array type"))
108                })?;
109            add_bytes!(arr.iter());
110        }
111        DataType::Int8 => add_primitive!(Int8Array, sign_extend::from_i8),
112        DataType::Int16 => add_primitive!(Int16Array, sign_extend::from_i16),
113        DataType::Int32 => add_primitive!(Int32Array, sign_extend::from_i32),
114        DataType::UInt8 => add_primitive!(UInt8Array, sign_extend::from_u8),
115        DataType::UInt16 => add_primitive!(UInt16Array, sign_extend::from_u16),
116        DataType::UInt32 => add_primitive!(UInt32Array, sign_extend::from_u32),
117        DataType::Int64 => add_primitive!(Int64Array, |v: i64| v),
118        // Values above i64::MAX wrap; the hash only needs the 64 bits.
119        DataType::UInt64 => add_primitive!(UInt64Array, |v: u64| v as i64),
120        DataType::Float32 => add_primitive!(Float32Array, canonical_float::from_f32),
121        DataType::Float64 => add_primitive!(Float64Array, canonical_float::from_f64),
122        other => return plan_err!("{NAME}: unsupported input type {other}"),
123    }
124    Ok(())
125}

127/// Aggregate that builds a serialized HyperLogLog sketch from the values of a column.
128#[derive(Debug, PartialEq, Eq, Hash)]
129pub struct MakeHll {
130    signature: Signature,
131}
132
133impl MakeHll {
134    pub fn new() -> Self {
135        Self {
136            signature: Signature::any(1, Volatility::Immutable),
137        }
138    }
139}
140
141impl Default for MakeHll {
142    fn default() -> Self {
143        Self::new()
144    }
145}
146
147impl AggregateUDFImpl for MakeHll {
148    fn name(&self) -> &str {
149        NAME
150    }
151
152    fn signature(&self) -> &Signature {
153        &self.signature
154    }
155
156    fn return_type(&self, arg_types: &[DataType]) -> Result<DataType> {
157        match arg_types {
158            [t] if is_supported(t) || matches!(t, DataType::Null) => Ok(binary_type()),
159            [t] => plan_err!("{NAME}: unsupported input type {t}"),
160            _ => plan_err!("{NAME} takes exactly one argument"),
161        }
162    }
163
164    fn accumulator(&self, _args: AccumulatorArgs) -> Result<Box<dyn Accumulator>> {
165        Ok(Box::new(HllAccumulator::new(NAME, add_values)?))
166    }
167
168    fn state_fields(&self, args: StateFieldsArgs) -> Result<Vec<FieldRef>> {
169        Ok(vec![Arc::new(Field::new(
170            format_state_name(args.name, NAME),
171            binary_type(),
172            true,
173        ))])
174    }
175}

177/// Creates the `make_hll` aggregate function.
178pub fn make_hll_udaf() -> AggregateUDF {
179    AggregateUDF::from(MakeHll::new())
