1//! `ViewRegistry`: a shared `Arc<ViewFactory>` (the code-driven base plus every DDL-defined global
2//! view) rebuilt from `lakehouse_view_set_definitions` on a timer, so `telemetry-maintenance-srv`
3//! and `flight-sql-srv` both start reading/materializing a new or changed view set without a
4//! restart. Shaped after `QueryDenyList`: a Postgres-backed store, an immutable compiled snapshot
5//! behind a lock, a `refresh`/`reload` that logs and meters its own failures, and
6//! `spawn_refresh_task`.
7

26/// `{prefix}`-less env var: how often `spawn_refresh_task` calls `reload()`, and the bound on
27/// cross-replica propagation of a definition change made on a different node (the node that
28/// executed the DDL statement reloads inline instead, see `rust/public/src/servers/view_ddl.rs`).
29pub const MICROMEGAS_VIEW_DEFINITION_REFRESH_SECONDS: &str =
30    "MICROMEGAS_VIEW_DEFINITION_REFRESH_SECONDS";
31/// Default value of [`MICROMEGAS_VIEW_DEFINITION_REFRESH_SECONDS`].
32pub const DEFAULT_REFRESH_SECONDS: u64 = 60;
33
34fn refresh_interval() -> Duration {
35    let secs = std::env::var(MICROMEGAS_VIEW_DEFINITION_REFRESH_SECONDS)
36        .ok()
37        .and_then(|s| s.parse::<u64>().ok())
38        .unwrap_or(DEFAULT_REFRESH_SECONDS)
39        .max(1);
40    Duration::from_secs(secs)
41}

43fn failure_tags(view_set_name: &str) -> &'static PropertySet {
44    PropertySet::find_or_create(vec![Property::new(
45        "view_set_name",
46        intern_string(view_set_name),

50/// Hashes the `(view_set_name, updated_at)` tuples of `rows`, in the order given (already
51/// `(update_group, view_set_name)`-ordered by `ViewDefinitionStore::list`) -- what `reload()`
52/// compares against `loaded_digest` to short-circuit a steady-state tick to one `SELECT`.
53fn digest_of(rows: &[ViewDefinitionRow]) -> u64 {
54    let mut hasher = DefaultHasher::new();
55    for row in rows {
56        row.view_set_name.hash(&mut hasher);
57        row.updated_at
58            .timestamp_nanos_opt()
59            .unwrap_or(0)
60            .hash(&mut hasher);
61    }
62    hasher.finish()
63}

65/// What a [`build_factory`] call is for, which decides how a skipped definition is reported.
66#[derive(Clone, Copy, Debug, PartialEq, Eq)]
67pub enum BuildPurpose {
68    /// The snapshot `current()` will serve: a skipped definition is real degradation -- something
69    /// that used to work no longer does, or never did -- so it is `warn!`-logged and metered.
70    LiveSnapshot,
71    /// A speculative build the DDL executor runs inside its own transaction, on row sets that may
72    /// deliberately be missing a view another one reads. A skip here is the expected signal, not
73    /// degradation: it is returned in [`BuildFailure`] for the caller to act on -- either to
74    /// refuse the statement, naming the definition and reason, or to ignore a row that was already
75    /// broken before the mutation. Logging it would report a rejected statement, or a pre-existing
76    /// breakage the live build already reports every tick, as a fresh registry failure.
77    Probe,
78}
79
80/// A definition [`build_factory`] skipped, with the reason it was skipped, so a [`BuildPurpose::
81/// Probe`] caller can name both in the error it raises.
82#[derive(Clone, Debug)]
83pub struct BuildFailure {
84    pub view_set_name: String,
85    /// The `{:#}` rendering of the parse/build/validation error, alternate-formatted to keep the
86    /// anyhow context chain.
87    pub error: String,
88}
89
90impl std::fmt::Display for BuildFailure {
91    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
92        write!(f, "{} ({})", self.view_set_name, self.error)
93    }
94}
95
96fn skip(
97    purpose: BuildPurpose,
98    view_set_name: String,
99    stage: &str,
100    e: anyhow::Error,
101) -> BuildFailure {
102    let error = format!("{e:#}");
103    if purpose == BuildPurpose::LiveSnapshot {
104        warn!("view_registry: '{view_set_name}' failed {stage}, skipping: {error}");
105        imetric!(
106            "view_definition_load_failure",
107            "count",
108            failure_tags(&view_set_name),
109            1_u64
110        );
111    }
112    BuildFailure {
113        view_set_name,
114        error,
115    }
116}

118/// Builds a factory from an explicit row set instead of `store.list()`, so a caller already
119/// holding rows inside an open transaction (the DDL executor's dependent-protection check) can
120/// validate against them without a second, pool-backed read that would not see its own
121/// uncommitted write. `reload()` calls this too, after its own `store.list()`. `async` because
122/// building each row's `SqlBatchView` is itself `async` and needs `runtime`/`lake`/
123/// `session_configurator` to plan the extract query.
124///
125/// Builds incrementally, in the order `rows` is given (`(update_group, view_set_name)`): `let mut
126/// factory = (**base).clone();` then, per row, build and validate its `SqlBatchView` against
127/// `Arc::new(factory.clone())` -- the built-ins plus every already-folded-in row -- and only then
128/// `factory.add_global_view(...)`. A row that fails to parse, build, or validate is **skipped, not
129/// fatal**: collected into the returned failed-set with its reason, so one broken definition
130/// cannot take out every other view set. Whether that skip is also logged and metered is
131/// `purpose`'s call.
132pub async fn build_factory(
133    base: &Arc<ViewFactory>,
134    rows: &[ViewDefinitionRow],
135    runtime: Arc<RuntimeEnv>,
136    lake: Arc<DataLakeConnection>,
137    session_configurator: Arc<dyn SessionConfigurator>,
138    purpose: BuildPurpose,
139) -> Result<(Arc<ViewFactory>, Vec<BuildFailure>)> {
140    let mut factory = (**base).clone();
141    let mut failed = Vec::new();
142    for row in rows {
143        let name = row.view_set_name.clone();
144        let def = match row.clone().into_definition() {
145            Ok(def) => def,
146            Err(e) => {
147                failed.push(skip(purpose, name, "to parse", e));
148                continue;
149            }
150        };
151        let factory_so_far = Arc::new(factory.clone());
152        let view = match build_sql_batch_view(
153            &def,
154            runtime.clone(),
155            lake.clone(),
156            factory_so_far.clone(),
157            session_configurator.clone(),
158        )
159        .await
160        {
161            Ok(view) => view,
162            Err(e) => {
163                failed.push(skip(purpose, name, "to build", e));
164                continue;
165            }
166        };
167        if let Err(e) = validate_view_definition(
168            &def,
169            &view,
170            &factory_so_far,
171            runtime.clone(),
172            lake.clone(),
173            session_configurator.clone(),
174        )
175        .await
176        {
177            failed.push(skip(purpose, name, "validation", e));
178            continue;
179        }
180        factory.add_global_view(Arc::new(view));
181    }
182    Ok((Arc::new(factory), failed))
183}

185/// A shared `Arc<ViewFactory>` (base + N DDL-defined `SqlBatchView`s) rebuilt from
186/// `lakehouse_view_set_definitions` on a timer. See the module doc comment.
187pub struct ViewRegistry {
188    base: Arc<ViewFactory>,
189    store: Arc<dyn ViewDefinitionStore>,
190    runtime: Arc<RuntimeEnv>,
191    lake: Arc<DataLakeConnection>,
192    session_configurator: Arc<dyn SessionConfigurator>,
193    current: RwLock<Arc<ViewFactory>>,
194    loaded_digest: RwLock<Option<u64>>,
195    failed_view_sets: RwLock<Vec<String>>,
196    /// Held across all of `reload()` -- list, build, and swap -- not just the swap. Without this,
197    /// a periodic reload that listed rows before a DDL-triggered reload could block behind the
198    /// DDL's own reload and then swap the pre-mutation factory back in over the DDL executor's
199    /// post-commit one, a lost update that would stay live for up to
200    /// `MICROMEGAS_VIEW_DEFINITION_REFRESH_SECONDS`. Mirrors `QueryDenyList::write_lock`'s
201    /// rationale.
202    reload_mutex: tokio::sync::Mutex<()>,
203}
204
205impl std::fmt::Debug for ViewRegistry {
206    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
207        f.debug_struct("ViewRegistry")
208            .field(
209                "loaded_views",
210                &self.current.read().expect("lock").get_global_views().len(),
211            )
212            .field(
213                "failed_view_sets",
214                &self.failed_view_sets.read().expect("lock").len(),
215            )
216            .finish()
217    }
218}

220impl ViewRegistry {
221    pub fn new(
222        base: Arc<ViewFactory>,
223        store: Arc<dyn ViewDefinitionStore>,
224        runtime: Arc<RuntimeEnv>,
225        lake: Arc<DataLakeConnection>,
226        session_configurator: Arc<dyn SessionConfigurator>,
227    ) -> Self {
228        Self {
229            current: RwLock::new(base.clone()),
230            base,
231            store,
232            runtime,
233            lake,
234            session_configurator,
235            loaded_digest: RwLock::new(None),
236            failed_view_sets: RwLock::new(Vec::new()),
237            reload_mutex: tokio::sync::Mutex::new(()),
238        }
239    }
240
241    /// The current snapshot: the base factory before the first successful `reload()`, or the base
242    /// plus every definition that built and validated on the last successful build afterward.
243    pub fn current(&self) -> Arc<ViewFactory> {
244        self.current.read().expect("lock").clone()
245    }
246
247    /// Names skipped by the last build, for introspection/tests.
248    pub fn failed_view_sets(&self) -> Vec<String> {
249        self.failed_view_sets.read().expect("lock").clone()
250    }
251
252    /// Thin wrapper over [`build_factory`] supplying this registry's `base`/`runtime`/`lake`/
253    /// `session_configurator`.
254    pub async fn build_from_rows(
255        &self,
256        rows: &[ViewDefinitionRow],
257        purpose: BuildPurpose,
258    ) -> Result<(Arc<ViewFactory>, Vec<BuildFailure>)> {
259        build_factory(
260            &self.base,
261            rows,
262            self.runtime.clone(),
263            self.lake.clone(),
264            self.session_configurator.clone(),
265            purpose,
266        )
267        .await
268    }
269
270    /// The dependent-protection refusal: builds a factory from `pre_rows` and one from
271    /// `post_rows`, then refuses with a named error if any definition present in `post_rows` --
272    /// including the row just written -- both fails to build against `post_rows` and was not
273    /// already in the pre-mutation failed set. Exact rather than textual: it uses the real
274    /// planner, so it catches a removed column as readily as a removed view set.
275    ///
276    /// Both builds are [`BuildPurpose::Probe`]s: a skip is what this check is looking for, so it
277    /// belongs in the refusal the caller returns -- with the planner's own reason, the only place
278    /// the admin can now read it -- not in the service log as a registry failure.
279    pub async fn check_dependents_survive(
280        &self,
281        pre_rows: &[ViewDefinitionRow],
282        post_rows: &[ViewDefinitionRow],
283    ) -> Result<()> {
284        let (_, pre_failed) = self.build_from_rows(pre_rows, BuildPurpose::Probe).await?;
285        let (_, post_failed) = self.build_from_rows(post_rows, BuildPurpose::Probe).await?;
286        let pre_failed_set: HashSet<&str> = pre_failed
287            .iter()
288            .map(|f| f.view_set_name.as_str())
289            .collect();
290        let newly_broken: Vec<String> = post_failed
291            .iter()
292            .filter(|f| !pre_failed_set.contains(f.view_set_name.as_str()))
293            .map(BuildFailure::to_string)
294            .collect();
295        if !newly_broken.is_empty() {
296            anyhow::bail!(
297                "this change would break the following view definition(s), which built \
298                 successfully before it: {}",
299                newly_broken.join(", ")
300            );
301        }
302        Ok(())
303    }
304
305    /// Refreshes `current()` from `lakehouse_view_set_definitions`.
306    ///
307    /// 0. Acquires `reload_mutex` and holds it through every step below.
308    /// 1. `store.list()` -> rows ordered by `(update_group, view_set_name)`.
309    /// 2. Hashes the `(view_set_name, updated_at)` tuples; if unchanged from `loaded_digest`
310    ///    **and** the previous build's failed set was empty, returns -- one `SELECT`, no
311    ///    planning. A digest match after a failed build does *not* short-circuit, so a row that
312    ///    failed for a transient reason is retried every tick.
313    /// 3. `build_factory(...)`.
314    /// 4. A row that fails to build or validate is skipped, not fatal (see [`build_factory`]).
315    /// 5. Swaps `current`, updates `loaded_digest` and `failed_view_sets`.
316    pub async fn reload(&self) -> Result<()> {
317        let _guard = self.reload_mutex.lock().await;
318        let rows = self.store.list().await?;
319        let digest = digest_of(&rows);
320        let previous_build_was_healthy = self.failed_view_sets.read().expect("lock").is_empty();
321        if previous_build_was_healthy && Some(digest) == *self.loaded_digest.read().expect("lock") {
322            return Ok(());
323        }
324        let (factory, failed) = self
325            .build_from_rows(&rows, BuildPurpose::LiveSnapshot)
326            .await?;
327        *self.current.write().expect("lock") = factory;
328        *self.loaded_digest.write().expect("lock") = Some(digest);
329        *self.failed_view_sets.write().expect("lock") = failed
330            .into_iter()
331            .map(|f| f.view_set_name)
332            .collect::<Vec<_>>();
333        Ok(())
334    }
335
336    /// Spawns the background task that calls [`Self::reload`] once immediately and then on every
337    /// [`MICROMEGAS_VIEW_DEFINITION_REFRESH_SECONDS`] tick, until `shutdown` resolves. A failed
338    /// `reload()` is `warn!`-logged here and discarded -- the previous snapshot is kept (the same
339    /// fail-open treatment `QueryDenyList::spawn_refresh_task` gives a failed periodic refresh).
340    pub fn spawn_refresh_task(
341        self: Arc<Self>,
342        shutdown: impl Future<Output = ()> + Send + 'static,
343    ) {
344        tokio::spawn(async move {
345            let mut shutdown = Box::pin(shutdown);
346            loop {
347                if let Err(e) = self.reload().await {
348                    warn!("view_registry: reload failed, keeping previous snapshot: {e:#}");
349                }
350                tokio::select! {
351                    _ = &mut shutdown => break,
352                    _ = tokio::time::sleep(refresh_interval()) => {}
353                }
354            }
355        });
356    }
