1//! [`default_view_factory`] makes the default [`ViewFactory`], giving users access to view instances, grouped in sets.
2//!
3//! # View sets
4//!
5//! A ViewFactory defines the available view sets and can instanciate view instances.
6//! All view instances in a set have the same schema.
7//! Some view instances are global (their view_instance_id is 'global').
8//! Global view instances are implicitly accessible to SQL queries.
9//! Non-global view instances are accessible using the table function `view_instance`. See [ViewInstanceTableFunction](super::view_instance_table_function::ViewInstanceTableFunction).
10//!
11//! ## log_entries
12//!
13//! | field        | type                        | description                                               |
14//! |------------- |-----------------------------|-----------------------------------------------------------|
15//! |process_id    |Utf8                         | unique id of the process, references the processes table  |
16//! |exe           |Utf8                         | filename of the process                                   |
17//! |username      |Utf8                         | username of the process                                   |
18//! |computer      |Utf8                         | computer name of the process                              |
19//! |time          |UTC Timestamp (nanoseconds)  | time of the log entry event                               |
20//! |target        |Utf8                         | category or module name of the log entry                  |
21//! |level         |int32                        | verbosity level (Fatal=1, Error=2, Warning=3, Info=4, Debug=5, Trace=6)|                                           |
22//! |msg           |Utf8                         | message                                                   |
23//! |audience      |Dictionary(Int32, Utf8)      | the audience of the block this log entry came from (the block's own stamp), never `NULL` |
24//!
25//! ### log_entries view instances
26//!
27//! The implicit use of the `log_entries` table corresponds to the 'global' instance, which contains the log entries of all the processes.
28//!
29//! Except the 'global' instance, the instance_id refers to any process_id. `view_instance('log_entries', process_id)` contains that process's log. Process-specific views are materialized just-in-time and can provide much better query performance compared to the 'global' instance.
30//!
31//! ## measures
32//!
33//! | field        | type                        | description                                               |
34//! |------------- |-----------------------------|-----------------------------------------------------------|
35//! |process_id    |Utf8                         | unique id of the process, references the processes table  |
36//! |exe           |Utf8                         | filename of the process                                   |
37//! |username      |Utf8                         | username of the process                                   |
38//! |computer      |Utf8                         | computer name of the process                              |
39//! |time          |UTC Timestamp (nanoseconds)  | time of the measure event                                 |
40//! |target        |Utf8                         | category or module name of the measure                    |
41//! |name          |Utf8                         | name of the measure                                       |
42//! |unit          |Utf8                         | unit of measure                                           |
43//! |value         |Float64                      | value measured                                            |
44//! |audience      |Dictionary(Int32, Utf8)      | the audience of the block this measure came from (the block's own stamp), never `NULL` |
45//!
46//!
47//! ### measures view instances
48//!
49//! The implicit use of the `measures` table corresponds to the 'global' instance, which contains the metrics of all the processes.
50//!
51//! Except the 'global' instance, the instance_id refers to any process_id. `view_instance('measures', process_id)` contains that process's metrics. Process-specific views are materialized just-in-time and can provide much better query performance compared to the 'global' instance.
52//!
53//! ## log_stats
54//!
55//! Materialized aggregate of `log_entries` by process, minute, level, and target -- one global
56//! instance, implicitly available (no `view_instance(...)` form).
57//!
58//! | field        | type                        | description                                                |
59//! |------------- |-----------------------------|------------------------------------------------------------|
60//! |time_bin      |UTC Timestamp (nanoseconds)  | 1-minute time bucket for aggregation                        |
61//! |process_id    |Dictionary(Int32, Utf8)      | process unique id                                          |
62//! |level         |Int32                        | log level (see `log_entries` above)                        |
63//! |target        |Dictionary(Int32, Utf8)      | category or module name of the aggregated logs             |
64//! |count         |Int64                        | number of log entries in this aggregation                  |
65//! |audience      |Dictionary(Int32, Utf8)      | the audience of the blocks aggregated into this row (the blocks' own stamp), never `NULL` |
66//!
67//! ## thread_spans
68//!
69//! | field        | type                        | description                                                |
70//! |------------- |-----------------------------|------------------------------------------------------------|
71//! |id            |Int64                        | span id, unique within this thread                         |
72//! |parent        |Int64                        | span id of the calling span                                |
73//! |depth         |UInt32                       | call stack depth                                           |
74//! |hash          |UInt32                       | identifies a call site (name, filename, line)              |
75//! |begin         |UTC Timestamp (nanoseconds)  | when the span started its execution                        |
76//! |end           |UTC Timestamp (nanoseconds)  | when the span finished its execution                       |
77//! |duration      |Int64 (nanoseconds)          | end-begin                                                  |
78//! |name          |Utf8                         | name of the span, usually a function name                  |
79//! |target        |Utf8                         | category or module name                                    |
80//! |filename      |Utf8                         | name or path of the source file where the span is coded    |
81//! |line          |UInt32                       | line number in the file where the span can be found        |
82//!
83//! ### thread_spans view instances
84//!
85//! There is no 'global' instance in the 'thread_spans' view set, there is therefore no implicit thread_spans table availble.
86//! Users can call the table function `view_instance('thread_spans', stream_id)` to query the spans in the thread associated with the specified stream_id.
87//!
88//! ## async_events
89//!
90//! | field        | type                        | description                                               |
91//! |------------- |-----------------------------|-----------------------------------------------------------|
92//! |stream_id     |Dictionary(Int32, Utf8)      | identifier of the thread stream that emitted the event   |
93//! |block_id      |Dictionary(Int32, Utf8)      | unique identifier of the event block                     |
94//! |time          |UTC Timestamp (nanoseconds)  | time when the async event occurred                       |
95//! |event_type    |Dictionary(Int32, Utf8)      | type of event: "begin" or "end"                          |
96//! |span_id       |Int64                        | unique async span identifier                             |
97//! |parent_span_id|Int64                        | span id of the parent async span                         |
98//! |name          |Dictionary(Int32, Utf8)      | name of the async span, usually a function name          |
99//! |filename      |Dictionary(Int32, Utf8)      | name or path of the source file where the span is coded  |
100//! |target        |Dictionary(Int32, Utf8)      | category or module name                                   |
101//! |line          |UInt32                       | line number in the file where the span can be found      |
102//!
103//! ### async_events view instances
104//!
105//! There is no 'global' instance in the 'async_events' view set, there is therefore no implicit async_events table available.
106//! Users can call the table function `view_instance('async_events', process_id)` to query the async events in all thread streams associated with the specified process_id.
107//! Process-specific views are materialized just-in-time and can provide good query performance.
108//!
109//! The schema is optimized for high-frequency data. Process information (exe, username, computer, etc.) can be joined when needed:
110//! ```sql
111//! SELECT ae.*, p.exe, p.username, p.computer
112//! FROM view_instance('async_events', process_id) ae
113//! JOIN streams s ON ae.stream_id = s.stream_id  
114//! JOIN processes p ON s.process_id = p.process_id
115//! ```
116//!
117//! ## net_spans
118//!
119//! | field            | type                        | description                                                             |
120//! |------------------|-----------------------------|-------------------------------------------------------------------------|
121//! |process_id        |Dictionary(Int32, Utf8)      | process unique id                                                       |
122//! |stream_id         |Dictionary(Int32, Utf8)      | net stream unique id                                                    |
123//! |span_id           |Int64                        | unique span id within the stream                                        |
124//! |parent_span_id    |Int64                        | span id of the enclosing span (-1 sentinel at Connection roots)         |
125//! |depth             |UInt32                       | tree depth (0 = Connection)                                             |
126//! |kind              |Dictionary(Int32, Utf8)      | one of `connection` / `object` / `property` / `rpc`                     |
127//! |name              |Dictionary(Int32, Utf8)      | span name (connection / object / property / function name)              |
128//! |connection_name   |Dictionary(Int32, Utf8)      | enclosing connection name (denormalized onto every row)                 |
129//! |is_outgoing       |Boolean                      | direction inherited from the enclosing connection                       |
130//! |begin_bits        |Int64                        | cumulative bit offset within the parent span (0 at the Connection root) |
131//! |end_bits          |Int64                        | `begin_bits + bit_size`                                                 |
132//! |bit_size          |Int64                        | inclusive number of bits attributed to this span                        |
133//! |begin_time        |UTC Timestamp (nanoseconds)  | timestamp of the span's Begin event                                     |
134//! |end_time          |UTC Timestamp (nanoseconds)  | timestamp of the span's End event (equals `begin_time` for properties)  |
135//!
136//! ### net_spans view instances
137//!
138//! There is no 'global' instance in the 'net_spans' view set. Users call
139//! `view_instance('net_spans', process_id)` to query the network bandwidth spans emitted by a
140//! given process's net stream. Partitions are materialized just-in-time.
141//!
142//! ## processes
143//!
144//! | field        | type                        | description                                                |
145//! |------------- |-----------------------------|------------------------------------------------------------|
146//! |process_id    |Utf8                         | process unique id                                          |
147//! |exe           |Utf8                         | filename of the process                                    |
148//! |username      |Utf8                         | username of the process                                    |
149//! |realname      |Utf8                         | real name of the user launching the process                |
150//! |computer      |Utf8                         | name of the computer or vm                                 |
151//! |distro        |Utf8                         | name of operating system                                   |
152//! |cpu_brand     |Utf8                         | identifies the cpu                                         |
153//! |tsc frequency |Int64                        | number of ticks per second                                 |
154//! |start_time    |UTC Timestamp (nanoseconds)  | when the process started (as reported by the instrumented process) |
155//! |start_ticks   |Int64                        | tick count associated with start_time                      |
156//! |insert_time   |UTC Timestamp (nanoseconds)  | server-side timestamp when the process metedata was received |
157//! |parent_process_id |Utf8                     | unique id of the parent process                            |
158//! |properties | Array of {key: utf8, value: utf8} | self-reported metadata by the process                   |
159//! |last_update_time |UTC Timestamp (nanoseconds) | when this process's data was last updated                |
160//! |last_block_end_ticks |Int64                   | tick count when the last block ended                       |
161//! |last_block_end_time |UTC Timestamp (nanoseconds) | timestamp when the last block ended                     |
162//! |audience      |Dictionary(Int32, Utf8)      | this process's write audience, resolved from its blocks' own stamps via `max(audience)`; never `NULL` |
163//!
164//! There is only one instance in this view set and it is implicitly available.

238/// A trait for creating views.
239pub trait ViewMaker: Send + Sync + Debug {
240    /// Creates a new view with the given instance ID.
241    fn make_view(&self, view_instance_id: &str) -> Result<Arc<dyn View>>;
242
243    /// Returns the schema hash for views created by this maker.
244    fn get_schema_hash(&self) -> Vec<u8>;
245
246    /// Returns the schema for views created by this maker.
247    fn get_schema(&self) -> Arc<Schema>;
248}
249
250/// A factory for creating and managing views.
251#[derive(Debug, Clone)]
252pub struct ViewFactory {
253    view_sets: HashMap<String, Arc<dyn ViewMaker>>,
254    global_views: Vec<Arc<dyn View>>,
255}
256
257impl ViewFactory {
258    pub fn new(global_views: Vec<Arc<dyn View>>) -> Self {
259        Self {
260            view_sets: HashMap::new(),
261            global_views,
262        }
263    }
264
265    pub fn get_global_views(&self) -> &[Arc<dyn View>] {
266        &self.global_views
267    }
268
269    pub fn get_global_view(&self, view_name: &str) -> Option<Arc<dyn View>> {
270        self.global_views
271            .iter()
272            .find(|v| *(v.get_view_set_name()) == view_name)
273            .cloned()
274    }
275
276    pub fn add_global_view(&mut self, view: Arc<dyn View>) {
277        self.global_views.push(view);
278    }
279
280    pub fn add_view_set(&mut self, view_set_name: String, maker: Arc<dyn ViewMaker>) {
281        self.view_sets.insert(view_set_name, maker);
282    }
283
284    pub fn get_view_sets(&self) -> &HashMap<String, Arc<dyn ViewMaker>> {
285        &self.view_sets
286    }
287
288    pub fn make_view(&self, view_set_name: &str, view_instance_id: &str) -> Result<Arc<dyn View>> {
289        if let Some(maker) = self.view_sets.get(view_set_name) {
290            maker.make_view(view_instance_id)
291        } else {
292            anyhow::bail!("view set {view_set_name} not found");
293        }
294    }
295}

297/// Creates the default `ViewFactory` with all built-in views.
298///
299/// `default_audience` is the deployment's `MICROMEGAS_DEFAULT_AUDIENCE`, which the
300/// `blocks` view binds when it materializes. Callers source it from
301/// `LakehouseContext::default_audience` so every role that builds a factory bakes the same value
302/// into partitions -- notably the maintenance role, which is the one that actually materializes
303/// the six global views.
304pub async fn default_view_factory(
305    runtime: Arc<RuntimeEnv>,
306    lake: Arc<DataLakeConnection>,
307    default_audience: Arc<str>,
308) -> Result<ViewFactory> {
309    let blocks_view = Arc::new(BlocksView::new(default_audience)?);
310    let processes_view = Arc::new(
311        make_processes_view(
312            runtime.clone(),
313            lake.clone(),
314            Arc::new(ViewFactory::new(vec![blocks_view.clone()])),
315        )
316        .await?,
317    );
318    let streams_view = Arc::new(
319        make_streams_view(
320            runtime.clone(),
321            lake.clone(),
322            Arc::new(ViewFactory::new(vec![blocks_view.clone()])),
323        )
324        .await?,
325    );
326    let images_view_maker = Arc::new(ImagesViewMaker {});
327    let log_view_maker = Arc::new(LogViewMaker {});
328    let metrics_view_maker = Arc::new(MetricsViewMaker {});
329
330    // Create base views first
331    let global_views = vec![
332        log_view_maker.make_view("global")?,
333        metrics_view_maker.make_view("global")?,
334        processes_view,
335        streams_view,
336        blocks_view,
337    ];
338    let mut factory = ViewFactory::new(global_views);
339    factory.add_view_set(String::from("images"), images_view_maker);
340    factory.add_view_set(String::from("log_entries"), log_view_maker.clone());
341    factory.add_view_set(String::from("measures"), metrics_view_maker);
342
343    // Create the factory as Arc to pass to other view makers
344    let factory_arc = Arc::new(factory);
345
346    // Create log_stats view with access to the complete factory (including log_entries)
347    let log_stats_view =
348        Arc::new(make_log_stats_view(runtime.clone(), lake.clone(), factory_arc.clone()).await?);
349
350    // Clone factory and add log_stats view
351    let mut updated_factory = (*factory_arc).clone();
352    updated_factory.add_global_view(log_stats_view);
353
354    // Add async_events view maker
355    updated_factory.add_view_set(
356        String::from("async_events"),
357        Arc::new(AsyncEventsViewMaker::new(Arc::new(updated_factory.clone()))),
358    );
359
360    // Add net_spans view maker
361    updated_factory.add_view_set(
362        String::from("net_spans"),
363        Arc::new(NetSpansViewMaker::new(Arc::new(updated_factory.clone()))),
364    );
365
366    // Add thread_spans view maker
367    updated_factory.add_view_set(
368        String::from("thread_spans"),
369        Arc::new(ThreadSpansViewMaker::new(Arc::new(updated_factory.clone()))),
370    );
371
372    // Add otel_spans view maker (per-process JIT only — no global instance)
373    updated_factory.add_view_set(String::from("otel_spans"), Arc::new(OtelSpansViewMaker {}));
374
375    Ok(updated_factory)
376}
