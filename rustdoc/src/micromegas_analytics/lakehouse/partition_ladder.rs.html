1//! Aligned bucket ladder (minute, 5 min, hour, 6 h, day) used to size JIT partitions from the
2//! query range, and the planner that turns per-minute block counts into the buckets to
3//! materialize.
4//!
5//! Every rung divides the next one and `duration_trunc` aligns to the Unix epoch, so buckets of
6//! different rungs nest and never straddle each other.
7
8use crate::time::TimeRange;
9use chrono::{DateTime, DurationRound, TimeDelta, Utc};
10
11/// Bucket widths, finest first. Each rung divides the next one.
12pub const LADDER: [TimeDelta; 5] = [
13    TimeDelta::seconds(60),
14    TimeDelta::seconds(5 * 60),
15    TimeDelta::seconds(60 * 60),
16    TimeDelta::seconds(6 * 60 * 60),
17    TimeDelta::seconds(24 * 60 * 60),
18];
19
20/// How many buckets a query should be spread over, give or take a rung.
21pub const TARGET_BUCKETS_PER_QUERY: i64 = 24;
22
23/// Largest rung `r` with `range / r >= TARGET_BUCKETS_PER_QUERY`; the finest rung if none.
24pub fn select_rung(query_range: TimeRange) -> TimeDelta {
25    let range = query_range.end - query_range.begin;
26    LADDER
27        .iter()
28        .rev()
29        .find(|rung| range.num_seconds() / rung.num_seconds().max(1) >= TARGET_BUCKETS_PER_QUERY)
30        .copied()
31        .unwrap_or(LADDER[0])
32}
33
34/// The rung right below `rung`, if any.
35pub fn finer_rung(rung: TimeDelta) -> Option<TimeDelta> {
36    let index = LADDER.iter().position(|r| *r == rung)?;
37    index.checked_sub(1).map(|i| LADDER[i])
38}

40/// One insert-time bucket the planner decided to materialize.
41#[derive(Debug, Clone, Copy, PartialEq, Eq)]
42pub struct PlannedBucket {
43    /// Half-open, rung-aligned.
44    pub range: TimeRange,
45    pub rung: TimeDelta,
46    /// True when no new block can land in the bucket anymore (see `plan_buckets`).
47    pub closed: bool,
48    pub nb_blocks: i64,
49    pub nb_objects: i64,
50}
51
52/// Block and object counts of one non-empty `finest`-wide bucket, keyed by its start.
53pub type BucketCount = (DateTime<Utc>, i64, i64);
54
55fn sum_counts(counts: &[BucketCount], range: TimeRange) -> (i64, i64) {
56    let first = counts.partition_point(|c| c.0 < range.begin);
57    counts[first..]
58        .iter()
59        .take_while(|c| c.0 < range.end)
60        .fold((0, 0), |(blocks, objects), c| (blocks + c.1, objects + c.2))
61}
62
63#[allow(clippy::too_many_arguments)]
64fn plan_bucket(
65    bucket: TimeRange,
66    rung: TimeDelta,
67    finest: TimeDelta,
68    closed_before: DateTime<Utc>,
69    counts: &[BucketCount],
70    single_finest_bucket: bool,
71    max_nb_objects: i64,
72    out: &mut Vec<PlannedBucket>,
73) {
74    let (nb_blocks, nb_objects) = sum_counts(counts, bucket);
75    // The count query is skipped for a single-bucket range, which cannot need descent.
76    if nb_blocks == 0 && !single_finest_bucket {
77        return;
78    }
79    let closed = bucket.end <= closed_before;
80    let finer = if rung > finest {
81        finer_rung(rung)
82    } else {
83        None
84    };
85    match finer {
86        Some(finer) if !closed || nb_objects > max_nb_objects => {
87            let mut begin = bucket.begin;
88            while begin < bucket.end {
89                plan_bucket(
90                    TimeRange::new(begin, begin + finer),
91                    finer,
92                    finest,
93                    closed_before,
94                    counts,
95                    false,
96                    max_nb_objects,
97                    out,
98                );
99                begin += finer;
100            }
101        }
102        _ => out.push(PlannedBucket {
103            range: bucket,
104            rung,
105            closed,
106            nb_blocks,
107            nb_objects,
108        }),
109    }
110}

112/// Plans the buckets to materialize over `insert_range` (rung-aligned) for a query at `rung`.
113///
114/// A bucket is emitted at `rung` when it is closed (`end <= now - grace`, `grace` absorbing the
115/// commit lag between the insert-time stamp and the row becoming visible) and holds at most
116/// `max_nb_objects` objects. Otherwise it descends one rung at a time, down to `finest`, where it
117/// is emitted as is. The open edge is therefore only ever materialized at `finest`, so every
118/// coarser partition was written while its bucket was closed.
119///
120/// `counts` holds one entry per non-empty `finest`-wide bucket, ascending. It may be empty when
121/// `insert_range` is a single `finest` bucket: the planner then emits that bucket as non-empty.
122pub fn plan_buckets(
123    insert_range: TimeRange,
124    rung: TimeDelta,
125    finest: TimeDelta,
126    now: DateTime<Utc>,
127    grace: TimeDelta,
128    counts: &[BucketCount],
129    max_nb_objects: i64,
130) -> Vec<PlannedBucket> {
131    let mut out = vec![];
132    let single_finest_bucket = insert_range.end - insert_range.begin <= finest;
133    let closed_before = now - grace;
134    let mut begin = insert_range.begin;
135    while begin < insert_range.end {
136        plan_bucket(
137            TimeRange::new(begin, begin + rung),
138            rung,
139            finest,
140            closed_before,
141            counts,
142            single_finest_bucket,
143            max_nb_objects,
144            &mut out,
145        );
146        begin += rung;
147    }
148    out
149}

151/// Truncates `range` outward to whole `rung` buckets.
152pub fn align_outward(range: TimeRange, rung: TimeDelta) -> anyhow::Result<TimeRange> {
153    Ok(TimeRange::new(
154        range.begin.duration_trunc(rung)?,
155        range.end.duration_trunc(rung)? + rung,
156    ))
