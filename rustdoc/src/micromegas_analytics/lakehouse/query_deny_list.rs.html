1//! Admin-managed query deny list (`tasks/query_deny_list_plan.md`): a small set of admin-authored
2//! rules, stored in Postgres, cached in every `flight-sql-srv` replica, and evaluated at the front
3//! of `execute_query` -- before the session context is built and before planning -- so a matching
4//! query is rejected for a few microseconds of work instead of a memory-pool reservation and a
5//! wave of object-store reads.
6//!
7//! A rule is a boolean SQL expression (§3 of the plan) over a fixed, documented match context
8//! ([`match_schema`]) -- parsed and evaluated by DataFusion itself, so there is no grammar of our
9//! own to specify or keep in sync, and no evaluator of our own to get subtly wrong. Rules stand
10//! until an admin removes them explicitly; there is no expiry.

34/// `{prefix}`-less env var: snapshot refresh / `last_hit_at` flush interval, and the bound on
35/// cross-replica propagation of a newly inserted or removed rule.
36pub const MICROMEGAS_QUERY_DENY_REFRESH_SECONDS: &str = "MICROMEGAS_QUERY_DENY_REFRESH_SECONDS";
37/// Default value of [`MICROMEGAS_QUERY_DENY_REFRESH_SECONDS`].
38pub const DEFAULT_REFRESH_SECONDS: u64 = 10;
39/// Env var: cap on the number of rules in force at once (bounds per-query evaluation cost).
40pub const MICROMEGAS_QUERY_DENY_MAX_RULES: &str = "MICROMEGAS_QUERY_DENY_MAX_RULES";
41/// Default value of [`MICROMEGAS_QUERY_DENY_MAX_RULES`].
42pub const DEFAULT_MAX_RULES: usize = 100;
43
44fn refresh_interval() -> Duration {
45    let secs = std::env::var(MICROMEGAS_QUERY_DENY_REFRESH_SECONDS)
46        .ok()
47        .and_then(|s| s.parse::<u64>().ok())
48        .unwrap_or(DEFAULT_REFRESH_SECONDS)
49        .max(1);
50    Duration::from_secs(secs)
51}

66/// Literal-stripped fingerprint of a statement: the first 16 hex chars of the SHA-256 of the
67/// normalized token stream. Tokenizes internally; the token stream is an implementation detail
68/// and does not appear in the signature.
69///
70/// This is what makes the dashboard case work: consecutive refreshes of the same panel differ
71/// only in their time-range literals, so they collapse to one fingerprint. Tokenization failure
72/// (unparseable SQL) falls back to hashing the whitespace-collapsed raw text, so a fingerprint
73/// always exists.
74pub fn fingerprint_of(sql: &str) -> String {
75    let normalized = match Tokenizer::new(&GenericDialect {}, sql).tokenize() {
76        Ok(tokens) => normalize_tokens(&tokens),
77        Err(_) => collapse_whitespace(sql),
78    };
79    hash_hex16(&normalized)
80}
81
82/// Joins the normalized token text with a single space: every `Token::Number`/string-literal
83/// variant becomes `?`; whitespace and comments (both represented as `Token::Whitespace`) are
84/// dropped; a `Token::Word` is lowercased only when it carries no `quote_style` -- a quoted word
85/// (e.g. `"processes.exe"`) is kept verbatim, case included, via its own `Display` impl, which is
86/// what keeps two differently-cased quoted identifiers from colliding into the same fingerprint.
87/// Everything else uses the token's own `Display` impl unchanged.
88fn normalize_tokens(tokens: &[Token]) -> String {
89    let mut parts: Vec<String> = Vec::with_capacity(tokens.len());
90    for tok in tokens {
91        match tok {
92            Token::Whitespace(_) => continue,
93            Token::Number(_, _)
94            | Token::SingleQuotedString(_)
95            | Token::NationalStringLiteral(_)
96            | Token::HexStringLiteral(_)
97            | Token::EscapedStringLiteral(_)
98            | Token::TripleSingleQuotedString(_)
99            | Token::TripleDoubleQuotedString(_) => parts.push("?".to_string()),
100            Token::Word(w) if w.quote_style.is_none() => parts.push(w.value.to_lowercase()),
101            other => parts.push(other.to_string()),
102        }
103    }
104    parts.join(" ")
105}
106
107fn collapse_whitespace(sql: &str) -> String {
108    sql.split_whitespace().collect::<Vec<_>>().join(" ")
109}
110
111fn hash_hex16(s: &str) -> String {
112    use sha2::{Digest, Sha256};
113    let digest = Sha256::digest(s.as_bytes());
114    let mut hex = hex::encode(digest);
115    hex.truncate(16);
116    hex
117}

123/// The fixed, documented schema every deny rule is a predicate over -- every attribute
124/// `execute_query` has already resolved by the time the check runs, in the same column order as
125/// [`QueryAttribution::to_batch`]. Every field is nullable `Utf8`: NULL semantics come from SQL
126/// and are the ones we want (e.g. `notebook = 'fleet-overview'` evaluates to NULL, not true, for
127/// a query that carried no notebook header, so the rule does not fire).
128///
129/// The identity column is named `user_id`, not `user`: DataFusion's default SQL dialect parses a
130/// bare `user` in expression position as the zero-argument function call `user()`, not a column
131/// reference, and no such scalar function is registered here -- `user = 'jean'` would fail at
132/// planning with "Invalid function 'user'".
133///
134/// Built once, in [`MATCH_SCHEMA`]: constructing a `DFSchema` runs `check_names()` (a `BTreeSet`
135/// insert per field) on top of allocating the `Fields`/`Schema` themselves, and this is called on
136/// every query while any deny rule stands -- so `match_schema()` and [`QueryAttribution::to_batch`]
137/// both clone the cached `DFSchema`/`Arc<Schema>` rather than rebuilding it.
138static MATCH_SCHEMA: LazyLock<DFSchema> = LazyLock::new(|| {
139    let arrow_schema = Schema::new(vec![
140        Field::new("user_id", DataType::Utf8, true),
141        Field::new("email", DataType::Utf8, true),
142        Field::new("service_account", DataType::Utf8, true),
143        Field::new("client", DataType::Utf8, true),
144        Field::new("agent", DataType::Utf8, true),
145        Field::new("entrypoint", DataType::Utf8, true),
146        Field::new("session", DataType::Utf8, true),
147        Field::new("notebook", DataType::Utf8, true),
148        Field::new("cell", DataType::Utf8, true),
149        Field::new("client_ip", DataType::Utf8, true),
150        Field::new("sql", DataType::Utf8, true),
151        Field::new("sql_hash", DataType::Utf8, true),
152    ]);
153    DFSchema::try_from(arrow_schema).expect("match_schema is a fixed, valid schema")
154});
155
156pub fn match_schema() -> DFSchema {
157    MATCH_SCHEMA.clone()
158}

177}

225/// Parses, validates, and plans `match_expr` into an `Arc<dyn PhysicalExpr>` -- the whole
226/// evaluator, correct by construction because DataFusion does the parsing and the evaluation.
227///
228/// 1. **Parse** with `ctx.parse_sql_expr`, against [`match_schema`]. `ctx` is expected to be a
229///    bare `SessionContext::new()` (no lakehouse tables or catalog registered) -- see
230///    [`QueryDenyList`]'s doc comment for why the same bare context is used both at refresh and
231///    from `deny_queries`.
232/// 2. **Validate** with exactly two checks: the result type is `Boolean`, and at least one column
233///    is referenced. An unknown column or function already fails at step 1; a subquery,
234///    aggregate, or window function cannot be lowered to a scalar `PhysicalExpr` and fails at
235///    step 3 -- each with DataFusion's own diagnostic.
236/// 3. **Plan** the validated `Expr` into a `PhysicalExpr` with the free-function
237///    `datafusion::physical_expr::create_physical_expr` -- deliberately *not*
238///    `SessionContext::create_physical_expr`/`SessionState::create_physical_expr`, both of which
239///    run the expression through `ExprSimplifier::coerce` first and would silently insert a
240///    `CAST` for a type-mismatched comparison (`client = 42` becomes `CAST(client AS Int64) =
241///    42`, which compiles, then either errors on every real value or -- worse -- happens to
242///    parse and quietly matches something the admin never intended). The free function's own
243///    doc comment is explicit that it performs no coercion ("There should be no coercion during
244///    physical planning") -- but it also performs no operand-type *checking* at this stage:
245///    `binary()` happily builds a `BinaryExpr` node comparing a `Utf8` column against an
246///    `Int64`/`Timestamp` literal, and Arrow's comparison kernels are the ones that actually
247///    refuse it, only once the node is evaluated. Left there, that would be exactly the "silent
248///    per-query failure" this design wants to avoid -- so step 4 forces that evaluation now.
249/// 4. **Canary-evaluate** the compiled expression once against [`QueryAttribution::probe`], a
250///    fixed one-row batch that exists for no other reason. Every match-context column is
251///    `Utf8`, so the only way this can fail is the type mismatch step 3 doesn't catch; catching
252///    it here turns "never fires" into a loud compile-time error, at the one-time cost of a
253///    single evaluate() call per rule (refresh/`deny_queries` time, never per query).
254pub fn compile_match_expr(
255    ctx: &SessionContext,
256    match_expr: &str,
257) -> datafusion::error::Result<Arc<dyn PhysicalExpr>> {
258    let schema = match_schema();
259    let expr = ctx.parse_sql_expr(match_expr, &schema)?;
260    let ty = expr.get_type(&schema)?;
261    if ty != DataType::Boolean {
262        return plan_err!("query deny rule must be a boolean expression, got {ty}: {match_expr}");
263    }
264    if expr.column_refs().is_empty() {
265        return plan_err!(
266            "query deny rule must reference at least one match-context column -- a rule with \
267             no column reference would deny every query in the deployment: {match_expr}"
268        );
269    }
270    let physical = datafusion::physical_expr::create_physical_expr(
271        &expr,
272        &schema,
273        &datafusion::execution::context::ExecutionProps::new(),
274    )?;
275    let probe_batch = QueryAttribution::probe().to_batch();
276    physical.evaluate(&probe_batch).map_err(|e| {
277        datafusion::error::DataFusionError::Plan(format!(
278            "query deny rule failed a compile-time check, likely comparing a column against a \
279             mismatched type: {e}: {match_expr}"
280        ))
281    })?;
282    Ok(physical)
283}

285/// The primary escape hatch (plan §5): the deny-list check is skipped when the caller can reach
286/// the admin functions at all -- the same predicate `register_lakehouse_functions` gates
287/// `deny_queries`/`remove_query_denial` on -- **and** the statement mentions one of
288/// `deny_queries` / `remove_query_denial` / `list_query_denials`: three `contains` checks over
289/// the lowercased SQL text, nothing more. No call-position or token analysis: this only ever
290/// runs for a query that is *already* being denied, and the gate it sits behind means any caller
291/// it opens for could call those three functions directly anyway.
292pub fn skip_for_admin_recovery(sql: &str, is_admin: bool, admin_principal_possible: bool) -> bool {
293    if !(is_admin || !admin_principal_possible) {
294        return false;
295    }
296    let lowered = sql.to_lowercase();
297    lowered.contains("deny_queries")
298        || lowered.contains("remove_query_denial")
299        || lowered.contains("list_query_denials")
300}

306/// The DB row, exactly as `list_query_denials()` returns it and as `insert` echoes back. No
307/// compiled expression or in-process timestamp here -- those exist only for a rule that has been
308/// compiled into a [`DenySnapshot`], and a freshly inserted or listed row has not necessarily
309/// been through that yet on every replica.
310#[derive(Debug, Clone)]
311pub struct QueryDenyRow {
312    pub rule_id: Uuid,
313    pub created_at: DateTime<Utc>,
314    pub created_by: String,
315    pub reason: String,
316    pub match_expr: String,
317    /// `None` until the rule first fires; accurate to within one refresh tick.
318    pub last_hit_at: Option<DateTime<Utc>>,
319}
320
321/// A row compiled into one snapshot: the row, its planned expression, and the one piece of
322/// in-process state a rule accumulates. Lives only inside a [`DenySnapshot`].
323#[derive(Debug)]
324pub struct QueryDenyRule {
325    pub row: QueryDenyRow,
326    expr: Arc<dyn PhysicalExpr>,
327    /// Unix seconds of the most recent hit since the last flush; `0` means "not hit since the
328    /// last flush". Flushed (and reset to `0`) by [`QueryDenyList::refresh`].
329    last_hit: AtomicI64,
330}
331
332impl QueryDenyRule {
333    /// `pub` so tests can build a rule directly from a [`compile_match_expr`] result, without
334    /// going through Postgres -- see [`QueryDenyList::with_snapshot`].
335    pub fn new(row: QueryDenyRow, expr: Arc<dyn PhysicalExpr>) -> Self {
336        Self {
337            row,
338            expr,
339            last_hit: AtomicI64::new(0),
340        }
341    }
342
343    /// Records that this rule matched a query, for the next [`QueryDenyList::refresh`] tick to
344    /// flush into `last_hit_at`. Racing hits within one tick are fine -- `fetch_max` keeps the
345    /// most recent, and that's all "is this rule still firing?" needs.
346    pub fn record_hit(&self) {
347        let now = Utc::now().timestamp();
348        self.last_hit.fetch_max(now, Ordering::Relaxed);
349    }
350}

352/// Tags a `query_denied`/`query_deny_compile_error_count` metric with the rule id it concerns.
353/// Interned via [`intern_string`] (same pattern as `maintenance.rs`'s per-view-set metric tags):
354/// bounded cardinality (a rule id exists only while it's part of a deployment's rule set, capped
355/// at [`max_rules`]), so leaking one interned string per rule id that has ever existed is cheap.
356pub fn rule_tags(rule_id: &Uuid) -> &'static PropertySet {
357    PropertySet::find_or_create(vec![Property::new(
358        "rule_id",
359        intern_string(&rule_id.to_string()),
360    )])

363/// The compiled rules, ordered by `(created_at, rule_id)`, oldest first -- so the first match is
364/// always the oldest matching rule, on every replica. An alias, not a newtype: nothing hangs
365/// behind it (see the plan's §4 for the full reasoning). `Arc<[_]>` rather than `Arc<Vec<_>>`:
366/// the snapshot is immutable once built, and this drops a pointer hop off the per-query path.
367pub type DenySnapshot = Arc<[Arc<QueryDenyRule>]>;
368
369fn empty_snapshot() -> DenySnapshot {
370    Arc::from(Vec::new())
371}

373/// Sorts `rules` by `(created_at, rule_id)`, oldest first, and wraps them into a [`DenySnapshot`]
374/// -- `pub` so tests can build one directly, matching the invariant [`QueryDenyList::refresh`]
375/// and [`QueryDenyList::insert`] maintain internally.
376pub fn sorted_snapshot(mut rules: Vec<Arc<QueryDenyRule>>) -> DenySnapshot {
377    rules.sort_by_key(|r| (r.row.created_at, r.row.rule_id));
378    Arc::from(rules)
379}

381/// Owns the Postgres-backed rule store and the compiled, in-memory snapshot every query is
382/// checked against. One instance is owned by `LakehouseContext`, mirroring `AudienceIndex`.
383///
384/// `ctx` is a bare `SessionContext::new()` -- no lakehouse tables or catalog registered against
385/// it, just DataFusion's parser and name resolution -- held only so [`compile_match_expr`] can be
386/// called against it. Both `refresh` and `deny_queries`'s `call_with_args` compile through this
387/// same context, so rule compilation produces the same result regardless of whose session
388/// triggered it.
389pub struct QueryDenyList {
390    pool: sqlx::Pool<sqlx::Postgres>,
391    ctx: SessionContext,
392    snapshot: RwLock<DenySnapshot>,
393    /// Serializes `insert`/`delete`/`refresh`'s DB-operation-plus-snapshot-swap against each
394    /// other, held across the whole sequence (including the `await`s) rather than just the
395    /// final snapshot edit. Without this, `refresh` reading the table in `load_rows` and a
396    /// concurrent `delete` performing its own DELETE-plus-snapshot-edit could interleave so that
397    /// `refresh`'s stale, pre-delete row set overwrites the snapshot `delete` just fixed up,
398    /// silently re-enforcing a rule the admin was just told was removed (or the mirror case,
399    /// losing a freshly inserted rule) until the next tick. Serializing the three methods means
400    /// a `delete`/`insert` either finishes entirely before a `refresh` starts reading the table,
401    /// or starts entirely after `refresh`'s swap -- either way it sees, and leaves, a consistent
402    /// snapshot. `flush_hits` is not covered: it only updates each rule's `last_hit` in place and
403    /// never replaces the snapshot, so it cannot lose an insert/delete.
404    write_lock: tokio::sync::Mutex<()>,
405}

425    /// Builds a `QueryDenyList` around an already-compiled `snapshot`, bypassing Postgres
426    /// entirely -- for tests exercising [`Self::check`]'s in-memory semantics without a live DB
427    /// (`pool` is unused by `check`; a `connect_lazy` pool that never touches the network is
428    /// enough). Production code should use [`Self::new`] plus [`Self::refresh`].
429    pub fn with_snapshot(pool: sqlx::Pool<sqlx::Postgres>, snapshot: DenySnapshot) -> Self {
430        Self {
431            pool,
432            ctx: SessionContext::new(),
433            snapshot: RwLock::new(snapshot),
434            write_lock: tokio::sync::Mutex::new(()),
435        }
436    }
437
438    /// A bare `SessionContext` with no lakehouse tables or catalog registered, for compiling a
439    /// `match_expr` outside this `QueryDenyList` (e.g. from `deny_queries`'s `call_with_args`,
440    /// which is handed the caller's own session and needs a SQL-expression parser instead).
441    pub fn compile_ctx(&self) -> &SessionContext {
442        &self.ctx
443    }
444
445    /// The number of rules currently compiled into the snapshot -- what `deny_queries` checks
446    /// against [`max_rules`] before inserting one more.
447    pub fn rule_count(&self) -> usize {
448        self.snapshot.read().expect("lock").len()
449    }
450
451    /// Checks `q` against every compiled rule, in order, returning the first (oldest) match.
452    /// Returns immediately on an empty snapshot -- the steady state of a deployment that is not
453    /// mid-incident -- before building the one-row batch every non-empty check needs.
454    pub fn check(&self, q: &QueryAttribution<'_>) -> Option<Arc<QueryDenyRule>> {
455        let snapshot = self.snapshot.read().expect("lock").clone();
456        if snapshot.is_empty() {
457            return None;
458        }
459        let batch = q.to_batch();
460        for rule in snapshot.iter() {
461            match rule.expr.evaluate(&batch) {
462                Ok(ColumnarValue::Array(array)) => {
463                    if let Some(bools) = array.as_any().downcast_ref::<BooleanArray>()
464                        && bools.len() == 1
465                        && !bools.is_null(0)
466                        && bools.value(0)
467                    {
468                        return Some(rule.clone());
469                    }
470                }
471                Ok(ColumnarValue::Scalar(datafusion::scalar::ScalarValue::Boolean(Some(true)))) => {
472                    return Some(rule.clone());
473                }
474                Ok(ColumnarValue::Scalar(_)) => {}
475                Err(e) => {
476                    warn!(
477                        "query_deny_list: rule {} failed to evaluate, skipping: {e:#}",
478                        rule.row.rule_id
479                    );
480                }
481            }
482        }
483        None
484    }

486    async fn load_rows(&self) -> Result<Vec<QueryDenyRow>> {
487        let records = sqlx::query(
488            "SELECT rule_id, created_at, created_by, reason, match_expr, last_hit_at \
489             FROM query_deny_list \
490             ORDER BY created_at, rule_id",
491        )
492        .fetch_all(&self.pool)
493        .await
494        .context("loading query_deny_list rows")?;
495        records
496            .into_iter()
497            .map(|row| {
498                Ok(QueryDenyRow {
499                    rule_id: row.try_get("rule_id")?,
500                    created_at: row.try_get("created_at")?,
501                    created_by: row.try_get("created_by")?,
502                    reason: row.try_get("reason")?,
503                    match_expr: row.try_get("match_expr")?,
504                    last_hit_at: row.try_get("last_hit_at")?,
505                })
506            })
507            .collect()
508    }
509
510    /// Every rule currently in force, read fresh from Postgres (not the in-memory snapshot) --
511    /// what `list_query_denials()` returns.
512    pub async fn list(&self) -> Result<Vec<QueryDenyRow>> {
513        self.load_rows().await
514    }

516    /// Inserts a new rule: `compiled` must already be the result of a successful
517    /// [`compile_match_expr`] call against `match_expr` (typically the same call
518    /// `deny_queries`'s `call_with_args` just made to validate it) -- `insert` stores
519    /// `match_expr` verbatim and does not re-validate it. Refreshes the local snapshot
520    /// synchronously before returning, so the admin who created the rule sees it in their own
521    /// `list_query_denials()` immediately; other replicas pick it up within one refresh tick.
522    pub async fn insert(
523        &self,
524        match_expr: &str,
525        compiled: Arc<dyn PhysicalExpr>,
526        reason: &str,
527        created_by: &str,
528    ) -> Result<QueryDenyRow> {
529        let row = QueryDenyRow {
530            rule_id: Uuid::new_v4(),
531            created_at: Utc::now(),
532            created_by: created_by.to_string(),
533            reason: reason.to_string(),
534            match_expr: match_expr.to_string(),
535            last_hit_at: None,
536        };
537        // Held across the INSERT and the snapshot edit -- see `write_lock`'s doc comment -- so a
538        // concurrent `refresh` cannot read the table before this INSERT and then overwrite the
539        // snapshot edit below with that stale, pre-insert row set.
540        let _guard = self.write_lock.lock().await;
541        sqlx::query(
542            "INSERT INTO query_deny_list (rule_id, created_at, created_by, reason, match_expr) \
543             VALUES ($1, $2, $3, $4, $5)",
544        )
545        .bind(row.rule_id)
546        .bind(row.created_at)
547        .bind(&row.created_by)
548        .bind(&row.reason)
549        .bind(&row.match_expr)
550        .execute(&self.pool)
551        .await
552        .context("inserting query_deny_list row")?;
553        let rule = Arc::new(QueryDenyRule::new(row.clone(), compiled));
554        {
555            let mut snapshot = self.snapshot.write().expect("lock");
556            let mut rules: Vec<Arc<QueryDenyRule>> = snapshot.iter().cloned().collect();
557            rules.push(rule);
558            *snapshot = sorted_snapshot(rules);
559        }
560        Ok(row)
561    }

563    /// Deletes a rule by id, refreshing the local snapshot synchronously. Returns `true` if a row
564    /// was actually deleted -- `false` lets the caller (`remove_query_denial`) return a clear
565    /// "no such rule" message rather than a silent no-op.
566    pub async fn delete(&self, rule_id: Uuid) -> Result<bool> {
567        // Held across the DELETE and the snapshot edit -- see `write_lock`'s doc comment -- so a
568        // concurrent `refresh` cannot read the table before this DELETE and then overwrite the
569        // snapshot edit below with that stale, pre-delete row set.
570        let _guard = self.write_lock.lock().await;
571        let result = sqlx::query("DELETE FROM query_deny_list WHERE rule_id = $1")
572            .bind(rule_id)
573            .execute(&self.pool)
574            .await
575            .context("deleting query_deny_list row")?;
576        let deleted = result.rows_affected() > 0;
577        if deleted {
578            let mut snapshot = self.snapshot.write().expect("lock");
579            let rules: Vec<Arc<QueryDenyRule>> = snapshot
580                .iter()
581                .filter(|rule| rule.row.rule_id != rule_id)
582                .cloned()
583                .collect();
584            *snapshot = Arc::from(rules);
585        }
586        Ok(deleted)
587    }

589    /// Flushes each rule's accumulated `last_hit`, then reloads the table and recompiles every
590    /// `match_expr`. **A rule that fails to compile is skipped, never fatal** -- dropped from the
591    /// snapshot with a `warn!` and a `query_deny_compile_error_count` metric, so an older replica
592    /// that cannot compile a newer rule declines to enforce it rather than denying everything or
593    /// crashing. **Fail-open on a failed reload**: the previous snapshot is kept, with a `warn!`
594    /// and a `query_deny_refresh_error_count` metric -- the deny list is an availability valve,
595    /// not a security control, and failing closed would deny every query on a Postgres blip.
596    pub async fn refresh(&self) -> Result<()> {
597        self.flush_hits().await;
598        // Held from before `load_rows` through the final snapshot swap below -- see
599        // `write_lock`'s doc comment -- so a concurrent `insert`/`delete` cannot land its own
600        // DB-op-plus-snapshot-edit inside that window and have it clobbered by this stale read.
601        let _guard = self.write_lock.lock().await;
602        let rows = match self.load_rows().await {
603            Ok(rows) => rows,
604            Err(e) => {
605                imetric!("query_deny_refresh_error_count", "count", 1_u64);
606                warn!("query_deny_list: refresh failed, keeping previous snapshot: {e:#}");
607                return Err(e);
608            }
609        };
610        let mut compiled = Vec::with_capacity(rows.len());
611        for row in rows {
612            let rule_id = row.rule_id;
613            match compile_match_expr(&self.ctx, &row.match_expr) {
614                Ok(expr) => compiled.push(Arc::new(QueryDenyRule::new(row, expr))),
615                Err(e) => {
616                    imetric!(
617                        "query_deny_compile_error_count",
618                        "count",
619                        rule_tags(&rule_id),
620                        1_u64
621                    );
622                    warn!(
623                        "query_deny_list: rule {rule_id} failed to compile, skipping (not \
624                         enforced): {e:#}"
625                    );
626                }
627            }
628        }
629        {
630            // Carry each new rule's `last_hit` forward from its predecessor in the outgoing
631            // snapshot, so a `record_hit()` landing on the old `Arc` between `flush_hits` and this
632            // swap (two DB round-trips plus the compile loop above) isn't discarded along with it.
633            // Held across the read-merge-write so the window in which such a hit could still be
634            // lost is just the in-memory work below, not the whole refresh.
635            let mut snapshot = self.snapshot.write().expect("lock");
636            for rule in &compiled {
637                if let Some(previous) = snapshot.iter().find(|p| p.row.rule_id == rule.row.rule_id)
638                {
639                    let carried = previous.last_hit.load(Ordering::Relaxed);
640                    rule.last_hit.fetch_max(carried, Ordering::Relaxed);
641                }
642            }
643            *snapshot = sorted_snapshot(compiled);
644        }
645        Ok(())
646    }
647
648    async fn flush_hits(&self) {
649        let snapshot = self.snapshot.read().expect("lock").clone();
650        for rule in snapshot.iter() {
651            let last_hit = rule.last_hit.swap(0, Ordering::Relaxed);
652            if last_hit == 0 {
653                continue;
654            }
655            let hit_at = DateTime::<Utc>::from_timestamp(last_hit, 0).unwrap_or_else(Utc::now);
656            if let Err(e) = sqlx::query(
657                "UPDATE query_deny_list \
658                 SET last_hit_at = greatest(coalesce(last_hit_at, $1), $1) \
659                 WHERE rule_id = $2",
660            )
661            .bind(hit_at)
662            .bind(rule.row.rule_id)
663            .execute(&self.pool)
664            .await
665            {
666                // Re-arm rather than leaving the atomic at the `swap(0)` above: a failed UPDATE
667                // must not permanently lose the hit, since it's the next tick's only chance to
668                // retry the write. `fetch_max` so a hit recorded concurrently (after our swap,
669                // above) isn't clobbered by this re-arm.
670                rule.last_hit.fetch_max(last_hit, Ordering::Relaxed);
671                warn!(
672                    "query_deny_list: failed to flush last_hit for rule {}: {e:#}",
673                    rule.row.rule_id
674                );
675            }
676        }
677    }

679    /// Spawns the background task that calls [`Self::refresh`] once immediately and then on
680    /// every [`MICROMEGAS_QUERY_DENY_REFRESH_SECONDS`] tick, until `shutdown` resolves. Only the
681    /// FlightSQL server builder spawns this; other `LakehouseContext` holders (maintenance
682    /// daemon, tests) keep an empty snapshot and never deny anything.
683    pub fn spawn_refresh_task(
684        self: Arc<Self>,
685        shutdown: impl Future<Output = ()> + Send + 'static,
686    ) {
687        tokio::spawn(async move {
688            let mut shutdown = Box::pin(shutdown);
689            loop {
690                // Ignore the `Err`: `refresh()` has already warn!-logged and metered a failed
691                // reload before returning it.
692                let _ = self.refresh().await;
693                tokio::select! {
694                    _ = &mut shutdown => break,
695                    _ = tokio::time::sleep(refresh_interval()) => {}
696                }
697            }
698        });
699    }
