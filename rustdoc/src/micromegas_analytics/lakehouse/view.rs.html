48/// A trait for defining a view.
49#[async_trait]
50pub trait View: std::fmt::Debug + Send + Sync {
51    /// name of the table from the user's perspective
52    fn get_view_set_name(&self) -> Arc<String>;
53
54    /// get_view_instance_id can be a process_id, a stream_id or 'global'.
55    fn get_view_instance_id(&self) -> Arc<String>;
56
57    /// make_batch_partition_spec determines what should be found in an up to date partition.
58    /// The resulting PartitionSpec can be used to validate existing partitions are create a new one.
59    async fn make_batch_partition_spec(
60        &self,
61        lakehouse: Arc<LakehouseContext>,
62        existing_partitions: Arc<PartitionCache>,
63        insert_range: TimeRange,
64    ) -> Result<Arc<dyn PartitionSpec>>;
65
66    /// get_file_schema_hash returns a hash (can be a version number, version string, etc.) that allows
67    /// to identify out of date partitions.
68    fn get_file_schema_hash(&self) -> Vec<u8>;
69
70    /// get_file_schema returns the schema of the partition file in object storage
71    fn get_file_schema(&self) -> Arc<Schema>;
72
73    /// jit_update creates or updates process-specific partitions before a query
74    async fn jit_update(
75        &self,
76        lakehouse: Arc<LakehouseContext>,
77        query_range: Option<TimeRange>,
78    ) -> Result<()>;
79
80    /// make_time_filter returns a set of expressions that will filter out the rows of the partition
81    /// outside the time range requested.
82    fn make_time_filter(&self, _begin: DateTime<Utc>, _end: DateTime<Utc>) -> Result<Vec<Expr>>;
83
84    // a view must provide a way to compute the time bounds of a DataFrame corresponding to its schema
85    fn get_time_bounds(&self) -> Arc<dyn DataFrameTimeBounds>;
86
87    /// register the table in the SessionContext
88    async fn register_table(&self, ctx: &SessionContext, table: MaterializedView) -> Result<()> {
89        let view_set_name = self.get_view_set_name().to_string();
90        ctx.register_table(
91            TableReference::Bare {
92                table: view_set_name.into(),
93            },
94            Arc::new(table),
95        )?;
96        Ok(())
97    }
98
99    async fn merge_partitions(
100        &self,
101        lakehouse: Arc<LakehouseContext>,
102        partitions_to_merge: Arc<Vec<Partition>>,
103        partitions_all_views: Arc<PartitionCache>,
104        insert_range: TimeRange,
105    ) -> Result<MergeQueryResult> {
106        let merge_query = Arc::new(String::from("SELECT * FROM source;"));
107        let empty_view_factory = Arc::new(ViewFactory::new(vec![]));
108        let merger = QueryMerger::new(
109            empty_view_factory,
110            Arc::new(NoOpSessionConfigurator),
111            self.get_file_schema(),
112            merge_query,
113        );
114        merger
115            .execute_merge_query(
116                lakehouse,
117                partitions_to_merge,
118                partitions_all_views,
119                insert_range,
120            )
121            .await
122    }
123
124    /// Returns the sort guarantee a merge of `partitions_to_merge` will actually produce, to be
125    /// recorded as the resulting partition's `Partition::sort_order` (see
126    /// `merge::create_merged_partition`).
127    ///
128    /// This is a distinct concept from `get_scan_output_ordering()`: that one is a trusted
129    /// scan-ordering declaration consumed during physical planning, while this one is a record of
130    /// what a specific merge actually produced, computed purely from the input partitions (before
131    /// `merge_partitions` runs) and independent of whether DataFusion's elision optimization
132    /// happened to succeed for this particular run.
133    ///
134    /// Default: `None` -- no guarantee recorded, ignoring the argument.
135    fn get_merged_partition_sort_order(
136        &self,
137        _partitions_to_merge: &[Partition],
138    ) -> Option<Vec<String>> {
139        None
140    }
141
142    /// tells the daemon which view should be materialized and in what order
143    fn get_update_group(&self) -> Option<i32>;
144
145    /// allow the view to subdivide the requested partition
146    fn get_max_partition_time_delta(&self, _strategy: &PartitionCreationStrategy) -> TimeDelta {
147        TimeDelta::days(1)
148    }
149
150    /// Declares an ordering the view's partition scan *already* emits, letting DataFusion
151    /// elide redundant `Sort` nodes for queries that `ORDER BY` these columns.
152    ///
153    /// Returning a non-`Unordered` value is a correctness contract the view must guarantee, and
154    /// its shape depends on which variant is returned:
155    /// - `Concatenated { columns, .. }`: rows within each partition file are already sorted by
156    ///   `columns`, the leading column is the view's min-event-time column, and partition
157    ///   event-time ranges are non-overlapping (so files concatenate in globally-sorted order).
158    ///   For `ThreadSpansView`, the *rows-sorted-within-a-file* half is obtained from JIT
159    ///   partitions being grouped under `jit_partitions::BlockOrder::EventTime`:
160    ///   `group_blocks_into_partitions` sorts a segment's blocks by event time and only cuts at
161    ///   insert-safe points (see its docs), so within a segment blocks land in event order rather
162    ///   than relying on an unenforced assumption about registration order, and
163    ///   `thread_spans_view::ensure_begin_non_decreasing` checks the resulting batch at write
164    ///   time.
165    ///
166    ///   The *partition-ranges-don't-overlap* half is **not** established by that grouping, and is
167    ///   not fully guaranteed. A partition's declared event-time bounds come from its blocks'
168    ///   `begin_ticks`/`end_ticks`, not from the rows, and whether two adjacent blocks' ticks
169    ///   overlap depends on the producer:
170    ///   - `micromegas_tracing::dispatch`'s flush paths stamp the replacement block's `begin`
171    ///     (`TracingBlock::new` -> `DualTime::now()`) *before* closing the outgoing block
172    ///     (`close()` -> `DualTime::now()`), so `block[k].end_ticks > block[k+1].begin_ticks` by the
173    ///     cost of the buffer swap. Any cut between two adjacent blocks then yields a previous
174    ///     partition whose `max_event_time` slightly exceeds the next partition's `min_event_time`,
175    ///     if the gap survives the microsecond resolution of the stored bounds.
176    ///   - The Unreal producer (`MicromegasTracing/Private/Dispatch.cpp`,
177    ///     `NetTraceWriter.cpp`) takes a single `DualTime::Now()` and uses it for both the new
178    ///     block's `begin` and the outgoing block's `Close`, so consecutive blocks *touch* exactly
179    ///     and no overlap arises.
180    ///
181    ///   Three residual caveats therefore remain, all backstopped by
182    ///   `sort_and_check_non_overlapping` (`partitioned_execution_plan.rs`) failing the query
183    ///   loudly rather than returning wrong rows: the block-boundary tick overlap just described
184    ///   (`micromegas_tracing`-produced streams only), an insert-time inversion straddling a JIT
185    ///   *segment* boundary (segments are still grouped independently, see
186    ///   `generate_stream_jit_partitions`), and TSC-frequency re-estimation drift across
187    ///   materialization epochs for `tsc_frequency == 0` processes.
188    /// - `PerFile { columns }`: rows within each partition file are already sorted, ascending, by
189    ///   `columns`, but partitions may overlap each other arbitrarily on those columns. A false
190    ///   declaration here is not merely mis-ordered rows but, under order-aware aggregation, wrong
191    ///   aggregate results (groups closed early, duplicate group keys) -- see
192    ///   `SqlBatchView::with_merge_sort_order`. What makes declaring this safe is the
193    ///   recorded-`sort_order` certification gate inside `make_partitioned_execution_plan`: every
194    ///   non-empty partition must certify `columns` before the declaration reaches DataFusion at
195    ///   all.
196    ///
197    /// Default: `Unordered` (no declared ordering — DataFusion sorts as usual).
198    fn get_scan_output_ordering(&self) -> ScanOrdering {
199        ScanOrdering::Unordered
200    }
201}
