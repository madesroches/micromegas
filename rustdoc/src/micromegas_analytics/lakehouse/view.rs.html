48/// A trait for defining a view.
49#[async_trait]
50pub trait View: std::fmt::Debug + Send + Sync {
51    /// name of the table from the user's perspective
52    fn get_view_set_name(&self) -> Arc<String>;
53
54    /// get_view_instance_id can be a process_id, a stream_id or 'global'.
55    fn get_view_instance_id(&self) -> Arc<String>;
56
57    /// make_batch_partition_spec determines what should be found in an up to date partition.
58    /// The resulting PartitionSpec can be used to validate existing partitions are create a new one.
59    async fn make_batch_partition_spec(
60        &self,

62        existing_partitions: Arc<PartitionCache>,
63        insert_range: TimeRange,

65
66    /// get_file_schema_hash returns a hash (can be a version number, version string, etc.) that allows
67    /// to identify out of date partitions.
68    fn get_file_schema_hash(&self) -> Vec<u8>;
69
70    /// get_file_schema returns the schema of the partition file in object storage
71    fn get_file_schema(&self) -> Arc<Schema>;
72
73    /// jit_update creates or updates process-specific partitions before a query
74    async fn jit_update(
75        &self,
76        lakehouse: Arc<LakehouseContext>,
77        query_range: Option<TimeRange>,
78    ) -> Result<()>;
79
80    /// make_time_filter returns a set of expressions that will filter out the rows of the partition
81    /// outside the time range requested.
82    fn make_time_filter(&self, _begin: DateTime<Utc>, _end: DateTime<Utc>) -> Result<Vec<Expr>>;
83
84    // a view must provide a way to compute the time bounds of a DataFrame corresponding to its schema

87    /// register the table in the SessionContext
88    async fn register_table(&self, ctx: &SessionContext, table: MaterializedView) -> Result<()> {
89        let view_set_name = self.get_view_set_name().to_string();
90        ctx.register_table(
91            TableReference::Bare {
92                table: view_set_name.into(),
93            },
94            Arc::new(table),
95        )?;
96        Ok(())
97    }

99    async fn merge_partitions(
100        &self,
101        lakehouse: Arc<LakehouseContext>,
102        partitions_to_merge: Arc<Vec<Partition>>,
103        partitions_all_views: Arc<PartitionCache>,
104        insert_range: TimeRange,
105    ) -> Result<MergeQueryResult> {
106        let merge_query = Arc::new(String::from("SELECT * FROM source;"));
107        let empty_view_factory = Arc::new(ViewFactory::new(vec![]));
108        let merger = QueryMerger::new(
109            empty_view_factory,
110            Arc::new(NoOpSessionConfigurator),
111            self.get_file_schema(),
112            merge_query,
113        );
114        merger
115            .execute_merge_query(
116                lakehouse,
117                partitions_to_merge,
118                partitions_all_views,
119                insert_range,
120            )
121            .await
122    }
123
124    /// Returns the sort guarantee a merge of `partitions_to_merge` will actually produce, to be
125    /// recorded as the resulting partition's `Partition::sort_order` (see
126    /// `merge::create_merged_partition`).
127    ///
128    /// This is a distinct concept from `get_scan_output_ordering()`: that one is a trusted
129    /// scan-ordering declaration consumed during physical planning, while this one is a record of
130    /// what a specific merge actually produced, computed purely from the input partitions (before
131    /// `merge_partitions` runs) and independent of whether DataFusion's elision optimization
132    /// happened to succeed for this particular run.
133    ///
134    /// Default: `None` -- no guarantee recorded, ignoring the argument.
135    fn get_merged_partition_sort_order(
136        &self,
137        _partitions_to_merge: &[Partition],
138    ) -> Option<Vec<String>> {
139        None
140    }
141
142    /// tells the daemon which view should be materialized and in what order
143    fn get_update_group(&self) -> Option<i32>;
144
145    /// allow the view to subdivide the requested partition
146    fn get_max_partition_time_delta(&self, _strategy: &PartitionCreationStrategy) -> TimeDelta {
147        TimeDelta::days(1)
148    }

150    /// Declares an ordering the view's partition scan *already* emits, letting DataFusion
151    /// elide redundant `Sort` nodes for queries that `ORDER BY` these columns.
152    ///
153    /// **A large or memory-heavy merge is not, on its own, a reason to declare an ordering here.**
154    /// Every `QueryMerger` merge scans its source partitions with a single sequential reader
155    /// regardless of what this method returns (`make_merge_session_context`,
156    /// `tasks/1491_merge_scan_memory_plan.md`), so a view with no natural sort contract gets a
157    /// bounded-memory merge for free, with no ordering guarantee to maintain. Reach for a declared
158    /// ordering only when a *query-side* need justifies it -- eliding a `Sort` for queries that
159    /// already `ORDER BY` these columns, or (via `PerFile`) a streaming k-way merge that must
160    /// preserve per-file order across an aggregation -- not to bound a merge's memory.
161    ///
162    /// Returning a non-`Unordered` value is a correctness contract the view must guarantee, and
163    /// its shape depends on which variant is returned:
164    /// - `Concatenated { columns, .. }`: rows within each partition file are already sorted by
165    ///   `columns`, the leading column is the view's min-event-time column, and partition
166    ///   event-time ranges are non-overlapping (so files concatenate in globally-sorted order).
167    ///   For `ThreadSpansView`, the *rows-sorted-within-a-file* half is obtained from JIT
168    ///   partitions being grouped under `jit_partitions::BlockOrder::EventTime`:
169    ///   `group_blocks_into_partitions` sorts a segment's blocks by event time and only cuts at
170    ///   insert-safe points (see its docs), so within a segment blocks land in event order rather
171    ///   than relying on an unenforced assumption about registration order, and
172    ///   `thread_spans_view::ensure_begin_non_decreasing` checks the resulting batch at write
173    ///   time.
174    ///
175    ///   The *partition-ranges-don't-overlap* half is **not** established by that grouping alone --
176    ///   a partition's declared event-time bounds come from its blocks' `begin_ticks`/`end_ticks`,
177    ///   not from the rows -- but by design both producers' consecutive blocks now *touch* exactly:
178    ///   - Both `micromegas_tracing::dispatch`'s four flush paths and the Unreal producer
179    ///     (`MicromegasTracing/Private/Dispatch.cpp`) take a single shared timestamp per flush and
180    ///     use it for both the outgoing block's close and the replacement block's `begin`, so
181    ///     `block[k].end_ticks == block[k+1].begin_ticks` by construction. This is the design intent
182    ///     the shared boundary stamp exists for: it lets call trees merge seamlessly across the cut
183    ///     (`group_contiguous_block_chains` treats touching blocks as chain-connected).
184    ///   - Data from `micromegas_tracing` versions predating that fix (or any other producer that
185    ///     stamps two separate timestamps) can still strictly overlap at block boundaries -- this is
186    ///     a legacy producer bug, not a supported producer shape, but the server tolerates it: the
187    ///     view records `max_sort_key_time` (the true max leading-sort-column value, exact per
188    ///     partition), and `partitioned_execution_plan::partition_bounds` reads that in preference
189    ///     to `max_event_time` for the non-overlap check, which the swap-window argument in
190    ///     `tasks/completed/thread_spans_segment_boundary_overlap_plan.md` shows can never trip for cuts at
191    ///     block boundaries of either producer. Partitions written before that column existed (or by
192    ///     a view that never populates it) fall back to the old, looser `max_event_time` bound, which
193    ///     is self-healing rather than a permanent gap: `ThreadSpansView::SCHEMA_VERSION`'s bump makes
194    ///     every pre-existing partition stale by schema hash, so it rebuilds -- carrying
195    ///     `max_sort_key_time` -- automatically on its next query, no admin `retire_partitions` call
196    ///     required.
197    ///
198    ///   Two residual caveats therefore remain, both backstopped by
199    ///   `sort_and_check_non_overlapping` (`partitioned_execution_plan.rs`) failing the query
200    ///   loudly rather than returning wrong rows: an insert-time inversion straddling a JIT
201    ///   *segment* boundary (segments are still grouped independently, see
202    ///   `generate_stream_jit_partitions`), which produces a genuine row-level overlap that the
203    ///   check correctly rejects; and TSC-frequency re-estimation drift across materialization
204    ///   epochs for `tsc_frequency == 0` processes, which can skew bounds written under different
205    ///   converters.
206    /// - `PerFile { columns }`: rows within each partition file are already sorted, ascending, by
207    ///   `columns`, but partitions may overlap each other arbitrarily on those columns. A false
208    ///   declaration here is not merely mis-ordered rows but, under order-aware aggregation, wrong
209    ///   aggregate results (groups closed early, duplicate group keys) -- see
210    ///   `SqlBatchView::with_merge_sort_order`. What makes declaring this safe is the
211    ///   recorded-`sort_order` certification gate inside `make_partitioned_execution_plan`: every
212    ///   non-empty partition must certify `columns` before the declaration reaches DataFusion at
213    ///   all.
214    ///
215    /// Default: `Unordered` (no declared ordering — DataFusion sorts as usual).
216    fn get_scan_output_ordering(&self) -> ScanOrdering {
217        ScanOrdering::Unordered
218    }
219}
