18/// Default metadata cache size in MB
19const DEFAULT_METADATA_CACHE_SIZE_MB: u64 = 50;

21/// Bundles all runtime resources needed for lakehouse query execution.
22///
23/// This struct holds the data lake connection, metadata cache, and DataFusion runtime,
24/// providing a single context object that can be passed through the query path. Parquet
25/// byte-range caching is handled by the in-process L1 cache wrapped around the reader
26/// factory's object store (see `object_cache::l1_wrap`), not by this struct.
27#[derive(Clone)]
28pub struct LakehouseContext {
29    lake: Arc<DataLakeConnection>,
30    metadata_cache: Arc<MetadataCache>,
31    runtime: Arc<RuntimeEnv>,
32    reader_factory: Arc<ReaderFactory>,
33    /// The call-level guard -- resolves *any telemetry id -> its
34    /// owning process's audience* from Postgres, size- and TTL-bounded. Fixed shape, no
35    /// operational knob (see [`DEFAULT_AUDIENCE_CACHE_ENTRIES`]'s doc comment for why).
36    audience_index: Arc<AudienceIndex>,
37    /// Admin-managed query deny list. The refresh task that
38    /// keeps its snapshot warm is spawned only by the FlightSQL server builder -- every other
39    /// holder of a `LakehouseContext` (maintenance daemon, tests) keeps an empty snapshot and
40    /// `check` never denies anything.
41    query_denials: Arc<QueryDenyList>,
42    /// The audience a never-stamped process is read as (`MICROMEGAS_DEFAULT_AUDIENCE`).
43    /// Resolved once here and handed to all three sites that read an audience out of Postgres --
44    /// `BlocksView`'s `data_sql`, `metadata::find_process`, and [`AudienceIndex`]'s
45    /// `owner_query_sql` -- so one process can never resolve two different defaults depending on
46    /// which path reached it. It lives here rather than on `IsolationConfig` because the
47    /// maintenance daemon needs it and never builds one of those.
48    default_audience: Arc<str>,
49}
50
51impl LakehouseContext {
52    /// Builds a lakehouse context from an already-connected `DataLakeConnection`.
53    ///
54    /// Runs `migrate_lakehouse` on the supplied connection (idempotent) and
55    /// creates the DataFusion runtime.  The caller is responsible for running
56    /// `migrate_db` (ingestion schema) before this if both migrations are needed
57    /// — the monolith does this via `connect_to_remote_data_lake`.
58    pub async fn from_connection(lake: Arc<DataLakeConnection>) -> Result<Arc<Self>> {
59        migrate_lakehouse(lake.db_pool.clone())
60            .await
61            .with_context(|| "migrate_lakehouse")?;
62        let runtime = Arc::new(make_runtime_env()?);
63        Ok(Arc::new(Self::new(lake, runtime)?))
64    }
65
66    /// Reads MICROMEGAS_SQL_CONNECTION_STRING and MICROMEGAS_OBJECT_STORE_URI,
67    /// connects to the data lake, runs lakehouse migrations, and creates the
68    /// runtime environment.
69    pub async fn from_env() -> Result<Arc<Self>> {
70        let cfg = DataLakeConfig::from_env()?;
71        let data_lake = Arc::new(
72            connect_to_data_lake(&cfg.sql_connection_string, &cfg.object_store_uri).await?,
73        );
74        migrate_lakehouse(data_lake.db_pool.clone())
75            .await
76            .with_context(|| "migrate_lakehouse")?;
77        let runtime = Arc::new(make_runtime_env()?);
78        Ok(Arc::new(Self::new(data_lake, runtime)?))
79    }
80
81    /// Creates a new lakehouse context with a default-sized metadata cache.
82    ///
83    /// Fails if `MICROMEGAS_DEFAULT_AUDIENCE` is set to a malformed value -- every caller is a
84    /// startup path, so a typo in that knob stops the role rather than silently relabelling
85    /// legacy data (see [`default_audience_from_env`]).
86    pub fn new(lake: Arc<DataLakeConnection>, runtime: Arc<RuntimeEnv>) -> Result<Self> {
87        let metadata_cache_mb = match std::env::var("MICROMEGAS_METADATA_CACHE_MB") {
88            Ok(s) => s.parse::<u64>().unwrap_or_else(|_| {
89                warn!(
90                    "Invalid MICROMEGAS_METADATA_CACHE_MB value '{s}', using default {DEFAULT_METADATA_CACHE_SIZE_MB} MB"
91                );
92                DEFAULT_METADATA_CACHE_SIZE_MB
93            }),
94            Err(_) => DEFAULT_METADATA_CACHE_SIZE_MB,
95        };
96
97        let metadata_cache = Arc::new(MetadataCache::new(metadata_cache_mb * 1024 * 1024));
98        let default_audience: Arc<str> = Arc::from(default_audience_from_env()?.as_str());
99
100        let reader_factory = Arc::new(ReaderFactory::new(
101            micromegas_object_cache::l1_wrap(lake.blob_storage.inner(), "lakehouse"),
102            metadata_cache.clone(),
103        ));
104        let audience_index = Arc::new(AudienceIndex::new(
105            lake.db_pool.clone(),
106            DEFAULT_AUDIENCE_CACHE_ENTRIES,
107            DEFAULT_AUDIENCE_CACHE_TTL,
108            default_audience.clone(),
109        ));
110        let query_denials = Arc::new(QueryDenyList::new(lake.db_pool.clone()));
111        Ok(Self {
112            lake,
113            metadata_cache,
114            runtime,
115            reader_factory,
116            audience_index,
117            query_denials,
118            default_audience,
119        })
120    }
121
122    /// Creates a new lakehouse context with a custom metadata cache.
123    ///
124    /// Takes `default_audience` rather than re-reading the environment: a context built this way
125    /// must resolve the same default as the one it is derived from (see the field's doc comment).
126    pub fn with_caches(
127        lake: Arc<DataLakeConnection>,
128        runtime: Arc<RuntimeEnv>,
129        metadata_cache: Arc<MetadataCache>,
130        default_audience: Arc<str>,
131    ) -> Self {
132        let reader_factory = Arc::new(ReaderFactory::new(
133            micromegas_object_cache::l1_wrap(lake.blob_storage.inner(), "lakehouse"),
134            metadata_cache.clone(),
135        ));
136        let audience_index = Arc::new(AudienceIndex::new(
137            lake.db_pool.clone(),
138            DEFAULT_AUDIENCE_CACHE_ENTRIES,
139            DEFAULT_AUDIENCE_CACHE_TTL,
140            default_audience.clone(),
141        ));
142        let query_denials = Arc::new(QueryDenyList::new(lake.db_pool.clone()));
143        Self {
144            lake,
145            metadata_cache,
146            runtime,
147            reader_factory,
148            audience_index,
149            query_denials,
150            default_audience,
151        }
152    }
153
154    /// Returns the audience a never-stamped process is read as
155    /// (`MICROMEGAS_DEFAULT_AUDIENCE`).
156    pub fn default_audience(&self) -> Arc<str> {
157        self.default_audience.clone()
158    }

160    /// Returns the data lake connection.
161    pub fn lake(&self) -> &Arc<DataLakeConnection> {
162        &self.lake
163    }

167        &self.metadata_cache
168    }

170    /// Returns the DataFusion runtime environment.
171    pub fn runtime(&self) -> &Arc<RuntimeEnv> {
172        &self.runtime
173    }
174
175    /// Clones this context with `runtime` swapped, sharing the metadata cache and reader
176    /// factory. Deliberately a struct-update clone rather than a call to `new()`/
177    /// `with_caches()` — those rebuild the `MetadataCache` and `ReaderFactory`, which would
178    /// throw away the shared metadata cache per query.
179    pub fn with_runtime(&self, runtime: Arc<RuntimeEnv>) -> Arc<Self> {
180        Arc::new(Self {
181            runtime,
182            ..self.clone()
183        })
184    }

186    /// Returns the shared `ReaderFactory`.

188        &self.reader_factory
189    }

192    pub fn audience_index(&self) -> &Arc<AudienceIndex> {
193        &self.audience_index
194    }

196    /// Returns the shared query deny list.
197    pub fn query_denials(&self) -> &Arc<QueryDenyList> {
198        &self.query_denials
199    }
