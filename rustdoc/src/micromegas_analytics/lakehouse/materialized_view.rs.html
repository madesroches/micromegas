20/// A DataFusion `TableProvider` for materialized views.
21#[derive(Debug)]
22pub struct MaterializedView {
23    lakehouse: Arc<LakehouseContext>,
24    reader_factory: Arc<ReaderFactory>,
25    view: Arc<dyn View>,
26    part_provider: Arc<dyn QueryPartitionProvider>,
27    query_range: Option<TimeRange>,
28    /// The `view_instance(...)` scan-time audience check, run before `jit_update`.
29    /// `Some` only when this provider was built by `ViewInstanceTableFunction` -- a
30    /// caller-named instance. `None` for every server-constructed `MaterializedView`: the
31    /// implicitly-registered global tables (`query.rs::register_table`) and `OwnershipRewrite`'s
32    /// own `processes`/`streams` sources, which are the row-level filter's job to filter row-by-row and must
33    /// never be denied wholesale.
34    instance_guard: Option<Arc<AudienceGuard>>,
35}
36
37impl MaterializedView {
38    pub fn new(
39        lakehouse: Arc<LakehouseContext>,
40        reader_factory: Arc<ReaderFactory>,
41        view: Arc<dyn View>,
42        part_provider: Arc<dyn QueryPartitionProvider>,
43        query_range: Option<TimeRange>,
44        instance_guard: Option<Arc<AudienceGuard>>,
45    ) -> Self {
46        Self {
47            lakehouse,
48            reader_factory,
49            view,
50            part_provider,
51            query_range,
52            instance_guard,
53        }
54    }
55
56    pub fn get_view(&self) -> Arc<dyn View> {
57        self.view.clone()
58    }
59
60    /// Whether `AudienceGuard::authorize_view_instance` will resolve this view's instance id
61    /// against the caller's scope, and deny, before `scan` yields a row. True only for a
62    /// caller-named, non-`'global'` `view_instance(...)`: those take the guard's Uuid arm (or its
63    /// fail-closed fallthrough). `'global'` is excluded because the guard passes it unconditionally
64    /// -- global instances are row-filtered instead.
65    ///
66    /// Deliberately does not mirror the guard's public-view-set arm: this accessor would still
67    /// return `true` for a public view set's instance, even though the guard passes it
68    /// unconditionally before ever reaching the Uuid arm. Harmless --
69    /// `OwnershipRewrite::predicate_for`'s own public-view-set check already returns `Ok(None)`
70    /// before this accessor is ever consulted, and both checks read
71    /// `IsolationConfig::public_view_sets`.
72    ///
73    /// Kept in step with `AudienceGuard::authorize_view_instance`'s arms; changing those means
74    /// revisiting this.
75    pub fn instance_is_audience_guarded(&self) -> bool {
76        self.instance_guard.is_some() && self.view.get_view_instance_id().as_str() != "global"
77    }
78}
79
80#[async_trait]
81impl TableProvider for MaterializedView {
82    fn schema(&self) -> SchemaRef {
83        self.view.get_file_schema()
84    }
85
86    fn table_type(&self) -> TableType {
87        TableType::Base
88    }
89
90    #[span_fn]
91    async fn scan(
92        &self,
93        state: &dyn Session,
94        projection: Option<&Vec<usize>>,
95        filters: &[Expr],
96        limit: Option<usize>,
97    ) -> datafusion::error::Result<Arc<dyn ExecutionPlan>> {
98        if let Some(guard) = &self.instance_guard {
99            guard
100                .authorize_view_instance(
101                    &self.view.get_view_set_name(),
102                    &self.view.get_view_instance_id(),
103                )
104                .await?;
105        }
106
107        self.view
108            .jit_update(self.lakehouse.clone(), self.query_range)
109            .await
110            .map_err(|e| DataFusionError::External(format!("{e:#}").into()))?;
111
112        let partitions = self
113            .part_provider
114            .fetch(
115                &self.view.get_view_set_name(),
116                &self.view.get_view_instance_id(),
117                self.query_range,
118                self.view.get_file_schema_hash(),
119            )
120            .await
121            .map_err(|e| datafusion::error::DataFusionError::External(e.into()))?;
122        trace!("MaterializedView::scan nb_partitions={}", partitions.len());
123
124        make_partitioned_execution_plan(
125            self.schema(),
126            self.reader_factory.clone(),
127            state,
128            projection,
129            filters,
130            limit,
131            Arc::new(partitions),
132            &self.view.get_scan_output_ordering(),
133        )
134    }
135
136    /// Tell DataFusion to push filters down to the scan method
137    fn supports_filters_pushdown(
138        &self,
139        filters: &[&Expr],
140    ) -> datafusion::error::Result<Vec<TableProviderFilterPushDown>> {
141        // Inexact because the pruning can't handle all expressions and pruning
142        // is not done at the row level -- there may be rows in returned files
143        // that do not pass the filter
144        Ok(vec![TableProviderFilterPushDown::Inexact; filters.len()])
145    }
146}
