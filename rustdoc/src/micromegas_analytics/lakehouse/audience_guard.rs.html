1//! Query Enforcement Prong B (#1371, AbAC Stage 3) -- arg-addressed guards for the span/metadata
2//! UDTFs and the `get_payload` UDF that [`super::ownership_rewrite::OwnershipRewrite`] (Prong A)
3//! structurally cannot reach: they bake their target id into a provider at plan time, return
4//! schemas with no `process_id` column to filter on, and some build their own inner session
5//! under `ReadScope::All`. See `tasks/1371_udtf_udf_guards_plan.md` for the full design rationale;
6//! this comment records only what a future reader of this file needs close at hand.
7//!
8//! ## One cache, one question
9//!
10//! [`AudienceIndex`] answers exactly one question, for three id kinds ([`IdKind`]): which
11//! audience is stamped on the process that owns this id? It resolves from **Postgres** (the
12//! origin of `micromegas.audience`), by primary-key point query -- fresher and independent of
13//! materialization, unlike [`super::ownership_rewrite::OwnershipRewrite`], which reads a
14//! daemon-materialized snapshot (see the plan's §11 for the consequences of the two prongs
15//! reading different copies).
16//!
17//! ## Fail-closed
18//!
19//! [`is_readable`] is the whole authorization rule, pure and offline-testable: `ReadScope::All`
20//! passes everything; `ReadScope::Audiences` denies [`OwnerAudience::Unknown`] unconditionally,
21//! passes [`OwnerAudience::Unstamped`] only when `unstamped_audience` is both configured and in
22//! scope, and matches [`OwnerAudience::Audience`] byte-exactly. An id ambiguous between a
23//! `process_id` and a `stream_id` interpretation ([`OwnerAudience::Ambiguous`]) is readable only
24//! when every interpretation is -- never by picking one arm over the other. A resolution *error*
25//! (Postgres unreachable) is a denial too -- [`AudienceGuard::authorize`]/
26//! [`AudienceGuard::readable_ids`] map it to a query failure, never to a readable verdict.
27//!
28//! ## No existence oracle
29//!
30//! Every guard denial and every "no such id" produce the same error text (e.g. `process_spans:
31//! 'xxx' not found or not accessible`) -- a distinct "permission denied" would let a caller
32//! enumerate which ids exist in other audiences. The server log ([`debug!`]) records the real
33//! reason so an operator can tell the two apart; the client cannot.

46/// The process property carrying the data-isolation audience. Single definition, shared with
47/// [`super::ownership_rewrite::OwnershipRewrite`] (which reads it via [`Self::audience_col`] --
48/// formerly its own inlined literal, before Prong B introduced this shared constant).
49pub const AUDIENCE_PROPERTY: &str = "micromegas.audience";
50
51/// What a resolution attempt found. `Unknown` is *not* `Unstamped`: an id with no row at all is
52/// always denied, while an unstamped process is subject to the `unstamped_audience` knob.
53/// "No row" covers both "not yet ingested" and "retention already deleted it" (plan §11) -- both
54/// deny, on the same fail-closed reasoning.
55#[derive(Clone, Debug, PartialEq, Eq)]
56pub enum OwnerAudience {
57    Unknown,
58    Unstamped,
59    Audience(Arc<str>),
60    /// The id resolved to more than one distinct owner under `IdKind::ProcessOrStream`'s two
61    /// arms -- a `process_id`/`stream_id` collision (see the variant's doc comment on
62    /// [`IdKind::ProcessOrStream`]). Fail-closed: [`is_readable`] treats this as readable only
63    /// when *every* one of the distinct owners would independently be readable, never by picking
64    /// one arm over the other.
65    Ambiguous(Vec<OwnerAudience>),
66}

68/// Which table resolves the id to its owning process. Also the cache's key discriminator: the
69/// same `Uuid` can legitimately be a `process_id` in one audience and a `stream_id`/`block_id` in
70/// another (all three are client-supplied at ingestion, with no cross-table uniqueness
71/// constraint), so caching by kind keeps those disjoint instead of relying on downstream
72/// emptiness to make a wrong-kind hit harmless.
73#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash)]
74pub enum IdKind {
75    /// `processes.process_id`.
76    Process,
77    /// `blocks.block_id` -> `blocks.process_id` -> `processes`.
78    Block,
79    /// `list_partitions`' `view_instance_id`: either a `process_id` or a `stream_id`, resolved in
80    /// one round trip. Cached under its own key rather than reusing `Process`/`Block` entries or a
81    /// separate `streams` kind, so the `UNION ALL` result -- fail-closed on a collision between the
82    /// two arms, see [`merge_owner_rows`] -- is what actually gets cached.
83    ProcessOrStream,
84}
85
86/// Default entry-count bound for [`AudienceIndex`]'s cache -- see [`super::lakehouse_context`].
87/// One `Uuid` + a short audience string is roughly 100 bytes, so 100k entries is ~10 MB: a fixed
88/// shape with no operational knob, mirroring `analytics-web-srv/src/data_source_cache.rs`'s
89/// hardcoded `.max_capacity(1000)` rather than `MICROMEGAS_METADATA_CACHE_MB`'s per-entry-weight
90/// variability.
91pub const DEFAULT_AUDIENCE_CACHE_ENTRIES: u64 = 100_000;
92
93/// Default freshness bound for [`AudienceIndex`]'s cache entries. The *only* invalidation
94/// mechanism: a process's Postgres row is written once and never updated in place, but a
95/// retention-then-re-export cycle can recreate the same `process_id` (deterministic UUIDv5 for
96/// OTLP) under fresh `properties`, and nothing else ever expires a stale cached answer. Bounds
97/// how long a re-derived process's audience can serve a stale answer.
98pub const DEFAULT_AUDIENCE_CACHE_TTL: Duration = Duration::from_secs(5 * 60);

100/// Merges possibly-duplicated `(id, audience)` rows into one [`OwnerAudience`] per id.
101/// Fail-closed on a collision: when `ProcessOrStream`'s two arms resolve the same id to
102/// *different* audiences (a `process_id`/`stream_id` collision -- both ids are client-supplied at
103/// ingestion, with no cross-table uniqueness constraint), neither arm wins over the other --
104/// the id maps to [`OwnerAudience::Ambiguous`], which [`is_readable`] only passes when every
105/// resolved audience is independently readable. `Process`/`Block` queries never produce more than
106/// one row per id, so this never triggers for them.
107fn merge_owner_rows(rows: Vec<(Uuid, Option<String>)>) -> HashMap<Uuid, OwnerAudience> {
108    let mut by_id: HashMap<Uuid, Vec<OwnerAudience>> = HashMap::new();
109    for (id, audience) in rows {
110        let owner = match audience {
111            Some(a) => OwnerAudience::Audience(Arc::from(a)),
112            None => OwnerAudience::Unstamped,
113        };
114        let distinct_owners = by_id.entry(id).or_default();
115        if !distinct_owners.contains(&owner) {
116            distinct_owners.push(owner);
117        }
118    }
119    by_id
120        .into_iter()
121        .map(|(id, mut owners)| {
122            let owner = if owners.len() == 1 {
123                owners.pop().expect("checked len() == 1 above")
124            } else {
125                debug!(
126                    "audience_guard: '{id}' resolved to {} distinct owners across process_id/stream_id \
127                     collision ({owners:?}), treating as Ambiguous (fail-closed)",
128                    owners.len()
129                );
130                OwnerAudience::Ambiguous(owners)
131            };
132            (id, owner)
133        })
134        .collect()
135}

137/// The SQL shape for each [`IdKind`]. `LEFT JOIN LATERAL` (not an inner `unnest` in the `FROM`
138/// list) is what keeps *unstamped* rows in the result -- an inner unnest would silently drop
139/// them, collapsing `Unstamped` into `Unknown`. `$2` is [`AUDIENCE_PROPERTY`]; `$1` is the batch
140/// of ids to resolve, bound as a `uuid[]` array so `resolve_many` is always one query.
141fn owner_query_sql(kind: IdKind) -> &'static str {
142    match kind {
143        IdKind::Process => {
144            "SELECT p.process_id AS id, a.value AS audience
145             FROM processes p
146             LEFT JOIN LATERAL (
147                 SELECT value FROM unnest(p.properties) WHERE key = $2 LIMIT 1
148             ) a ON TRUE
149             WHERE p.process_id = ANY($1::uuid[])"
150        }
151        IdKind::Block => {
152            "SELECT b.block_id AS id, a.value AS audience
153             FROM blocks b
154             JOIN processes p ON p.process_id = b.process_id
155             LEFT JOIN LATERAL (
156                 SELECT value FROM unnest(p.properties) WHERE key = $2 LIMIT 1
157             ) a ON TRUE
158             WHERE b.block_id = ANY($1::uuid[])"
159        }
160        IdKind::ProcessOrStream => {
161            "SELECT p.process_id AS id, a.value AS audience
162             FROM processes p
163             LEFT JOIN LATERAL (
164                 SELECT value FROM unnest(p.properties) WHERE key = $2 LIMIT 1
165             ) a ON TRUE
166             WHERE p.process_id = ANY($1::uuid[])
167             UNION ALL
168             SELECT s.stream_id AS id, a.value AS audience
169             FROM streams s
170             JOIN processes p ON p.process_id = s.process_id
171             LEFT JOIN LATERAL (
172                 SELECT value FROM unnest(p.properties) WHERE key = $2 LIMIT 1
173             ) a ON TRUE
174             WHERE s.stream_id = ANY($1::uuid[])"
175        }
176    }
177}

179async fn fetch_owner_rows(
180    pool: &sqlx::Pool<sqlx::Postgres>,
181    ids: &[Uuid],
182    kind: IdKind,
183) -> anyhow::Result<Vec<(Uuid, Option<String>)>> {
184    let rows = sqlx::query(owner_query_sql(kind))
185        .bind(ids)
186        .bind(AUDIENCE_PROPERTY)
187        .fetch_all(pool)
188        .await
189        .with_context(|| format!("resolving owning process' audience for {kind:?}"))?;
190    rows.into_iter()
191        .map(|row| {
192            let id: Uuid = row.try_get("id").context("reading id column")?;
193            let audience: Option<String> =
194                row.try_get("audience").context("reading audience column")?;
195            Ok((id, audience))
196        })
197        .collect()
198}

200/// Resolves *any telemetry id -> its owning process's audience*, from Postgres, cached and
201/// TTL-bounded. See the module doc comment for the full rationale.
202pub struct AudienceIndex {
203    pool: sqlx::Pool<sqlx::Postgres>,
204    cache: moka::future::Cache<(IdKind, Uuid), OwnerAudience>,
205}
206
207impl AudienceIndex {
208    pub fn new(pool: sqlx::Pool<sqlx::Postgres>, max_entries: u64, ttl: Duration) -> Self {
209        let cache = moka::future::Cache::builder()
210            .max_capacity(max_entries)
211            .time_to_live(ttl)
212            .build();
213        Self { pool, cache }
214    }
215
216    /// Resolves a single id. A thin wrapper over [`Self::resolve_many`] -- always at least as
217    /// cheap, and the batch path is where the actual query lives.
218    pub async fn resolve(&self, id: Uuid, kind: IdKind) -> anyhow::Result<OwnerAudience> {
219        let mut resolved = self.resolve_many(&[id], kind).await?;
220        Ok(resolved.remove(&id).unwrap_or(OwnerAudience::Unknown))
221    }
222
223    /// Resolves a batch of ids of the same `kind` in at most one round trip (cache misses only).
224    /// `Unknown` is never cached -- a miss may just mean "not ingested yet", and caching it would
225    /// both pin a wrong answer and let a caller pollute the cache with random ids.
226    pub async fn resolve_many(
227        &self,
228        ids: &[Uuid],
229        kind: IdKind,
230    ) -> anyhow::Result<HashMap<Uuid, OwnerAudience>> {
231        let mut result = HashMap::with_capacity(ids.len());
232        let mut misses: Vec<Uuid> = Vec::new();
233        let mut seen_miss: HashSet<Uuid> = HashSet::new();
234        for &id in ids {
235            if let Some(cached) = self.cache.get(&(kind, id)).await {
236                result.insert(id, cached);
237            } else if seen_miss.insert(id) {
238                misses.push(id);
239            }
240        }
241        if misses.is_empty() {
242            return Ok(result);
243        }
244        let rows = fetch_owner_rows(&self.pool, &misses, kind).await?;
245        let resolved = merge_owner_rows(rows);
246        for id in misses {
247            let owner = resolved.get(&id).cloned().unwrap_or(OwnerAudience::Unknown);
248            if owner != OwnerAudience::Unknown {
249                self.cache.insert((kind, id), owner.clone()).await;
250            }
251            result.insert(id, owner);
252        }
253        imetric!(
254            "audience_cache_entry_count",
255            "count",
256            self.cache.entry_count()
257        );
258        Ok(result)
259    }
260}

262impl std::fmt::Debug for AudienceIndex {
263    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
264        f.debug_struct("AudienceIndex")
265            .field("entries", &self.cache.entry_count())
266            .finish()
267    }
268}

270/// Pure, offline-testable: the whole authorization rule, with no I/O in it. `ReadScope::All` is
271/// the only branch that passes `Unknown` -- every other combination denies it unconditionally.
272pub fn is_readable(
273    scope: &ReadScope,
274    unstamped_audience: Option<&str>,
275    owner: &OwnerAudience,
276) -> bool {
277    match scope {
278        ReadScope::All => true,
279        ReadScope::Audiences(auds) => match owner {
280            OwnerAudience::Unknown => false,
281            OwnerAudience::Unstamped => {
282                unstamped_audience.is_some_and(|u| auds.iter().any(|a| a == u))
283            }
284            OwnerAudience::Audience(a) => auds.iter().any(|x| x.as_str() == &**a),
285            OwnerAudience::Ambiguous(owners) => {
286                !owners.is_empty()
287                    && owners
288                        .iter()
289                        .all(|owner| is_readable(scope, unstamped_audience, owner))
290            }
291        },
292    }
293}

295/// Witness that some id's owning process was resolved and found readable under the caller's
296/// scope. Constructible only by [`AudienceGuard::authorize`] -- a future call site that wants an
297/// inner, unscoped session must hold one of these first, which is what keeps a fourth such UDTF
298/// from silently reproducing the bypass this stage closes by calling `CallerContext::internal()`
299/// directly. Not airtight: `internal()` stays public for genuinely non-user-reachable callers.
300#[derive(Debug)]
301pub struct Authorized {
302    id: Uuid,
303}
304
305impl Authorized {
306    pub fn id(&self) -> Uuid {
307        self.id
308    }
309
310    /// The `CallerContext` an inner, post-authorization session may run under. Deliberately
311    /// `ReadScope::All` (not the caller's own scope) -- see
312    /// `tasks/1371_udtf_udf_guards_plan.md` §6 for why guard-then-internal is the chosen shape
313    /// here rather than scope inheritance: every inner statement these three call sites run is
314    /// server-constructed and confined to `self.id`, so if `self.id`'s process is readable,
315    /// everything those statements can reach is readable too.
316    pub fn internal_caller(&self) -> CallerContext {
317        CallerContext::internal()
318    }
319}

321/// The whole guard: pure decision (`is_readable`) plus async resolution (`AudienceIndex`),
322/// fail-closed throughout. One instance is built per request (in `query.rs`'s
323/// `register_lakehouse_functions`) and shared, via `Arc`, across every arg-addressed UDTF/UDF the
324/// request's session registers.
325#[derive(Debug)]
326pub struct AudienceGuard {
327    read_scope: ReadScope,
328    unstamped_audience: Option<String>,
329    public_view_sets: Vec<String>,
330    index: Arc<AudienceIndex>,
331}
332
333impl AudienceGuard {
334    pub fn new(
335        read_scope: ReadScope,
336        unstamped_audience: Option<String>,
337        public_view_sets: Vec<String>,
338        index: Arc<AudienceIndex>,
339    ) -> Self {
340        Self {
341            read_scope,
342            unstamped_audience,
343            public_view_sets,
344            index,
345        }
346    }
347
348    /// The caller's scope. Exposed so callers with their own row-level filtering
349    /// (`list_partitions`) can decide up front whether to even attempt an optimization (e.g.
350    /// `LIMIT` pushdown) that's only valid under `ReadScope::All`.
351    pub fn read_scope(&self) -> &ReadScope {
352        &self.read_scope
353    }
354
355    /// `ReadScope::All` ⇒ `Ok` with no I/O at all. Otherwise resolves `id`'s owning process's
356    /// audience and denies with a uniform, existence-oracle-proof message
357    /// (`{fname}: '{id}' not found or not accessible`) on anything but a readable verdict -- a
358    /// resolution error included, never a pass. The real reason (resolved `OwnerAudience`, the
359    /// caller's scope) is logged at `debug!`, for operators only.
360    pub async fn authorize(
361        &self,
362        id: Uuid,
363        kind: IdKind,
364        fname: &str,
365    ) -> datafusion::error::Result<Authorized> {
366        if self.read_scope == ReadScope::All {
367            return Ok(Authorized { id });
368        }
369        let owner = match self.index.resolve(id, kind).await {
370            Ok(owner) => owner,
371            Err(e) => {
372                debug!("{fname}: audience resolution failed for '{id}' ({kind:?}): {e:#}");
373                return Err(DataFusionError::External(e.into()));
374            }
375        };
376        if is_readable(&self.read_scope, self.unstamped_audience.as_deref(), &owner) {
377            Ok(Authorized { id })
378        } else {
379            debug!(
380                "{fname}: denying '{id}' ({kind:?}): owner={owner:?}, scope={:?}",
381                self.read_scope
382            );
383            plan_err!("{fname}: '{id}' not found or not accessible")
384        }
385    }
386
387    /// Batch form of the same rule, for the callers that row-filter (`list_partitions`) or need
388    /// an all-or-nothing verdict over a batch (`get_payload`) rather than a single witness.
389    /// `ReadScope::All` ⇒ every id is readable, with no I/O.
390    pub async fn readable_ids(
391        &self,
392        ids: &[Uuid],
393        kind: IdKind,
394    ) -> datafusion::error::Result<HashSet<Uuid>> {
395        if self.read_scope == ReadScope::All {
396            return Ok(ids.iter().copied().collect());
397        }
398        let owners = self
399            .index
400            .resolve_many(ids, kind)
401            .await
402            .map_err(|e| DataFusionError::External(e.into()))?;
403        Ok(ids
404            .iter()
405            .copied()
406            .filter(|id| {
407                let owner = owners.get(id).unwrap_or(&OwnerAudience::Unknown);
408                is_readable(&self.read_scope, self.unstamped_audience.as_deref(), owner)
409            })
410            .collect())
411    }
412
413    /// `list_partitions`' `'global'`-row rule (plan §8): a global aggregate has no single
414    /// audience, so it is treated like unstamped data -- visible under `ReadScope::All`, when
415    /// `view_set_name` is on the public allowlist, or when the configured `unstamped_audience` is
416    /// itself in the caller's scope.
417    pub fn global_rows_visible(&self, view_set_name: &str) -> bool {
418        match &self.read_scope {
419            ReadScope::All => true,
420            ReadScope::Audiences(auds) => {
421                self.public_view_sets.iter().any(|s| s == view_set_name)
422                    || self
423                        .unstamped_audience
424                        .as_deref()
425                        .is_some_and(|u| auds.iter().any(|a| a == u))
426            }
427        }
428    }
