37/// Builds the ascending `ScanSortColumn` list corresponding to a declared sort order.
38fn sort_order_as_scan_columns(columns: &[Arc<String>]) -> Vec<ScanSortColumn> {
39    columns
40        .iter()
41        .map(|column| ScanSortColumn {
42            column: column.clone(),
43            descending: false,
44        })
45        .collect()
46}

75    #[expect(clippy::too_many_arguments)]
76    /// # Arguments
77    ///
78    /// * `runtime` - datafusion runtime
79    /// * `view_set_name` - name of the table
80    /// * `min_event_time_column` - min(column) should result in the first timestamp in a dataframe
81    /// * `max_event_time_column` - max(column) should result in the last timestamp in a dataframe
82    /// * `count_src_query` - used to count the rows of the underlying data to know if a cached partition is up to date
83    /// * `extract_query` - used to extract the source data into a cached partition
84    /// * `merge_partitions_query` - used to merge multiple partitions into a single one (and user queries which are one multiple partitions by default)
85    /// * `lake` - data lake
86    /// * `view_factory` - all views accessible to the `count_src_query`
87    /// * `session_configurator` - configurator for registering custom tables (e.g., JSON files)
88    /// * `update_group` - tells the daemon which view should be materialized and in what order
89    pub async fn new(
90        runtime: Arc<RuntimeEnv>,
91        view_set_name: Arc<String>,
92        min_event_time_column: Arc<String>,
93        max_event_time_column: Arc<String>,
94        count_src_query: Arc<String>,
95        extract_query: Arc<String>,
96        merge_partitions_query: Arc<String>,
97        lake: Arc<DataLakeConnection>,
98        view_factory: Arc<ViewFactory>,
99        session_configurator: Arc<dyn SessionConfigurator>,
100        update_group: Option<i32>,
101        max_partition_delta_from_source: TimeDelta,
102        max_partition_delta_from_merge: TimeDelta,
103        merger_maker: Option<&MergerMaker>,
104    ) -> Result<Self> {
105        let null_part_provider = Arc::new(NullPartitionProvider {});
106        let lakehouse = Arc::new(LakehouseContext::new(lake.clone(), runtime.clone()));
107        let ctx = make_session_context(
108            lakehouse,
109            null_part_provider,
110            None,
111            view_factory.clone(),
112            session_configurator.clone(),
113            CallerContext::maintenance(),
114        )
115        .await
116        .with_context(|| "make_session_context")?;
117        let now_str = Utc::now().to_rfc3339();
118        let sql = extract_query
119            .replace("{begin}", &now_str)
120            .replace("{end}", &now_str);
121        let extracted_df = ctx.sql(&sql).await?;
122        let schema = extracted_df.schema().inner().clone();
123        let session_configurator_for_merger = session_configurator.clone();
124        let merger = merger_maker.unwrap_or(&|_runtime, schema| {
125            let merge_query = Arc::new(merge_partitions_query.replace("{source}", "source"));
126            Arc::new(QueryMerger::new(
127                view_factory.clone(),
128                session_configurator_for_merger.clone(),
129                schema,
130                merge_query,
131            ))
132        })(runtime.clone(), schema.clone());
133
134        Ok(Self {
135            view_set_name,
136            view_instance_id: Arc::new(String::from("global")),
137            min_event_time_column,
138            max_event_time_column,
139            count_src_query,
140            extract_query,
141            merge_partitions_query,
142            schema,
143            merger,
144            view_factory,
145            session_configurator,
146            update_group,
147            max_partition_delta_from_source,
148            max_partition_delta_from_merge,
149            sort_order: None,
150            ordered_merger: None,
151        })
152    }

154    /// Declares that this view's partitions are internally sorted, ascending, by `columns` in
155    /// order. `columns` must be non-empty and every name must exist in this view's schema.
156    ///
157    /// This is a four-item view-author contract (see
158    /// `tasks/completed/1392_kway_merge_sorted_partitions_plan.md` Design §5):
159    /// 1. Every declared sort column must appear among the merge query's `GROUP BY` keys (order
160    ///    within `GROUP BY` is irrelevant; extra keys degrade to `PartiallySorted`, not a blocking
161    ///    sort).
162    /// 2. Any enrichment join in the merge query must put the dimension table on the left and the
163    ///    ordered stream on the right -- `CollectLeft` buffers its left (build) input and inherits
164    ///    its output ordering from its right (probe) input, so the natural phrasing
165    ///    `(<ordered agg>) a LEFT JOIN dim d` reinstates a blocking sort.
166    /// 3. The extract query needs a top-level `ORDER BY` matching the declared columns -- and it
167    ///    must be top-level: a CTE-internal `ORDER BY` that is later joined does not count, since
168    ///    the join discards it. The merge query needs no author-written `ORDER BY` at all: this
169    ///    builder forwards `columns` to `with_merge_scan_ordering`, and `QueryMerger` applies the
170    ///    sort as a DataFusion logical-plan node -- no SQL text is derived or rewritten.
171    /// 4. The merge query's aggregates must be composable over already-aggregated rows (e.g.
172    ///    `sum(count)`, not `count(*)`; no bare `avg` -- carry `sum` and `count` and divide at
173    ///    read time). `log_stats` is the in-repo model.
174    ///
175    /// A custom `merger_maker` passed to `new` remains fully supported alongside this
176    /// declaration: it stays the fallback merger, used whenever any input to a merge does not
177    /// certify the declared sort order, while merges whose inputs all certify take the ordered
178    /// `QueryMerger` this builder installs. This is the intended rollout path (plan Rollout steps
179    /// 1 and 4): keep e.g. a `BatchPartitionMerger` as the bounded fallback until every live
180    /// partition certifies, then drop it.
181    pub fn with_merge_sort_order(mut self, columns: Vec<Arc<String>>) -> Result<Self> {
182        if columns.is_empty() {
183            anyhow::bail!("with_merge_sort_order: columns must be non-empty");
184        }
185        for column in &columns {
186            self.schema.index_of(column.as_str()).with_context(|| {
187                format!(
188                    "with_merge_sort_order: column {column:?} not found in view {}'s schema",
189                    self.view_set_name
190                )
191            })?;
192        }
193        let scan_columns = sort_order_as_scan_columns(&columns);
194        let merge_query = Arc::new(self.merge_partitions_query.replace("{source}", "source"));
195        let ordered_merger: Arc<dyn PartitionMerger> = Arc::new(
196            QueryMerger::new(
197                self.view_factory.clone(),
198                self.session_configurator.clone(),
199                self.schema.clone(),
200                merge_query,
201            )
202            .with_merge_scan_ordering(ScanOrdering::PerFile {
203                columns: scan_columns,
204            }),
205        );
206        self.ordered_merger = Some(ordered_merger);
207        self.sort_order = Some(columns);
208        Ok(self)
209    }

221    /// True when a merge sort order is declared and every partition in `partitions_to_merge`
222    /// certifies it (empty partitions certify vacuously -- see `Partition::certifies_sort_order`).
223    fn all_inputs_certify(&self, partitions_to_merge: &[Partition]) -> bool {
224        let Some(sort_order) = &self.sort_order else {
225            return false;
226        };
227        let columns = sort_order_as_scan_columns(sort_order);
228        partitions_to_merge
229            .iter()
230            .all(|p| p.certifies_sort_order(&columns))
231    }

244    async fn make_batch_partition_spec(
245        &self,
246        lakehouse: Arc<LakehouseContext>,
247        existing_partitions: Arc<PartitionCache>,
248        insert_range: TimeRange,
249    ) -> Result<Arc<dyn PartitionSpec>> {
250        let view_meta = ViewMetadata {
251            view_set_name: self.get_view_set_name(),
252            view_instance_id: self.get_view_instance_id(),
253            file_schema_hash: self.get_file_schema_hash(),
254        };
255        let partitions_in_range = Arc::new(existing_partitions.filter_insert_range(insert_range));
256        let ctx = make_session_context(
257            lakehouse,
258            partitions_in_range.clone(),
259            None,
260            self.view_factory.clone(),
261            self.session_configurator.clone(),
262            CallerContext::maintenance(),
263        )
264        .await
265        .with_context(|| "make_session_context")?;
266
267        let count_src_sql = self
268            .count_src_query
269            .replace("{begin}", &insert_range.begin.to_rfc3339())
270            .replace("{end}", &insert_range.end.to_rfc3339());
271
272        let extract_sql = self
273            .extract_query
274            .replace("{begin}", &insert_range.begin.to_rfc3339())
275            .replace("{end}", &insert_range.end.to_rfc3339());
276
277        Ok(Arc::new(
278            fetch_sql_partition_spec(
279                ctx,
280                Arc::new(TrivialRecordBatchTransformer {}),
281                self.get_time_bounds(),
282                self.schema.clone(),
283                count_src_sql,
284                extract_sql,
285                view_meta,
286                insert_range,
287                self.sort_order
288                    .as_ref()
289                    .map(|columns| columns.iter().map(|c| c.to_string()).collect()),
290            )
291            .await
292            .with_context(|| "fetch_sql_partition_spec")?,
293        ))
294    }

328    async fn register_table(&self, ctx: &SessionContext, table: MaterializedView) -> Result<()> {
329        let view_name = self.get_view_set_name().to_string();
330        let partitions_table_name = format!("__{view_name}__partitions");
331        ctx.register_table(
332            TableReference::Bare {
333                table: partitions_table_name.clone().into(),
334            },
335            Arc::new(table),
336        )?;
337        let df = ctx
338            .sql(
339                &self
340                    .merge_partitions_query
341                    .replace("{source}", &partitions_table_name),
342            )
343            .await?;
344        ctx.register_table(
345            TableReference::Bare {
346                table: view_name.into(),
347            },
348            df.into_view(),
349        )?;
350        Ok(())
351    }

353    async fn merge_partitions(
354        &self,
355        lakehouse: Arc<LakehouseContext>,
356        partitions_to_merge: Arc<Vec<Partition>>,
357        partitions_all_views: Arc<PartitionCache>,
358        insert_range: TimeRange,
359    ) -> Result<MergeQueryResult> {
360        // An all-empty source scans as an EmptyExec, whose SortExec is never elided -- taking the
361        // ordered path there would trip the plan-shape check's memory-regression warning on every
362        // quiet-day retry. So the ordered path additionally requires at least one non-empty input.
363        let any_non_empty = partitions_to_merge.iter().any(|p| !p.is_empty());
364        let merger = match &self.ordered_merger {
365            Some(ordered_merger)
366                if any_non_empty && self.all_inputs_certify(&partitions_to_merge) =>
367            {
368                ordered_merger
369            }
370            _ => &self.merger,
371        };
372        let res = merger
373            .execute_merge_query(
374                lakehouse,
375                partitions_to_merge,
376                partitions_all_views,
377                insert_range,
378            )
379            .await;
380        if let Err(e) = &res {
381            error!("{e:?}");
382        }
383        res
384    }
