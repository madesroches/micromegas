31/// A `PartitionSpec` implementation for SQL-defined partitions.
32pub struct SqlPartitionSpec {
33    ctx: SessionContext,
34    transformer: Arc<dyn RecordBatchTransformer>,
35    compute_time_bounds: Arc<dyn DataFrameTimeBounds>,
36    schema: Arc<Schema>,
37    extract_query: String,
38    view_metadata: ViewMetadata,
39    insert_range: TimeRange,
40    record_count: i64,
41    /// The sort guarantee to record on the fresh partition this extract query writes, if any (see
42    /// `SqlBatchView::with_merge_sort_order`). When set, `write` applies it to the extract
43    /// query's `DataFrame` as a `DataFrame::sort` before executing it (see
44    /// `plan_sorted_extract`), so the guarantee it records always matches what was actually
45    /// written.
46    sort_order: Option<Vec<String>>,
47}
48
49impl SqlPartitionSpec {
50    #[expect(clippy::too_many_arguments)]
51    pub fn new(
52        ctx: SessionContext,
53        transformer: Arc<dyn RecordBatchTransformer>,
54        compute_time_bounds: Arc<dyn DataFrameTimeBounds>,
55        schema: Arc<Schema>,
56        extract_query: String,
57        view_metadata: ViewMetadata,
58        insert_range: TimeRange,
59        record_count: i64,
60        sort_order: Option<Vec<String>>,
61    ) -> Self {
62        Self {
63            ctx,
64            transformer,
65            compute_time_bounds,
66            schema,
67            extract_query,
68            view_metadata,
69            insert_range,
70            record_count,
71            sort_order,
72        }
73    }
74
75    /// Converts the declared `sort_order` (if any) to `ScanSortColumn`s and delegates to
76    /// `plan_sorted_extract`, which applies it to the extract query's `DataFrame` before building
77    /// the physical plan -- the same apply-then-plan discipline as
78    /// `QueryMerger::execute_sorted_merge` -- then executes the plan it returns. The undeclared
79    /// path (`sort_order: None`) plans and executes the query unsorted.
80    async fn execute_extract_query(&self, df: DataFrame) -> Result<SendableRecordBatchStream> {
81        let columns: Option<Vec<ScanSortColumn>> = self.sort_order.as_ref().map(|sort_order| {
82            sort_order
83                .iter()
84                .map(|c| ScanSortColumn {
85                    column: Arc::new(c.clone()),
86                    descending: false,
87                })
88                .collect()
89        });
90        let subject = format!("extract query for {}", self.view_metadata.view_set_name);
91        let (plan, task_ctx) =
92            plan_sorted_extract(df, columns.as_deref(), &subject, self.insert_range).await?;
93        execute_stream(plan, task_ctx).with_context(|| "executing extract query plan")
94    }
95}

97/// Applies `sort_order` (if declared) to `df` as a `DataFrame::sort` over its `ScanSortColumn`s,
98/// builds the physical plan, and -- only when a sort order was declared -- runs
99/// `assert_single_partition` and `assert_ordering_satisfied` against it, mirroring
100/// `QueryMerger::execute_sorted_merge`'s apply-then-plan sequence (`merge.rs:227-270`; its four
101/// merge-specific optimizer settings have no counterpart here). With `sort_order: None` the plan
102/// is built unsorted and neither assertion runs. `subject` names the query for the assertions'
103/// error messages (e.g. `format!("extract query for {view}")`); `insert_range` is the range being
104/// written. Returns the physical plan and the `TaskContext` the caller needs to execute it --
105/// callers are left to `execute_stream` it themselves, since only they know what to do with the
106/// resulting stream.
107pub async fn plan_sorted_extract(
108    df: DataFrame,
109    sort_order: Option<&[ScanSortColumn]>,
110    subject: &str,
111    insert_range: TimeRange,
112) -> Result<(Arc<dyn ExecutionPlan>, Arc<TaskContext>)> {
113    let Some(columns) = sort_order else {
114        let task_ctx = Arc::new(df.task_ctx());
115        let plan = df
116            .create_physical_plan()
117            .await
118            .with_context(|| "creating physical plan for extract query")?;
119        return Ok((plan, task_ctx));
120    };
121
122    let df = df.sort(sort_exprs(columns))?;
123    let task_ctx = Arc::new(df.task_ctx());
124    let plan = df
125        .create_physical_plan()
126        .await
127        .with_context(|| "creating physical plan for extract query")?;
128
129    assert_single_partition(
130        &plan,
131        subject,
132        insert_range,
133        "a declared sort_order requires a single-partition, globally-ordered output.",
134    )?;
135
136    assert_ordering_satisfied(
137        &plan,
138        columns,
139        "extract-query",
140        subject,
141        insert_range,
142        &format!(
143            "the declared sort_order {columns:?} even after applying it as a DataFrame::sort; \
144             this indicates the physical plan silently dropped or reordered the applied sort, \
145             not an author mistake."
146        ),
147    )?;
148
149    Ok((plan, task_ctx))
150}

159impl PartitionSpec for SqlPartitionSpec {
160    fn is_empty(&self) -> bool {
161        self.record_count < 1
162    }
163
164    fn get_source_data_hash(&self) -> Vec<u8> {
165        self.record_count.to_le_bytes().to_vec()
166    }
167
168    #[span_fn(view = self.view_metadata.view_set_name)]
169    async fn write(&self, lake: Arc<DataLakeConnection>, logger: Arc<dyn Logger>) -> Result<()> {
170        let stream = self.begin_write(&logger).await?;
171
172        let (tx, rx) = tokio::sync::mpsc::channel(1);
173        let join_handle = spawn_with_context(write_partition_from_rows(
174            lake.clone(),
175            self.view_metadata.clone(),
176            self.schema.clone(),
177            self.insert_range,
178            self.get_source_data_hash(),
179            self.sort_order.clone(),
180            RetireMatch::Containment,
181            Vec::new(),
182            rx,
183            logger.clone(),
184        ));
185
186        self.extract_stream(stream, tx, join_handle).await
187    }
188
189    #[span_fn(view = self.view_metadata.view_set_name)]
190    async fn write_file(
191        &self,
192        lake: Arc<DataLakeConnection>,
193        logger: Arc<dyn Logger>,
194    ) -> Result<Partition> {
195        let stream = self.begin_write(&logger).await?;
196
197        let (tx, rx) = tokio::sync::mpsc::channel(1);
198        let view_metadata = self.view_metadata.clone();
199        let schema = self.schema.clone();
200        let insert_range = self.insert_range;
201        let source_data_hash = self.get_source_data_hash();
202        let sort_order = self.sort_order.clone();
203        let join_handle = spawn_with_context(async move {
204            write_partition_file(
205                &lake,
206                view_metadata,
207                schema,
208                insert_range,
209                source_data_hash,
210                sort_order,
211                rx,
212                logger,
213            )
214            .await
215        });
216
217        self.extract_stream(stream, tx, join_handle).await
218    }

221impl SqlPartitionSpec {
222    /// Logs the start of a write and plans the extract. An empty `record_count` is allowed: the
223    /// writer creates an empty partition record when no data is sent through the channel.
224    async fn begin_write(&self, logger: &Arc<dyn Logger>) -> Result<SendableRecordBatchStream> {
225        let desc = format!(
226            "[{}, {}] {} {}",
227            self.view_metadata.view_set_name,
228            self.view_metadata.view_instance_id,
229            self.insert_range.begin.to_rfc3339(),
230            self.insert_range.end.to_rfc3339()
231        );
232        logger.write_log_entry(format!("writing {desc}")).await?;
233        self.extract_plan().await
234    }
235
236    #[span_fn(view = self.view_metadata.view_set_name)]
237    async fn extract_plan(&self) -> Result<SendableRecordBatchStream> {
238        let df = self.ctx.sql(&self.extract_query).await?;
239        self.execute_extract_query(df).await
240    }
241
242    async fn transform_batch(
243        &self,
244        rb_res: datafusion::error::Result<RecordBatch>,
245    ) -> Result<PartitionRowSet> {
246        let rb = self.transformer.transform(rb_res?).await?;
247        let event_time_range = self
248            .compute_time_bounds
249            .get_time_bounds(self.ctx.read_batch(rb.clone())?)
250            .await?;
251        Ok(PartitionRowSet::new(event_time_range, rb, None))
252    }
253
254    // Takes `tx` by value so the writer task sees the channel close before it is awaited.
255    #[span_fn(view = self.view_metadata.view_set_name)]
256    async fn extract_stream<T>(
257        &self,
258        mut stream: SendableRecordBatchStream,
259        tx: tokio::sync::mpsc::Sender<Result<PartitionRowSet>>,
260        join_handle: tokio::task::JoinHandle<Result<T>>,
261    ) -> Result<T> {
262        // A bare early return would close the channel, which the writer reads as a clean end of
263        // data and would upload a partial file nobody registers. Forwarding the error makes it
264        // take its failure path and delete the file.
265        let mut extract_err = None;
266        while let Some(rb_res) = stream.next().await {
267            match self.transform_batch(rb_res).await {
268                Ok(row_set) => tx.send(Ok(row_set)).await?,
269                Err(e) => {
270                    extract_err = Some(e);
271                    break;
272                }
273            }
274        }
275        if let Some(e) = extract_err {
276            let msg = format!("{e:#}");
277            let _ = tx.send(Err(e)).await;
278            drop(tx);
279            return match join_handle.await? {
280                Ok(_) => Err(anyhow::anyhow!(msg)),
281                Err(writer_err) => Err(writer_err.context(msg)),
282            };
283        }
284        drop(tx);
285        join_handle.await?
286    }
287}

289/// Fetches a `SqlPartitionSpec` by executing a count query and an extract query. `sort_order`
290/// declares the ordering the resulting partition's rows are guaranteed to satisfy (see
291/// `SqlBatchView::with_merge_sort_order`); pass `None` for views that make no such guarantee.
292#[expect(clippy::too_many_arguments)]
293#[span_fn(view = view_metadata.view_set_name)]
294pub async fn fetch_sql_partition_spec(
295    ctx: SessionContext,
296    transformer: Arc<dyn RecordBatchTransformer>,
297    compute_time_bounds: Arc<dyn DataFrameTimeBounds>,
298    schema: Arc<Schema>,
299    count_src_sql: String,
300    extract_query: String,
301    view_metadata: ViewMetadata,
302    insert_range: TimeRange,
303    sort_order: Option<Vec<String>>,
304) -> Result<SqlPartitionSpec> {
305    let batches = count_query(&ctx, &count_src_sql, &view_metadata.view_set_name).await?;
306    if batches.len() != 1 {
307        anyhow::bail!("fetch_sql_partition_spec: query should return a single batch");
308    }
309    let rb = &batches[0];
310    let count_column: &Int64Array = typed_column_by_name(rb, "count")?;
311    if count_column.len() != 1 {
312        anyhow::bail!("fetch_sql_partition_spec: query should return a single row");
313    }
314    let count = count_column.value(0);
315    if count > 0 {
316        trace!(
317            "fetch_sql_partition_spec for view {}, count={count}",
318            &*view_metadata.view_set_name
319        );
320    }
321    Ok(SqlPartitionSpec::new(
322        ctx,
323        transformer,
324        compute_time_bounds,
325        schema,
326        extract_query,
327        view_metadata,
328        insert_range,
329        count,
330        sort_order,
331    ))
332}

337    count_src_sql: &str,
