1//! `list_query_denials()` -- admin UDTF listing every query-deny-list rule currently in force
2//! (`tasks/query_deny_list_plan.md` §8). Registered inside the same admin-gated block as
3//! `list_partitions()`'s mutating siblings.

20fn schema() -> SchemaRef {

23        Field::new(

29        Field::new("reason", DataType::Utf8, false),

37}

62#[derive(Debug)]
63struct ListQueryDenialsTableProvider {
64    query_denials: Arc<QueryDenyList>,
65}
66
67#[async_trait]
68impl TableProvider for ListQueryDenialsTableProvider {
69    fn schema(&self) -> SchemaRef {
70        schema()
71    }
72
73    fn table_type(&self) -> TableType {
74        TableType::Temporary
75    }
76
77    async fn scan(
78        &self,
79        _state: &dyn Session,
80        projection: Option<&Vec<usize>>,
81        _filters: &[Expr],
82        limit: Option<usize>,
83    ) -> datafusion::error::Result<Arc<dyn ExecutionPlan>> {
84        let mut rows = self
85            .query_denials
86            .list()
87            .await
88            .map_err(|e| DataFusionError::External(e.into()))?;
89        if let Some(n) = limit {
90            rows.truncate(n);
91        }
92        let rule_id: StringArray = rows.iter().map(|r| Some(r.rule_id.to_string())).collect();
93        let created_at: TimestampNanosecondArray = rows
94            .iter()
95            .map(|r| r.created_at.timestamp_nanos_opt())
96            .collect::<TimestampNanosecondArray>()
97            .with_timezone("+00:00".to_string());
98        let created_by: StringArray = rows.iter().map(|r| Some(r.created_by.clone())).collect();
99        let reason: StringArray = rows.iter().map(|r| Some(r.reason.clone())).collect();
100        let match_expr: StringArray = rows.iter().map(|r| Some(r.match_expr.clone())).collect();
101        let last_hit_at: TimestampNanosecondArray = rows
102            .iter()
103            .map(|r| r.last_hit_at.and_then(|t| t.timestamp_nanos_opt()))
104            .collect::<TimestampNanosecondArray>()
105            .with_timezone("+00:00".to_string());
106        let rb = RecordBatch::try_new(
107            schema(),
108            vec![
109                Arc::new(rule_id),
110                Arc::new(created_at),
111                Arc::new(created_by),
112                Arc::new(reason),
113                Arc::new(match_expr),
114                Arc::new(last_hit_at),
115            ],
116        )?;
117        let source =
118            MemorySourceConfig::try_new(&[vec![rb]], schema(), projection.map(|v| v.to_owned()))?;
119        Ok(DataSourceExec::from_data_source(source))
120    }
121}
