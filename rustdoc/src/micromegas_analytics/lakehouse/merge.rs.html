32/// The outcome of running a merge query.
33pub struct MergeQueryResult {
34    /// The merged rows.
35    pub stream: SendableRecordBatchStream,
36    /// Whether the merger's declared scan ordering (if any) was honored by the physical plan
37    /// without falling back to a buffering `Sort` node. Always `true` when no ordering was
38    /// declared to DataFusion in the first place -- it is only ever computed dynamically by an
39    /// ordering-declaring `QueryMerger`. This drives only a memory-regression warning; it never
40    /// gates the recorded `sort_order` (see `View::get_merged_partition_sort_order`).
41    ///
42    /// For `ScanOrdering::PerFile`, a surviving `SortPreservingMergeExec` is the *expected*
43    /// operator, not a regression -- only a surviving `SortExec` means the streaming shape was
44    /// lost. For `ScanOrdering::Concatenated`, either operator surviving is a regression.
45    pub ordering_honored: bool,
46}

61/// A `PartitionMerger` that executes a SQL query to merge partitions.
62#[derive(Debug)]
63pub struct QueryMerger {
64    view_factory: Arc<ViewFactory>,
65    session_configurator: Arc<dyn SessionConfigurator>,
66    file_schema: Arc<Schema>,
67    query: Arc<String>,
68    merge_scan_ordering: ScanOrdering,
69}

71impl QueryMerger {
72    pub fn new(
73        view_factory: Arc<ViewFactory>,
74        session_configurator: Arc<dyn SessionConfigurator>,
75        file_schema: Arc<Schema>,
76        query: Arc<String>,
77    ) -> Self {
78        Self {
79            view_factory,
80            session_configurator,
81            file_schema,
82            query,
83            merge_scan_ordering: ScanOrdering::Unordered,
84        }
85    }
86
87    /// Declares an ordering the merge's source scan already satisfies (see
88    /// `PartitionedTableProvider::with_scan_ordering`), letting DataFusion elide the merge query's
89    /// `Sort` node instead of buffering. Default: `Unordered` (matching today's behavior for every
90    /// existing caller).
91    pub fn with_merge_scan_ordering(mut self, ordering: ScanOrdering) -> Self {
92        self.merge_scan_ordering = ordering;
93        self
94    }
95
96    /// The `Concatenated` merge path (unchanged from before `ScanOrdering` existed): force the
97    /// source scan into a single sequential file group so the declared ordering can be elided
98    /// instead of re-sorted, then build the physical plan once, inspect it, and execute that exact
99    /// plan -- never planning or building twice.
100    async fn execute_concatenated_merge(
101        &self,
102        ctx: &SessionContext,
103        insert_range: TimeRange,
104    ) -> Result<MergeQueryResult> {
105        ctx.state_ref()
106            .write()
107            .config_mut()
108            .options_mut()
109            .optimizer
110            .repartition_file_scans = false;
111
112        let df = ctx.sql(&self.query).await?;
113        let task_ctx = Arc::new(df.task_ctx());
114        let plan = df
115            .create_physical_plan()
116            .await
117            .with_context(|| "creating physical plan for merge query")?;
118
119        assert_single_partition(
120            &plan,
121            &format!("merge query {:?}", self.query),
122            insert_range,
123            "executing it would coalesce partitions and destroy the declared ordering. This \
124             likely means repartition_file_scans did not take effect.",
125        )?;
126
127        let plan_str = displayable(plan.as_ref()).indent(true).to_string();
128        let ordering_honored =
129            !plan_str.contains("SortExec") && !plan_str.contains("SortPreservingMergeExec");
130        if !ordering_honored {
131            warn!(
132                "merge query {:?} (insert_range=[{}, {}]) did not elide its declared ordering -- \
133                 the merge will still produce a correctly ordered result, but it will buffer in \
134                 memory instead of streaming. Plan:\n{plan_str}",
135                self.query,
136                insert_range.begin.to_rfc3339(),
137                insert_range.end.to_rfc3339()
138            );
139        }
140
141        let stream =
142            execute_stream(plan, task_ctx).with_context(|| "executing merge query plan")?;
143        Ok(MergeQueryResult {
144            stream,
145            ordering_honored,
146        })
147    }
148
149    /// The `PerFile` k-way merge path (Design §2 of
150    /// `tasks/completed/1392_kway_merge_sorted_partitions_plan.md`): five optimizer settings keep the merge
151    /// a bounded-memory streaming k-way merge instead of fanning out to `target_partitions`, an
152    /// unconditional logical-plan `DataFrame::sort` makes the declared ordering a property of the
153    /// query plan rather than of an optimizer preference (so a wrong or missing merge-query sort
154    /// is not representable), and three checks -- two hard, one warn-only -- run against the
155    /// physical plan before it is executed.
156    async fn execute_per_file_merge(
157        &self,
158        ctx: &SessionContext,
159        columns: &[ScanSortColumn],
160        insert_range: TimeRange,
161    ) -> Result<MergeQueryResult> {
162        {
163            let state = ctx.state_ref();
164            let mut state = state.write();
165            let optimizer = &mut state.config_mut().options_mut().optimizer;
166            optimizer.repartition_file_scans = false;
167            optimizer.enable_round_robin_repartition = false;
168            optimizer.repartition_aggregations = false;
169            optimizer.prefer_existing_sort = true;
170            optimizer.repartition_joins = false;
171        }
172
173        let df = ctx.sql(&self.query).await?;
174        let df = df.sort(
175            columns
176                .iter()
177                .map(|c| {
178                    Expr::Column(datafusion::common::Column::new_unqualified(
179                        c.column.as_str(),
180                    ))
181                    .sort(!c.descending, c.descending)
182                })
183                .collect(),
184        )?;
185        let task_ctx = Arc::new(df.task_ctx());
186        let plan = df
187            .create_physical_plan()
188            .await
189            .with_context(|| "creating physical plan for per-file merge query")?;
190
191        // Check 1 (hard): execute_stream would otherwise coalesce partitions and destroy order.
192        assert_single_partition(
193            &plan,
194            &format!("merge query {:?}", self.query),
195            insert_range,
196            "executing it would coalesce partitions and destroy the declared ordering. This \
197             likely means repartition_aggregations or enable_round_robin_repartition did not \
198             take effect, or the merge query is missing a collapsing operator (e.g. a \
199             SortPreservingMergeExec) to bring the per-file partitions back to one.",
200        )?;
201
202        // Check 2 (hard): the plan's output ordering must satisfy the declared columns -- a
203        // defensive assertion that the sort_order about to be recorded is truthful. This has no
204        // author-reachable failure path under this design (the sort above is applied
205        // unconditionally): it is a backstop against a future DataFusion regression, not against
206        // author error.
207        assert_ordering_satisfied(
208            &plan,
209            columns,
210            "per-file merge",
211            &format!("merge query {:?}", self.query),
212            insert_range,
213            &format!(
214                "the declared per-file merge columns {columns:?}; refusing to record a false \
215                 sort_order guarantee."
216            ),
217        )?;
218
219        // Check 3 (warn-only): a SortPreservingMergeExec is the *expected* operator here -- only
220        // a surviving SortExec signals the streaming shape regressed (memory, not correctness --
221        // check 2 already proved the output order).
222        let plan_str = displayable(plan.as_ref()).indent(true).to_string();
223        let ordering_honored = !plan_str.contains("SortExec");
224        if !ordering_honored {
225            warn!(
226                "merge query {:?} (insert_range=[{}, {}]) did not elide its declared per-file \
227                 ordering -- the merge will still produce a correctly ordered result, but it will \
228                 buffer in memory instead of streaming. Plan:\n{plan_str}",
229                self.query,
230                insert_range.begin.to_rfc3339(),
231                insert_range.end.to_rfc3339()
232            );
233        }
234
235        let stream = execute_stream(plan, task_ctx)
236            .with_context(|| "executing per-file merge query plan")?;
237        Ok(MergeQueryResult {
238            stream,
239            ordering_honored,
240        })
241    }
242}

244#[async_trait]
245impl PartitionMerger for QueryMerger {
246    async fn execute_merge_query(
247        &self,
248        lakehouse: Arc<LakehouseContext>,
249        partitions_to_merge: Arc<Vec<Partition>>,
250        partitions_all_views: Arc<PartitionCache>,
251        insert_range: TimeRange,
252    ) -> Result<MergeQueryResult> {
253        let reader_factory = lakehouse.reader_factory().clone();
254        let ctx = make_session_context(
255            lakehouse.clone(),
256            partitions_all_views,
257            Some(insert_range),
258            self.view_factory.clone(),
259            self.session_configurator.clone(),
260            true,
261        )
262        .await?;
263        let src_table = PartitionedTableProvider::with_scan_ordering(
264            self.file_schema.clone(),
265            reader_factory,
266            partitions_to_merge,
267            self.merge_scan_ordering.clone(),
268        );
269        ctx.register_table(
270            TableReference::Bare {
271                table: "source".into(),
272            },
273            Arc::new(src_table),
274        )?;
275
276        match &self.merge_scan_ordering {
277            ScanOrdering::Unordered => {
278                let stream = ctx
279                    .sql(&self.query)
280                    .await?
281                    .execute_stream()
282                    .await
283                    .with_context(|| "merged_df.execute_stream")?;
284                Ok(MergeQueryResult {
285                    stream,
286                    ordering_honored: true,
287                })
288            }
289            ScanOrdering::Concatenated { .. } => {
290                self.execute_concatenated_merge(&ctx, insert_range).await
291            }
292            ScanOrdering::PerFile { columns } => {
293                self.execute_per_file_merge(&ctx, columns, insert_range)
294                    .await
295            }
296        }
297    }
298}

300fn partition_set_stats(
301    view: Arc<dyn View>,
302    filtered_partitions: &[Partition],
303) -> Result<(i64, i64)> {
304    let mut sum_size: i64 = 0;
305    let mut source_hash: i64 = 0;
306    let latest_file_schema_hash = view.get_file_schema_hash();
307    for p in filtered_partitions {
308        // for some time all the hashes will actually be the number of events in the source data
309        // when views have different hash algos, we should delegate to the view the creation of the merged hash
310        source_hash = if p.source_data_hash.len() == std::mem::size_of::<i64>() {
311            source_hash + hash_to_object_count(&p.source_data_hash)?
312        } else {
313            //previous hash algo
314            xxh32(&p.source_data_hash, source_hash as u32).into()
315        };
316
317        sum_size += p.file_size;
318
319        if p.view_metadata.file_schema_hash != latest_file_schema_hash {
320            anyhow::bail!(
321                "incompatible file schema with [{},{}]",
322                p.begin_insert_time().to_rfc3339(),
323                p.end_insert_time().to_rfc3339()
324            );
325        }
326    }
327    Ok((sum_size, source_hash))
328}

330/// Creates a merged partition from a set of existing partitions.
331pub async fn create_merged_partition(
332    partitions_to_merge: Arc<PartitionCache>,
333    partitions_all_views: Arc<PartitionCache>,
334    lakehouse: Arc<LakehouseContext>,
335    view: Arc<dyn View>,
336    insert_range: TimeRange,
337    logger: Arc<dyn Logger>,
338) -> Result<()> {
339    let view_set_name = &view.get_view_set_name();
340    let view_instance_id = &view.get_view_instance_id();
341    let desc = format!(
342        "[{}, {}] {view_set_name} {view_instance_id}",
343        insert_range.begin.to_rfc3339(),
344        insert_range.end.to_rfc3339()
345    );
346    // we are not looking for intersecting partitions, but only those that fit completely in the range
347    // otherwise we'd get duplicated records
348    let mut filtered_partitions = partitions_to_merge
349        .filter_inside_range(view_set_name, view_instance_id, insert_range)
350        .partitions;
351    if filtered_partitions.len() != partitions_to_merge.len() {
352        warn!("partitions_to_merge was not filtered properly");
353    }
354    if filtered_partitions.len() < 2 {
355        logger
356            .write_log_entry(format!("{desc}: not enough partitions to merge"))
357            .await
358            .with_context(|| "writing log")?;
359        return Ok(());
360    }
361    let (sum_size, source_hash) = partition_set_stats(view.clone(), &filtered_partitions)
362        .with_context(|| "partition_set_stats")?;
363    logger
364        .write_log_entry(format!(
365            "{desc}: merging {} partitions sum_size={sum_size}",
366            filtered_partitions.len()
367        ))
368        .await
369        .with_context(|| "write_log_entry")?;
370    filtered_partitions.sort_by_key(|p| p.begin_insert_time());
371    // Computed before merge_partitions runs: a pure function of the input slice alone (Design §4).
372    let merged_sort_order = view.get_merged_partition_sort_order(&filtered_partitions);
373    let merge_result = view
374        .merge_partitions(
375            lakehouse.clone(),
376            Arc::new(filtered_partitions),
377            partitions_all_views,
378            insert_range,
379        )
380        .await
381        .with_context(|| "view.merge_partitions")?;
382    // A defeated ordering elision (ordering_honored: false) is already warned about, with the
383    // offending plan, inside QueryMerger::execute_merge_query.
384    let mut merged_stream = merge_result.stream;
385    let (tx, rx) = tokio::sync::mpsc::channel(1);
386    let view_copy = view.clone();
387    let lake = lakehouse.lake().clone();
388    let join_handle = spawn_with_context(write_partition_from_rows(
389        lake,
390        view_copy.get_meta(),
391        view_copy.get_file_schema(),
392        insert_range,
393        source_hash.to_le_bytes().to_vec(),
394        merged_sort_order,
395        RetireMatch::Containment,
396        Vec::new(),
397        rx,
398        logger.clone(),
399    ));
400    let compute_time_bounds = view.get_time_bounds();
401    let ctx =
402        SessionContext::new_with_config_rt(SessionConfig::default(), lakehouse.runtime().clone());
403    let stream_result: Result<()> = async {
404        while let Some(rb_res) = merged_stream.next().await {
405            let rb = rb_res.with_context(|| "receiving record_batch from stream")?;
406            let event_time_range = compute_time_bounds
407                .get_time_bounds(ctx.read_batch(rb.clone()).with_context(|| "read_batch")?)
408                .await?;
409            tx.send(Ok(PartitionRowSet::new(event_time_range, rb)))
410                .await
411                .with_context(|| "sending partition row set")?;
412        }
413        Ok(())
414    }
415    .await;
416
417    match stream_result {
418        Ok(()) => {
419            drop(tx);
420            join_handle.await??;
421            Ok(())
422        }
423        Err(e) => {
424            warn!("aborting merge partition write for {desc}: {e:?}");
425            let _ = tx.send(Err(anyhow::anyhow!("merge stream aborted"))).await;
426            drop(tx);
427            match join_handle.await {
428                Ok(Ok(())) => {}
429                Ok(Err(writer_err)) => {
430                    debug!("merge writer task error during abort: {writer_err:?}");
431                }
432                Err(join_err) => {
433                    warn!("merge writer task panicked during abort: {join_err:?}");
434                }
435            }
436            Err(e)
437        }
438    }
