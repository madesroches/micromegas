56async fn register_table(
57    lakehouse: Arc<LakehouseContext>,
58    reader_factory: Arc<ReaderFactory>,
59    part_provider: Arc<dyn QueryPartitionProvider>,
60    query_range: Option<TimeRange>,
61    ctx: &SessionContext,
62    view: Arc<dyn View>,
63) -> Result<()> {
64    let table = MaterializedView::new(
65        lakehouse,

69        query_range,
70        None,
71    );
72    view.register_table(ctx, table).await
73}

75/// query_partitions_context returns a context to run queries using the partitions as the "source" table
76#[span_fn]
77pub async fn query_partitions_context(
78    lakehouse: &LakehouseContext,
79    schema: SchemaRef,
80    partitions: Arc<Vec<Partition>>,
81) -> Result<SessionContext> {
82    let table =
83        PartitionedTableProvider::new(schema, lakehouse.reader_factory().clone(), partitions);
84    let object_store_url = ObjectStoreUrl::parse("obj://lakehouse/").unwrap();
85    let ctx = SessionContext::new_with_config_rt(
86        lakehouse.session_config().await?,
87        lakehouse.runtime().clone(),
88    );
89    ctx.register_object_store(
90        object_store_url.as_ref(),
91        lakehouse.lake().blob_storage.inner(),
92    );
93    ctx.register_table(
94        TableReference::Bare {
95            table: "source".into(),
96        },
97        Arc::new(table),
98    )?;
99    register_extension_functions(&ctx);
100    Ok(ctx)
101}

103// query_partitions returns a dataframe, leaving the option of streaming the results
104#[span_fn]
105pub async fn query_partitions(
106    lakehouse: &LakehouseContext,
107    schema: SchemaRef,
108    partitions: Arc<Vec<Partition>>,
109    sql: &str,
110) -> Result<DataFrame> {
111    let ctx = query_partitions_context(lakehouse, schema, partitions).await?;
112    Ok(ctx.sql(sql).await?)
113}

115/// register functions that are part of the lakehouse architecture
116#[span_fn]
117pub fn register_lakehouse_functions(
118    ctx: &SessionContext,
119    lakehouse: Arc<LakehouseContext>,
120    part_provider: Arc<dyn QueryPartitionProvider>,
121    query_range: Option<TimeRange>,
122    view_factory: Arc<ViewFactory>,
123    caller: &CallerContext,
124    configurator: Arc<dyn SessionConfigurator>,
125) {
126    // The call-level guard: one guard, shared (via `Arc`) across every
127    // arg-addressed UDTF/UDF this call registers below -- `view_instance`, `process_spans`,
128    // `perfetto_trace_chunks`, `parse_block`, `get_payload`, and `list_partitions`' row filter.
129    // `ReadScope::All` makes every one of its checks a no-op, so this costs nothing for
130    // internal/maintenance callers or an auth-unset deployment.
131    let lakehouse_admin = caller.is_admin;
132    let audience_guard = Arc::new(AudienceGuard::new(
133        caller.read_scope.clone(),
134        lakehouse_admin,
135        caller.isolation_config.public_view_sets.clone(),
136        lakehouse.audience_index().clone(),
137    ));
138    ctx.register_udtf(
139        "view_instance",
140        Arc::new(ViewInstanceTableFunction::new(
141            lakehouse.clone(),
142            view_factory.clone(),
143            part_provider.clone(),
144            query_range,
145            audience_guard.clone(),
146        )),
147    );
148    ctx.register_udtf(
149        "list_partitions",
150        Arc::new(ListPartitionsTableFunction::new(
151            lakehouse.lake().clone(),
152            audience_guard.clone(),
153        )),
154    );
155    ctx.register_udtf(
156        "list_view_sets",
157        Arc::new(ListViewSetsTableFunction::new(view_factory.clone())),
158    );
159    // Caller-scoped, not admin-gated: registered for every caller, admin or not, with the
160    // visibility split decided here rather than inside the function -- an admin sees every
161    // row, a non-admin only the pairs their `grant_selectors` hold.
162    ctx.register_udtf(
163        "list_audience_grants",
164        Arc::new(ListAudienceGrantsTableFunction::new(
165            lakehouse.lake().db_pool.clone(),
166            if caller.is_admin {
167                GrantVisibility::All
168            } else {
169                GrantVisibility::Held(caller.grant_selectors.clone())
170            },
171        )),
172    );
173    ctx.register_udtf(
174        "perfetto_trace_chunks",
175        Arc::new(PerfettoTraceTableFunction::new(
176            lakehouse.clone(),
177            view_factory.clone(),
178            part_provider.clone(),
179            audience_guard.clone(),
180        )),
181    );
182    ctx.register_udtf(
183        "parse_block",
184        Arc::new(ParseBlockTableFunction::new(
185            lakehouse.clone(),
186            view_factory.clone(),
187            part_provider.clone(),
188            query_range,
189            audience_guard.clone(),
190        )),
191    );
192    ctx.register_udtf(
193        "process_spans",
194        Arc::new(ProcessSpansTableFunction::new(
195            lakehouse.clone(),
196            view_factory.clone(),
197            part_provider.clone(),
198            query_range,
199            audience_guard.clone(),
200        )),
201    );
202    ctx.register_udf(
203        AsyncScalarUDF::new(Arc::new(GetPayload::new(
204            lakehouse.lake().clone(),
205            audience_guard.clone(),
206        )))
207        .into_scalar_udf(),
208    );
209    if lakehouse_admin {
210        ctx.register_udtf(
211            "retire_partitions",
212            Arc::new(RetirePartitionsTableFunction::new(lakehouse.lake().clone())),
213        );
214        ctx.register_udtf(
215            "materialize_partitions",
216            Arc::new(MaterializePartitionsTableFunction::new(
217                lakehouse.clone(),
218                view_factory.clone(),
219            )),
220        );
221        ctx.register_udtf(
222            "regenerate_partitions",
223            Arc::new(RegeneratePartitionsTableFunction::new(
224                lakehouse.clone(),
225                view_factory.clone(),
226            )),
227        );
228        ctx.register_udf(
229            make_retire_partition_by_file_udf(lakehouse.lake().clone()).into_scalar_udf(),
230        );
231        ctx.register_udf(
232            make_retire_partition_by_metadata_udf(lakehouse.lake().clone()).into_scalar_udf(),
233        );
234        // Admin-managed query deny list: same admin gate as the five functions above.
235        ctx.register_udtf(
236            "list_query_denials",
237            Arc::new(ListQueryDenialsTableFunction::new(
238                lakehouse.query_denials().clone(),
239            )),
240        );
241        ctx.register_udtf(
242            "deny_queries",
243            Arc::new(DenyQueriesTableFunction::new(
244                lakehouse.query_denials().clone(),
245                caller.identity.clone(),
246            )),
247        );
248        ctx.register_udf(
249            make_remove_query_denial_udf(lakehouse.query_denials().clone()).into_scalar_udf(),
250        );
251        // Reads `lakehouse_view_set_definitions` straight from Postgres, so an admin can see a
252        // definition that exists on disk but failed to load, and adds the built-in view sets.
253        ctx.register_udtf(
254            "list_view_set_definitions",
255            Arc::new(ListViewSetDefinitionsTableFunction::new(
256                lakehouse.lake().clone(),
257                view_factory.clone(),
258            )),
259        );
260        // Plans a definition's merge as the daemon would, so it needs the same session
261        // configurator the real sessions use.
262        ctx.register_udtf(
263            "check_view_merge",
264            Arc::new(CheckViewMergeTableFunction::new(
265                lakehouse.clone(),
266                view_factory.clone(),
267                configurator,
268            )),
269        );
270    }
271}

286    query_range: Option<TimeRange>,

303#[span_fn]
304pub async fn make_session_context(
305    lakehouse: Arc<LakehouseContext>,
306    part_provider: Arc<dyn QueryPartitionProvider>,
307    query_range: Option<TimeRange>,
308    view_factory: Arc<ViewFactory>,
309    configurator: Arc<dyn SessionConfigurator>,
310    caller: CallerContext,
311) -> Result<SessionContext> {
312    // Disable page index reading for backward compatibility with legacy Parquet files
313    // Legacy files may have incomplete ColumnIndex metadata (missing null_pages field)
314    // which causes errors in DataFusion 51+ with Arrow 57.0 when reading page indexes
315    let config = lakehouse
316        .session_config()
317        .await?
318        .set_bool("datafusion.execution.parquet.enable_page_index", false)
319        // Populate `Diagnostic`/`Span` on plan-time errors (unknown column, ambiguous
320        // reference, type mismatch, ...) so a caller-facing message can point at a
321        // line/column in their SQL text instead of just naming the problem. Off by
322        // default in DataFusion.
323        .set_bool("datafusion.sql_parser.collect_spans", true)
324        .with_information_schema(true);
325    let ctx = SessionContext::new_with_config_rt(config, lakehouse.runtime().clone());
326    if let Some(range) = &query_range {
327        ctx.add_analyzer_rule(Arc::new(TableScanRewrite::new(*range)));
328    }
329    let object_store_url = ObjectStoreUrl::parse("obj://lakehouse/").unwrap();
330    let object_store = lakehouse.lake().blob_storage.inner();
331    ctx.register_object_store(object_store_url.as_ref(), object_store);
332    let reader_factory = lakehouse.reader_factory().clone();
333    register_functions(
334        &ctx,
335        lakehouse.clone(),
336        part_provider.clone(),
337        query_range,
338        view_factory.clone(),
339        &caller,
340        configurator.clone(),
341    );
342    for view in view_factory.get_global_views() {
343        register_table(
344            lakehouse.clone(),
345            reader_factory.clone(),
346            part_provider.clone(),
347            query_range,
348            &ctx,
349            view.clone(),
350        )
351        .await?;
352    }
353    if caller.read_scope != ReadScope::All {
354        // ReadScope::All is the internal/maintenance marker -- OwnershipRewrite would no-op for
355        // it anyway, so skip resolving `processes`/`streams` sources and registering the rule
356        // entirely rather than requiring every ReadScope::All caller's ViewFactory to carry them.
357        //
358        // Must be registered *after* the TableScanRewrite registration above (`query_range.is_some()`
359        // block): TableScanRewrite::analyze walks the whole plan with transform_up_with_subqueries
360        // and would time-bound the audience lookup's own processes/streams scans if it ran after
361        // OwnershipRewrite injected them (analyzer rules each run exactly once, in registration
362        // order). The audience lookup must stay time-unbounded (query_range: None below).
363        let processes_view = view_factory.get_global_view("processes").with_context(
364            || "OwnershipRewrite requires the `processes` global view to be registered",
365        )?;
366        let streams_view = view_factory.get_global_view("streams").with_context(
367            || "OwnershipRewrite requires the `streams` global view to be registered",
368        )?;
369        let processes_source: Arc<dyn TableSource> =
370            Arc::new(DefaultTableSource::new(Arc::new(MaterializedView::new(
371                lakehouse.clone(),
372                reader_factory.clone(),
373                processes_view,
374                part_provider.clone(),
375                None,
376                None,
377            ))));
378        let streams_source: Arc<dyn TableSource> =
379            Arc::new(DefaultTableSource::new(Arc::new(MaterializedView::new(
380                lakehouse.clone(),
381                reader_factory.clone(),
382                streams_view,
383                part_provider.clone(),
384                None,
385                None,
386            ))));
387        ctx.add_analyzer_rule(Arc::new(OwnershipRewrite::new(
388            caller.read_scope.clone(),
389            caller.isolation_config.public_view_sets.clone(),
390            processes_source,
391            streams_source,
392        )));
393    }
394    // Apply custom configuration
395    configurator.configure(&ctx).await?;
396    Ok(ctx)
397}

399#[span_fn]
400pub async fn query(
401    lakehouse: Arc<LakehouseContext>,
402    part_provider: Arc<dyn QueryPartitionProvider>,
403    query_range: Option<TimeRange>,
404    sql: &str,
405    view_factory: Arc<ViewFactory>,
406    configurator: Arc<dyn SessionConfigurator>,
407    caller: CallerContext,
408) -> Result<Answer> {
409    info!("query sql={sql}");
410    let ctx = make_session_context(
411        lakehouse,
412        part_provider,
413        query_range,
414        view_factory,
415        configurator,
416        caller,
417    )
418    .await
419    .with_context(|| "make_session_context")?;
420    let df = ctx.sql(sql).await?;
421    let schema = df.schema().inner().clone();
422    let batches: Vec<RecordBatch> = df.collect().await?;
423    Ok(Answer::new(schema, batches))
424}
