24/// Which pair of bounds on `Partition` a declared ordering's leading column is checked against.
25#[derive(Clone, Copy, Debug)]
26pub enum OrderingBounds {
27    /// `min_event_time()` / `max_event_time()` -- `Option`, absent for empty partitions.
28    EventTime,
29    /// `begin_insert_time()` / `end_insert_time()` -- always present.
30    InsertTime,
31}
32
33/// Reads the pair of bounds a declared ordering's leading column is checked against, per
34/// `OrderingBounds`. `InsertTime` bounds are always present; `EventTime` bounds are `None` for
35/// empty partitions (callers are expected to have already filtered those out).
36fn partition_bounds(
37    p: &Partition,
38    bounds: OrderingBounds,
39) -> Option<(DateTime<Utc>, DateTime<Utc>)> {
40    match bounds {
41        OrderingBounds::EventTime => p.min_event_time().zip(p.max_event_time()),
42        OrderingBounds::InsertTime => Some((p.begin_insert_time(), p.end_insert_time())),
43    }
44}

46/// Sorts the non-empty partitions by their leading-column bound ascending (tiebreak `file_path`)
47/// and verifies that adjacent partitions' ranges do not overlap. This makes the declared scan
48/// ordering self-contained: the file group is guaranteed to concatenate in globally-sorted order,
49/// independent of the order the partition cache returned.
50///
51/// Returns an error if any adjacent pair overlaps: the declared ordering cannot be honored, so we
52/// fail loudly instead of silently emitting a mis-ordered scan. For `OrderingBounds::EventTime` the
53/// causes are: an insert-time inversion straddling a JIT segment boundary; TSC-frequency estimation
54/// drift across materialization epochs (for `tsc_frequency == 0` processes whose blocks were
55/// materialized under different clock estimates); and, for `micromegas_tracing`-produced streams
56/// only, block-boundary tick overlap (a partition's event bounds come from block ticks, and that
57/// producer's consecutive blocks overlap by the cost of the buffer swap -- the Unreal producer
58/// stamps one timestamp for both sides, so its blocks touch exactly). See the ordering-invariant
59/// notes on `View::get_scan_output_ordering`. The first two are fixed by retiring the affected
60/// stream's partitions so they rebuild with a single, consistent converter. For
61/// `OrderingBounds::InsertTime` an overlap indicates a genuine partitioning bug -- input
62/// partitions are expected to be non-overlapping in insert_time by construction.
63fn sort_and_check_non_overlapping(
64    mut partitions: Vec<&Partition>,
65    bounds: OrderingBounds,
66) -> datafusion::error::Result<Vec<&Partition>> {
67    partitions.sort_by(|a, b| {
68        partition_bounds(a, bounds)
69            .map(|(begin, _)| begin)
70            .cmp(&partition_bounds(b, bounds).map(|(begin, _)| begin))
71            .then_with(|| a.file_path.cmp(&b.file_path))
72    });
73    for pair in partitions.windows(2) {
74        let prev = pair[0];
75        let next = pair[1];
76        if let (Some((_, prev_max)), Some((next_min, _))) = (
77            partition_bounds(prev, bounds),
78            partition_bounds(next, bounds),
79        ) && prev_max > next_min
80        {
81            return Err(datafusion::error::DataFusionError::Internal(format!(
82                "declared scan ordering violated: partition {:?} (range ending {prev_max}) overlaps partition {:?} (range starting {next_min}). \
83                 For event-time ordering the usual causes are an insert-time inversion straddling a JIT segment boundary, or -- for tsc_frequency == 0 processes -- TSC-frequency \
84                 re-estimation drift across materialization epochs spanning a clock adjustment. Both are fixed by retiring the affected stream's partitions so they rebuild with a \
85                 single, consistent time converter. See the rustdoc on sort_and_check_non_overlapping (partitioned_execution_plan.rs) and the ordering-invariant notes on \
86                 View::get_scan_output_ordering in view.rs for the full cause list.",
87                prev.file_path, next.file_path
88            )));
89        }
90    }
91    Ok(partitions)
92}

94/// Attaches the leading `output_ordering` column's min/max statistics to a `PartitionedFile`,
95/// using `Precision::Inexact` since the bounds read from `Partition` (per `OrderingBounds`) are
96/// not necessarily the column's exact min/max. DataFusion's multi-file-group ordering validation
97/// (`is_ordering_valid_for_file_groups`) requires these statistics to be present -- without them
98/// the declared ordering is silently dropped for any file group with more than one file.
99fn attach_ordering_statistics(
100    mut file: PartitionedFile,
101    schema: &SchemaRef,
102    leading_column: &ScanSortColumn,
103    partition: &Partition,
104    bounds: OrderingBounds,
105) -> datafusion::error::Result<PartitionedFile> {
106    let mut stats = Statistics::new_unknown(schema);
107    if let Some((min_time, max_time)) = partition_bounds(partition, bounds) {
108        let idx = schema.index_of(&leading_column.column)?;
109        stats.column_statistics[idx] = ColumnStatistics::new_unknown()
110            .with_min_value(Precision::Inexact(datetime_to_scalar(min_time)))
111            .with_max_value(Precision::Inexact(datetime_to_scalar(max_time)));
112    }
113    file = file.with_statistics(Arc::new(stats));
114    Ok(file)
115}

117/// How a partition scan's declared output ordering is realized.
118#[derive(Clone, Debug)]
119pub enum ScanOrdering {
120    /// No declared ordering (today's default).
121    Unordered,
122    /// All files form one sequential file group that concatenates in globally-sorted order.
123    /// Requires non-overlapping leading-column bounds (checked against `bounds`).
124    Concatenated {
125        columns: Vec<ScanSortColumn>,
126        bounds: OrderingBounds,
127    },
128    /// Each file is internally sorted by `columns`; files may overlap arbitrarily. The scan
129    /// yields one ordered plan partition per file, for a downstream `SortPreservingMergeExec`.
130    /// `columns` should be non-empty -- `SqlBatchView::with_merge_sort_order` rejects an empty
131    /// list at construction, but this enum and `View::get_scan_output_ordering` are public, so
132    /// that is not the only construction path. An empty `columns` is still safe: it never
133    /// certifies (see `Partition::certifies_sort_order`), so `make_partitioned_execution_plan`
134    /// degrades it to `Unordered` for any non-empty partition rather than planning and recording a
135    /// vacuous ordering.
136    PerFile { columns: Vec<ScanSortColumn> },
137}

139/// Builds the `LexOrdering` declaring the already-satisfied output ordering of the scan, matching
140/// DataFusion's default `ORDER BY` semantics (ASC NULLS LAST unless `descending`).
141pub fn make_lex_ordering(
142    schema: &SchemaRef,
143    output_ordering: &[ScanSortColumn],
144) -> datafusion::error::Result<Option<LexOrdering>> {
145    let sort_exprs = output_ordering
146        .iter()
147        .map(|c| {
148            let col =
149                datafusion::physical_expr::expressions::Column::new_with_schema(&c.column, schema)?;
150            Ok(PhysicalSortExpr::new(
151                Arc::new(col),
152                SortOptions {
153                    descending: c.descending,
154                    // Match DataFusion's default ORDER BY semantics: ASC NULLS LAST, DESC NULLS
155                    // FIRST. Hardcoding `false` here would declare `DESC NULLS LAST`, which fails
156                    // to satisfy a descending query's `DESC NULLS FIRST` requirement and silently
157                    // keeps a redundant Sort.
158                    nulls_first: c.descending,
159                },
160            ))
161        })
162        .collect::<datafusion::error::Result<Vec<_>>>()?;
163    Ok(LexOrdering::new(sort_exprs))
164}

166/// Bails unless `plan` is a single-partition physical plan. `subject` names the query for the
167/// error message (e.g. `format!("merge query {:?}", query)` or `format!("extract query for
168/// {view}")`); `reason` supplies the full, call-site-specific trailing sentence(s) explaining what
169/// a non-single-partition plan means here -- executing such a plan would coalesce partitions and
170/// silently destroy a declared ordering before it is safe to record or execute. Shared by the
171/// three query-execution paths that must verify this before executing (Design §2/§3 of
172/// `tasks/completed/1392_kway_merge_sorted_partitions_plan.md`): `QueryMerger::execute_concatenated_merge`,
173/// `QueryMerger::execute_per_file_merge`, and `SqlPartitionSpec::execute_extract_query`.
174pub fn assert_single_partition(
175    plan: &Arc<dyn ExecutionPlan>,
176    subject: &str,
177    insert_range: TimeRange,
178    reason: &str,
179) -> anyhow::Result<()> {
180    let partition_count = plan.properties().output_partitioning().partition_count();
181    if partition_count != 1 {
182        anyhow::bail!(
183            "{subject} (insert_range=[{}, {}]) produced a {partition_count}-partition physical \
184             plan; {reason}",
185            insert_range.begin.to_rfc3339(),
186            insert_range.end.to_rfc3339()
187        );
188    }
189    Ok(())
190}

192/// Bails unless `plan`'s output ordering satisfies the declared `columns`, defensively
193/// re-verifying that a `sort_order` guarantee about to be recorded on a fresh partition is
194/// truthful. `label` is a short noun phrase used in the intermediate error-context messages (e.g.
195/// `"per-file merge"` or `"extract-query"`); `subject` names the query for the bail message;
196/// `reason` supplies the full, call-site-specific trailing text of the bail message (what was
197/// declared, and any guidance for diagnosing a mismatch). Shared by the two paths that record a
198/// `sort_order` guarantee (Design §2/§3): `QueryMerger::execute_per_file_merge` and
199/// `SqlPartitionSpec::execute_extract_query`. `QueryMerger::execute_concatenated_merge` does not
200/// call this -- its ordering is a structural property of the sorted, non-overlapping file group
201/// rather than a query-plan sort DataFusion could get wrong.
202pub fn assert_ordering_satisfied(
203    plan: &Arc<dyn ExecutionPlan>,
204    columns: &[ScanSortColumn],
205    label: &str,
206    subject: &str,
207    insert_range: TimeRange,
208    reason: &str,
209) -> anyhow::Result<()> {
210    let lex = make_lex_ordering(&plan.schema(), columns)
211        .with_context(|| format!("building the declared {label} ordering"))?
212        .with_context(|| format!("declared {label} columns must be non-empty"))?;
213    let ordering_satisfied = plan
214        .properties()
215        .equivalence_properties()
216        .ordering_satisfy(lex)
217        .with_context(|| format!("checking {label} plan output ordering"))?;
218    if !ordering_satisfied {
219        anyhow::bail!(
220            "{subject} (insert_range=[{}, {}]) produced a physical plan whose output ordering \
221             does not satisfy {reason}",
222            insert_range.begin.to_rfc3339(),
223            insert_range.end.to_rfc3339()
224        );
225    }
226    Ok(())
227}

229/// Creates a partitioned execution plan for scanning Parquet files.
230///
231/// `scan_ordering` declares how the scan's already-satisfied ordering (see
232/// `View::get_scan_output_ordering`), if any, is realized:
233/// - `Unordered`: behavior is unchanged from before this parameter existed.
234/// - `Concatenated { columns, bounds }`: the file group is sorted by the leading column's bound
235///   (read per `bounds`) and checked for non-overlap (erroring if violated), per-file min/max
236///   statistics are attached so DataFusion accepts the declared ordering, and the ordering is
237///   attached to the resulting `FileScanConfig` so `EnforceSorting` can elide a redundant `Sort`
238///   node.
239/// - `PerFile { columns }`: gated by `Partition::certifies_sort_order` -- if every non-empty
240///   partition's recorded `sort_order` certifies `columns`, each non-empty partition becomes its
241///   own single-file group (no overlap check, no per-file statistics -- DataFusion's multi-file-group
242///   ordering validation only needs those to prove cross-file order *within* a group, and
243///   single-file groups pass trivially), all declaring the same `columns` ordering for a
244///   downstream `SortPreservingMergeExec`. If any non-empty partition fails to certify, this
245///   degrades to the same plan shape as `Unordered`.
246#[span_fn]
247#[expect(clippy::too_many_arguments)]
248pub fn make_partitioned_execution_plan(
249    schema: SchemaRef,
250    reader_factory: Arc<ReaderFactory>,
251    state: &dyn Session,
252    projection: Option<&Vec<usize>>,
253    filters: &[Expr],
254    limit: Option<usize>,
255    partitions: Arc<Vec<Partition>>,
256    scan_ordering: &ScanOrdering,
257) -> datafusion::error::Result<Arc<dyn ExecutionPlan>> {
258    let predicate = filters_to_predicate(schema.clone(), state, filters)?;
259
260    let non_empty_partitions: Vec<&Partition> =
261        partitions.iter().filter(|p| !p.is_empty()).collect();
262
263    // A PerFile declaration that any non-empty partition does not certify degrades to Unordered:
264    // both consumers (user-query scans and merge scans) fall back to sorting rather than trust a
265    // stale or uncertified guarantee.
266    let scan_ordering = match scan_ordering {
267        ScanOrdering::PerFile { columns }
268            if !non_empty_partitions
269                .iter()
270                .all(|p| p.certifies_sort_order(columns)) =>
271        {
272            &ScanOrdering::Unordered
273        }
274        other => other,
275    };
276
277    match scan_ordering {
278        ScanOrdering::Unordered => build_unordered_or_concatenated_plan(
279            schema,
280            reader_factory,
281            predicate,
282            projection,
283            limit,
284            non_empty_partitions,
285            None,
286        ),
287        ScanOrdering::Concatenated { columns, bounds } => {
288            let non_empty_partitions =
289                sort_and_check_non_overlapping(non_empty_partitions, *bounds)?;
290            build_unordered_or_concatenated_plan(
291                schema,
292                reader_factory,
293                predicate,
294                projection,
295                limit,
296                non_empty_partitions,
297                Some((columns.as_slice(), *bounds)),
298            )
299        }
300        ScanOrdering::PerFile { columns } => build_per_file_plan(
301            schema,
302            reader_factory,
303            predicate,
304            projection,
305            limit,
306            non_empty_partitions,
307            columns,
308        ),
309    }
310}

312/// Builds the `Unordered` (`ordering: None`) and `Concatenated` (`ordering: Some((columns,
313/// bounds))`) scan shapes: every non-empty partition in one sequential file group.
314fn build_unordered_or_concatenated_plan(
315    schema: SchemaRef,
316    reader_factory: Arc<ReaderFactory>,
317    predicate: Arc<dyn datafusion::physical_expr::PhysicalExpr>,
318    projection: Option<&Vec<usize>>,
319    limit: Option<usize>,
320    non_empty_partitions: Vec<&Partition>,
321    ordering: Option<(&[ScanSortColumn], OrderingBounds)>,
322) -> datafusion::error::Result<Arc<dyn ExecutionPlan>> {
323    let mut file_group = vec![];
324    for part in &non_empty_partitions {
325        let file_path = part.file_path.as_ref().ok_or_else(|| {
326            datafusion::error::DataFusionError::Internal(format!(
327                "non-empty partition has no file_path: num_rows={}",
328                part.num_rows
329            ))
330        })?;
331        let mut pf = PartitionedFile::new(file_path, part.file_size as u64);
332        if let Some((output_ordering, ordering_bounds)) = ordering
333            && let Some(leading_column) = output_ordering.first()
334        {
335            pf = attach_ordering_statistics(pf, &schema, leading_column, part, ordering_bounds)?;
336        }
337        file_group.push(pf);
338    }
339
340    // If all partitions are empty, return EmptyExec with projected schema
341    if file_group.is_empty() {
342        return empty_exec(schema, projection);
343    }
344
345    let object_store_url = ObjectStoreUrl::parse("obj://lakehouse/").unwrap();
346    let source = Arc::new(
347        ParquetSource::new(schema.clone())
348            .with_predicate(predicate)
349            .with_parquet_file_reader_factory(reader_factory),
350    );
351    let mut builder = FileScanConfigBuilder::new(object_store_url, source)
352        .with_limit(limit)
353        .with_projection_indices(projection.cloned())?
354        .with_file_groups(vec![file_group.into()]);
355
356    if let Some((output_ordering, _)) = ordering
357        && let Some(lex) = make_lex_ordering(&schema, output_ordering)?
358    {
359        builder = builder.with_output_ordering(vec![lex]);
360    }
361    let file_scan_config = builder.build();
362    Ok(Arc::new(DataSourceExec::new(Arc::new(file_scan_config))))
363}

365/// Builds the `PerFile` scan shape: one single-file file group per non-empty partition, all
366/// declaring the same `columns` ordering. No overlap check and no per-file statistics -- see the
367/// module-level rustdoc on `ScanOrdering::PerFile`.
368fn build_per_file_plan(
369    schema: SchemaRef,
370    reader_factory: Arc<ReaderFactory>,
371    predicate: Arc<dyn datafusion::physical_expr::PhysicalExpr>,
372    projection: Option<&Vec<usize>>,
373    limit: Option<usize>,
374    non_empty_partitions: Vec<&Partition>,
375    columns: &[ScanSortColumn],
376) -> datafusion::error::Result<Arc<dyn ExecutionPlan>> {
377    let mut file_groups = vec![];
378    for part in &non_empty_partitions {
379        let file_path = part.file_path.as_ref().ok_or_else(|| {
380            datafusion::error::DataFusionError::Internal(format!(
381                "non-empty partition has no file_path: num_rows={}",
382                part.num_rows
383            ))
384        })?;
385        let pf = PartitionedFile::new(file_path, part.file_size as u64);
386        file_groups.push(vec![pf].into());
387    }
388
389    if file_groups.is_empty() {
390        return empty_exec(schema, projection);
391    }
392
393    let object_store_url = ObjectStoreUrl::parse("obj://lakehouse/").unwrap();
394    let source = Arc::new(
395        ParquetSource::new(schema.clone())
396            .with_predicate(predicate)
397            .with_parquet_file_reader_factory(reader_factory),
398    );
399    let mut builder = FileScanConfigBuilder::new(object_store_url, source)
400        .with_limit(limit)
401        .with_projection_indices(projection.cloned())?
402        .with_file_groups(file_groups);
403
404    if let Some(lex) = make_lex_ordering(&schema, columns)? {
405        builder = builder.with_output_ordering(vec![lex]);
406    }
407    let file_scan_config = builder.build();
408    Ok(Arc::new(DataSourceExec::new(Arc::new(file_scan_config))))
409}

411fn empty_exec(
412    schema: SchemaRef,
413    projection: Option<&Vec<usize>>,
414) -> datafusion::error::Result<Arc<dyn ExecutionPlan>> {
415    use datafusion::physical_plan::empty::EmptyExec;
416    let projected_schema = if let Some(projection) = projection {
417        Arc::new(schema.project(projection)?)
418    } else {
419        schema
420    };
421    Ok(Arc::new(EmptyExec::new(projected_schema)))
