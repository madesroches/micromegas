24/// Which pair of bounds on `Partition` a declared ordering's leading column is checked against.
25#[derive(Clone, Copy, Debug)]
26pub enum OrderingBounds {
27    /// `min_event_time()` / `max_event_time()` -- `Option`, absent for empty partitions.
28    EventTime,
29    /// `begin_insert_time()` / `end_insert_time()` -- always present.
30    InsertTime,
31}
32
33/// Reads the pair of bounds a declared ordering's leading column is checked against, per
34/// `OrderingBounds`. `InsertTime` bounds are always present; `EventTime` bounds are `None` for
35/// empty partitions (callers are expected to have already filtered those out).
36///
37/// The `EventTime` upper bound prefers `max_sort_key_time` (the partition's true recorded max of
38/// the leading sort column, e.g. `begin` for `thread_spans`) and falls back to `max_event_time`
39/// (the max span *end*, a merely conservative stand-in) when it is `None` -- true for every
40/// partition written before that column existed, and for any view that never declares a
41/// `Concatenated` event-time ordering. This single change point upgrades all three consumers of
42/// `partition_bounds` coherently: the sort key stays `min_event_time`; the non-overlap check below
43/// compares the previous partition's *true* max `begin` against the next's block-derived min
44/// (which can never strictly overlap for cuts at block boundaries of a producer using a shared
45/// flush timestamp); and `attach_ordering_statistics` attaches a tighter, exact `begin` max
46/// statistic. The fallback preserves bit-for-bit behavior for legacy partitions.
47fn partition_bounds(
48    p: &Partition,
49    bounds: OrderingBounds,
50) -> Option<(DateTime<Utc>, DateTime<Utc>)> {
51    match bounds {
52        OrderingBounds::EventTime => p
53            .min_event_time()
54            .zip(p.max_sort_key_time().or(p.max_event_time())),
55        OrderingBounds::InsertTime => Some((p.begin_insert_time(), p.end_insert_time())),
56    }
57}

59/// Sorts the non-empty partitions by their leading-column bound ascending (tiebreak `file_path`)
60/// and verifies that adjacent partitions' ranges do not overlap. This makes the declared scan
61/// ordering self-contained: the file group is guaranteed to concatenate in globally-sorted order,
62/// independent of the order the partition cache returned.
63///
64/// Returns an error if any adjacent pair overlaps: the declared ordering cannot be honored, so we
65/// fail loudly instead of silently emitting a mis-ordered scan. For `OrderingBounds::EventTime`,
66/// block-boundary tick overlap no longer trips this check for partitions carrying
67/// `max_sort_key_time` (`partition_bounds` reads it in preference to `max_event_time`): the
68/// producer now stamps one shared timestamp per flush (matching the Unreal producer), so
69/// consecutive blocks touch exactly, and the recorded bound reflects that. Legacy partitions
70/// written before that column existed fall back to the old, looser `max_event_time` bound and so
71/// remain a residual until they are rebuilt -- which happens automatically on their next query,
72/// since `ThreadSpansView::SCHEMA_VERSION`'s bump makes every pre-existing partition stale by
73/// schema hash; this is a self-healing residual, not an admin-retire dependency. The two remaining
74/// residual causes are: an insert-time inversion straddling a JIT segment boundary (a genuine
75/// row-level overlap, correctly rejected); and TSC-frequency estimation drift across
76/// materialization epochs (for `tsc_frequency == 0` processes whose blocks were materialized under
77/// different clock estimates) -- fixed the same way, by retiring the affected stream's partitions
78/// so they rebuild with a single, consistent converter. See the ordering-invariant notes on
79/// `View::get_scan_output_ordering`. For `OrderingBounds::InsertTime` an overlap indicates a
80/// genuine partitioning bug -- input partitions are expected to be non-overlapping in insert_time
81/// by construction.
82fn sort_and_check_non_overlapping(
83    mut partitions: Vec<&Partition>,
84    bounds: OrderingBounds,
85) -> datafusion::error::Result<Vec<&Partition>> {
86    partitions.sort_by(|a, b| {
87        partition_bounds(a, bounds)
88            .map(|(begin, _)| begin)
89            .cmp(&partition_bounds(b, bounds).map(|(begin, _)| begin))
90            .then_with(|| a.file_path.cmp(&b.file_path))
91    });
92    for pair in partitions.windows(2) {
93        let prev = pair[0];
94        let next = pair[1];
95        if let (Some((_, prev_max)), Some((next_min, _))) = (
96            partition_bounds(prev, bounds),
97            partition_bounds(next, bounds),
98        ) && prev_max > next_min
99        {
100            return Err(datafusion::error::DataFusionError::Internal(format!(
101                "declared scan ordering violated: partition {:?} (range ending {prev_max}) overlaps partition {:?} (range starting {next_min}). \
102                 If either partition predates max_sort_key_time (schema v8), this heals itself on its next query: a schema-hash bump makes it stale and it \
103                 rebuilds automatically, carrying the exact bound, with no admin action needed. Otherwise, for event-time ordering the remaining causes are an \
104                 insert-time inversion straddling a JIT segment boundary, or -- for tsc_frequency == 0 processes -- TSC-frequency re-estimation drift across \
105                 materialization epochs spanning a clock adjustment; both are fixed by retiring the affected stream's partitions so they rebuild with a single, \
106                 consistent time converter. See the rustdoc on sort_and_check_non_overlapping (partitioned_execution_plan.rs) and the ordering-invariant notes on \
107                 View::get_scan_output_ordering in view.rs for the full cause list.",
108                prev.file_path, next.file_path
109            )));
110        }
111    }
112    Ok(partitions)
113}

115/// Attaches the leading `output_ordering` column's min/max statistics to a `PartitionedFile`,
116/// using `Precision::Inexact` since the bounds read from `Partition` (per `OrderingBounds`) are
117/// not necessarily the column's exact min/max -- though for `EventTime` bounds when
118/// `max_sort_key_time` is recorded, the attached max happens to be exact; `Precision::Inexact` is
119/// still correct to declare, since an exact statistic is a legal special case of an inexact one.
120/// DataFusion's multi-file-group ordering validation (`is_ordering_valid_for_file_groups`)
121/// requires these statistics to be present -- without them the declared ordering is silently
122/// dropped for any file group with more than one file.
123fn attach_ordering_statistics(
124    mut file: PartitionedFile,
125    schema: &SchemaRef,
126    leading_column: &ScanSortColumn,
127    partition: &Partition,
128    bounds: OrderingBounds,
129) -> datafusion::error::Result<PartitionedFile> {
130    let mut stats = Statistics::new_unknown(schema);
131    if let Some((min_time, max_time)) = partition_bounds(partition, bounds) {
132        let idx = schema.index_of(&leading_column.column)?;
133        stats.column_statistics[idx] = ColumnStatistics::new_unknown()
134            .with_min_value(Precision::Inexact(datetime_to_scalar(min_time)))
135            .with_max_value(Precision::Inexact(datetime_to_scalar(max_time)));
136    }
137    file = file.with_statistics(Arc::new(stats));
138    Ok(file)
139}

141/// How a partition scan's declared output ordering is realized.
142///
143/// Two scan shapes exist: a single sequential file group spanning every input partition
144/// (`Unordered`, `Concatenated`) and one file group per input partition (`PerFile`, scanned by k
145/// readers for a downstream `SortPreservingMergeExec`). `Unordered` and `Concatenated` build and
146/// execute the identical single-file-group scan -- they differ only in whether the resulting order
147/// is declared to DataFusion, not in scan shape or execution strategy (see
148/// `ScanOrdering::declares_concatenated_ordering`).
149#[derive(Clone, Debug)]
150pub enum ScanOrdering {
151    /// No declared ordering (today's default).
152    Unordered,
153    /// All files form one sequential file group that concatenates in globally-sorted order.
154    /// Requires non-overlapping leading-column bounds (checked against `bounds`).
155    Concatenated {
156        columns: Vec<ScanSortColumn>,
157        bounds: OrderingBounds,
158    },
159    /// Each file is internally sorted by `columns`; files may overlap arbitrarily. The scan
160    /// yields one ordered plan partition per file, for a downstream `SortPreservingMergeExec`.
161    /// `columns` should be non-empty -- `SqlBatchView::with_merge_sort_order` rejects an empty
162    /// list at construction, but this enum and `View::get_scan_output_ordering` are public, so
163    /// that is not the only construction path. An empty `columns` is still safe: it never
164    /// certifies (see `Partition::certifies_sort_order`), so `make_partitioned_execution_plan`
165    /// degrades it to `Unordered` for any non-empty partition rather than planning and recording a
166    /// vacuous ordering.
167    PerFile { columns: Vec<ScanSortColumn> },
168}
169
170impl ScanOrdering {
171    /// True when this ordering declares a concatenating scan's global order (`Concatenated`).
172    /// `PerFile` returns `false` because a per-file ordering only becomes a global one through a
173    /// downstream merge -- it is the other strategy, not an undeclared version of this one. A
174    /// bool, not the columns themselves: nothing on the concatenating path inspects the declared
175    /// columns, only whether an ordering was declared at all (see
176    /// `QueryMerger::execute_concatenated_merge` in `merge.rs`).
177    pub fn declares_concatenated_ordering(&self) -> bool {
178        matches!(self, ScanOrdering::Concatenated { .. })
179    }
180}

182/// Builds the `LexOrdering` declaring the already-satisfied output ordering of the scan, matching
183/// DataFusion's default `ORDER BY` semantics (ASC NULLS LAST unless `descending`).
184pub fn make_lex_ordering(
185    schema: &SchemaRef,
186    output_ordering: &[ScanSortColumn],
187) -> datafusion::error::Result<Option<LexOrdering>> {
188    let sort_exprs = output_ordering
189        .iter()
190        .map(|c| {
191            let col =
192                datafusion::physical_expr::expressions::Column::new_with_schema(&c.column, schema)?;
193            Ok(PhysicalSortExpr::new(
194                Arc::new(col),
195                SortOptions {
196                    descending: c.descending,
197                    // Match DataFusion's default ORDER BY semantics: ASC NULLS LAST, DESC NULLS
198                    // FIRST. Hardcoding `false` here would declare `DESC NULLS LAST`, which fails
199                    // to satisfy a descending query's `DESC NULLS FIRST` requirement and silently
200                    // keeps a redundant Sort.
201                    nulls_first: c.descending,
202                },
203            ))
204        })
205        .collect::<datafusion::error::Result<Vec<_>>>()?;
206    Ok(LexOrdering::new(sort_exprs))
207}

209/// Bails unless `plan` is a single-partition physical plan. `subject` names the query for the
210/// error message (e.g. `format!("merge query {:?}", query)` or `format!("extract query for
211/// {view}")`); `reason` supplies the full, call-site-specific trailing sentence(s) explaining what
212/// a non-single-partition plan means here -- executing such a plan would coalesce partitions and
213/// silently destroy a declared ordering before it is safe to record or execute. Shared by the
214/// query-execution paths that must verify this before executing:
215/// `QueryMerger::execute_concatenated_merge` (only when its ordering is declared),
216/// `QueryMerger::execute_sorted_merge`, and `SqlPartitionSpec::execute_extract_query`.
217pub fn assert_single_partition(
218    plan: &Arc<dyn ExecutionPlan>,
219    subject: &str,
220    insert_range: TimeRange,
221    reason: &str,
222) -> anyhow::Result<()> {
223    let partition_count = plan.properties().output_partitioning().partition_count();
224    if partition_count != 1 {
225        anyhow::bail!(
226            "{subject} (insert_range=[{}, {}]) produced a {partition_count}-partition physical \
227             plan; {reason}",
228            insert_range.begin.to_rfc3339(),
229            insert_range.end.to_rfc3339()
230        );
231    }
232    Ok(())
233}

235/// Bails unless `plan`'s output ordering satisfies the declared `columns`, defensively
236/// re-verifying that a `sort_order` guarantee about to be recorded on a fresh partition is
237/// truthful. `label` is a short noun phrase used in the intermediate error-context messages (e.g.
238/// `"per-file merge"` or `"extract-query"`); `subject` names the query for the bail message;
239/// `reason` supplies the full, call-site-specific trailing text of the bail message (what was
240/// declared, and any guidance for diagnosing a mismatch). Shared by the two paths that record a
241/// `sort_order` guarantee: `QueryMerger::execute_sorted_merge` and
242/// `SqlPartitionSpec::execute_extract_query`. `QueryMerger::execute_concatenated_merge` does not
243/// call this -- its ordering is a structural property of the sorted, non-overlapping file group
244/// rather than a query-plan sort DataFusion could get wrong.
245pub fn assert_ordering_satisfied(
246    plan: &Arc<dyn ExecutionPlan>,
247    columns: &[ScanSortColumn],
248    label: &str,
249    subject: &str,
250    insert_range: TimeRange,
251    reason: &str,
252) -> anyhow::Result<()> {
253    let lex = make_lex_ordering(&plan.schema(), columns)
254        .with_context(|| format!("building the declared {label} ordering"))?
255        .with_context(|| format!("declared {label} columns must be non-empty"))?;
256    let ordering_satisfied = plan
257        .properties()
258        .equivalence_properties()
259        .ordering_satisfy(lex)
260        .with_context(|| format!("checking {label} plan output ordering"))?;
261    if !ordering_satisfied {
262        anyhow::bail!(
263            "{subject} (insert_range=[{}, {}]) produced a physical plan whose output ordering \
264             does not satisfy {reason}",
265            insert_range.begin.to_rfc3339(),
266            insert_range.end.to_rfc3339()
267        );
268    }
269    Ok(())
270}

272/// Creates a partitioned execution plan for scanning Parquet files.
273///
274/// `scan_ordering` declares how the scan's already-satisfied ordering (see
275/// `View::get_scan_output_ordering`), if any, is realized:
276/// - `Unordered`: no ordering is declared to DataFusion.
277/// - `Concatenated { columns, bounds }`: the file group is sorted by the leading column's bound
278///   (read per `bounds`) and checked for non-overlap (erroring if violated), per-file min/max
279///   statistics are attached so DataFusion accepts the declared ordering, and the ordering is
280///   attached to the resulting `FileScanConfig` so `EnforceSorting` can elide a redundant `Sort`
281///   node.
282/// - `PerFile { columns }`: gated by `Partition::certifies_sort_order` -- if every non-empty
283///   partition's recorded `sort_order` certifies `columns`, each non-empty partition becomes its
284///   own single-file group (no overlap check, no per-file statistics -- DataFusion's multi-file-group
285///   ordering validation only needs those to prove cross-file order *within* a group, and
286///   single-file groups pass trivially), all declaring the same `columns` ordering for a
287///   downstream `SortPreservingMergeExec`. If any non-empty partition fails to certify, this
288///   degrades to the same plan shape as `Unordered`.
289#[span_fn]
290#[expect(clippy::too_many_arguments)]
291pub fn make_partitioned_execution_plan(
292    schema: SchemaRef,
293    reader_factory: Arc<ReaderFactory>,
294    state: &dyn Session,
295    projection: Option<&Vec<usize>>,
296    filters: &[Expr],
297    limit: Option<usize>,
298    partitions: Arc<Vec<Partition>>,
299    scan_ordering: &ScanOrdering,
300) -> datafusion::error::Result<Arc<dyn ExecutionPlan>> {
301    let predicate = filters_to_predicate(schema.clone(), state, filters)?;
302
303    let non_empty_partitions: Vec<&Partition> =
304        partitions.iter().filter(|p| !p.is_empty()).collect();
305
306    // A PerFile declaration that any non-empty partition does not certify degrades to Unordered:
307    // both consumers (user-query scans and merge scans) fall back to sorting rather than trust a
308    // stale or uncertified guarantee.
309    let scan_ordering = match scan_ordering {
310        ScanOrdering::PerFile { columns }
311            if !non_empty_partitions
312                .iter()
313                .all(|p| p.certifies_sort_order(columns)) =>
314        {
315            &ScanOrdering::Unordered
316        }
317        other => other,
318    };
319
320    match scan_ordering {
321        ScanOrdering::Unordered => build_unordered_or_concatenated_plan(
322            schema,
323            reader_factory,
324            predicate,
325            projection,
326            limit,
327            non_empty_partitions,
328            None,
329        ),
330        ScanOrdering::Concatenated { columns, bounds } => {
331            let non_empty_partitions =
332                sort_and_check_non_overlapping(non_empty_partitions, *bounds)?;
333            build_unordered_or_concatenated_plan(
334                schema,
335                reader_factory,
336                predicate,
337                projection,
338                limit,
339                non_empty_partitions,
340                Some((columns.as_slice(), *bounds)),
341            )
342        }
343        ScanOrdering::PerFile { columns } => build_per_file_plan(
344            schema,
345            reader_factory,
346            predicate,
347            projection,
348            limit,
349            non_empty_partitions,
350            columns,
351        ),
352    }
353}

355/// Builds the `Unordered` (`ordering: None`) and `Concatenated` (`ordering: Some((columns,
356/// bounds))`) scan shapes: every non-empty partition in one sequential file group.
357fn build_unordered_or_concatenated_plan(
358    schema: SchemaRef,
359    reader_factory: Arc<ReaderFactory>,
360    predicate: Arc<dyn datafusion::physical_expr::PhysicalExpr>,
361    projection: Option<&Vec<usize>>,
362    limit: Option<usize>,
363    non_empty_partitions: Vec<&Partition>,
364    ordering: Option<(&[ScanSortColumn], OrderingBounds)>,
365) -> datafusion::error::Result<Arc<dyn ExecutionPlan>> {
366    let mut file_group = vec![];
367    for part in &non_empty_partitions {
368        let file_path = part.file_path.as_ref().ok_or_else(|| {
369            datafusion::error::DataFusionError::Internal(format!(
370                "non-empty partition has no file_path: num_rows={}",
371                part.num_rows
372            ))
373        })?;
374        let mut pf = PartitionedFile::new(file_path, part.file_size as u64);
375        if let Some((output_ordering, ordering_bounds)) = ordering
376            && let Some(leading_column) = output_ordering.first()
377        {
378            pf = attach_ordering_statistics(pf, &schema, leading_column, part, ordering_bounds)?;
379        }
380        file_group.push(pf);
381    }
382
383    // If all partitions are empty, return EmptyExec with projected schema
384    if file_group.is_empty() {
385        return empty_exec(schema, projection);
386    }
387
388    let object_store_url = ObjectStoreUrl::parse("obj://lakehouse/").unwrap();
389    let source = Arc::new(
390        ParquetSource::new(schema.clone())
391            .with_predicate(predicate)
392            .with_parquet_file_reader_factory(reader_factory),
393    );
394    let mut builder = FileScanConfigBuilder::new(object_store_url, source)
395        .with_limit(limit)
396        .with_projection_indices(projection.cloned())?
397        .with_file_groups(vec![file_group.into()]);
398
399    if let Some((output_ordering, _)) = ordering
400        && let Some(lex) = make_lex_ordering(&schema, output_ordering)?
401    {
402        builder = builder.with_output_ordering(vec![lex]);
403    }
404    let file_scan_config = builder.build();
405    Ok(Arc::new(DataSourceExec::new(Arc::new(file_scan_config))))
406}

408/// Builds the `PerFile` scan shape: one single-file file group per non-empty partition, all
409/// declaring the same `columns` ordering. No overlap check and no per-file statistics -- see the
410/// module-level rustdoc on `ScanOrdering::PerFile`.
411fn build_per_file_plan(
412    schema: SchemaRef,
413    reader_factory: Arc<ReaderFactory>,
414    predicate: Arc<dyn datafusion::physical_expr::PhysicalExpr>,
415    projection: Option<&Vec<usize>>,
416    limit: Option<usize>,
417    non_empty_partitions: Vec<&Partition>,
418    columns: &[ScanSortColumn],
419) -> datafusion::error::Result<Arc<dyn ExecutionPlan>> {
420    let mut file_groups = vec![];
421    for part in &non_empty_partitions {
422        let file_path = part.file_path.as_ref().ok_or_else(|| {
423            datafusion::error::DataFusionError::Internal(format!(
424                "non-empty partition has no file_path: num_rows={}",
425                part.num_rows
426            ))
427        })?;
428        let pf = PartitionedFile::new(file_path, part.file_size as u64);
429        file_groups.push(vec![pf].into());
430    }
431
432    if file_groups.is_empty() {
433        return empty_exec(schema, projection);
434    }
435
436    let object_store_url = ObjectStoreUrl::parse("obj://lakehouse/").unwrap();
437    let source = Arc::new(
438        ParquetSource::new(schema.clone())
439            .with_predicate(predicate)
440            .with_parquet_file_reader_factory(reader_factory),
441    );
442    let mut builder = FileScanConfigBuilder::new(object_store_url, source)
443        .with_limit(limit)
444        .with_projection_indices(projection.cloned())?
445        .with_file_groups(file_groups);
446
447    if let Some(lex) = make_lex_ordering(&schema, columns)? {
448        builder = builder.with_output_ordering(vec![lex]);
449    }
450    let file_scan_config = builder.build();
451    Ok(Arc::new(DataSourceExec::new(Arc::new(file_scan_config))))
452}

454fn empty_exec(
455    schema: SchemaRef,
456    projection: Option<&Vec<usize>>,
457) -> datafusion::error::Result<Arc<dyn ExecutionPlan>> {
458    use datafusion::physical_plan::empty::EmptyExec;
459    let projected_schema = if let Some(projection) = projection {
460        Arc::new(schema.project(projection)?)
461    } else {
462        schema
463    };
464    Ok(Arc::new(EmptyExec::new(projected_schema)))
