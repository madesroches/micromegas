23/// Builder for the default (env-driven) authentication provider stack, plus an
24/// optional DB-backed API key store and an optional local-group store.
25///
26/// The env factory (`provider()` / `provider_with_prefix()` below) is a builder
27/// so that adding the DB store — and later a policy — does not re-break the
28/// signature.
29pub struct ProviderBuilder {
30    prefix: String,
31    key_store: Option<(PgPool, ApiKeyTable)>,
32    group_store: Option<PgPool>,
33}
34
35impl ProviderBuilder {
36    /// Starts a builder scoped to `prefix` (e.g. `"MICROMEGAS_INGESTION"`, or `""`
37    /// for the unprefixed default), following the same `{prefix}_*`-with-fallback
38    /// convention as `provider_with_prefix`.
39    pub fn new(prefix: &str) -> Self {
40        Self {
41            prefix: prefix.to_string(),
42            key_store: None,
43            group_store: None,
44        }
45    }
46
47    /// Attaches a DB-backed key store bound to `table`, looked up through `pool`
48    /// (expected to be a [`crate::db_api_key::dedicated_key_store_pool`], not a
49    /// clone of the caller's lake pool).
50    pub fn with_db_key_store(mut self, pool: PgPool, table: ApiKeyTable) -> Self {
51        self.key_store = Some((pool, table));
52        self
53    }
54
55    /// Attaches a local-group store, looked up through `pool` (same dedicated-pool
56    /// expectation as [`Self::with_db_key_store`]). `compose()` wraps the finished chain in a
57    /// [`MembershipProvider`] when this is attached.
58    pub fn with_group_store(mut self, pool: PgPool) -> Self {
59        self.group_store = Some(pool);
60        self
61    }
62
63    /// Resolves the API-keys env var name for this builder's prefix, with
64    /// fallback to the unprefixed name.
65    fn api_keys_json(&self) -> Option<String> {
66        std::env::var(resolve_prefixed_var(&self.prefix, "API_KEYS")).ok()
67    }
68
69    /// Resolves the OIDC config env var name for this builder's prefix, with
70    /// fallback to the unprefixed name.
71    fn oidc_config_var(&self) -> String {
72        resolve_prefixed_var(&self.prefix, "OIDC_CONFIG")
73    }
74
75    /// Composes the chain and reports whether env keys or OIDC counted as
76    /// "configured". Shared by `build()` and `build_chain()`, which each
77    /// document the provider order and the DB-provider guarantee.
78    ///
79    /// Takes `&self` rather than `self` so `build()` can still reach the
80    /// attached key store's pool for its existence query afterwards.
81    async fn compose(&self) -> Result<(MultiAuthProvider, bool)> {
82        let oidc_config_var = self.oidc_config_var();
83        let api_keys_json = self.api_keys_json();
84
85        let mut multi = MultiAuthProvider::new();
86        let mut configured = false;
87
88        if let Some(keys_json) = api_keys_json {
89            let keyring = parse_key_ring(&keys_json)?;
90            info!("API key authentication enabled");
91            multi = multi.with_provider(Arc::new(ApiKeyAuthProvider::new(keyring)));
92            configured = true;
93        } else {
94            info!("API key auth not configured");
95        }
96
97        match OidcConfig::from_env_var(&oidc_config_var) {
98            Ok(config) => {
99                info!("Initializing OIDC authentication");
100                let oidc_provider = OidcAuthProvider::new(config).await?;
101                multi = multi.with_provider(Arc::new(oidc_provider));
102                configured = true;
103            }
104            Err(e) => {
105                info!("OIDC not configured ({e}) - OIDC auth disabled");
106            }
107        }
108
109        if let Some((pool, table)) = &self.key_store {
110            let db_config = DbApiKeyConfig::from_env_with_prefix(&self.prefix);
111            let db_provider = DbApiKeyAuthProvider::new(pool.clone(), *table, db_config);
112            multi = multi.with_provider(Arc::new(db_provider));
113        }
114
115        Ok((multi, configured))
116    }
117
118    /// Wraps `multi` in a [`MembershipProvider`] when a group store is attached, and runs one
119    /// eager `current()` at startup: on `Ok`, `warn!`s while `has_wildcard_admin()` is true
120    /// ("every authenticated caller is an admin; add a `user:` member to `admins` and remove
121    /// `*`"); on `Err`, `warn!`s and continues -- the first request will 503, and a split
122    /// deployment may legitimately start flight-sql before the migration runner is up, as with
123    /// the v5 key store.
124    async fn wrap_with_membership(&self, multi: MultiAuthProvider) -> Arc<dyn AuthProvider> {
125        let Some(pool) = &self.group_store else {
126            return Arc::new(multi) as Arc<dyn AuthProvider>;
127        };
128        let group_config = DbGroupsConfig::from_env_with_prefix(&self.prefix);
129        let groups = Arc::new(DbGroupsSource::new(
130            pool.clone(),
131            Duration::from_secs(group_config.cache_ttl_secs),
132        ));
133        match groups.current().await {
134            Ok(graph) => {
135                if graph.has_wildcard_admin() {
136                    warn!(
137                        "every authenticated caller is an admin; add a `user:` member to \
138                         `admins` and remove `*`"
139                    );
140                }
141            }
142            Err(e) => {
143                warn!("group store not yet reachable at startup, continuing: {e:#}");
144            }
145        }
146        Arc::new(MembershipProvider::new(
147            Arc::new(multi) as Arc<dyn AuthProvider>,
148            groups,
149        )) as Arc<dyn AuthProvider>
150    }
151
152    /// Builds the composed provider.
153    ///
154    /// Composes, in order: env `ApiKeyAuthProvider` → `OidcAuthProvider` →
155    /// `DbApiKeyAuthProvider`; `MultiAuthProvider` tries providers in order, so
156    /// putting the DB provider last means only tokens that are neither an env
157    /// key nor a valid JWT ever reach it. The DB provider is always pushed onto
158    /// the chain whenever a key store is attached, so a key minted into a
159    /// previously empty table authenticates on the very next request, with no
160    /// restart.
161    ///
162    /// **A DB key store with at least one live key counts as "auth configured";
163    /// an empty one does not.** When a key store is attached, this runs one cheap
164    /// startup existence query (`key_store_has_live_rows`) and treats a non-empty
165    /// result the same as env keys or OIDC being present. A failure of that query
166    /// (e.g. a missing relation because the schema has not reached migration v5
167    /// yet) is propagated as an `Err` from `build()` — unless env keys or OIDC
168    /// already configured auth, in which case the failure is only `warn!`-logged
169    /// and `has_live_rows` is treated as `false`, since the query's result would
170    /// be unused either way.
171    ///
172    /// The chain is discarded when nothing counted as configured; use
173    /// `build_chain()` to skip the guard.
174    pub async fn build(self) -> Result<Option<Arc<dyn AuthProvider>>> {
175        let (multi, mut configured) = self.compose().await?;
176
177        if let Some((pool, table)) = &self.key_store {
178            let has_live_rows = match key_store_has_live_rows(pool, *table).await {
179                Ok(has_live_rows) => has_live_rows,
180                Err(e) if !configured => {
181                    return Err(e.context(format!(
182                        "checking whether {} has any live key — has the schema reached migration v5? \
183                         (rust/ingestion/src/sql_migration.rs; the ingestion binary or monolith must run \
184                         the migration before flight-sql starts in a split deployment)",
185                        table.table_name()
186                    )));
187                }
188                Err(e) => {
189                    // Auth is already configured via another provider (env keys or
190                    // OIDC); this query's only purpose is deciding whether an
191                    // otherwise-unconfigured deployment counts as configured, so a
192                    // failure here (e.g. schema not yet at v5) must not abort
193                    // startup.
194                    warn!(
195                        "checking whether {} has any live key failed, ignoring (auth already configured): {e:#}",
196                        table.table_name()
197                    );
198                    false
199                }
200            };
201            if has_live_rows {
202                configured = true;
203            }
204        }
205
206        if !configured {
207            return Ok(None);
208        }
209
210        Ok(Some(self.wrap_with_membership(multi).await))
211    }
212
213    /// Composes the provider chain with **no** "is anything configured?" guard.
214    ///
215    /// Composes, in order: env `ApiKeyAuthProvider` → `OidcAuthProvider` →
216    /// `DbApiKeyAuthProvider`. The DB provider is always pushed onto the chain
217    /// whenever a key store is attached, so a key minted into a previously
218    /// empty table authenticates on the very next request, with no restart.
219    ///
220    /// Always returns the chain — never `None`, because there is no `Option` —
221    /// and never runs `key_store_has_live_rows`, so unlike `build()` it cannot
222    /// fail on a schema short of migration v5. This is the entry point for a
223    /// caller that wants the no-restart property above without inheriting
224    /// `build()`'s startup guard or its existence-query failure mode — e.g. an
225    /// embedder folding this chain into a larger `MultiAuthProvider` via
226    /// `FlightSqlServer::with_auth_provider`.
227    ///
228    /// When nothing at all is configured (no env keys, no OIDC, no key store)
229    /// the returned chain is an empty `MultiAuthProvider`, which rejects every
230    /// request — fail-closed, since the caller asked for no guard. Logs a
231    /// `warn!` in that case, since an empty chain is useless to any caller.
232    pub async fn build_chain(self) -> Result<Arc<dyn AuthProvider>> {
233        let (multi, _) = self.compose().await?;
234        if multi.is_empty() {
235            warn!("no auth provider configured: env keys, OIDC, and DB key store are all absent");
236        }
237        Ok(self.wrap_with_membership(multi).await)
238    }
239}

241/// Initializes the default authentication provider with API key and OIDC from environment.
242///
243/// Reads configuration from:
244/// - `MICROMEGAS_API_KEYS`: JSON array of API keys
245/// - `MICROMEGAS_OIDC_CONFIG`: OIDC configuration JSON
246///
247/// Admin-ness is no longer an env-driven list here: it lives in the `admins` group (schema v10),
248/// resolved by a [`crate::membership::MembershipProvider`] when a group store is attached via
249/// [`ProviderBuilder::with_group_store`].
250///
251/// Returns `Ok(Some(...))` if at least one provider is configured.
252/// Returns `Ok(None)` if no providers are configured (auth disabled).
253/// Returns `Err` on configuration errors.
254///
255/// Kept as a thin env-only wrapper around [`ProviderBuilder`] — this crate's
256/// documented, published entry point for a caller that wants API-key + OIDC
257/// composition without a DB pool.
258///
259/// # Example
260///
261/// ```rust,no_run
262/// use micromegas_auth::default_provider::provider;
263///
264/// # async fn example() -> anyhow::Result<()> {
265/// let auth_provider = provider().await?;
266/// if let Some(provider) = auth_provider {
267///     println!("Authentication enabled");
268/// } else {
269///     println!("No authentication configured");
270/// }
271/// # Ok(())
272/// # }
273/// ```
274pub async fn provider() -> Result<Option<Arc<dyn AuthProvider>>> {
275    provider_with_prefix("").await
276}

278/// Initializes auth providers using env vars scoped to a prefix.
279///
280/// For prefix `"MICROMEGAS_INGESTION"`:
281/// - API keys: tries `MICROMEGAS_INGESTION_API_KEYS`, falls back to `MICROMEGAS_API_KEYS`
282/// - OIDC:     tries `MICROMEGAS_INGESTION_OIDC_CONFIG`, falls back to `MICROMEGAS_OIDC_CONFIG`
283///
284/// With an empty prefix the behaviour is identical to [`provider`]. Kept as a
285/// thin env-only wrapper around [`ProviderBuilder`] — no DB key store is attached.
286pub async fn provider_with_prefix(prefix: &str) -> Result<Option<Arc<dyn AuthProvider>>> {
287    ProviderBuilder::new(prefix).build().await
