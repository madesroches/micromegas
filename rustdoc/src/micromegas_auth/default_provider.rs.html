1//! Default authentication provider initialization for Micromegas services.
2//!
3//! This module provides the standard way to initialize authentication with
4//! API key, OIDC, and (optionally) a DB-backed API key store from environment
5//! variables.

19/// Builder for the default (env-driven) authentication provider stack, plus an
20/// optional DB-backed API key store.
21///
22/// The env factory (`provider()` / `provider_with_prefix()` below) is a builder
23/// so that adding the DB store — and later a policy — does not re-break the
24/// signature.
25pub struct ProviderBuilder {
26    prefix: String,
27    key_store: Option<(PgPool, ApiKeyTable)>,
28}
29
30impl ProviderBuilder {
31    /// Starts a builder scoped to `prefix` (e.g. `"MICROMEGAS_INGESTION"`, or `""`
32    /// for the unprefixed default), following the same `{prefix}_*`-with-fallback
33    /// convention as `provider_with_prefix`.
34    pub fn new(prefix: &str) -> Self {
35        Self {
36            prefix: prefix.to_string(),
37            key_store: None,
38        }
39    }
40
41    /// Attaches a DB-backed key store bound to `table`, looked up through `pool`
42    /// (expected to be a [`crate::db_api_key::dedicated_key_store_pool`], not a
43    /// clone of the caller's lake pool).
44    pub fn with_db_key_store(mut self, pool: PgPool, table: ApiKeyTable) -> Self {
45        self.key_store = Some((pool, table));
46        self
47    }
48
49    /// Resolves the API-keys env var name for this builder's prefix, with
50    /// fallback to the unprefixed name.
51    fn api_keys_json(&self) -> Option<String> {
52        if self.prefix.is_empty() {
53            std::env::var("MICROMEGAS_API_KEYS").ok()
54        } else {
55            std::env::var(format!("{}_API_KEYS", self.prefix))
56                .or_else(|_| std::env::var("MICROMEGAS_API_KEYS"))
57                .ok()
58        }
59    }
60
61    /// Resolves the OIDC config env var name for this builder's prefix, with
62    /// fallback to the unprefixed name.
63    fn oidc_config_var(&self) -> String {
64        if self.prefix.is_empty() {
65            "MICROMEGAS_OIDC_CONFIG".to_string()
66        } else if std::env::var(format!("{}_OIDC_CONFIG", self.prefix)).is_ok() {
67            format!("{}_OIDC_CONFIG", self.prefix)
68        } else {
69            "MICROMEGAS_OIDC_CONFIG".to_string()
70        }
71    }
72
73    /// Resolves the admin-users env var name for this builder's prefix, with
74    /// fallback to the unprefixed name.
75    fn admin_var(&self) -> String {
76        if self.prefix.is_empty() {
77            "MICROMEGAS_ADMINS".to_string()
78        } else {
79            let prefixed = format!("{}_ADMINS", self.prefix);
80            if std::env::var(&prefixed).is_ok() {
81                prefixed
82            } else {
83                "MICROMEGAS_ADMINS".to_string()
84            }
85        }
86    }
87
88    /// Builds the composed provider.
89    ///
90    /// Composes, in this order: env `ApiKeyAuthProvider` (in-memory, cheapest,
91    /// preserves today's precedence) → `OidcAuthProvider` → `DbApiKeyAuthProvider`.
92    /// `MultiAuthProvider` tries providers in order, so putting the DB provider
93    /// last means only tokens that are neither an env key nor a valid JWT ever
94    /// reach it.
95    ///
96    /// **The DB provider is always pushed onto the chain whenever a key store is
97    /// attached** — registration never depends on the existence query below, so a
98    /// deployment that mints its first key through `POST /auth/api_keys` into a
99    /// previously empty table authenticates it on the very next request, with no
100    /// restart.
101    ///
102    /// **A DB key store with at least one live key counts as "auth configured";
103    /// an empty one does not.** When a key store is attached, this runs one cheap
104    /// startup existence query (`key_store_has_live_rows`) and treats a non-empty
105    /// result the same as env keys or OIDC being present. A failure of that query
106    /// (e.g. a missing relation because the schema has not reached migration v5
107    /// yet) is propagated as an `Err` from `build()` — unless env keys or OIDC
108    /// already configured auth, in which case the failure is only `warn!`-logged
109    /// and `has_live_rows` is treated as `false`, since the query's result would
110    /// be unused either way.
111    ///
112    /// Returns `Ok(None)` when nothing is configured at all (preserving the
113    /// "genuinely empty deployment" startup guard every caller relies on).
114    pub async fn build(self) -> Result<Option<Arc<dyn AuthProvider>>> {
115        let admin_var = self.admin_var();
116        let oidc_config_var = self.oidc_config_var();
117        let api_keys_json = self.api_keys_json();
118
119        let mut multi = MultiAuthProvider::new();
120        let mut configured = false;
121
122        if let Some(keys_json) = api_keys_json {
123            let keyring = parse_key_ring(&keys_json)?;
124            info!("API key authentication enabled");
125            multi = multi.with_provider(Arc::new(ApiKeyAuthProvider::new(keyring)));
126            configured = true;
127        } else {
128            info!("API key auth not configured");
129        }
130
131        match OidcConfig::from_env_var(&oidc_config_var) {
132            Ok(config) => {
133                info!("Initializing OIDC authentication");
134                let oidc_provider = OidcAuthProvider::new(config, &admin_var).await?;
135                multi = multi.with_provider(Arc::new(oidc_provider));
136                configured = true;
137            }
138            Err(e) => {
139                info!("OIDC not configured ({e}) - OIDC auth disabled");
140            }
141        }
142
143        if let Some((pool, table)) = self.key_store {
144            let db_config = DbApiKeyConfig::from_env_with_prefix(&self.prefix);
145            let db_provider = DbApiKeyAuthProvider::new(pool.clone(), table, db_config);
146            multi = multi.with_provider(Arc::new(db_provider));
147
148            let has_live_rows = match key_store_has_live_rows(&pool, table).await {
149                Ok(has_live_rows) => has_live_rows,
150                Err(e) if !configured => {
151                    return Err(e.context(format!(
152                        "checking whether {} has any live key — has the schema reached migration v5? \
153                         (rust/ingestion/src/sql_migration.rs; the ingestion binary or monolith must run \
154                         the migration before flight-sql starts in a split deployment)",
155                        table.table_name()
156                    )));
157                }
158                Err(e) => {
159                    // Auth is already configured via another provider (env keys or
160                    // OIDC); this query's only purpose is deciding whether an
161                    // otherwise-unconfigured deployment counts as configured, so a
162                    // failure here (e.g. schema not yet at v5) must not abort
163                    // startup.
164                    warn!(
165                        "checking whether {} has any live key failed, ignoring (auth already configured): {e:#}",
166                        table.table_name()
167                    );
168                    false
169                }
170            };
171            if has_live_rows {
172                configured = true;
173            }
174        }
175
176        if !configured {
177            return Ok(None);
178        }
179
180        Ok(Some(Arc::new(multi) as Arc<dyn AuthProvider>))
181    }
182}

184/// Initializes the default authentication provider with API key and OIDC from environment.
185///
186/// Reads configuration from:
187/// - `MICROMEGAS_API_KEYS`: JSON array of API keys
188/// - `MICROMEGAS_OIDC_CONFIG`: OIDC configuration JSON
189/// - `MICROMEGAS_ADMINS`: JSON array of admin user emails/subjects
190///
191/// Returns `Ok(Some(...))` if at least one provider is configured.
192/// Returns `Ok(None)` if no providers are configured (auth disabled).
193/// Returns `Err` on configuration errors.
194///
195/// Kept as a thin env-only wrapper around [`ProviderBuilder`] — this crate's
196/// documented, published entry point for a caller that wants API-key + OIDC
197/// composition without a DB pool.
198///
199/// # Example
200///
201/// ```rust,no_run
202/// use micromegas_auth::default_provider::provider;
203///
204/// # async fn example() -> anyhow::Result<()> {
205/// let auth_provider = provider().await?;
206/// if let Some(provider) = auth_provider {
207///     println!("Authentication enabled");
208/// } else {
209///     println!("No authentication configured");
210/// }
211/// # Ok(())
212/// # }
213/// ```
214pub async fn provider() -> Result<Option<Arc<dyn AuthProvider>>> {
215    provider_with_prefix("").await
216}

218/// Initializes auth providers using env vars scoped to a prefix.
219///
220/// For prefix `"MICROMEGAS_INGESTION"`:
221/// - API keys: tries `MICROMEGAS_INGESTION_API_KEYS`, falls back to `MICROMEGAS_API_KEYS`
222/// - OIDC:     tries `MICROMEGAS_INGESTION_OIDC_CONFIG`, falls back to `MICROMEGAS_OIDC_CONFIG`
223/// - Admins:   tries `MICROMEGAS_INGESTION_ADMINS`, falls back to `MICROMEGAS_ADMINS`
224///
225/// With an empty prefix the behaviour is identical to [`provider`]. Kept as a
226/// thin env-only wrapper around [`ProviderBuilder`] — no DB key store is attached.
227pub async fn provider_with_prefix(prefix: &str) -> Result<Option<Arc<dyn AuthProvider>>> {
228    ProviderBuilder::new(prefix).build().await
