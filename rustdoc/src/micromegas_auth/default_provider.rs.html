1//! Default authentication provider initialization for Micromegas services.
2//!
3//! This module provides the standard way to initialize authentication with
4//! API key, OIDC, and (optionally) a DB-backed API key store from environment
5//! variables.

20/// Builder for the default (env-driven) authentication provider stack, plus an
21/// optional DB-backed API key store.
22///
23/// The env factory (`provider()` / `provider_with_prefix()` below) is a builder
24/// so that adding the DB store — and later a policy — does not re-break the
25/// signature.
26pub struct ProviderBuilder {
27    prefix: String,
28    key_store: Option<(PgPool, ApiKeyTable)>,
29}
30
31impl ProviderBuilder {
32    /// Starts a builder scoped to `prefix` (e.g. `"MICROMEGAS_INGESTION"`, or `""`
33    /// for the unprefixed default), following the same `{prefix}_*`-with-fallback
34    /// convention as `provider_with_prefix`.
35    pub fn new(prefix: &str) -> Self {
36        Self {
37            prefix: prefix.to_string(),
38            key_store: None,
39        }
40    }
41
42    /// Attaches a DB-backed key store bound to `table`, looked up through `pool`
43    /// (expected to be a [`crate::db_api_key::dedicated_key_store_pool`], not a
44    /// clone of the caller's lake pool).
45    pub fn with_db_key_store(mut self, pool: PgPool, table: ApiKeyTable) -> Self {
46        self.key_store = Some((pool, table));
47        self
48    }
49
50    /// Resolves the API-keys env var name for this builder's prefix, with
51    /// fallback to the unprefixed name.
52    fn api_keys_json(&self) -> Option<String> {
53        std::env::var(resolve_prefixed_var(&self.prefix, "API_KEYS")).ok()
54    }
55
56    /// Resolves the OIDC config env var name for this builder's prefix, with
57    /// fallback to the unprefixed name.
58    fn oidc_config_var(&self) -> String {
59        resolve_prefixed_var(&self.prefix, "OIDC_CONFIG")
60    }
61
62    /// Resolves the admin-users env var name for this builder's prefix, with
63    /// fallback to the unprefixed name.
64    fn admin_var(&self) -> String {
65        resolve_prefixed_var(&self.prefix, "ADMINS")
66    }
67
68    /// Builds the composed provider.
69    ///
70    /// Composes, in this order: env `ApiKeyAuthProvider` (in-memory, cheapest,
71    /// preserves today's precedence) → `OidcAuthProvider` → `DbApiKeyAuthProvider`.
72    /// `MultiAuthProvider` tries providers in order, so putting the DB provider
73    /// last means only tokens that are neither an env key nor a valid JWT ever
74    /// reach it.
75    ///
76    /// **The DB provider is always pushed onto the chain whenever a key store is
77    /// attached** — registration never depends on the existence query below, so a
78    /// deployment that mints its first key through `analytics-web-srv`'s
79    /// `POST {base_path}/api/ingestion-api-keys` into a previously empty table
80    /// authenticates it on the very next request, with no restart.
81    ///
82    /// **A DB key store with at least one live key counts as "auth configured";
83    /// an empty one does not.** When a key store is attached, this runs one cheap
84    /// startup existence query (`key_store_has_live_rows`) and treats a non-empty
85    /// result the same as env keys or OIDC being present. A failure of that query
86    /// (e.g. a missing relation because the schema has not reached migration v5
87    /// yet) is propagated as an `Err` from `build()` — unless env keys or OIDC
88    /// already configured auth, in which case the failure is only `warn!`-logged
89    /// and `has_live_rows` is treated as `false`, since the query's result would
90    /// be unused either way.
91    ///
92    /// Returns `Ok(None)` when nothing is configured at all (preserving the
93    /// "genuinely empty deployment" startup guard every caller relies on).
94    pub async fn build(self) -> Result<Option<Arc<dyn AuthProvider>>> {
95        let admin_var = self.admin_var();
96        let oidc_config_var = self.oidc_config_var();
97        let api_keys_json = self.api_keys_json();
98
99        let mut multi = MultiAuthProvider::new();
100        let mut configured = false;
101
102        if let Some(keys_json) = api_keys_json {
103            let keyring = parse_key_ring(&keys_json)?;
104            info!("API key authentication enabled");
105            multi = multi.with_provider(Arc::new(ApiKeyAuthProvider::new(keyring)));
106            configured = true;
107        } else {
108            info!("API key auth not configured");
109        }
110
111        match OidcConfig::from_env_var(&oidc_config_var) {
112            Ok(config) => {
113                info!("Initializing OIDC authentication");
114                let oidc_provider = OidcAuthProvider::new(config, &admin_var).await?;
115                multi = multi.with_provider(Arc::new(oidc_provider));
116                configured = true;
117            }
118            Err(e) => {
119                info!("OIDC not configured ({e}) - OIDC auth disabled");
120            }
121        }
122
123        if let Some((pool, table)) = self.key_store {
124            let db_config = DbApiKeyConfig::from_env_with_prefix(&self.prefix);
125            let db_provider = DbApiKeyAuthProvider::new(pool.clone(), table, db_config);
126            multi = multi.with_provider(Arc::new(db_provider));
127
128            let has_live_rows = match key_store_has_live_rows(&pool, table).await {
129                Ok(has_live_rows) => has_live_rows,
130                Err(e) if !configured => {
131                    return Err(e.context(format!(
132                        "checking whether {} has any live key — has the schema reached migration v5? \
133                         (rust/ingestion/src/sql_migration.rs; the ingestion binary or monolith must run \
134                         the migration before flight-sql starts in a split deployment)",
135                        table.table_name()
136                    )));
137                }
138                Err(e) => {
139                    // Auth is already configured via another provider (env keys or
140                    // OIDC); this query's only purpose is deciding whether an
141                    // otherwise-unconfigured deployment counts as configured, so a
142                    // failure here (e.g. schema not yet at v5) must not abort
143                    // startup.
144                    warn!(
145                        "checking whether {} has any live key failed, ignoring (auth already configured): {e:#}",
146                        table.table_name()
147                    );
148                    false
149                }
150            };
151            if has_live_rows {
152                configured = true;
153            }
154        }
155
156        if !configured {
157            return Ok(None);
158        }
159
160        Ok(Some(Arc::new(multi) as Arc<dyn AuthProvider>))
161    }
162}

164/// Initializes the default authentication provider with API key and OIDC from environment.
165///
166/// Reads configuration from:
167/// - `MICROMEGAS_API_KEYS`: JSON array of API keys
168/// - `MICROMEGAS_OIDC_CONFIG`: OIDC configuration JSON
169/// - `MICROMEGAS_ADMINS`: JSON array of admin user emails/subjects
170///
171/// Returns `Ok(Some(...))` if at least one provider is configured.
172/// Returns `Ok(None)` if no providers are configured (auth disabled).
173/// Returns `Err` on configuration errors.
174///
175/// Kept as a thin env-only wrapper around [`ProviderBuilder`] — this crate's
176/// documented, published entry point for a caller that wants API-key + OIDC
177/// composition without a DB pool.
178///
179/// # Example
180///
181/// ```rust,no_run
182/// use micromegas_auth::default_provider::provider;
183///
184/// # async fn example() -> anyhow::Result<()> {
185/// let auth_provider = provider().await?;
186/// if let Some(provider) = auth_provider {
187///     println!("Authentication enabled");
188/// } else {
189///     println!("No authentication configured");
190/// }
191/// # Ok(())
192/// # }
193/// ```
194pub async fn provider() -> Result<Option<Arc<dyn AuthProvider>>> {
195    provider_with_prefix("").await
196}

198/// Initializes auth providers using env vars scoped to a prefix.
199///
200/// For prefix `"MICROMEGAS_INGESTION"`:
201/// - API keys: tries `MICROMEGAS_INGESTION_API_KEYS`, falls back to `MICROMEGAS_API_KEYS`
202/// - OIDC:     tries `MICROMEGAS_INGESTION_OIDC_CONFIG`, falls back to `MICROMEGAS_OIDC_CONFIG`
203/// - Admins:   tries `MICROMEGAS_INGESTION_ADMINS`, falls back to `MICROMEGAS_ADMINS`
204///
205/// With an empty prefix the behaviour is identical to [`provider`]. Kept as a
206/// thin env-only wrapper around [`ProviderBuilder`] — no DB key store is attached.
207pub async fn provider_with_prefix(prefix: &str) -> Result<Option<Arc<dyn AuthProvider>>> {
208    ProviderBuilder::new(prefix).build().await
209}
