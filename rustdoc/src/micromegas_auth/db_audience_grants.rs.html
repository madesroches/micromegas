1//! DB-backed audience grant store (#1489, AbAC Stage 6a): a whole-table snapshot cache over the
2//! `audience_grants` table (migration v7, `rust/ingestion/src/sql_migration.rs`), checked
3//! alongside the existing `{prefix}_AUDIENCE_GRANTS` env map by
4//! [`crate::policy::AudienceReadPolicy`]/[`crate::policy::AudienceMintPolicy`] -- a selector
5//! present in either source grants access, without either side being deep-cloned or merged into
6//! a combined map (`current()` hands back the cached grants behind an `Arc`). This is what makes
7//! a grant creatable without a service restart -- the env map stays the static/bootstrap layer.
8//!
9//! Modeled on `db_api_key.rs`'s config/pool conventions, but a single cached value rather than a
10//! per-key `moka` cache: the issue is explicit that the whole map is small enough to hold as one
11//! snapshot, so `moka`'s eviction/LRU machinery has nothing to do here.
12

22/// Cache-TTL knob for [`DbAudienceGrantsSource`], read from env with a default.
23#[derive(Clone, Copy, Debug)]
24pub struct DbAudienceGrantsConfig {
25    /// `{prefix}_AUDIENCE_GRANT_CACHE_TTL_SECONDS`, falling back to
26    /// `MICROMEGAS_AUDIENCE_GRANT_CACHE_TTL_SECONDS`, default 60 -- mirrors
27    /// `MICROMEGAS_API_KEY_CACHE_TTL_SECONDS`'s name, default, and prefix-fallback shape.
28    pub cache_ttl_secs: u64,
29}
30
31impl DbAudienceGrantsConfig {
32    /// Resolves the knob as `{prefix}_AUDIENCE_GRANT_CACHE_TTL_SECONDS` first, falling back to
33    /// the unprefixed name -- the same `resolve_u64`-based pattern
34    /// `DbApiKeyConfig::from_env_with_prefix` (`db_api_key.rs`) already uses for its four knobs,
35    /// so this config follows the same prefix contract every other knob at its wiring sites does.
36    /// With an empty prefix this is identical to the unprefixed var.
37    pub fn from_env_with_prefix(prefix: &str) -> Self {
38        Self {
39            cache_ttl_secs: resolve_u64(prefix, "AUDIENCE_GRANT_CACHE_TTL_SECONDS", 60),
40        }
41    }
42}

44#[derive(Debug)]
45struct Snapshot {
46    grants: Arc<AudienceGrants>,
47    /// Time of the last *successful* load -- never advanced on a failed refresh. Feeds the age
48    /// reported in the refresh-failure log/error context ("how stale is the grant view right
49    /// now"), independent of `fetched_at`; it is not itself emitted as a metric.
50    loaded_at: Instant,
51    /// Time of the last refresh *attempt*, successful or not -- gates how often a failing DB is
52    /// re-queried once at least one load has succeeded, so a post-first-success outage costs one
53    /// query per TTL, not one per request.
54    fetched_at: Instant,
55}
56
57/// The whole-table snapshot cache described in the module doc comment.
58#[derive(Debug)]
59pub struct DbAudienceGrantsSource {
60    pool: PgPool,
61    ttl: Duration,
62    snapshot: tokio::sync::RwLock<Option<Snapshot>>,
63    /// Process-start baseline every `last_attempt_at` reading is measured from -- captured once
64    /// here, not `Instant::now()` at each read, so `last_attempt_at` values are comparable across
65    /// calls the same way a wall-clock timestamp would be, but monotonic: unlike
66    /// `Utc::now().timestamp()`, a backwards clock step (NTP correction, VM migration, manual
67    /// clock set) can never make a later reading smaller than an earlier one.
68    start: Instant,
69    /// Milliseconds since `start` of the last refresh *attempt*, recorded outside `Snapshot` so
70    /// it exists even before any load has ever succeeded -- mirrors `db_api_key.rs`'s
71    /// `last_logged_at`, but monotonic rather than a Unix-epoch-seconds value (see [`Self::start`]).
72    /// This is what gates cold-start retries: with no `Snapshot` yet, there is nowhere else to
73    /// remember "we just tried and failed a moment ago", so without this field every `current()`
74    /// call during process startup against a still-coming-up DB would re-query with no throttling
75    /// at all, unlike the post-success path which `fetched_at` already gates. Initialized to
76    /// `i64::MIN` -- not `0`, which (being only milliseconds after `start`) would be indistinguishable
77    /// from a real attempt made moments after construction and would incorrectly throttle the very
78    /// first cold-start call -- so the first real attempt's `saturating_sub` against it is always
79    /// far past `ttl` regardless of how soon after construction it lands.
80    last_attempt_at: AtomicI64,
81}
82
83impl DbAudienceGrantsSource {
84    /// Builds a source over `pool` (expected to be a [`crate::db_api_key::dedicated_key_store_pool`],
85    /// not the caller's lake pool directly), with no snapshot loaded yet -- the first `current()`
86    /// call performs the first query.
87    pub fn new(pool: PgPool, config: DbAudienceGrantsConfig) -> Self {
88        Self {
89            pool,
90            ttl: Duration::from_secs(config.cache_ttl_secs),
91            snapshot: tokio::sync::RwLock::new(None),
92            start: Instant::now(),
93            last_attempt_at: AtomicI64::new(i64::MIN),
94        }
95    }
96
97    /// Milliseconds elapsed since `self.start` -- the monotonic clock `last_attempt_at` is
98    /// measured against.
99    fn elapsed_millis(&self) -> i64 {
100        i64::try_from(self.start.elapsed().as_millis()).unwrap_or(i64::MAX)
101    }
102
103    /// Queries the whole table and builds an `AudienceGrants` via
104    /// [`AudienceGrants::from_rows`]. The one place a malformed row (one that slipped past the
105    /// table's own `CHECK` constraints, e.g. via a direct `psql` session) surfaces as a load
106    /// failure rather than a silently-inert or silently-unreadable grant.
107    async fn fetch(&self) -> Result<AudienceGrants> {
108        let rows = sqlx::query("SELECT audience, axis, selector FROM audience_grants")
109            .fetch_all(&self.pool)
110            .await
111            .context("querying audience_grants")?;
112        let mut triples = Vec::with_capacity(rows.len());
113        for row in rows {
114            let audience: String = row.try_get("audience").context("reading audience")?;
115            let axis: String = row.try_get("axis").context("reading axis")?;
116            let selector: String = row.try_get("selector").context("reading selector")?;
117            let axis = match axis.as_str() {
118                "read" => GrantAxis::Read,
119                "mint" => GrantAxis::Mint,
120                other => {
121                    return Err(anyhow!(
122                        "audience_grants row for {audience:?}/{selector:?} has unrecognized \
123                         axis {other:?}"
124                    ));
125                }
126            };
127            triples.push((audience, axis, selector));
128        }
129        AudienceGrants::from_rows(triples)
130    }
131
132    /// The synthesized error for a throttled cold-start window (see [`Self::current`]'s doc
133    /// comment) -- never a stored, re-handed-out `anyhow::Error` from a prior attempt, since that
134    /// type isn't `Clone` and this store deliberately carries none of `db_api_key.rs`'s
135    /// moka/`Arc<E>` sharing machinery to make it so.
136    ///
137    /// `last_attempt_millis` is measured against `self.start` (see that field's doc comment), not
138    /// a wall-clock instant, so there is no absolute timestamp left to report here -- the message
139    /// reports the throttle window's remaining duration instead, which stays meaningful (and never
140    /// goes backwards) regardless of the system clock.
141    fn throttled_error(&self, last_attempt_millis: i64) -> anyhow::Error {
142        let since_last_attempt = Duration::from_millis(
143            self.elapsed_millis()
144                .saturating_sub(last_attempt_millis)
145                .max(0) as u64,
146        );
147        let retry_in = self.ttl.saturating_sub(since_last_attempt);
148        // Wrapped in `ProviderUnavailable` (not a bare `anyhow!`) so `caller_context`
149        // (`flight_sql_service_impl.rs`) downcasts this to `Status::unavailable` rather than
150        // `Status::permission_denied` -- a throttled cold start is a store-outage symptom, not an
151        // authorization decision.
152        ProviderUnavailable(anyhow!(
153            "audience grant store unavailable; cold-start retry throttled, retry available in \
154             {} seconds",
155            retry_in.as_secs()
156        ))
157        .into()
158    }
159
160    /// Returns the current grant snapshot, refreshing it first if stale.
161    ///
162    /// Both the cold-start path (no `Snapshot` yet) and the post-success path are throttled to at
163    /// most one DB query per TTL window: cold-start via `last_attempt_at` (checked-and-set the
164    /// same compare-exchange way `db_api_key.rs::maybe_log_error` rate-limits its own log line),
165    /// the post-success path via `Snapshot::fetched_at` as before. A `current()` call that lands
166    /// inside an already-throttled window returns the last snapshot if one exists; if not, it
167    /// returns a freshly synthesized error describing the throttled cold-start state rather than
168    /// storing and re-handing-out the prior attempt's `anyhow::Error`. A concurrent caller that
169    /// loses the `last_attempt_at` compare-exchange while the very first cold-start attempt is
170    /// still in flight re-checks `self.snapshot` once more before giving up: if the winning
171    /// caller has since populated it, the loser serves that snapshot instead of failing; only if
172    /// it is still empty does it get the synthesized throttled-state error. It never blocks on
173    /// another caller's still-in-flight query, and it never skips the query silently with nothing
174    /// to show for it.
175    ///
176    /// `Err` only when there has never been one successful load -- a fresh process whose first
177    /// query hits a down DB has no "last good" to serve, so it fails closed like everything else
178    /// on this seam, at a rate capped by `last_attempt_at` rather than once per request. Once any
179    /// load has succeeded, a later refresh failure is logged + counted
180    /// (`imetric!("audience_grant_refresh_error_count", ...)`) and the last good snapshot keeps
181    /// serving, unbounded -- this store has no per-item TTL eviction to fall back on the way
182    /// `db_api_key.rs`'s cache does, so an outage degrades to staleness for as long as it lasts,
183    /// not just one TTL window.
184    pub async fn current(&self) -> Result<Arc<AudienceGrants>> {
185        // Fast path: an unexpired snapshot needs no refresh attempt at all.
186        {
187            let guard = self.snapshot.read().await;
188            if let Some(snap) = guard.as_ref()
189                && snap.fetched_at.elapsed() < self.ttl
190            {
191                return Ok(Arc::clone(&snap.grants));
192            }
193        }
194
195        let had_snapshot = self.snapshot.read().await.is_some();
196
197        if had_snapshot {
198            // Post-success path: no single-flight/dedup lock -- a plain `SELECT` with no side
199            // effect to de-duplicate, gated only by `fetched_at` above. A few concurrent callers
200            // landing right at the TTL boundary can each re-run this; strictly simpler and still
201            // cheap (the whole point of "the map is small").
202            match self.fetch().await {
203                Ok(grants) => {
204                    let grants = Arc::new(grants);
205                    let mut guard = self.snapshot.write().await;
206                    *guard = Some(Snapshot {
207                        grants: Arc::clone(&grants),
208                        loaded_at: Instant::now(),
209                        fetched_at: Instant::now(),
210                    });
211                    Ok(grants)
212                }
213                Err(err) => {
214                    micromegas_tracing::imetric!(
215                        "audience_grant_refresh_error_count",
216                        "count",
217                        1_u64
218                    );
219                    let mut guard = self.snapshot.write().await;
220                    match guard.as_mut() {
221                        Some(snap) => {
222                            let age_secs = snap.loaded_at.elapsed().as_secs();
223                            micromegas_tracing::error!(
224                                "audience grant store refresh failed, serving stale snapshot \
225                                 (age={age_secs}s): {err:#}"
226                            );
227                            snap.fetched_at = Instant::now();
228                            Ok(Arc::clone(&snap.grants))
229                        }
230                        // Snapshot vanished between the check above and now -- this store never
231                        // clears a snapshot once set, so this is unreachable in practice; treat
232                        // it as a cold-start failure rather than panicking. Wrapped in
233                        // `ProviderUnavailable` for the same reason `throttled_error` is: a real
234                        // DB failure here must surface to `caller_context` as
235                        // `Status::unavailable`, not `Status::permission_denied`.
236                        None => Err(ProviderUnavailable(err).into()),
237                    }
238                }
239            }
240        } else {
241            // Cold-start path: `last_attempt_at`'s compare-exchange *is* the dedup mechanism, so
242            // only one caller per TTL window fires the query and the rest get the synthesized
243            // throttled-state error instead of racing in. Measured against `self.start`
244            // (monotonic), not `Utc::now()` -- a backwards wall-clock step can never widen this
245            // window the way it could with an epoch-seconds comparison.
246            let now = self.elapsed_millis();
247            let prev = self.last_attempt_at.load(Ordering::Relaxed);
248            let ttl_millis = self.ttl.as_millis().max(1) as i64;
249            // `throttled_millis`, when set, is the timestamp that actually explains the
250            // remaining wait: on the throttle-hit branch that's `prev` (still accurate, no
251            // exchange attempted); on the CAS-loss branch it's the value the failed
252            // `compare_exchange` reports as *currently* stored, not the stale `prev` this
253            // caller read before attempting the exchange -- using `prev` there would report
254            // the wrong (often ancient/sentinel) retry window instead of the window the
255            // winning caller just started.
256            let throttled_millis = if now.saturating_sub(prev) < ttl_millis {
257                Some(prev)
258            } else {
259                self.last_attempt_at
260                    .compare_exchange(prev, now, Ordering::Relaxed, Ordering::Relaxed)
261                    .err()
262            };
263            if let Some(last_attempt_millis) = throttled_millis {
264                // The winner of the compare-exchange may have already finished its first query
265                // and populated the snapshot by the time we lost the race -- re-check before
266                // treating this as a throttled cold start, so concurrent callers during the very
267                // first load don't get denied purely from contention on a healthy DB. A cheap
268                // re-check, not a full single-flight/wait: if the winner hasn't finished yet,
269                // this still falls through to the throttled error below.
270                if let Some(snap) = self.snapshot.read().await.as_ref() {
271                    return Ok(Arc::clone(&snap.grants));
272                }
273                return Err(self.throttled_error(last_attempt_millis));
274            }
275
276            match self.fetch().await {
277                Ok(grants) => {
278                    let grants = Arc::new(grants);
279                    let mut guard = self.snapshot.write().await;
280                    *guard = Some(Snapshot {
281                        grants: Arc::clone(&grants),
282                        loaded_at: Instant::now(),
283                        fetched_at: Instant::now(),
284                    });
285                    Ok(grants)
286                }
287                Err(err) => {
288                    micromegas_tracing::imetric!(
289                        "audience_grant_refresh_error_count",
290                        "count",
291                        1_u64
292                    );
293                    micromegas_tracing::error!(
294                        "audience grant store cold-start load failed: {err:#}"
295                    );
296                    // Wrapped in `ProviderUnavailable` for the same reason `throttled_error` is:
297                    // a real DB failure here must surface to `caller_context` as
298                    // `Status::unavailable`, not `Status::permission_denied`.
299                    Err(ProviderUnavailable(err).into())
300                }
301            }
302        }
303    }
