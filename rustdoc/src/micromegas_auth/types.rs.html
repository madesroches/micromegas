4/// A DB-backed auth provider (e.g. `DbApiKeyAuthProvider`) could not reach its
5/// key store — distinguishes a store *outage* from a rejected credential all
6/// the way out to the HTTP/gRPC response.
7///
8/// Wraps the underlying `anyhow::Error` rather than replacing it: callers that
9/// only need to log or propagate the error keep using it as an `anyhow::Error`
10/// (via `?`/`From`), while `MultiAuthProvider`, `auth_middleware`, and the gRPC
11/// `AuthService` downcast (`anyhow::Error::downcast_ref`) to detect this specific
12/// kind and map it to a retryable status (503 / `Status::unavailable`) instead of
13/// the blanket "invalid credential" response every other error gets.
14///
15/// Per `rust/CLAUDE.md`'s anyhow-vs-thiserror rule, the retryable/terminal
16/// distinction *is* modeled as an explicit type rather than an ad-hoc string —
17/// that's the part of the rule this satisfies. What it doesn't satisfy: callers
18/// still reach that type via `downcast_ref` instead of matching on it directly,
19/// because `AuthProvider::validate_request` returns `anyhow::Result` and this
20/// plan chose not to change that trait's public return type. Treat this as a
21/// deliberate, scoped exception rather than a full application of the rule.
22#[derive(thiserror::Error, Debug)]
23#[error("auth provider unavailable: {0}")]
24pub struct ProviderUnavailable(#[source] pub anyhow::Error);

35/// Authentication context containing user identity and metadata
36#[derive(Debug, Clone)]
37pub struct AuthContext {
38    /// Unique subject identifier (e.g., user ID, service account ID)
39    pub subject: String,
40    /// Email address (if available)
41    pub email: Option<String>,
42    /// Issuer (for OIDC) or "api_key" for API key auth
43    pub issuer: String,
44    /// Audience that was matched during token validation
45    /// - For OIDC access tokens: API audience (e.g., "<https://api.example.com>")
46    /// - For OIDC ID tokens: client ID
47    /// - For API key auth: None
48    pub audience: Option<String>,
49    /// Token expiration time (if applicable)
50    pub expires_at: Option<DateTime<Utc>>,
51    /// Authentication type
52    pub auth_type: AuthType,
53    /// Whether this user has admin privileges
54    pub is_admin: bool,
55    /// Whether this authentication allows user delegation (acting on behalf of others)
56    /// - OIDC user tokens: false (user cannot impersonate others)
57    /// - Analytics API keys/service accounts: true (can act on behalf of users)
58    /// - Ingestion API keys: false (AbAC Stage 4, #1372) — a write credential is not a
59    ///   delegating service account, and it can never reach the gRPC path this flag governs
60    ///   anyway (`ingestion_api_keys` never crosses `flight_sql_service_impl.rs`)
61    pub allow_delegation: bool,
62    /// The write audience an ingestion key is immutably bound to (AbAC Stage 4, #1372).
63    /// `Some(..)` for every ingestion API key (the `ingestion_api_keys.audience` column is
64    /// `NOT NULL` as of migration v6); `None` for every other principal kind. Write-side
65    /// only; never consulted by `ReadPolicy`.
66    pub bound_audience: Option<String>,
67    /// The set of audiences an analytics service-account key is granted read access to (AbAC
68    /// Stage 4b). Empty for every principal today — populated by the analytics key provider once
69    /// Stage 4b lands. Read-side only; folded into `AudienceReadPolicy::resolve`'s union but never
70    /// into `AudienceMintPolicy`'s mintable set (a read grant confers no mint authority).
71    pub read_audiences: Vec<String>,
72    /// IdP-asserted **leaf** group membership — an input to policy resolution, possibly
73    /// incomplete. This is *not* the caller's effective groups: in the AbAC plan's recorded target
74    /// state the IdP supplies direct memberships only, while nesting (group-in-group) and
75    /// group→audience grants live in a micromegas-owned store, so the effective, transitive
76    /// closure is what the policy computes from this vector, not this vector itself. Raw claim
77    /// values, not yet namespaced — `AudienceReadPolicy`/`AudienceMintPolicy` match each entry
78    /// against `group:<id>` grant-map *selectors*, so this general-purpose auth type stays free
79    /// of the AbAC-specific convention. Empty for API keys (no groups claim) and for OIDC
80    /// callers whose token carries no `groups` claim.
81    pub groups: Vec<String>,
82}

84/// Trait for extracting authentication-relevant data from requests
85pub trait RequestParts: Send + Sync {
86    /// Extract Authorization header as string
87    fn authorization_header(&self) -> Option<&str>;
88
89    /// Extract Bearer token from Authorization header
90    fn bearer_token(&self) -> Option<&str> {
91        self.authorization_header()
92            .and_then(|h| h.strip_prefix("Bearer "))
93    }
94
95    /// Get custom header value by name
96    fn get_header(&self, name: &str) -> Option<&str>;
97
98    /// Get request method (if applicable)
99    fn method(&self) -> Option<&str>;
100
101    /// Get request URI (if applicable)
102    fn uri(&self) -> Option<&str>;
103}

105/// HTTP request validation input
106pub struct HttpRequestParts {
107    /// HTTP headers
108    pub headers: http::HeaderMap,
109    /// HTTP method
110    pub method: http::Method,
111    /// Request URI
112    pub uri: http::Uri,
113}
114
115impl RequestParts for HttpRequestParts {
116    fn authorization_header(&self) -> Option<&str> {
117        self.headers
118            .get(http::header::AUTHORIZATION)
119            .and_then(|h| h.to_str().ok())
120    }
121
122    fn get_header(&self, name: &str) -> Option<&str> {
123        self.headers.get(name).and_then(|h| h.to_str().ok())
124    }
125
126    fn method(&self) -> Option<&str> {
127        Some(self.method.as_str())
128    }
129
130    fn uri(&self) -> Option<&str> {
131        Some(self.uri.path())
132    }
133}

161/// Trait for authentication providers
162#[async_trait::async_trait]
163pub trait AuthProvider: Send + Sync {
164    /// Validate a request and return authentication context
165    async fn validate_request(&self, parts: &dyn RequestParts) -> Result<AuthContext>;
166
167    /// Whether this provider can ever produce an `AuthContext { is_admin: true, .. }`.
168    ///
169    /// **Invariant**: must stay `true` for any provider that can ever return an admin
170    /// `AuthContext` -- a wrong `false` here silently widens who may call destructive
171    /// maintenance functions (the mutating lakehouse UDTFs/UDFs get registered for *any*
172    /// authenticated caller when no provider in the deployment can grant admin). Defaults to
173    /// `false`, matching the two API-key providers, which hardcode `is_admin: false` and can
174    /// never produce an admin principal.
175    fn can_grant_admin(&self) -> bool {
176        false
177    }
178}
