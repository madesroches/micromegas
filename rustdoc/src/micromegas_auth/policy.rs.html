1//! Authorization seam (#1369, AbAC Stage 1; grant map rewrite #1372, Stage 4): `MintPolicy`,
2//! `ReadPolicy`, and the audience-based implementations that resolve them from a caller's
3//! `AuthContext` and a JSON grant map keyed by audience name.
4//!
5//! **No enforcement lands with this module itself.** This module fixes the *shape* of
6//! authorization -- every caller of these traits must deny on `Err`, and `ReadPolicy` cannot
7//! express "grant everything" at all -- while the resolved `ReadableAudiences`/`ReadScope` is
8//! consumed downstream by `OwnershipRewrite` (#1370, AbAC Stage 2; Prong A) and, still pending
9//! (#1371, Stage 3), Prong B's UDTF/UDF guards. See `rust/analytics/src/lakehouse/read_scope.rs`
10//! and `tasks/1372_audience_on_keys_plan.md`.
11//!
12//! **An audience is an opaque label, not a principal encoding.** `is_valid_audience` and
13//! `PUBLIC_AUDIENCE` define what a name is; `AudienceGrants` defines who may read or mint into
14//! it. No code here derives an audience from a caller's identity -- see `AudienceGrants` and
15//! `tasks/1372_audience_on_keys_plan.md` §1-§2 for the reasoning.
16

31/// The reserved audience every authenticated principal may read.
32pub const PUBLIC_AUDIENCE: &str = "public";
33
34/// `true` if `aud` is a valid audience name: `[A-Za-z0-9_-]{1,255}`, checked in bytes -- the
35/// ASCII-only charset makes byte length and char length agree for every value that can pass, so
36/// this doubles as the length check the `ingestion_api_keys.audience` `VARCHAR(255)` column
37/// enforces on the SQL side.
38///
39/// An audience is an opaque label on data, not a principal encoding: this charset makes an email
40/// (`@`, `.`) or a `user:`/`group:`-prefixed value (`:`) unrepresentable, on purpose -- see
41/// `AudienceGrants` for why there is no derivation from a caller's identity to replace it with.
42///
43/// Deliberately **not** normalizing: no case folding, no trimming. The value is stored and
44/// compared verbatim, which is what makes uniqueness meaningful -- `team-alpha` and `Team-Alpha`
45/// are two audiences, and neither is a typo the system silently repairs into the other.
46pub fn is_valid_audience(aud: &str) -> bool {
47    !aud.is_empty()
48        && aud.len() <= 255
49        && aud
50            .bytes()
51            .all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'-')
52}

54/// Resolves `{prefix}_DEFAULT_KEY_AUDIENCE` (falling back to `MICROMEGAS_DEFAULT_KEY_AUDIENCE`) --
55/// resolved **once at startup** (`web_server.rs`) so a typo fails fast rather than surfacing as a
56/// per-request 400. `None` when neither is set or the value is empty. Invalid ⇒ `Err`.
57///
58/// Consumed by `ingestion_keys.rs::resolve_audience`'s `import` fallback (`Some(PUBLIC_AUDIENCE)`)
59/// and `mint`'s (`None` -- an unresolved mint is a 400, never a silent `public`). See
60/// `tasks/1372_audience_on_keys_plan.md` §5 for why the two routes differ only when this knob is
61/// unset.
62pub fn default_key_audience_from_env(prefix: &str) -> Result<Option<String>> {
63    let var = resolve_prefixed_var(prefix, "DEFAULT_KEY_AUDIENCE");
64    let resolved = match std::env::var(&var) {
65        Ok(raw) => {
66            let raw = raw.trim().to_string();
67            if raw.is_empty() {
68                None
69            } else if is_valid_audience(&raw) {
70                Some(raw)
71            } else {
72                return Err(anyhow!(
73                    "{var}: {raw:?} is not a valid audience name -- must match [A-Za-z0-9_-]{{1,255}}"
74                ));
75            }
76        }
77        Err(_) => None,
78    };
79    match &resolved {
80        Some(aud) => info!("{var}: default key audience = {aud}"),
81        None => info!("{var}: no default key audience configured"),
82    }
83    Ok(resolved)
84}

86/// A principal selector on either axis of an [`AudienceGrants`] entry: `*` (any authenticated
87/// principal), `user:<email>` (matches `AuthContext.email`), or `group:<g>` (matches any raw
88/// value in `AuthContext.groups`). Validated at parse time; `resolve`/`resolve_audience` never
89/// see an unrecognized shape.
90///
91/// `pub` (#1489, AbAC Stage 6a) -- `analytics-web-srv`'s admin grant-write route (a separate
92/// crate) needs to run this exact same selector-shape check `parse`/`from_rows` run, not a
93/// re-implementation of it.
94pub fn valid_selector(selector: &str) -> bool {
95    selector == "*"
96        || selector
97            .strip_prefix("user:")
98            .is_some_and(|rest| !rest.is_empty())
99        || selector
100            .strip_prefix("group:")
101            .is_some_and(|rest| !rest.is_empty())
102}
103
104fn selector_matches(selector: &str, caller: &AuthContext) -> bool {
105    if selector == "*" {
106        return true;
107    }
108    if let Some(email) = selector.strip_prefix("user:") {
109        return caller.email.as_deref() == Some(email);
110    }
111    if let Some(group) = selector.strip_prefix("group:") {
112        return caller.groups.iter().any(|g| g == group);
113    }
114    // Unreachable given `valid_selector` gates every selector at parse time; false rather than
115    // panicking keeps this fail-closed if that invariant is ever violated.
116    false
117}

119/// One audience's read/mint selector lists, both already validated by [`AudienceGrants::parse`].
120#[derive(Debug, Clone, Default)]
121struct GrantEntry {
122    read: Vec<String>,
123    mint: Vec<String>,
124}

126/// The Rust side of `audience_grants.axis` (#1489, AbAC Stage 6a) -- which selector list a DB row
127/// contributes to, mirroring the JSON grant map's `"read"`/`"mint"` keys.
128#[derive(Debug, Clone, Copy, PartialEq, Eq)]
129pub enum GrantAxis {
130    Read,
131    Mint,
132}

134/// The bare-array-or-object shape a single grant-map value may take on the wire, before content
135/// validation. An untagged enum rather than a hand-rolled visitor: JSON arrays and objects are
136/// syntactically distinct, so serde can dispatch on shape alone, and any value that is neither
137/// (a number, a string, `null`, ...) fails here with serde's own type-mismatch error.
138#[derive(Debug, Deserialize)]
139#[serde(untagged)]
140enum RawGrantValue {
141    /// Read-only shorthand: a bare array is the audience's `read` list, with an implicitly empty
142    /// `mint` list -- never derived from `read`.
143    Bare(Vec<String>),
144    /// The explicit form, needed only when an audience also grants mint authority.
145    Object(RawGrantObject),
146}
147
148/// The object form of a single grant-map value, broken out of [`RawGrantValue::Object`] so
149/// `#[serde(deny_unknown_fields)]` -- which serde cannot apply directly to an enum variant --
150/// has somewhere to attach: without it, a misspelled key (`"raed"` for `"read"`) would silently
151/// parse into an empty grant instead of failing startup.
152#[derive(Debug, Deserialize)]
153#[serde(deny_unknown_fields)]
154struct RawGrantObject {
155    #[serde(default)]
156    read: Vec<String>,
157    #[serde(default)]
158    mint: Vec<String>,
159}
160
161/// Deserializes the top-level `{prefix}_AUDIENCE_GRANTS` object while rejecting a repeated key --
162/// `serde_json`'s own `Map` deserialization silently keeps the *last* value for a duplicate key,
163/// which would discard an earlier grant list without a word (see `AudienceGrants`'s doc comment).
164/// Pulling keys one at a time through `MapAccess`, rather than deserializing straight into a
165/// `BTreeMap`, is what makes that duplicate visible before it is lost.
166struct RawAudienceGrants(BTreeMap<String, RawGrantValue>);
167
168impl<'de> Deserialize<'de> for RawAudienceGrants {
169    fn deserialize<D>(deserializer: D) -> std::result::Result<Self, D::Error>
170    where
171        D: serde::Deserializer<'de>,
172    {
173        struct GrantsVisitor;
174
175        impl<'de> Visitor<'de> for GrantsVisitor {
176            type Value = RawAudienceGrants;
177
178            fn expecting(&self, f: &mut fmt::Formatter) -> fmt::Result {
179                write!(f, "a JSON object keyed by audience name")
180            }
181
182            fn visit_map<A>(self, mut map: A) -> std::result::Result<Self::Value, A::Error>
183            where
184                A: MapAccess<'de>,
185            {
186                let mut result = BTreeMap::new();
187                while let Some(key) = map.next_key::<String>()? {
188                    let value: RawGrantValue = map.next_value()?;
189                    if result.contains_key(&key) {
190                        return Err(de::Error::custom(format!(
191                            "duplicate audience key {key:?} in grant map"
192                        )));
193                    }
194                    result.insert(key, value);
195                }
196                Ok(RawAudienceGrants(result))
197            }
198        }
199
200        deserializer.deserialize_map(GrantsVisitor)
201    }
202}

204/// The parsed, validated `{prefix}_AUDIENCE_GRANTS` map: "who can access this audience", keyed by
205/// audience name, one relation per axis (read/mint -- AbAC plan's `group_read_grants` /
206/// `group_mint_grants`, kept as one env map only because there is no store yet to split them
207/// across).
208///
209/// A bare-array value is read-only shorthand: an omitted `"mint"` list is therefore always empty,
210/// never defaulted from `"read"` -- a read grant confers no mint authority, by construction.
211///
212/// `public` is not stored here: it is the sole built-in read grant, applied by
213/// `AudienceReadPolicy::resolve` directly rather than needing a `{"public": ["*"]}` entry (though
214/// writing one changes nothing).
215#[derive(Debug, Clone, Default)]
216pub struct AudienceGrants {
217    entries: BTreeMap<String, GrantEntry>,
218}
219
220impl AudienceGrants {
221    /// No grants at all -- every audience but `public` is unreadable and unmintable by anyone
222    /// non-admin. Used as the disabled-auth / no-explicit-policy default.
223    pub fn empty() -> Self {
224        Self::default()
225    }
226
227    /// Parses a `{prefix}_AUDIENCE_GRANTS` JSON string directly. Split out from
228    /// [`Self::from_env`] so tests can exercise parsing without mutating the environment.
229    ///
230    /// Validates both axes of the map, not just its JSON shape -- same reason every other knob in
231    /// this crate fails startup on a typo rather than shipping inert: every key must satisfy
232    /// [`is_valid_audience`] (`{"group:everyone": [...]}`, the value operators are migrating
233    /// *from*, can never match a stamped audience once the charset excludes `:`), and every
234    /// selector on either axis must be `*`, `user:<non-empty>`, or `group:<non-empty>` (an
235    /// unprefixed or mis-prefixed selector matches no identity axis and would otherwise be a
236    /// silently-inert entry).
237    pub fn parse(raw: &str) -> Result<Self> {
238        let raw: RawAudienceGrants =
239            serde_json::from_str(raw).map_err(|e| anyhow!("invalid audience grant map: {e}"))?;
240        let mut entries = BTreeMap::new();
241        for (audience, value) in raw.0 {
242            if !is_valid_audience(&audience) {
243                return Err(anyhow!(
244                    "invalid audience grant map: {audience:?} is not a valid audience name -- \
245                     must match [A-Za-z0-9_-]{{1,255}}"
246                ));
247            }
248            let (read, mint) = match value {
249                RawGrantValue::Bare(read) => (read, vec![]),
250                RawGrantValue::Object(RawGrantObject { read, mint }) => (read, mint),
251            };
252            for selector in read.iter().chain(mint.iter()) {
253                if !valid_selector(selector) {
254                    return Err(anyhow!(
255                        "invalid audience grant map: selector {selector:?} for audience \
256                         {audience:?} must be '*', 'user:<id>', or 'group:<id>'"
257                    ));
258                }
259            }
260            entries.insert(audience, GrantEntry { read, mint });
261        }
262        Ok(Self { entries })
263    }
264
265    /// Resolves [`audience_grants_var`] and parses it. Unset ⇒ [`Self::empty`].
266    pub fn from_env(prefix: &str) -> Result<Self> {
267        let var = resolve_prefixed_var(prefix, "AUDIENCE_GRANTS");
268        let grants = match std::env::var(&var) {
269            Ok(raw) if raw.trim().is_empty() => Self::empty(),
270            Ok(raw) => Self::parse(&raw).map_err(|e| anyhow!("{var}: {e}"))?,
271            Err(_) => Self::empty(),
272        };
273        if grants.entries.is_empty() {
274            info!("{var}: no audience grants configured");
275        } else {
276            let names = grants.entries.keys().cloned().collect::<Vec<_>>().join(",");
277            info!("{var}: {} audience grants ({names})", grants.entries.len());
278        }
279        Ok(grants)
280    }
281
282    /// Builds an `AudienceGrants` from `(audience, axis, selector)` triples -- the DB-store
283    /// analogue of [`Self::parse`] (#1489, AbAC Stage 6a). Runs the *same* [`is_valid_audience`]/
284    /// [`valid_selector`] checks `parse` runs, so a row that slipped past `audience_grants`'s own
285    /// `CHECK` constraints (e.g. via a direct `psql` session) still can't reach a policy decision:
286    /// this is the one place both the JSON path and the DB path fail closed on a malformed row.
287    pub fn from_rows(rows: impl IntoIterator<Item = (String, GrantAxis, String)>) -> Result<Self> {
288        let mut entries: BTreeMap<String, GrantEntry> = BTreeMap::new();
289        for (audience, axis, selector) in rows {
290            if !is_valid_audience(&audience) {
291                return Err(anyhow!(
292                    "invalid audience grant row: {audience:?} is not a valid audience name -- \
293                     must match [A-Za-z0-9_-]{{1,255}}"
294                ));
295            }
296            if !valid_selector(&selector) {
297                return Err(anyhow!(
298                    "invalid audience grant row: selector {selector:?} for audience \
299                     {audience:?} must be '*', 'user:<id>', or 'group:<id>'"
300                ));
301            }
302            let entry = entries.entry(audience).or_default();
303            match axis {
304                GrantAxis::Read => entry.read.push(selector),
305                GrantAxis::Mint => entry.mint.push(selector),
306            }
307        }
308        Ok(Self { entries })
309    }
310
311    /// Unions each audience's `read`/`mint` selector lists across `self` and `other` (#1489, AbAC
312    /// Stage 6a): a selector present in either map grants exactly the same access as being
313    /// present in both. No dedup -- `selector_matches` is called with `.any()`, so a selector
314    /// present in both sources costs one redundant comparison, never a wrong answer.
315    ///
316    /// A public utility for combining two `AudienceGrants` maps, exercised by this crate's own
317    /// integration tests (`rust/auth/tests/policy_tests.rs`); `resolve`/`resolve_audience` do
318    /// **not** call it -- they check the env map and the DB store snapshot as two separate
319    /// sources instead, specifically to avoid a per-request deep clone (see the comment in
320    /// [`AudienceReadPolicy::resolve`]).
321    pub fn merge(&self, other: &Self) -> Self {
322        let mut entries = self.entries.clone();
323        for (audience, other_entry) in &other.entries {
324            let entry = entries.entry(audience.clone()).or_default();
325            entry.read.extend(other_entry.read.iter().cloned());
326            entry.mint.extend(other_entry.mint.iter().cloned());
327        }
328        Self { entries }
329    }
330
331    fn readers(&self) -> impl Iterator<Item = (&String, &[String])> {
332        self.entries.iter().map(|(a, e)| (a, e.read.as_slice()))
333    }
334
335    fn mint_selectors(&self, audience: &str) -> &[String] {
336        self.entries
337            .get(audience)
338            .map(|e| e.mint.as_slice())
339            .unwrap_or(&[])
340    }
341}

364/// Resolves the set of audiences an authenticated caller may read.
365///
366/// `resolve` can **never** return "all" -- `ReadScope::All` is deliberately not a policy output.
367/// It is the marker only internal (non-request) callers pass, which is what keeps "who granted
368/// themselves `All`?" a greppable question with a small, auditable answer set. A policy that
369/// wants to be maximally permissive still has to enumerate the audiences that make it so.
370///
371/// `async` and fallible because the AbAC plan's recorded *Long-term model* resolves grants from a
372/// store -- nested groups, plus group→audience grants -- which cannot live behind a sync,
373/// infallible signature. This is no longer just future-proofing: `AudienceReadPolicy` is fallible
374/// today whenever a [`DbAudienceGrantsSource`] is attached via `with_store` -- a cold-start store
375/// outage with no prior successful load returns `Err`, which `caller_context`
376/// (`flight_sql_service_impl.rs`) maps to `Status::unavailable`. Without a store attached it still
377/// cannot fail.
378///
379/// **Every caller must deny on `Err`.** An `Err` here means "the policy could not be evaluated" --
380/// never soften it into `ReadableAudiences::new(Arc::from([]))` (that would read as a legitimate,
381/// audited fail-closed decision) and never into `ReadScope::All` (a silent fail-open bypass).
382#[async_trait]
383pub trait ReadPolicy: Send + Sync + Debug {
384    /// Resolves `caller`'s readable-audience set, or `Err` if the policy could not be evaluated
385    /// (e.g. a backing store outage once a store-backed policy lands).
386    async fn resolve(&self, caller: &AuthContext) -> Result<ReadableAudiences>;
387}

389/// Resolves the audience a newly minted key may be stamped with.
390///
391/// `requested` is the caller-supplied audience (e.g. `"team-alpha"`); `None` means "no audience
392/// requested and none can be defaulted" -- under the opaque-label model there is no "myself"
393/// audience for a caller's identity to default to, so every implementation must return `Err` for
394/// `requested: None`, admin or not. Async and fallible for the same forward-looking reason as
395/// `ReadPolicy` -- wired by `mint_key` (`analytics-web-srv/src/ingestion_keys.rs`, AbAC Stage 6,
396/// #1374), its first production caller, gated by `MintGate`/`AuthenticatedUser` rather than
397/// `AdminUser`.
398#[async_trait]
399pub trait MintPolicy: Send + Sync + Debug {
400    /// Resolves the audience to stamp on a newly minted key, or `Err` if `requested` is outside
401    /// what `caller` may mint (including `requested: None` -- see the trait doc comment).
402    async fn resolve_audience(
403        &self,
404        caller: &AuthContext,
405        requested: Option<&str>,
406    ) -> Result<String>;
407}

409/// The shipped `ReadPolicy`: a lookup over an [`AudienceGrants`] map -- and, when a
410/// [`DbAudienceGrantsSource`] is attached via `with_store`, a second lookup over that store's
411/// snapshot -- with no identity derivation anywhere. Resolves the readable set:
412///
413/// ```text
414/// { PUBLIC_AUDIENCE }
415///   ∪ { a : "*"            ∈ grants[a].read }
416///   ∪ { a : "user:<email>" ∈ grants[a].read }                  if email present
417///   ∪ { a : "group:<g>"    ∈ grants[a].read for some g ∈ caller.groups }
418///   ∪ { a : selector       ∈ store.readers(a) matches caller } if a store is attached
419///   ∪ caller.read_audiences                                    // Stage 4b per-key direct grant
420/// ```
421///
422/// There is **no self-audience rule** -- a caller is granted no audience merely for being named
423/// like one. See `tasks/1372_audience_on_keys_plan.md` §2 for why: the charset makes an email
424/// unrepresentable as an audience name, and keying on `subject` instead would let an admin mint
425/// themselves read access by naming a key after an audience.
426///
427/// `read_audiences` is folded in here but deliberately **excluded** from
428/// `AudienceMintPolicy`'s mintable set -- a service-account key's ability to *see* an audience
429/// must not imply it can *stamp new keys* with that audience (mint is integrity, read is
430/// confidentiality).
431#[derive(Debug, Clone, Default)]
432pub struct AudienceReadPolicy {
433    grants: AudienceGrants,
434    /// The DB-backed grant store (#1489, AbAC Stage 6a), when this process has one configured --
435    /// checked alongside `grants` on every `resolve` call, as a separate source rather than
436    /// merged into it (see `resolve`). `None` for every disabled-auth/test caller that has no DB
437    /// pool to back one.
438    store: Option<Arc<DbAudienceGrantsSource>>,
439}
440
441impl AudienceReadPolicy {
442    /// Builds a policy with an explicit grant map (bypassing env resolution).
443    pub fn new(grants: AudienceGrants) -> Self {
444        Self {
445            grants,
446            store: None,
447        }
448    }
449
450    /// Resolves the grant map from `{prefix}_AUDIENCE_GRANTS` (falling back to
451    /// `MICROMEGAS_AUDIENCE_GRANTS`) via [`AudienceGrants::from_env`]. Unset ⇒ an empty grant map
452    /// ⇒ the readable set degenerates to `{public}` plus `read_audiences`, which
453    /// `OwnershipRewrite` (#1370, AbAC Stage 2) enforces. A malformed grant map is `Err`, not a
454    /// silently-emptied one, so a startup `?` turns a typo into a fail-fast instead of a
455    /// silently-inactive knob.
456    pub fn from_env(prefix: &str) -> Result<Self> {
457        let grants = AudienceGrants::from_env(prefix)?;
458        Ok(Self {
459            grants,
460            store: None,
461        })
462    }
463
464    /// Attaches (or clears, with `None`) the DB-backed grant store (#1489, AbAC Stage 6a). A
465    /// builder method, not a constructor argument, so `new`/`from_env` keep working unchanged for
466    /// every caller with no DB pool to back one (disabled-auth, tests).
467    pub fn with_store(mut self, store: Option<Arc<DbAudienceGrantsSource>>) -> Self {
468        self.store = store;
469        self
470    }
471}

473#[async_trait]
474impl ReadPolicy for AudienceReadPolicy {
475    async fn resolve(&self, caller: &AuthContext) -> Result<ReadableAudiences> {
476        // `Err` only on a cold-start outage (no snapshot has ever loaded successfully) --
477        // propagated as-is, so `resolve` denies exactly as documented on its own trait.
478        let store_grants = match &self.store {
479            Some(store) => Some(store.current().await?),
480            None => None,
481        };
482        let mut set = BTreeSet::new();
483        set.insert(PUBLIC_AUDIENCE.to_string());
484        // The env map and the store snapshot are checked as two separate sources -- a selector
485        // present in either grants access -- rather than merged into one map, so neither side is
486        // deep-cloned on every request (#1489, AbAC Stage 6a).
487        for (audience, read) in self.grants.readers() {
488            if read.iter().any(|s| selector_matches(s, caller)) {
489                set.insert(audience.clone());
490            }
491        }
492        if let Some(grants) = &store_grants {
493            for (audience, read) in grants.readers() {
494                if read.iter().any(|s| selector_matches(s, caller)) {
495                    set.insert(audience.clone());
496                }
497            }
498        }
499        for audience in &caller.read_audiences {
500            set.insert(audience.clone());
501        }
502        let audiences: Arc<[String]> = set.into_iter().collect::<Vec<_>>().into();
503        Ok(ReadableAudiences::new(audiences))
504    }
505}

507/// The shipped `MintPolicy`. Non-admin callers may mint only an audience in their **mint** set --
508/// `grants[a].mint`, never `grants[a].read` (a read grant confers no mint authority, unchanged
509/// from `AudienceReadPolicy`'s split); being able to *read* `public`, which every authenticated
510/// principal is, does not imply being able to *mint into* it, unless some grant names `public` in
511/// a `"mint"` list.
512///
513/// `is_admin` callers may mint **any** valid audience, `public` included -- `mint_key`
514/// (`analytics-web-srv/src/ingestion_keys.rs`) delegates authorization to this policy for every
515/// caller, admin included, rather than gating admin callers separately of its own accord (its own
516/// gate, `MintGate`, only enforces the self-service knob against non-admins); without this arm the
517/// only shipped `MintPolicy` could not express the admin mint flow that route depends on; the arm
518/// grants no power the route's gate does not already grant. This is deliberately **asymmetric** to
519/// the read path, where `is_admin` is never a bypass (AbAC plan §5): mint is an integrity decision
520/// (who may stamp a credential), reads are a confidentiality decision (who may see data), and the
521/// two axes are allowed to disagree.
522#[derive(Debug, Clone, Default)]
523pub struct AudienceMintPolicy {
524    grants: AudienceGrants,
525    /// The DB-backed grant store (#1489, AbAC Stage 6a). Built for symmetry with
526    /// [`AudienceReadPolicy::with_store`] and unit-tested, but this stage wires nothing to it: no
527    /// production code constructs a `dyn MintPolicy` today -- see [`with_store`](Self::with_store).
528    store: Option<Arc<DbAudienceGrantsSource>>,
529}
530
531impl AudienceMintPolicy {
532    /// Builds a policy with an explicit grant map.
533    pub fn new(grants: AudienceGrants) -> Self {
534        Self {
535            grants,
536            store: None,
537        }
538    }
539
540    /// Attaches (or clears, with `None`) the DB-backed grant store (#1489, AbAC Stage 6a). Built
541    /// alongside [`AudienceReadPolicy::with_store`] for symmetry -- unlike the read side, no
542    /// production call site attaches a store through this method: `mint_key` (AbAC Stage 6,
543    /// #1374) constructs its `AudienceMintPolicy` via `new`, so mint grants stay resolved by a
544    /// fresh, uncached point query against `audience_grants`, never a `DbAudienceGrantsSource`
545    /// snapshot, in this stage.
546    pub fn with_store(mut self, store: Option<Arc<DbAudienceGrantsSource>>) -> Self {
547        self.store = store;
548        self
549    }
550}
551
552#[async_trait]
553impl MintPolicy for AudienceMintPolicy {
554    async fn resolve_audience(
555        &self,
556        caller: &AuthContext,
557        requested: Option<&str>,
558    ) -> Result<String> {
559        let Some(aud) = requested else {
560            return Err(anyhow!("no audience requested and none can be defaulted"));
561        };
562        if caller.is_admin {
563            return if is_valid_audience(aud) {
564                Ok(aud.to_string())
565            } else {
566                Err(anyhow!(
567                    "malformed audience {aud:?}: must match [A-Za-z0-9_-]{{1,255}}"
568                ))
569            };
570        }
571        // The env map and the store snapshot are checked as two separate sources -- a selector
572        // present in either grants access -- rather than merged into one map, so neither side is
573        // deep-cloned on every request (#1489, AbAC Stage 6a).
574        let store_grants = match &self.store {
575            Some(store) => Some(store.current().await?),
576            None => None,
577        };
578        let granted = self
579            .grants
580            .mint_selectors(aud)
581            .iter()
582            .any(|s| selector_matches(s, caller))
583            || store_grants.as_ref().is_some_and(|grants| {
584                grants
585                    .mint_selectors(aud)
586                    .iter()
587                    .any(|s| selector_matches(s, caller))
588            });
589        if granted {
590            Ok(aud.to_string())
591        } else {
592            Err(anyhow!(
593                "audience {aud:?} is not in the caller's mintable set"
594            ))
595        }
596    }
597}
