1//! Authorization seam: `MintPolicy`, `ReadPolicy`, and the audience-based implementations that
2//! resolve them from a caller's `AuthContext` and a JSON grant map keyed by audience name.
3//!
4//! **No enforcement lands with this module itself.** This module fixes the *shape* of
5//! authorization -- every caller of these traits must deny on `Err`, and `ReadPolicy` cannot
6//! express "grant everything" at all -- while the resolved `ReadableAudiences`/`ReadScope` is
7//! consumed downstream by `OwnershipRewrite` (the row-level filter) and the call-level guard's UDTF/UDF guards. See
8//! `rust/analytics/src/lakehouse/read_scope.rs`.
9//!
10//! **An audience is an opaque label, not a principal encoding.** `is_valid_audience` and
11//! `PUBLIC_AUDIENCE` define what a name is; `AudienceGrants` defines who may read or mint into
12//! it. No code here derives an audience from a caller's identity -- see `AudienceGrants` for the
13//! reasoning.
14

29/// The conventional shared audience. Not a built-in read grant -- every-principal readability
30/// comes from the seeded `('public', 'read', '*')` row (schema v9), an ordinary grant like any
31/// other, removable the same way.
32pub const PUBLIC_AUDIENCE: &str = "public";
33
34/// `true` if `aud` is a valid audience name: `[A-Za-z0-9_-]{1,255}`, checked in bytes -- the
35/// ASCII-only charset makes byte length and char length agree for every value that can pass, so
36/// this doubles as the length check the `ingestion_api_keys.audience` `VARCHAR(255)` column
37/// enforces on the SQL side.
38///
39/// An audience is an opaque label on data, not a principal encoding: this charset makes an email
40/// (`@`, `.`) or a `user:`/`group:`-prefixed value (`:`) unrepresentable, on purpose -- see
41/// `AudienceGrants` for why there is no derivation from a caller's identity to replace it with.
42///
43/// Deliberately **not** normalizing: no case folding, no trimming. The value is stored and
44/// compared verbatim, which is what makes uniqueness meaningful -- `team-alpha` and `Team-Alpha`
45/// are two audiences, and neither is a typo the system silently repairs into the other.
46pub fn is_valid_audience(aud: &str) -> bool {
47    !aud.is_empty()
48        && aud.len() <= 255
49        && aud
50            .bytes()
51            .all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'-')
52}

54/// Resolves `{prefix}_DEFAULT_AUDIENCE` (falling back to `MICROMEGAS_DEFAULT_AUDIENCE`) --
55/// **the** deployment default audience, defaulting to [`PUBLIC_AUDIENCE`] when unset. Resolved
56/// **once at startup** (`web_server.rs`) so a typo fails fast rather than surfacing as a
57/// per-request 400. Empty or otherwise invalid ⇒ `Err`: with one knob there is no "unset" state
58/// left for an empty string to mean, so a blank value is a misconfiguration, not an opt-out.
59///
60/// Leading/trailing whitespace is trimmed and `warn!`-logged, the same way the other two readers
61/// of this same variable handle it, so one env value can never be accepted by one role and
62/// rejected by another. Trimming keeps a value a deployment template rendered with stray
63/// padding working; the warning stays because padding is far more likely a templating slip than
64/// an intention.
65///
66/// One knob, one meaning: the audience anything that arrives without one gets. There are three
67/// code readers today (not deployment roles -- a role can build more than one of these):
68/// this function (key mint/import), `micromegas_analytics::audience::default_audience_from_env`
69/// (read once by `LakehouseContext::new` and handed to the three Postgres read sites), and the
70/// ingestion HTTP edge's own call in `serve_ingestion`, which resolves the default a credential
71/// with no bound audience is stamped with at write time. This is what
72/// `ingestion_keys.rs::resolve_audience` falls back to on both the `mint` and `import` routes.
73pub fn default_audience_from_env(prefix: &str) -> Result<String> {
74    let var = resolve_prefixed_var(prefix, "DEFAULT_AUDIENCE");
75    let resolved = match std::env::var(&var) {
76        Ok(raw) => {
77            let trimmed = raw.trim();
78            if trimmed != raw {
79                warn!(
80                    "{var}: value {raw:?} has leading or trailing whitespace -- using {trimmed:?}"
81                );
82            }
83            let raw = trimmed.to_string();
84            if is_valid_audience(&raw) {
85                raw
86            } else {
87                return Err(anyhow!(
88                    "{var}: {raw:?} is not a valid audience name -- must match [A-Za-z0-9_-]{{1,255}}"
89                ));
90            }
91        }
92        Err(_) => PUBLIC_AUDIENCE.to_owned(),
93    };
94    info!("{var}: default audience = {resolved}");
95    Ok(resolved)
96}

98/// A principal selector on either axis of an [`AudienceGrants`] entry: `*` (any authenticated
99/// principal), `user:<email>` (matches `AuthContext.email`), or `group:<g>` (matches any entry in
100/// `AuthContext.memberships` -- the caller's resolved, transitive local-group closure). Validated
101/// at parse time; `resolve`/`resolve_audience` never see an unrecognized shape.
102///
103/// `pub` -- `analytics-web-srv`'s admin grant-write route (a separate crate) needs to run this
104/// exact same selector-shape check `parse`/`from_rows` run, not a re-implementation of it.
105pub fn valid_selector(selector: &str) -> bool {
106    selector == "*"
107        || selector
108            .strip_prefix("user:")
109            .is_some_and(|rest| !rest.is_empty())
110        || selector
111            .strip_prefix("group:")
112            .is_some_and(|rest| !rest.is_empty())
113}
114
115/// The grant selectors `caller` matches: a leading `"*"` (every caller matches it), then
116/// `"user:<email>"` when `caller.email` is `Some`, then one `"group:<g>"` per entry in
117/// `caller.memberships`. `pub` -- shared by three consumers: `flight_sql_service_impl.rs`'s
118/// `CallerContext::grant_selectors`, `analytics-web-srv`'s `my_audiences` handler, and the
119/// audience-grant write policy's per-pair hold check (which must strip the leading `"*"` before
120/// binding the result -- a `*` grant is not something a caller individually holds).
121pub fn caller_selectors(caller: &AuthContext) -> Vec<String> {
122    let mut selectors = vec!["*".to_string()];
123    if let Some(email) = &caller.email {
124        selectors.push(format!("user:{email}"));
125    }
126    selectors.extend(caller.memberships.iter().map(|g| format!("group:{g}")));
127    selectors
128}
129
130fn selector_matches(selector: &str, caller: &AuthContext) -> bool {
131    if selector == "*" {
132        return true;
133    }
134    if let Some(email) = selector.strip_prefix("user:") {
135        return caller.email.as_deref() == Some(email);
136    }
137    if let Some(group) = selector.strip_prefix("group:") {
138        return caller.memberships.iter().any(|g| g == group);
139    }
140    // Unreachable given `valid_selector` gates every selector at parse time; false rather than
141    // panicking keeps this fail-closed if that invariant is ever violated.
142    false
143}

145/// One audience's read/mint selector lists, both already validated by [`AudienceGrants::parse`].
146#[derive(Debug, Clone, Default)]
147struct GrantEntry {
148    read: Vec<String>,
149    mint: Vec<String>,
150}

152/// The Rust side of `audience_grants.axis` -- which selector list a DB row

154#[derive(Debug, Clone, Copy, PartialEq, Eq)]
155pub enum GrantAxis {
156    Read,

160/// The bare-array-or-object shape a single grant-map value may take on the wire, before content
161/// validation. An untagged enum rather than a hand-rolled visitor: JSON arrays and objects are
162/// syntactically distinct, so serde can dispatch on shape alone, and any value that is neither
163/// (a number, a string, `null`, ...) fails here with serde's own type-mismatch error.
164#[derive(Debug, Deserialize)]
165#[serde(untagged)]
166enum RawGrantValue {
167    /// Read-only shorthand: a bare array is the audience's `read` list, with an implicitly empty
168    /// `mint` list -- never derived from `read`.
169    Bare(Vec<String>),
170    /// The explicit form, needed only when an audience also grants mint authority.
171    Object(RawGrantObject),
172}
173
174/// The object form of a single grant-map value, broken out of [`RawGrantValue::Object`] so
175/// `#[serde(deny_unknown_fields)]` -- which serde cannot apply directly to an enum variant --
176/// has somewhere to attach: without it, a misspelled key (`"raed"` for `"read"`) would silently
177/// parse into an empty grant instead of failing startup.
178#[derive(Debug, Deserialize)]
179#[serde(deny_unknown_fields)]
180struct RawGrantObject {
181    #[serde(default)]
182    read: Vec<String>,
183    #[serde(default)]
184    mint: Vec<String>,
185}

187/// Deserializes the top-level `{prefix}_AUDIENCE_GRANTS` object while rejecting a repeated key --
188/// `serde_json`'s own `Map` deserialization silently keeps the *last* value for a duplicate key,
189/// which would discard an earlier grant list without a word (see `AudienceGrants`'s doc comment).
190/// Pulling keys one at a time through `MapAccess`, rather than deserializing straight into a
191/// `BTreeMap`, is what makes that duplicate visible before it is lost.
192struct RawAudienceGrants(BTreeMap<String, RawGrantValue>);
193
194impl<'de> Deserialize<'de> for RawAudienceGrants {
195    fn deserialize<D>(deserializer: D) -> std::result::Result<Self, D::Error>
196    where
197        D: serde::Deserializer<'de>,
198    {
199        struct GrantsVisitor;
200
201        impl<'de> Visitor<'de> for GrantsVisitor {
202            type Value = RawAudienceGrants;
203
204            fn expecting(&self, f: &mut fmt::Formatter) -> fmt::Result {
205                write!(f, "a JSON object keyed by audience name")
206            }
207
208            fn visit_map<A>(self, mut map: A) -> std::result::Result<Self::Value, A::Error>
209            where
210                A: MapAccess<'de>,
211            {
212                let mut result = BTreeMap::new();
213                while let Some(key) = map.next_key::<String>()? {
214                    let value: RawGrantValue = map.next_value()?;
215                    if result.contains_key(&key) {
216                        return Err(de::Error::custom(format!(
217                            "duplicate audience key {key:?} in grant map"
218                        )));
219                    }
220                    result.insert(key, value);
221                }
222                Ok(RawAudienceGrants(result))
223            }
224        }
225
226        deserializer.deserialize_map(GrantsVisitor)
227    }
228}

230/// The parsed, validated `{prefix}_AUDIENCE_GRANTS` map: "who can access this audience", keyed by
231/// audience name, one relation per axis (read/mint -- AbAC plan's `group_read_grants` /
232/// `group_mint_grants`, kept as one env map only because there is no store yet to split them
233/// across).
234///
235/// A bare-array value is read-only shorthand: an omitted `"mint"` list is therefore always empty,
236/// never defaulted from `"read"` -- a read grant confers no mint authority, by construction.
237///
238/// `public` is stored the same way as any other audience -- via a `{"public": ["*"]}` entry here,
239/// or, in production, the seeded `('public', 'read', '*')` DB row (schema v9). It has no built-in
240/// grant on either axis; nothing here or in `AudienceReadPolicy::resolve` treats it specially.
241#[derive(Debug, Clone, Default)]
242pub struct AudienceGrants {
243    entries: BTreeMap<String, GrantEntry>,
244}
245
246impl AudienceGrants {
247    /// No grants at all -- every audience, `public` included, is unreadable and unmintable by
248    /// anyone non-admin. Used as the disabled-auth / no-explicit-policy default.
249    pub fn empty() -> Self {
250        Self::default()
251    }
252
253    /// Parses a `{prefix}_AUDIENCE_GRANTS` JSON string directly. Split out from
254    /// [`Self::from_env`] so tests can exercise parsing without mutating the environment.
255    ///
256    /// Validates both axes of the map, not just its JSON shape -- same reason every other knob in
257    /// this crate fails startup on a typo rather than shipping inert: every key must satisfy
258    /// [`is_valid_audience`] (`{"group:everyone": [...]}`, the value operators are migrating
259    /// *from*, can never match a stamped audience once the charset excludes `:`), and every
260    /// selector on either axis must be `*`, `user:<non-empty>`, or `group:<non-empty>` (an
261    /// unprefixed or mis-prefixed selector matches no identity axis and would otherwise be a
262    /// silently-inert entry).
263    pub fn parse(raw: &str) -> Result<Self> {
264        let raw: RawAudienceGrants =
265            serde_json::from_str(raw).map_err(|e| anyhow!("invalid audience grant map: {e}"))?;
266        let mut entries = BTreeMap::new();
267        for (audience, value) in raw.0 {
268            if !is_valid_audience(&audience) {
269                return Err(anyhow!(
270                    "invalid audience grant map: {audience:?} is not a valid audience name -- \
271                     must match [A-Za-z0-9_-]{{1,255}}"
272                ));
273            }
274            let (read, mint) = match value {
275                RawGrantValue::Bare(read) => (read, vec![]),
276                RawGrantValue::Object(RawGrantObject { read, mint }) => (read, mint),
277            };
278            for selector in read.iter().chain(mint.iter()) {
279                if !valid_selector(selector) {
280                    return Err(anyhow!(
281                        "invalid audience grant map: selector {selector:?} for audience \
282                         {audience:?} must be '*', 'user:<id>', or 'group:<id>'"
283                    ));
284                }
285            }
286            entries.insert(audience, GrantEntry { read, mint });
287        }
288        Ok(Self { entries })
289    }
290
291    /// Resolves [`audience_grants_var`] and parses it. Unset ⇒ [`Self::empty`].
292    pub fn from_env(prefix: &str) -> Result<Self> {
293        let var = resolve_prefixed_var(prefix, "AUDIENCE_GRANTS");
294        let grants = match std::env::var(&var) {
295            Ok(raw) if raw.trim().is_empty() => Self::empty(),
296            Ok(raw) => Self::parse(&raw).map_err(|e| anyhow!("{var}: {e}"))?,
297            Err(_) => Self::empty(),
298        };
299        if grants.entries.is_empty() {
300            info!("{var}: no audience grants configured");
301        } else {
302            let names = grants.entries.keys().cloned().collect::<Vec<_>>().join(",");
303            info!("{var}: {} audience grants ({names})", grants.entries.len());
304        }
305        Ok(grants)
306    }
307
308    /// Builds an `AudienceGrants` from `(audience, axis, selector)` triples -- the DB-store
309    /// analogue of [`Self::parse`]. Runs the *same* [`is_valid_audience`]/[`valid_selector`]
310    /// checks `parse` runs, so a row that slipped past `audience_grants`'s own `CHECK`
311    /// constraints (e.g. via a direct `psql` session) still can't reach a policy decision: this
312    /// is the one place both the JSON path and the DB path fail closed on a malformed row.
313    pub fn from_rows(rows: impl IntoIterator<Item = (String, GrantAxis, String)>) -> Result<Self> {
314        let mut entries: BTreeMap<String, GrantEntry> = BTreeMap::new();
315        for (audience, axis, selector) in rows {
316            if !is_valid_audience(&audience) {
317                return Err(anyhow!(
318                    "invalid audience grant row: {audience:?} is not a valid audience name -- \
319                     must match [A-Za-z0-9_-]{{1,255}}"
320                ));
321            }
322            if !valid_selector(&selector) {
323                return Err(anyhow!(
324                    "invalid audience grant row: selector {selector:?} for audience \
325                     {audience:?} must be '*', 'user:<id>', or 'group:<id>'"
326                ));
327            }
328            let entry = entries.entry(audience).or_default();
329            match axis {
330                GrantAxis::Read => entry.read.push(selector),
331                GrantAxis::Mint => entry.mint.push(selector),
332            }
333        }
334        Ok(Self { entries })
335    }
336
337    /// Unions each audience's `read`/`mint` selector lists across `self` and `other`: a selector
338    /// present in either map grants exactly the same access as being present in both. No dedup
339    /// -- `selector_matches` is called with `.any()`, so a selector present in both sources costs
340    /// one redundant comparison, never a wrong answer.
341    ///
342    /// A public utility for combining two `AudienceGrants` maps, exercised by this crate's own
343    /// integration tests (`rust/auth/tests/policy_tests.rs`); `resolve`/`resolve_audience` do
344    /// **not** call it -- they check the env map and the DB store snapshot as two separate
345    /// sources instead, specifically to avoid a per-request deep clone (see the comment in
346    /// [`AudienceReadPolicy::resolve`]).
347    pub fn merge(&self, other: &Self) -> Self {
348        let mut entries = self.entries.clone();
349        for (audience, other_entry) in &other.entries {
350            let entry = entries.entry(audience.clone()).or_default();
351            entry.read.extend(other_entry.read.iter().cloned());
352            entry.mint.extend(other_entry.mint.iter().cloned());
353        }
354        Self { entries }
355    }
356
357    fn readers(&self) -> impl Iterator<Item = (&String, &[String])> {
358        self.entries.iter().map(|(a, e)| (a, e.read.as_slice()))
359    }
360
361    fn mint_selectors(&self, audience: &str) -> &[String] {
362        self.entries
363            .get(audience)
364            .map(|e| e.mint.as_slice())
365            .unwrap_or(&[])
366    }
367}

390/// Resolves the set of audiences an authenticated caller may read.
391///
392/// `resolve` can **never** return "all" -- `ReadScope::All` is deliberately not a policy output.
393/// It is the marker only internal (non-request) callers pass, which is what keeps "who granted
394/// themselves `All`?" a greppable question with a small, auditable answer set. A policy that
395/// wants to be maximally permissive still has to enumerate the audiences that make it so.
396///
397/// `async` and fallible because grants can be resolved from a store -- nested groups, plus
398/// group→audience grants -- which cannot live behind a sync, infallible signature.
399/// `AudienceReadPolicy` is fallible today whenever a [`DbAudienceGrantsSource`] is attached via
400/// `with_store` -- a cold-start store outage with no prior successful load returns `Err`, which
401/// `caller_context` (`flight_sql_service_impl.rs`) maps to `Status::unavailable`. Without a store
402/// attached it still cannot fail.
403///
404/// **Every caller must deny on `Err`.** An `Err` here means "the policy could not be evaluated" --
405/// never soften it into `ReadableAudiences::new(Arc::from([]))` (that would read as a legitimate,
406/// audited fail-closed decision) and never into `ReadScope::All` (a silent fail-open bypass).
407#[async_trait]
408pub trait ReadPolicy: Send + Sync + Debug {
409    /// Resolves `caller`'s readable-audience set, or `Err` if the policy could not be evaluated
410    /// (e.g. a backing store outage once a store-backed policy lands).
411    async fn resolve(&self, caller: &AuthContext) -> Result<ReadableAudiences>;
412}

414/// Resolves the audience a newly minted key may be stamped with.
415///
416/// `requested` is the caller-supplied audience (e.g. `"team-alpha"`); `None` means "no audience
417/// requested and none can be defaulted" -- under the opaque-label model there is no "myself"
418/// audience for a caller's identity to default to, so every implementation must return `Err` for
419/// `requested: None`, admin or not. Async and fallible for the same reason as `ReadPolicy` --
420/// wired by `mint_key` (`analytics-web-srv/src/ingestion_keys.rs`), gated by
421/// `MintGate`/`AuthenticatedUser` rather than `AdminUser`.
422#[async_trait]
423pub trait MintPolicy: Send + Sync + Debug {
424    /// Resolves the audience to stamp on a newly minted key, or `Err` if `requested` is outside
425    /// what `caller` may mint (including `requested: None` -- see the trait doc comment).
426    async fn resolve_audience(
427        &self,
428        caller: &AuthContext,
429        requested: Option<&str>,
430    ) -> Result<String>;
431}

433/// The shipped `ReadPolicy`: a lookup over an [`AudienceGrants`] map -- and, when a
434/// [`DbAudienceGrantsSource`] is attached via `with_store`, a second lookup over that store's
435/// snapshot -- with no identity derivation anywhere. Resolves the readable set:
436///
437/// ```text
438/// { a : "*"            ∈ grants[a].read }
439///   ∪ { a : "user:<email>" ∈ grants[a].read }                  if email present
440///   ∪ { a : "group:<g>"    ∈ grants[a].read for some g ∈ caller.memberships }
441///   ∪ { a : selector       ∈ store.readers(a) matches caller } if a store is attached
442///   ∪ caller.read_audiences                                    // per-key direct grant
443/// ```
444///
445/// There is **no self-audience rule** -- a caller is granted no audience merely for being named
446/// like one: the charset makes an email unrepresentable as an audience name, and keying on
447/// `subject` instead would let an admin mint themselves read access by naming a key after an
448/// audience.
449///
450/// `read_audiences` is folded in here but deliberately **excluded** from
451/// `AudienceMintPolicy`'s mintable set -- a service-account key's ability to *see* an audience
452/// must not imply it can *stamp new keys* with that audience (mint is integrity, read is
453/// confidentiality).
454#[derive(Debug, Clone, Default)]
455pub struct AudienceReadPolicy {
456    grants: AudienceGrants,
457    /// The DB-backed grant store, when this process has one configured -- checked alongside
458    /// `grants` on every `resolve` call, as a separate source rather than merged into it (see
459    /// `resolve`). `None` for every disabled-auth/test caller that has no DB pool to back one.
460    store: Option<Arc<DbAudienceGrantsSource>>,
461}
462
463impl AudienceReadPolicy {
464    /// Builds a policy with an explicit grant map (bypassing env resolution).
465    pub fn new(grants: AudienceGrants) -> Self {
466        Self {
467            grants,
468            store: None,
469        }
470    }
471
472    /// Resolves the grant map from `{prefix}_AUDIENCE_GRANTS` (falling back to
473    /// `MICROMEGAS_AUDIENCE_GRANTS`) via [`AudienceGrants::from_env`]. Unset ⇒ an empty grant map
474    /// ⇒ the readable set degenerates to `read_audiences` alone (`public` included only if a DB
475    /// store attached via `with_store` resolves the seeded row), which `OwnershipRewrite`
476    /// enforces. A malformed grant map is `Err`, not a silently-emptied one, so a startup `?`
477    /// turns a typo into a fail-fast instead of a silently-inactive knob.
478    pub fn from_env(prefix: &str) -> Result<Self> {
479        let grants = AudienceGrants::from_env(prefix)?;
480        Ok(Self {
481            grants,
482            store: None,
483        })
484    }
485
486    /// Attaches (or clears, with `None`) the DB-backed grant store. A builder method, not a
487    /// constructor argument, so `new`/`from_env` keep working unchanged for every caller with no
488    /// DB pool to back one (disabled-auth, tests).
489    pub fn with_store(mut self, store: Option<Arc<DbAudienceGrantsSource>>) -> Self {
490        self.store = store;
491        self
492    }
493}

495#[async_trait]
496impl ReadPolicy for AudienceReadPolicy {
497    async fn resolve(&self, caller: &AuthContext) -> Result<ReadableAudiences> {
498        // `Err` only on a cold-start outage (no snapshot has ever loaded successfully) --
499        // propagated as-is, so `resolve` denies exactly as documented on its own trait.
500        let store_grants = match &self.store {
501            Some(store) => Some(store.current().await?),
502            None => None,
503        };
504        let mut set = BTreeSet::new();
505        // The env map and the store snapshot are checked as two separate sources -- a selector
506        // present in either grants access -- rather than merged into one map, so neither side is
507        // deep-cloned on every request.
508        for (audience, read) in self.grants.readers() {
509            if read.iter().any(|s| selector_matches(s, caller)) {
510                set.insert(audience.clone());
511            }
512        }
513        if let Some(grants) = &store_grants {
514            for (audience, read) in grants.readers() {
515                if read.iter().any(|s| selector_matches(s, caller)) {
516                    set.insert(audience.clone());
517                }
518            }
519        }
520        for audience in &caller.read_audiences {
521            set.insert(audience.clone());
522        }
523        let audiences: Arc<[String]> = set.into_iter().collect::<Vec<_>>().into();
524        Ok(ReadableAudiences::new(audiences))
525    }
526}

528/// The shipped `MintPolicy`. Non-admin callers may mint only an audience in their **mint** set --
529/// `grants[a].mint`, never `grants[a].read` (a read grant confers no mint authority, unchanged
530/// from `AudienceReadPolicy`'s split); being able to *read* `public` via its seeded read grant
531/// does not imply being able to *mint into* it -- that requires its own grant naming `public` in
532/// a `"mint"` list (in production, the seeded `('public', 'mint', '*')` row, schema v9).
533///
534/// `is_admin` callers may mint **any** valid audience, `public` included -- `mint_key`
535/// (`analytics-web-srv/src/ingestion_keys.rs`) delegates authorization to this policy for every
536/// caller, admin included, rather than gating admin callers separately of its own accord (its own
537/// gate, `MintGate`, only enforces the self-service knob against non-admins); without this arm the
538/// only shipped `MintPolicy` could not express the admin mint flow that route depends on; the arm
539/// grants no power the route's gate does not already grant. This is deliberately **asymmetric** to
540/// the read path, where `is_admin` is never a bypass: mint is an integrity decision (who may
541/// stamp a credential), reads are a confidentiality decision (who may see data), and the two axes
542/// are allowed to disagree.
543#[derive(Debug, Clone, Default)]
544pub struct AudienceMintPolicy {
545    grants: AudienceGrants,
546    /// The DB-backed grant store. Built for symmetry with [`AudienceReadPolicy::with_store`] and
547    /// unit-tested, but nothing wires it in production yet: no production code constructs a `dyn
548    /// MintPolicy` today -- see [`with_store`](Self::with_store).
549    store: Option<Arc<DbAudienceGrantsSource>>,
550}
551
552impl AudienceMintPolicy {
553    /// Builds a policy with an explicit grant map.
554    pub fn new(grants: AudienceGrants) -> Self {
555        Self {
556            grants,
557            store: None,
558        }
559    }
560
561    /// Attaches (or clears, with `None`) the DB-backed grant store. Built alongside
562    /// [`AudienceReadPolicy::with_store`] for symmetry -- unlike the read side, no production
563    /// call site attaches a store through this method: `mint_key` constructs its
564    /// `AudienceMintPolicy` via `new`, so mint grants stay resolved by a fresh, uncached point
565    /// query against `audience_grants`, never a `DbAudienceGrantsSource` snapshot.
566    pub fn with_store(mut self, store: Option<Arc<DbAudienceGrantsSource>>) -> Self {
567        self.store = store;
568        self
569    }
570}
571
572#[async_trait]
573impl MintPolicy for AudienceMintPolicy {
574    async fn resolve_audience(
575        &self,
576        caller: &AuthContext,
577        requested: Option<&str>,
578    ) -> Result<String> {
579        let Some(aud) = requested else {
580            return Err(anyhow!("no audience requested and none can be defaulted"));
581        };
582        if caller.is_admin() {
583            return if is_valid_audience(aud) {
584                Ok(aud.to_string())
585            } else {
586                Err(anyhow!(
587                    "malformed audience {aud:?}: must match [A-Za-z0-9_-]{{1,255}}"
588                ))
589            };
590        }
591        // The env map and the store snapshot are checked as two separate sources -- a selector
592        // present in either grants access -- rather than merged into one map, so neither side is
593        // deep-cloned on every request.
594        let store_grants = match &self.store {
595            Some(store) => Some(store.current().await?),
596            None => None,
597        };
598        let granted = self
599            .grants
600            .mint_selectors(aud)
601            .iter()
602            .any(|s| selector_matches(s, caller))
603            || store_grants.as_ref().is_some_and(|grants| {
604                grants
605                    .mint_selectors(aud)
606                    .iter()
607                    .any(|s| selector_matches(s, caller))
608            });
609        if granted {
610            Ok(aud.to_string())
611        } else {
612            Err(anyhow!(
613                "audience {aud:?} is not in the caller's mintable set"
614            ))
615        }
616    }
617}
