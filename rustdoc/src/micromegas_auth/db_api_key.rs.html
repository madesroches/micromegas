1//! DB-backed API key authentication (#1383).
2//!
3//! Moves API keys out of `MICROMEGAS_API_KEYS` (plaintext JSON in an env var) into
4//! two Postgres tables — `ingestion_api_keys` and `analytics_api_keys` (migration
5//! v5, `rust/ingestion/src/sql_migration.rs`) — holding only a SHA-256 hash of each
6//! key, plus a `created_at`/`created_by`/`last_used_at`/`revoked_at`/`revoked_by`
7//! audit trail. [`DbApiKeyAuthProvider`] validates by hash-indexed lookup behind a
8//! short-TTL `moka` cache.
9

24/// Which key table a provider (or the management API) is bound to.
25///
26/// The table name is always a `&'static str` literal — never derived from caller
27/// input — so the SQL built from it (below) can never be injected into.
28#[derive(Clone, Copy, Debug, PartialEq, Eq)]
29pub enum ApiKeyTable {
30    /// `ingestion_api_keys` — write credentials, minted/listed/revoked over HTTP.
31    Ingestion,
32    /// `analytics_api_keys` — read credentials, issued only by hand (see the admin
33    /// runbook); never mintable through the HTTP API.
34    Analytics,
35}
36
37impl ApiKeyTable {
38    /// Static table name.
39    pub fn table_name(self) -> &'static str {
40        match self {
41            ApiKeyTable::Ingestion => "ingestion_api_keys",
42            ApiKeyTable::Analytics => "analytics_api_keys",
43        }
44    }
45}

47/// Cache and audit knobs for [`DbApiKeyAuthProvider`], read from env with defaults.
48#[derive(Clone, Copy, Debug)]
49pub struct DbApiKeyConfig {
50    /// `MICROMEGAS_API_KEY_CACHE_SIZE`, default 10_000.
51    pub cache_size: u64,
52    /// `MICROMEGAS_API_KEY_CACHE_TTL_SECONDS`, default 60. Also the bound on
53    /// revocation latency: a `DELETE` writes `revoked_at` but cannot invalidate a
54    /// remote process's cache, so a revoked key keeps authenticating until this
55    /// TTL elapses.
56    pub cache_ttl_secs: u64,
57    /// `MICROMEGAS_API_KEY_UNKNOWN_CACHE_TTL_SECONDS`, default 10. Shorter than
58    /// `cache_ttl_secs` so a freshly minted key is not masked by an earlier probe
59    /// of that same (not-yet-existing) string for longer than necessary.
60    pub unknown_cache_ttl_secs: u64,
61    /// `MICROMEGAS_API_KEY_UNKNOWN_CACHE_SIZE`, default 10_000.
62    pub unknown_cache_size: u64,
63}
64
65fn resolve_u64(prefix: &str, suffix: &str, default: u64) -> u64 {
66    let (var, raw) = if prefix.is_empty() {
67        let var = format!("MICROMEGAS_{suffix}");
68        let raw = std::env::var(&var).ok();
69        (var, raw)
70    } else {
71        let prefixed = format!("{prefix}_{suffix}");
72        match std::env::var(&prefixed) {
73            Ok(s) => (prefixed, Some(s)),
74            Err(_) => {
75                let var = format!("MICROMEGAS_{suffix}");
76                let raw = std::env::var(&var).ok();
77                (var, raw)
78            }
79        }
80    };
81    match raw {
82        Some(s) => s.parse::<u64>().unwrap_or_else(|_| {
83            micromegas_tracing::warn!("Invalid {var} value '{s}', using default {default}");
84            default
85        }),
86        None => default,
87    }
88}

90impl DbApiKeyConfig {
91    /// Resolves each of the four knobs as `{prefix}_API_KEY_CACHE_*` first, falling
92    /// back to the unprefixed name — the same fallback `provider_with_prefix`
93    /// already uses for `{prefix}_API_KEYS` / `{prefix}_OIDC_CONFIG` /
94    /// `{prefix}_ADMINS`. With an empty prefix this is identical to the unprefixed
95    /// vars, so an unprefixed caller just passes `""`.
96    pub fn from_env_with_prefix(prefix: &str) -> Self {
97        Self {
98            cache_size: resolve_u64(prefix, "API_KEY_CACHE_SIZE", 10_000),
99            cache_ttl_secs: resolve_u64(prefix, "API_KEY_CACHE_TTL_SECONDS", 60),
100            unknown_cache_ttl_secs: resolve_u64(prefix, "API_KEY_UNKNOWN_CACHE_TTL_SECONDS", 10),
101            unknown_cache_size: resolve_u64(prefix, "API_KEY_UNKNOWN_CACHE_SIZE", 10_000),
102        }
103    }
104}

106/// SHA-256 of the full key string. The only place a key is hashed; the mint route
107/// (`rust/public/src/servers/api_keys.rs`) and this provider's lookup go through
108/// the same digest definition.
109pub fn hash_key(key: &str) -> [u8; 32] {
110    let mut hasher = Sha256::new();
111    hasher.update(key.as_bytes());
112    hasher.finalize().into()
113}
114
115/// 256 bits of OS entropy, base64url-nopad, `mmk_`-prefixed. `mmk_` makes minted
116/// keys recognizable to secret scanners; it is cosmetic to validation, since
117/// [`hash_key`] covers the whole string (which is what lets imported legacy keys
118/// of any shape keep working).
119pub fn generate_key() -> String {
120    let mut rng = rand::rng();
121    let bytes: [u8; 32] = rng.random();
122    let encoded = base64::engine::general_purpose::URL_SAFE_NO_PAD.encode(bytes);
123    format!("mmk_{encoded}")
124}

126/// Builds a small, dedicated connection pool for key-store lookups from an
127/// existing pool's connect options — rather than sharing the caller's lake pool
128/// directly. `max_connections(4)`, `acquire_timeout(2s)`: a credential flood or a
129/// DB outage on the key-lookup path can therefore never starve the write path of
130/// connections, and a lookup fails fast into its 503 rather than blocking up to
131/// sqlx's default 30s `acquire_timeout`.
132pub fn dedicated_key_store_pool(lake_pool: &PgPool) -> PgPool {
133    let options = (*lake_pool.connect_options()).clone();
134    PgPoolOptions::new()
135        .max_connections(4)
136        .acquire_timeout(Duration::from_secs(2))
137        .connect_lazy_with(options)
138}

140/// `SELECT EXISTS(SELECT 1 FROM <table> WHERE revoked_at IS NULL)` — used by
141/// `ProviderBuilder::build()` (`rust/auth/src/default_provider.rs`) to decide
142/// whether a non-empty key store counts as "auth configured" on its own. A
143/// missing relation (e.g. a schema that has not reached migration v5 yet)
144/// surfaces as an `Err` here and must be propagated, never treated as "empty".
145pub async fn key_store_has_live_rows(pool: &PgPool, table: ApiKeyTable) -> Result<bool> {
146    let row = sqlx::query(&format!(
147        "SELECT EXISTS(SELECT 1 FROM {} WHERE revoked_at IS NULL) AS has_rows",
148        table.table_name()
149    ))
150    .fetch_one(pool)
151    .await
152    .with_context(|| format!("checking whether {} has any live key", table.table_name()))?;
153    let has_rows: bool = row
154        .try_get("has_rows")
155        .with_context(|| "reading has_rows")?;
156    Ok(has_rows)
157}

159/// hash -> (key_id, name) for a key known to be live.
160struct KeyRow {
161    key_id: uuid::Uuid,
162    name: String,
163}
164
165/// Distinguishes "the DB answered: no such live key" from "the DB could not be
166/// reached at all" — only the latter becomes a [`ProviderUnavailable`] and must
167/// never populate either cache.
168#[derive(thiserror::Error, Debug)]
169enum LookupError {
170    #[error("no such live key")]
171    NotFound,
172    #[error("{0}")]
173    Db(anyhow::Error),
174}

176fn table_tags(table: &'static str) -> &'static micromegas_tracing::property_set::PropertySet {
177    micromegas_tracing::property_set::PropertySet::find_or_create(vec![
178        micromegas_tracing::property_set::Property::new("table", table),
179    ])
180}

182/// Rate-limits the outage `error!` log to at most once per `window_secs` (per
183/// table), checked-and-set via a single `AtomicI64` "last logged at" timestamp —
184/// not once per rejected request. §2's design notes: `DbApiKeyAuthProvider` sits
185/// last in the auth chain, so during an outage every non-env-key, non-JWT request
186/// reaches it; an unconditional `error!` would flood `log_entries` with the
187/// outage's own noise on the highest-volume service in the deployment.
188fn maybe_log_error(last_logged_at: &AtomicI64, window_secs: i64, table: &str, err: &anyhow::Error) {
189    let now = Utc::now().timestamp();
190    let prev = last_logged_at.load(Ordering::Relaxed);
191    if now.saturating_sub(prev) >= window_secs
192        && last_logged_at
193            .compare_exchange(prev, now, Ordering::Relaxed, Ordering::Relaxed)
194            .is_ok()
195    {
196        micromegas_tracing::error!("db_api_key store error (table={table}): {err:#}");
197    }
198}

200/// DB-backed API key authentication provider, bound to one [`ApiKeyTable`].
201///
202/// Every DB error is wrapped in [`ProviderUnavailable`] before being returned and
203/// emits `imetric!("db_api_key_error_count", "count", {table}, 1)` unconditionally
204/// — the metric is unconditional even on requests whose `error!` line was
205/// suppressed by [`maybe_log_error`]'s rate limit.
206pub struct DbApiKeyAuthProvider {
207    pool: PgPool,
208    table: ApiKeyTable,
209    /// hash -> (key_id, name) for keys known to be live.
210    valid: Cache<[u8; 32], Arc<KeyRow>>,
211    /// hash -> () for tokens the DB answered "no such live key" for.
212    unknown: Cache<[u8; 32], ()>,
213    /// Rate-limit window for the outage `error!` log — mirrors `cache_ttl_secs`,
214    /// floored so a zero or very low cache TTL doesn't also disable log
215    /// rate-limiting.
216    error_log_window_secs: i64,
217    last_logged_at: Arc<AtomicI64>,
218}
219
220impl DbApiKeyAuthProvider {
221    /// Creates a new provider bound to `table`, using `pool` for lookups (the
222    /// caller is expected to pass a [`dedicated_key_store_pool`], not the lake
223    /// pool itself).
224    pub fn new(pool: PgPool, table: ApiKeyTable, config: DbApiKeyConfig) -> Self {
225        let valid = Cache::builder()
226            .max_capacity(config.cache_size)
227            .time_to_live(Duration::from_secs(config.cache_ttl_secs))
228            .build();
229        let unknown = Cache::builder()
230            .max_capacity(config.unknown_cache_size)
231            .time_to_live(Duration::from_secs(config.unknown_cache_ttl_secs))
232            .build();
233        Self {
234            pool,
235            table,
236            valid,
237            unknown,
238            error_log_window_secs: config.cache_ttl_secs.max(60) as i64,
239            last_logged_at: Arc::new(AtomicI64::new(0)),
240        }
241    }
242}

244#[async_trait::async_trait]
245impl AuthProvider for DbApiKeyAuthProvider {
246    async fn validate_request(&self, parts: &dyn RequestParts) -> Result<AuthContext> {
247        let token = parts
248            .bearer_token()
249            .ok_or_else(|| anyhow!("missing bearer token"))?;
250        let hash = hash_key(token);
251
252        // A repeated probe of the same bogus token is free after the first
253        // attempt; a flood of distinct bogus tokens still costs one DB round trip
254        // per request (see the plan's "A negative cache" trade-off).
255        if self.unknown.get(&hash).await.is_some() {
256            anyhow::bail!("invalid API token");
257        }
258
259        let pool = self.pool.clone();
260        let table = self.table;
261        let last_logged_at = self.last_logged_at.clone();
262        let window_secs = self.error_log_window_secs;
263
264        // `try_get_with`, not a plain `get`/`insert` pair: among concurrent
265        // callers for the same hash, exactly one runs this loader — the rest
266        // await its result. This is what rate-limits the `last_used_at` write to
267        // once per cache TTL per key, not once per concurrent request.
268        let result = self
269            .valid
270            .try_get_with(hash, async move {
271                let row = sqlx::query(&format!(
272                    "UPDATE {} SET last_used_at = now() WHERE key_hash = $1 AND revoked_at IS NULL RETURNING key_id, name",
273                    table.table_name()
274                ))
275                .bind(&hash[..])
276                .fetch_optional(&pool)
277                .await
278                .map_err(|e| {
279                    let err = anyhow::Error::from(e)
280                        .context(format!("looking up key in {}", table.table_name()));
281                    // Unconditional: fires on every DB error, independent of the
282                    // rate-limited `error!` line below.
283                    micromegas_tracing::imetric!(
284                        "db_api_key_error_count",
285                        "count",
286                        table_tags(table.table_name()),
287                        1_u64
288                    );
289                    maybe_log_error(&last_logged_at, window_secs, table.table_name(), &err);
290                    LookupError::Db(err)
291                })?;
292
293                match row {
294                    Some(row) => {
295                        let key_id: uuid::Uuid = row
296                            .try_get("key_id")
297                            .map_err(|e| LookupError::Db(anyhow::Error::from(e).context("reading key_id")))?;
298                        let name: String = row
299                            .try_get("name")
300                            .map_err(|e| LookupError::Db(anyhow::Error::from(e).context("reading name")))?;
301                        Ok(Arc::new(KeyRow { key_id, name }))
302                    }
303                    None => Err(LookupError::NotFound),
304                }
305            })
306            .await;
307
308        match result {
309            Ok(row) => {
310                micromegas_tracing::trace!(
311                    "db api key validated: table={} key_id={} name={}",
312                    self.table.table_name(),
313                    row.key_id,
314                    row.name
315                );
316                Ok(AuthContext {
317                    subject: row.name.clone(),
318                    email: None,
319                    issuer: "api_key".to_string(),
320                    audience: None,
321                    expires_at: None,
322                    auth_type: AuthType::ApiKey,
323                    // SECURITY: API keys can NEVER be admins.
324                    is_admin: false,
325                    allow_delegation: true,
326                })
327            }
328            // A DB *error* is propagated and cached in neither map: caching an
329            // outage as `unknown` would turn a transient failure into a
330            // TTL-long outage for every affected key.
331            Err(arc_err) => match arc_err.as_ref() {
332                LookupError::NotFound => {
333                    self.unknown.insert(hash, ()).await;
334                    Err(anyhow!("invalid API token"))
335                }
336                LookupError::Db(e) => Err(ProviderUnavailable(anyhow!(
337                    "{} key store unavailable: {e:#}",
338                    self.table.table_name()
339                ))
340                .into()),
341            },
342        }
343    }
