1//! C ABI for the Micromegas telemetry producer stack.
2//!
3//! Exposes init/shutdown/log/metric/flush as flat `extern "C"` functions that any
4//! non-Rust process can load via dlopen (cdylib) or link at build time (staticlib).
5//!
6//! Threading model: all functions are safe to call from any thread.  The transport
7//! runs on its own OS thread inside the library; callers need no async runtime.
8//!
9//! Everything a caller passes (metric names, units, log targets, property keys and values) is
10//! copied into the block being filled and freed with it; nothing is retained for the life of the
11//! process.  Cardinality still matters for queries: every distinct metric name is a series and
12//! every distinct property set is a dictionary entry, so keep them drawn from bounded sets.
13
14#![allow(unsafe_code, clippy::missing_safety_doc)]

27// Name and unit travel as `name` and `unit` property overrides, so one descriptor serves both paths.
28static METRIC_DESC: StaticMetricMetadata = StaticMetricMetadata {
29    lod: Verbosity::Min,
30    name: "",
31    unit: "",
32    target: "capi",
33    file: "micromegas-capi",
34    line: 0,
35};

37/// Reads pair `i` of parallel key/value arrays.  A pair is skipped when its key or value is
38/// null, or when the key is empty or not UTF-8; a value that is not UTF-8 reads as "".
39unsafe fn read_pair<'a>(
40    keys: *const *const c_char,
41    values: *const *const c_char,
42    i: usize,
43) -> Option<(&'a str, &'a str)> {
44    let key_ptr = unsafe { *keys.add(i) };
45    let val_ptr = unsafe { *values.add(i) };
46    if key_ptr.is_null() || val_ptr.is_null() {
47        return None;
48    }
49    let key = match unsafe { CStr::from_ptr(key_ptr) }.to_str() {
50        Ok(s) if !s.is_empty() => s,
51        _ => return None,
52    };
53    let val = unsafe { CStr::from_ptr(val_ptr) }.to_str().unwrap_or("");
54    Some((key, val))
55}

57/// Reads the property arrays of the `*_tagged` functions.  Keys are block-scoped names.
58#[doc(hidden)]
59pub unsafe fn read_properties<'a>(
60    keys: *const *const c_char,
61    values: *const *const c_char,
62    count: c_uint,
63) -> Vec<DynProperty<'a>> {
64    if keys.is_null() || values.is_null() {
65        return vec![];
66    }
67    (0..count as usize)
68        .filter_map(|i| unsafe { read_pair(keys, values, i) })
69        .map(|(key, val)| DynProperty::with_dyn_name(key, val))
70        .collect()
71}

73/// The properties of a metric: the caller's pairs, then `name` and `unit`.  They come last
74/// because `merge_properties` keeps the last entry of a name, so the arguments win over keys of
75/// the same name.
76#[doc(hidden)]
77pub unsafe fn metric_properties<'a>(
78    keys: *const *const c_char,
79    values: *const *const c_char,
80    count: c_uint,
81    name: &'a str,
82    unit: &'a str,
83) -> Vec<DynProperty<'a>> {
84    let mut properties = unsafe { read_properties(keys, values, count) };
85    properties.push(DynProperty::new("name", name));
86    properties.push(DynProperty::new("unit", unit));
87    properties
88}

90unsafe fn metric_name_and_unit<'a>(
91    handle: *mut MmHandle,
92    name: *const c_char,
93    unit: *const c_char,
94) -> Option<(&'a str, &'a str)> {
95    if handle.is_null() || name.is_null() {
96        return None;
97    }
98    let name_str = match unsafe { CStr::from_ptr(name) }.to_str() {
99        Ok(s) if !s.is_empty() => s,
100        _ => return None,
101    };
102    let unit_str = if unit.is_null() {
103        ""
104    } else {
105        unsafe { CStr::from_ptr(unit) }.to_str().unwrap_or("")
106    };
107    Some((name_str, unit_str))
108}

114/// Configuration passed to `mm_init`.
115///
116/// All pointer fields may be null:
117/// - `sink_url` null → reads `MICROMEGAS_TELEMETRY_URL` env var.
118/// - `property_keys` / `property_values` null → no extra process properties.
119/// - Auth is always taken from env: `MICROMEGAS_INGESTION_API_KEY` or OIDC vars.
120#[repr(C)]
121pub struct MmConfig {
122    /// HTTP endpoint of the Micromegas ingestion server, e.g. "http://host:9000".
123    pub sink_url: *const c_char,
124    /// Parallel arrays of NUL-terminated key/value process property strings.
125    pub property_keys: *const *const c_char,
126    pub property_values: *const *const c_char,
127    /// Number of key/value pairs in the arrays above.
128    pub property_count: c_uint,
129}
130
131/// Opaque handle returned by `mm_init` and consumed by `mm_shutdown`.
132pub struct MmHandle {
133    _guard: TelemetryGuard,
134}

140/// Initialize the telemetry system and return an opaque handle.
141///
142/// Returns null on failure (e.g., invalid sink URL, internal error).
143/// The caller must eventually call `mm_shutdown` to flush buffers and free resources.
144#[unsafe(no_mangle)]
145pub unsafe extern "C" fn mm_init(cfg: *const MmConfig) -> *mut MmHandle {
146    let cfg = if cfg.is_null() {
147        return std::ptr::null_mut();
148    } else {
149        unsafe { &*cfg }
150    };
151
152    let mut builder = TelemetryGuardBuilder::default()
153        .with_local_sink_enabled(false)
154        .with_install_tracing_capture(false)
155        .with_auth_from_env();
156
157    if !cfg.sink_url.is_null() {
158        // Always call with_telemetry_sink_url even for empty strings so the
159        // caller can explicitly opt out of the MICROMEGAS_TELEMETRY_URL env var.
160        // The builder will filter empty URLs internally.
161        let url = unsafe { CStr::from_ptr(cfg.sink_url) }
162            .to_str()
163            .unwrap_or_default()
164            .to_owned();
165        builder = builder.with_telemetry_sink_url(url);
166    }
167
168    if cfg.property_count > 0 && !cfg.property_keys.is_null() && !cfg.property_values.is_null() {
169        let mut props = HashMap::new();
170        for i in 0..cfg.property_count as usize {
171            if let Some((key, val)) =
172                unsafe { read_pair(cfg.property_keys, cfg.property_values, i) }
173            {
174                props.insert(key.to_owned(), val.to_owned());
175            }
176        }
177        if !props.is_empty() {
178            builder = builder.with_process_properties(props);
179        }
180    }
181
182    match builder.build() {
183        Ok(guard) => Box::into_raw(Box::new(MmHandle { _guard: guard })),
184        Err(e) => {
185            eprintln!("mm_init: failed to initialize telemetry: {e}");
186            std::ptr::null_mut()
187        }
188    }
189}

191/// Flush all pending events and shut down the telemetry system.
192///
193/// After this call `handle` is invalid and must not be used.
194/// Safe to call with a null handle (no-op).
195#[unsafe(no_mangle)]
196pub unsafe extern "C" fn mm_shutdown(handle: *mut MmHandle) {
197    if !handle.is_null() {
198        unsafe { drop(Box::from_raw(handle)) };
199    }
200}

202/// Emit a log event.
203///
204/// `level` maps to the Micromegas level constants defined in `micromegas.h`
205/// (1=Fatal … 6=Trace).  Null `target` defaults to "capi".  Null `msg` is a no-op.
206#[unsafe(no_mangle)]
207pub unsafe extern "C" fn mm_log(
208    handle: *mut MmHandle,
209    level: c_int,
210    target: *const c_char,
211    msg: *const c_char,
212) {
213    unsafe {
214        mm_log_tagged(
215            handle,
216            level,
217            target,
218            msg,
219            std::ptr::null(),
220            std::ptr::null(),
221            0,
222        )
223    }
224}

226/// Emit a log event with properties.
227///
228/// Same as `mm_log`, plus `property_count` key/value pairs read from the parallel arrays, the
229/// way `mm_init` reads process properties: a pair with a null key or value, or an empty or
230/// non-UTF-8 key, is skipped.  Keys and values are copied into the current block.  `target` and
231/// `level` keys are not applied: the arguments decide those.
232#[unsafe(no_mangle)]
233pub unsafe extern "C" fn mm_log_tagged(
234    handle: *mut MmHandle,
235    level: c_int,
236    target: *const c_char,
237    msg: *const c_char,
238    property_keys: *const *const c_char,
239    property_values: *const *const c_char,
240    property_count: c_uint,
241) {
242    if handle.is_null() {
243        return;
244    }
245    if msg.is_null() {
246        return;
247    }
248    let msg_str = match unsafe { CStr::from_ptr(msg) }.to_str() {
249        Ok(s) => s,
250        Err(_) => return,
251    };
252    let target_str = if target.is_null() {
253        "capi"
254    } else {
255        unsafe { CStr::from_ptr(target) }.to_str().unwrap_or("capi")
256    };
257    let level = match level {
258        1 => Level::Fatal,
259        2 => Level::Error,
260        3 => Level::Warn,
261        4 => Level::Info,
262        5 => Level::Debug,
263        _ => Level::Trace,
264    };
265    let metadata = LogMetadata {
266        level,
267        level_filter: AtomicU32::new(FILTER_LEVEL_UNSET_VALUE),
268        fmt_str: "",
269        target: target_str,
270        module_path: "",
271        file: "",
272        line: 0,
273    };
274    let properties = unsafe { read_properties(property_keys, property_values, property_count) };
275    if properties.is_empty() {
276        dispatch::log_interop(&metadata, format_args!("{msg_str}"));
277    } else {
278        dispatch::log_interop_dyn_tagged(&metadata, &properties, format_args!("{msg_str}"));
279    }
280}

282/// Emit an integer metric.
283///
284/// `name` must not be null or empty; a null `unit` is "".  The strings are copied into the
285/// current block.
286#[unsafe(no_mangle)]
287pub unsafe extern "C" fn mm_metric_i(
288    handle: *mut MmHandle,
289    name: *const c_char,
290    unit: *const c_char,
291    value: u64,
292) {
293    unsafe {
294        mm_metric_i_tagged(
295            handle,
296            name,
297            unit,
298            std::ptr::null(),
299            std::ptr::null(),
300            0,
301            value,
302        )
303    }
304}

306/// Emit an integer metric with properties.
307///
308/// Same as `mm_metric_i`, plus `property_count` key/value pairs read as in `mm_log_tagged`.
309/// A `target` key replaces the default "capi" target; `name` and `unit` keys are overridden by
310/// the arguments.
311#[unsafe(no_mangle)]
312pub unsafe extern "C" fn mm_metric_i_tagged(
313    handle: *mut MmHandle,
314    name: *const c_char,
315    unit: *const c_char,
316    property_keys: *const *const c_char,
317    property_values: *const *const c_char,
318    property_count: c_uint,
319    value: u64,
320) {
321    let Some((name_str, unit_str)) = (unsafe { metric_name_and_unit(handle, name, unit) }) else {
322        return;
323    };
324    let properties = unsafe {
325        metric_properties(
326            property_keys,
327            property_values,
328            property_count,
329            name_str,
330            unit_str,
331        )
332    };
333    dispatch::tagged_integer_metric_dyn(&METRIC_DESC, PropertySet::empty(), &properties, value);
334}

387/// Flush all in-memory log and metric buffers to the transport.
388///
389/// The transport thread will then ship them to the ingestion server.
390/// Safe to call with a null handle (operates on the global dispatch).
391#[unsafe(no_mangle)]
392pub unsafe extern "C" fn mm_flush(_handle: *mut MmHandle) {
393    dispatch::flush_log_buffer();
394    dispatch::flush_metrics_buffer();
