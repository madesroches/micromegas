1//! Audience grant admin API for `analytics-web-srv` (#1489, AbAC Stage 6a).
2//!
3//! Directly mirrors `ingestion_keys.rs`'s shape (`AudienceGrantsState { pool: Option<PgPool> }`,
4//! an `IntoResponse` error enum, `AdminUser`-gated handlers) over the new `audience_grants` table
5//! (migration v7, `rust/ingestion/src/sql_migration.rs`). This is the admin write surface for the
6//! grant store `micromegas-auth::db_audience_grants::DbAudienceGrantsSource` reads from -- the
7//! store's own snapshot cache picks up rows written here within its cache TTL.
8//!
9//! **Duplication, accepted.** This duplicates most of `ingestion_keys.rs`/`analytics_keys.rs`'s
10//! validation/SQL/error shape -- deliberately, per those modules' own doc comments: a generic
11//! abstraction over a handful of near-identical handlers differing only in which table they
12//! target is a shape this codebase already declines elsewhere
13//! (`data_sources.rs`/`screens.rs`/`folders.rs`).
14//!
15//! Every handler in this file is [`AdminUser`]-gated except one:
16//! `GET .../audience-grants/my-audiences` (AbAC Stage 6, #1374) is caller-scoped
17//! ([`AuthenticatedUser`]-gated instead) -- it answers only "which audiences does *this* caller's
18//! own identity match," which carries none of the confidentiality sensitivity `list_grants`'s
19//! admin gate exists for. See that handler's own doc comment.
20

33const DEFAULT_LIMIT: i64 = 100;

35/// `valid_selector` places no charset/length bound on a `group:<id>` selector (a hierarchical
36/// IdP group name can be arbitrarily long), but the `selector` column is `VARCHAR(255)` -- this
37/// check is what turns an over-long selector into a `400 BadRequest` instead of a `500` at the
38/// `INSERT`.
39const MAX_SELECTOR_BYTES: usize = 255;

73/// Errors this API returns.
74///
75/// `Forbidden` (AbAC Stage 6, #1374) is returned only by `/my-audiences`'s own knob-gate check --
76/// the one handler in this file [`AdminUser`] doesn't cover (see that handler's doc comment).
77/// Every other handler here is still `AdminUser`-gated, whose rejection renders as
78/// `AdminRequired`'s own 403 body before the handler even starts running, so `Forbidden` is never
79/// constructed by any of them.
80#[derive(Debug)]
81pub enum AudienceGrantError {
82    /// Request body/query failed validation.
83    BadRequest(String),
84    /// Unknown `(audience, axis, selector)` on `DELETE`.
85    NotFound,
86    /// A DB error.
87    Database(sqlx::Error),
88    /// `state.pool == None` -- the telemetry-DB pool was never configured
89    /// (`MICROMEGAS_SQL_CONNECTION_STRING` unset).
90    NotConfigured,
91    /// The create statement (see [`insert_or_get`]) returned zero rows twice in a row -- an
92    /// internal error, not a caller mistake (see that function's doc comment).
93    Internal(String),
94    /// `/my-audiences`'s off-by-default self-service gate denied a non-admin caller
95    /// (`MICROMEGAS_SELF_SERVICE_MINT` is off).
96    Forbidden(String),
97}

99impl IntoResponse for AudienceGrantError {
100    fn into_response(self) -> Response {
101        match self {
102            AudienceGrantError::BadRequest(msg) => (
103                StatusCode::BAD_REQUEST,
104                Json(ErrorResponse::new("BAD_REQUEST", msg)),
105            )
106                .into_response(),
107            AudienceGrantError::NotFound => (
108                StatusCode::NOT_FOUND,
109                Json(ErrorResponse::new("NOT_FOUND", "grant not found")),
110            )
111                .into_response(),
112            AudienceGrantError::Database(err) => {
113                error!("audience_grants: database error: {err}");
114                (
115                    StatusCode::INTERNAL_SERVER_ERROR,
116                    Json(ErrorResponse::new(
117                        "DATABASE_ERROR",
118                        "internal database error",
119                    )),
120                )
121                    .into_response()
122            }
123            AudienceGrantError::NotConfigured => (
124                StatusCode::SERVICE_UNAVAILABLE,
125                Json(ErrorResponse::new(
126                    "NOT_CONFIGURED",
127                    "audience grant store not configured: set MICROMEGAS_SQL_CONNECTION_STRING",
128                )),
129            )
130                .into_response(),
131            AudienceGrantError::Internal(msg) => {
132                error!("audience_grants: internal error: {msg}");
133                (
134                    StatusCode::INTERNAL_SERVER_ERROR,
135                    Json(ErrorResponse::new("INTERNAL_ERROR", "internal error")),
136                )
137                    .into_response()
138            }
139            AudienceGrantError::Forbidden(msg) => (
140                StatusCode::FORBIDDEN,
141                Json(ErrorResponse::new("FORBIDDEN", msg)),
142            )
143                .into_response(),
144        }
145    }
146}

158fn validate_audience(audience: &str) -> Result<(), AudienceGrantError> {
159    if !is_valid_audience(audience) {
160        return Err(AudienceGrantError::BadRequest(format!(
161            "invalid audience {audience:?}: must match [A-Za-z0-9_-]{{1,255}}"
162        )));
163    }
164    Ok(())
165}
166
167fn validate_axis(axis: &str) -> Result<(), AudienceGrantError> {
168    if axis != "read" && axis != "mint" {
169        return Err(AudienceGrantError::BadRequest(format!(
170            "invalid axis {axis:?}: must be 'read' or 'mint'"
171        )));
172    }
173    Ok(())
174}
175
176fn validate_selector(selector: &str) -> Result<(), AudienceGrantError> {
177    if !valid_selector(selector) {
178        return Err(AudienceGrantError::BadRequest(format!(
179            "invalid selector {selector:?}: must be '*', 'user:<id>', or 'group:<id>'"
180        )));
181    }
182    if selector.len() > MAX_SELECTOR_BYTES {
183        return Err(AudienceGrantError::BadRequest(format!(
184            "selector must be at most {MAX_SELECTOR_BYTES} bytes"
185        )));
186    }
187    Ok(())
188}

207/// Row shape shared by both branches of `insert_or_get`'s CTE (see its doc comment).
208#[derive(sqlx::FromRow)]
209struct UpsertedRow {
210    audience: String,
211    axis: String,
212    selector: String,
213    created_at: DateTime<Utc>,
214    created_by: String,
215    created: bool,
216}
217
218/// One round trip: a CTE that unions the just-inserted row with the pre-existing one, so there is
219/// no window between a failed insert and a re-`SELECT` for a concurrent `DELETE` to invalidate --
220/// unlike `ingestion_keys.rs::import_key`'s insert-then-re-`SELECT`, safe there only because that
221/// table never physically deletes rows.
222///
223/// This single statement can still return **zero rows**: Postgres data-modifying CTEs share one
224/// statement-level snapshot with the query around them, so when two callers race to create the
225/// same new `(audience, axis, selector)`, the loser's `ins` branch resolves to "do nothing" (its
226/// `INSERT ... ON CONFLICT` finds the winner's row already committed) while its plain-`SELECT`
227/// branch still runs against the snapshot taken before the winner committed -- neither branch
228/// sees the row. The caller retries the exact same statement once more (now that the winner's
229/// insert has definitely committed, the loser's re-`SELECT` branch will see it); a second
230/// zero-row result is treated as an internal error rather than looping further.
231const UPSERT_GRANT_SQL: &str = "
232    WITH ins AS (
233        INSERT INTO audience_grants (audience, axis, selector, created_at, created_by)
234        VALUES ($1, $2, $3, now(), $4)
235        ON CONFLICT (audience, axis, selector) DO NOTHING
236        RETURNING audience, axis, selector, created_at, created_by
237    )
238    SELECT audience, axis, selector, created_at, created_by, true AS created FROM ins
239    UNION ALL
240    SELECT audience, axis, selector, created_at, created_by, false AS created
241    FROM audience_grants
242    WHERE audience = $1 AND axis = $2 AND selector = $3
243      AND NOT EXISTS (SELECT 1 FROM ins)";
244
245async fn insert_or_get(
246    pool: &PgPool,
247    audience: &str,
248    axis: &str,
249    selector: &str,
250    created_by: &str,
251) -> Result<UpsertedRow, AudienceGrantError> {
252    for _ in 0..2 {
253        let row = sqlx::query_as::<_, UpsertedRow>(UPSERT_GRANT_SQL)
254            .bind(audience)
255            .bind(axis)
256            .bind(selector)
257            .bind(created_by)
258            .fetch_optional(pool)
259            .await?;
260        if let Some(row) = row {
261            return Ok(row);
262        }
263    }
264    Err(AudienceGrantError::Internal(format!(
265        "audience_grants upsert for ({audience:?}, {axis:?}, {selector:?}) returned no row \
266         after a retry"
267    )))
268}

270/// `POST {base_path}/api/audience-grants` -- creates (or reports the pre-existing) grant row.
271/// `201` when this call created it, `200` when it already existed.
272async fn create_grant(
273    Extension(state): Extension<AudienceGrantsState>,
274    AdminUser(user): AdminUser,
275    Json(body): Json<CreateGrantRequest>,
276) -> Result<(StatusCode, Json<GrantResponse>), AudienceGrantError> {
277    let pool = require_pool(&state)?;
278    validate_audience(&body.audience)?;
279    validate_axis(&body.axis)?;
280    validate_selector(&body.selector)?;
281
282    let created_by = user.email.clone().unwrap_or_else(|| user.subject.clone());
283    let row = insert_or_get(
284        &pool,
285        &body.audience,
286        &body.axis,
287        &body.selector,
288        &created_by,
289    )
290    .await?;
291
292    let status = if row.created {
293        StatusCode::CREATED
294    } else {
295        StatusCode::OK
296    };
297    info!(
298        "audience grant audience={} axis={} selector={} created={} created_by={}",
299        row.audience, row.axis, row.selector, row.created, row.created_by
300    );
301    Ok((
302        status,
303        Json(GrantResponse {
304            audience: row.audience,
305            axis: row.axis,
306            selector: row.selector,
307            created_at: row.created_at,
308            created_by: row.created_by,
309        }),
310    ))
311}

330/// `GET {base_path}/api/audience-grants?audience=&axis=&limit=&offset=` -- lists rows, optionally
331/// filtered, newest first. Admin-gated like the write side: revealing who can read which audience
332/// is itself confidentiality-sensitive.
333async fn list_grants(
334    Extension(state): Extension<AudienceGrantsState>,
335    AdminUser(_user): AdminUser,
336    Query(query): Query<ListQuery>,
337) -> Result<Json<Vec<GrantListEntry>>, AudienceGrantError> {
338    let pool = require_pool(&state)?;
339
340    let limit = query.limit.unwrap_or(DEFAULT_LIMIT);
341    if limit <= 0 {
342        return Err(AudienceGrantError::BadRequest(
343            "limit must be > 0".to_string(),
344        ));
345    }
346    // A read endpoint, so capping is safer than erroring: silently clamp rather than reject
347    // values above MAX_LIMIT.
348    let limit = limit.min(MAX_LIMIT);
349    let offset = query.offset.unwrap_or(0);
350    if offset < 0 {
351        return Err(AudienceGrantError::BadRequest(
352            "offset must be >= 0".to_string(),
353        ));
354    }
355    if let Some(axis) = &query.axis {
356        validate_axis(axis)?;
357    }
358
359    let rows = match (&query.audience, &query.axis) {
360        (None, None) => {
361            sqlx::query_as::<_, GrantListEntry>(
362                "SELECT audience, axis, selector, created_at, created_by
363                 FROM audience_grants
364                 ORDER BY created_at DESC
365                 LIMIT $1 OFFSET $2",
366            )
367            .bind(limit)
368            .bind(offset)
369            .fetch_all(&pool)
370            .await?
371        }
372        (Some(audience), None) => {
373            sqlx::query_as::<_, GrantListEntry>(
374                "SELECT audience, axis, selector, created_at, created_by
375                 FROM audience_grants
376                 WHERE audience = $1
377                 ORDER BY created_at DESC
378                 LIMIT $2 OFFSET $3",
379            )
380            .bind(audience)
381            .bind(limit)
382            .bind(offset)
383            .fetch_all(&pool)
384            .await?
385        }
386        (None, Some(axis)) => {
387            sqlx::query_as::<_, GrantListEntry>(
388                "SELECT audience, axis, selector, created_at, created_by
389                 FROM audience_grants
390                 WHERE axis = $1
391                 ORDER BY created_at DESC
392                 LIMIT $2 OFFSET $3",
393            )
394            .bind(axis)
395            .bind(limit)
396            .bind(offset)
397            .fetch_all(&pool)
398            .await?
399        }
400        (Some(audience), Some(axis)) => {
401            sqlx::query_as::<_, GrantListEntry>(
402                "SELECT audience, axis, selector, created_at, created_by
403                 FROM audience_grants
404                 WHERE audience = $1 AND axis = $2
405                 ORDER BY created_at DESC
406                 LIMIT $3 OFFSET $4",
407            )
408            .bind(audience)
409            .bind(axis)
410            .bind(limit)
411            .bind(offset)
412            .fetch_all(&pool)
413            .await?
414        }
415    };
416
417    Ok(Json(rows))
418}

427/// `DELETE {base_path}/api/audience-grants?audience=&axis=&selector=` -- natural key passed as
428/// query parameters, not path segments: `valid_selector` places no charset restriction on a
429/// `group:<id>` selector (a hierarchical IdP group name can contain `/`, `?`, or other
430/// URL-significant characters), so encoding it as a raw path segment the way every other route's
431/// `Uuid` id does would be unsafe here. `404` if no such row.
432async fn delete_grant(
433    Extension(state): Extension<AudienceGrantsState>,
434    AdminUser(user): AdminUser,
435    Query(query): Query<DeleteGrantQuery>,
436) -> Result<StatusCode, AudienceGrantError> {
437    let pool = require_pool(&state)?;
438    validate_axis(&query.axis)?;
439
440    let result = sqlx::query(
441        "DELETE FROM audience_grants WHERE audience = $1 AND axis = $2 AND selector = $3",
442    )
443    .bind(&query.audience)
444    .bind(&query.axis)
445    .bind(&query.selector)
446    .execute(&pool)
447    .await?;
448
449    if result.rows_affected() == 0 {
450        return Err(AudienceGrantError::NotFound);
451    }
452
453    let deleted_by = user.email.clone().unwrap_or_else(|| user.subject.clone());
454    info!(
455        "deleted audience grant audience={} axis={} selector={} deleted_by={deleted_by}",
456        query.audience, query.axis, query.selector
457    );
458    Ok(StatusCode::NO_CONTENT)
459}

461/// Derives the caller-scoped namespace prefix `micromegas-setup-telemetry` mints fresh audiences
462/// under (AbAC Stage 6, #1374, Design §5). `pub`, not module-private, and pure/sync -- no DB, no
463/// `AuthContext` needed beyond the plain `Option<String>` email -- so the whole sanitization is
464/// unit-testable directly, the same reason `ingestion_keys::resolve_audience` is `pub`.
465///
466/// Takes the local part of `email` (everything before the first `@`), lowercases it, replaces
467/// every character outside `[a-z0-9_-]` with `-`, collapses any run of `-` to a single `-`, trims
468/// leading/trailing `-`, and appends one more `-` as the separator. `None` when `email` is `None`
469/// (a client-credentials service-account caller -- the same condition the lazy claim path itself
470/// gates on) or when sanitizing leaves an empty string.
471///
472/// Deliberately not injective (`alice.smith@x` and `alice-smith@x` both yield `alice-smith-`,
473/// and two different domains with the same local part collide too) -- harmless, since
474/// authorization is still the exact `user:<email>` selector the lazy claim writes, indifferent to
475/// the prefix; a collision just means the second caller's claim attempt hits the ordinary
476/// "audience already exists" denial.
477pub fn mint_prefix_for(email: &Option<String>) -> Option<String> {
478    let email = email.as_deref()?;
479    let local = email.split('@').next().unwrap_or("");
480    let mut sanitized = String::with_capacity(local.len() + 1);
481    let mut last_was_dash = false;
482    for ch in local.chars() {
483        let lower = ch.to_ascii_lowercase();
484        let mapped = if lower.is_ascii_alphanumeric() || lower == '_' || lower == '-' {
485            lower
486        } else {
487            '-'
488        };
489        if mapped == '-' {
490            if last_was_dash {
491                continue;
492            }
493            last_was_dash = true;
494        } else {
495            last_was_dash = false;
496        }
497        sanitized.push(mapped);
498    }
499    let trimmed = sanitized.trim_matches('-');
500    if trimmed.is_empty() {
501        None
502    } else {
503        Some(format!("{trimmed}-"))
504    }
505}

508struct MyAudiencesResponse {
509    is_admin: bool,
510    audiences: Vec<String>,
511    mint_prefix: Option<String>,
512    email: Option<String>,
513}
514
515/// `GET {base_path}/api/audience-grants/my-audiences` -- audiences `caller` may mint into today,
516/// per the DB store's current rows (no cache -- this reads `pool` directly, same as
517/// `list_grants`), plus the caller's own `is_admin` flag, `mint_prefix`, and `email` (AbAC Stage
518/// 6, #1374, Design §5).
519///
520/// Caller-scoped, so [`AuthenticatedUser`] (any authenticated caller), not [`AdminUser`]: unlike
521/// `list_grants`, this can never reveal another principal's selector, only whether *this*
522/// caller's own email/groups match one, plus facts about the caller's own identity. `is_admin`,
523/// `mint_prefix`, and `email` all ride on this response because there is no other route
524/// reachable from a CLI caller (authenticated purely with a Bearer header) that exposes any of
525/// them -- `/auth/me` reads its ID token only from the browser's `id_token` cookie, with no
526/// `Authorization: Bearer` fallback.
527///
528/// Gated on the same off-by-default `self_service_mint_enabled` knob `MintGate` enforces for the
529/// mint route itself, for the same reason: this is new non-admin surface too, and must not widen
530/// on upgrade regardless of the knob. An admin caller is exempt, matching `MintGate`'s own
531/// `!caller.is_admin` condition.
532async fn my_audiences(
533    Extension(state): Extension<AudienceGrantsState>,
534    AuthenticatedUser(caller): AuthenticatedUser,
535) -> Result<Json<MyAudiencesResponse>, AudienceGrantError> {
536    if !caller.is_admin && !state.self_service_mint_enabled {
537        return Err(AudienceGrantError::Forbidden(
538            "self-service minting is disabled".to_string(),
539        ));
540    }
541    let pool = require_pool(&state)?;
542    // Push the selector test into SQL rather than pulling every `mint` row into Rust and
543    // filtering with `selector_matches`: `*` plus the caller's own `user:`/`group:` selectors
544    // are exactly the selectors `selector_matches` would accept, and binding them as an array
545    // lets Postgres do the filtering instead of materializing the whole (monotonically-growing)
546    // mint axis on every call.
547    let mut selectors: Vec<String> = vec!["*".to_string()];
548    if let Some(email) = &caller.email {
549        selectors.push(format!("user:{email}"));
550    }
551    selectors.extend(caller.groups.iter().map(|g| format!("group:{g}")));
552    let rows: Vec<(String,)> = sqlx::query_as(
553        "SELECT DISTINCT audience FROM audience_grants WHERE axis = 'mint' AND selector = ANY($1)",
554    )
555    .bind(&selectors)
556    .fetch_all(&pool)
557    .await?;
558    let mut audiences: Vec<String> = rows.into_iter().map(|(audience,)| audience).collect();
559    audiences.sort();
560    let mint_prefix = mint_prefix_for(&caller.email);
561    let email = caller.email.clone();
562    Ok(Json(MyAudiencesResponse {
563        is_admin: caller.is_admin,
564        audiences,
565        mint_prefix,
566        email,
567    }))
568}

570/// Routes only -- [`AudienceGrantsState`] is layered separately in
571/// `web_server.rs::build_protected_routes`, the same way `analytics_keys_state`/

573pub fn audience_grants_router(base_path: &str) -> Router {
574    Router::new()

576            &format!("{base_path}/api/audience-grants"),

580            &format!("{base_path}/api/audience-grants/my-audiences"),
581            get(my_audiences),
