1//! Audience grant API for `analytics-web-srv`.
2//!
3//! Directly mirrors `ingestion_keys.rs`'s shape (`AudienceGrantsState { pool: Option<PgPool> }`,
4//! an `IntoResponse` error enum) over the new `audience_grants` table (migration v7,
5//! `rust/ingestion/src/sql_migration.rs`). This is the write surface for the grant store
6//! `micromegas-auth::db_audience_grants::DbAudienceGrantsSource` reads from -- the store's own
7//! snapshot cache picks up rows written here within its cache TTL.
8//!
9//! **Duplication, accepted.** This duplicates most of `ingestion_keys.rs`/`analytics_keys.rs`'s
10//! validation/SQL/error shape -- deliberately, per those modules' own doc comments: a generic
11//! abstraction over a handful of near-identical handlers differing only in which table they
12//! target is a shape this codebase already declines elsewhere
13//! (`data_sources.rs`/`screens.rs`/`folders.rs`).
14//!
15//! **Gating, not admin-only.** `create_grant`/`delete_grant` are gated by [`GrantGate`]: an
16//! admin acts unconditionally (exactly the old `AdminUser` behavior); a non-admin is admitted
17//! only when `self_service_mint_enabled` is on (the same knob `MintGate` and `/my-audiences`
18//! already gate on -- sharing an audience is the second half of the self-service feature that
19//! introduced it), and then further constrained inside each handler by a per-pair hold/ownership
20//! check that needs the parsed body/query (see `create_grant`/`delete_grant`'s own doc comments).
21//! `GET .../my-audiences` stays [`AuthenticatedUser`]-gated with its own knob check, unchanged.
22//! `GET .../visible` is [`AuthenticatedUser`]-gated with no admin requirement at all -- it is the
23//! Audience Access page's own list read, and narrows for a non-admin when the knob is off (see
24//! its own doc comment); it carries none of the old paginated `list_grants`'s confidentiality
25//! concern differently -- that route is deleted outright, superseded by the caller-scoped
26//! `list_audience_grants()` SQL table function for ad-hoc auditing (`micromegas-query`) and by
27//! this route for the page's own display.

43/// `valid_selector` places no charset/length bound on a `group:<id>` selector (a hierarchical
44/// IdP group name can be arbitrarily long), but the `selector` column is `VARCHAR(255)` -- this
45/// check is what turns an over-long selector into a `400 BadRequest` instead of a `500` at the
46/// `INSERT`.
47const MAX_SELECTOR_BYTES: usize = 255;
48
49/// Holds the (possibly absent) telemetry-DB pool for the audience-grant routes. `None` only when
50/// `MICROMEGAS_SQL_CONNECTION_STRING` is unset -- the routes stay registered either way and
51/// return 503 per-request in that case, the same always-register-503-when-unconfigured shape
52/// `IngestionKeysState`/`AnalyticsKeysState` use.
53#[derive(Clone)]
54pub struct AudienceGrantsState {
55    pub pool: Option<PgPool>,
56    /// Off-by-default self-service gate. Resolved once at startup from `MICROMEGAS_SELF_SERVICE_MINT`
57    /// (`web_server.rs`, the same knob resolved onto `IngestionKeysState`), default `false`.
58    /// Gates [`GrantGate`] (`create_grant`/`delete_grant`), `GET .../my-audiences`, and the
59    /// non-admin narrowing on `GET .../visible` -- all new non-admin surface introduced by the
60    /// same self-service feature, so none of it must widen on upgrade any more than the mint
61    /// route itself does.
62    pub self_service_mint_enabled: bool,
63    /// `MICROMEGAS_SELF_SERVICE_MAX_GRANTS_PER_CALLER`, default 50 -- caps how many rows one
64    /// non-admin caller may have created in `audience_grants` (`created_by = <caller>`,
65    /// counted across every audience/axis/selector, not just the pair being shared into, but
66    /// excluding the caller's own `user:<email>` rows -- those are claim/self-access rows, not
67    /// shares), checked in `create_grant` before the insert. Mirrors `IngestionKeysState`'s
68    /// `max_claims_per_caller`/`max_keys_per_caller` bounds on the mint side of the same
69    /// `MICROMEGAS_SELF_SERVICE_MINT` knob: without it, a non-admin holding one grant on a pair
70    /// could plant unlimited `group:<arbitrary-id>` rows on it. Best-effort under concurrency,
71    /// not a hard ceiling -- same caveat as those two bounds. Skipped for an admin.
72    pub max_grants_per_caller: i64,
73}

92/// Errors this API returns.
93#[derive(Debug)]
94pub enum AudienceGrantError {
95    /// Request body/query failed validation.
96    BadRequest(String),
97    /// Unknown `(audience, axis, selector)` on `DELETE`.
98    NotFound,
99    /// `create_grant`'s `group:<name>` selector names a group that does not exist. Checked only
100    /// after authorization succeeds (see `create_grant`'s own doc comment), so this is never a
101    /// group-name existence oracle for a caller with no authority on the pair.
102    GroupNotFound(String),
103    /// A DB error.
104    Database(sqlx::Error),
105    /// `state.pool == None` -- the telemetry-DB pool was never configured
106    /// (`MICROMEGAS_SQL_CONNECTION_STRING` unset).
107    NotConfigured,
108    /// The create statement (see [`insert_or_get`]) returned zero rows twice in a row -- an
109    /// internal error, not a caller mistake (see that function's doc comment).
110    Internal(String),
111    /// `GrantGate`'s knob-off denial for a non-admin caller, `/my-audiences`'s identical check,
112    /// a non-admin `create_grant` naming `selector: "*"`, or a non-admin's per-pair
113    /// hold/ownership check failing on `create_grant`/`delete_grant`.
114    Forbidden(String),
115    /// [`GrantGate`]/[`AuthenticatedUser`] found no `AuthContext` extension in the request --
116    /// normally unreachable once routing is wired correctly; mirrors
117    /// `ingestion_keys.rs::IngestionKeyError::Unauthenticated`.
118    Unauthenticated(String),
119}
120
121impl IntoResponse for AudienceGrantError {
122    fn into_response(self) -> Response {
123        match self {
124            AudienceGrantError::BadRequest(msg) => (
125                StatusCode::BAD_REQUEST,
126                Json(ErrorResponse::new("BAD_REQUEST", msg)),
127            )
128                .into_response(),
129            AudienceGrantError::NotFound => (
130                StatusCode::NOT_FOUND,
131                Json(ErrorResponse::new("NOT_FOUND", "grant not found")),
132            )
133                .into_response(),
134            AudienceGrantError::GroupNotFound(msg) => (
135                StatusCode::NOT_FOUND,
136                Json(ErrorResponse::new("NOT_FOUND", msg)),
137            )
138                .into_response(),
139            AudienceGrantError::Database(err) => {
140                error!("audience_grants: database error: {err}");
141                (
142                    StatusCode::INTERNAL_SERVER_ERROR,
143                    Json(ErrorResponse::new(
144                        "DATABASE_ERROR",
145                        "internal database error",
146                    )),
147                )
148                    .into_response()
149            }
150            AudienceGrantError::NotConfigured => (
151                StatusCode::SERVICE_UNAVAILABLE,
152                Json(ErrorResponse::new(
153                    "NOT_CONFIGURED",
154                    "audience grant store not configured: set MICROMEGAS_SQL_CONNECTION_STRING",
155                )),
156            )
157                .into_response(),
158            AudienceGrantError::Internal(msg) => {
159                error!("audience_grants: internal error: {msg}");
160                (
161                    StatusCode::INTERNAL_SERVER_ERROR,
162                    Json(ErrorResponse::new("INTERNAL_ERROR", "internal error")),
163                )
164                    .into_response()
165            }
166            AudienceGrantError::Forbidden(msg) => (
167                StatusCode::FORBIDDEN,
168                Json(ErrorResponse::new("FORBIDDEN", msg)),
169            )
170                .into_response(),
171            AudienceGrantError::Unauthenticated(msg) => (
172                StatusCode::UNAUTHORIZED,
173                Json(ErrorResponse::new("UNAUTHENTICATED", msg)),
174            )
175                .into_response(),
176        }
177    }
178}

195fn require_pool(state: &AudienceGrantsState) -> Result<PgPool, AudienceGrantError> {

199fn validate_audience(audience: &str) -> Result<(), AudienceGrantError> {
200    if !is_valid_audience(audience) {
201        return Err(AudienceGrantError::BadRequest(format!(
202            "invalid audience {audience:?}: must match [A-Za-z0-9_-]{{1,255}}"
203        )));
204    }
205    Ok(())
206}
207
208fn validate_axis(axis: &str) -> Result<(), AudienceGrantError> {
209    if axis != "read" && axis != "mint" {
210        return Err(AudienceGrantError::BadRequest(format!(
211            "invalid axis {axis:?}: must be 'read' or 'mint'"
212        )));
213    }
214    Ok(())
215}
216
217fn validate_selector(selector: &str) -> Result<(), AudienceGrantError> {
218    if !valid_selector(selector) {
219        return Err(AudienceGrantError::BadRequest(format!(
220            "invalid selector {selector:?}: must be '*', 'user:<id>', or 'group:<id>'"
221        )));
222    }
223    if selector.len() > MAX_SELECTOR_BYTES {
224        return Err(AudienceGrantError::BadRequest(format!(
225            "selector must be at most {MAX_SELECTOR_BYTES} bytes"
226        )));
227    }
228    Ok(())
229}

231/// `caller`'s own identity for `created_by`/ownership purposes: email if present, else subject --
232/// the same resolution every mint/revoke/import handler in this crate uses.
233fn caller_identity(caller: &AuthContext) -> String {
234    caller
235        .email
236        .clone()
237        .unwrap_or_else(|| caller.subject.clone())
238}

240/// `FromRequestParts` extractor for `create_grant`/`delete_grant`, modeled directly on
241/// `ingestion_keys.rs`'s `MintGate`: yields the caller's `AuthContext` after enforcing the
242/// off-by-default self-service knob for a non-admin caller. Because this is a
243/// `FromRequestParts` impl, axum runs it -- like `AuthenticatedUser` itself -- before
244/// `Json<CreateGrantRequest>` (or the delete query) ever parses, so a knob-off non-admin caller
245/// is rejected before the body is ever touched (`auth/handlers.rs`'s own rationale for why
246/// `AdminUser` is an extractor, not an in-body check).
247///
248/// This only enforces the knob half of the gate. The per-pair hold/ownership check below it
249/// genuinely needs the parsed body/query, so it stays in each handler, run after `GrantGate` has
250/// already admitted the request.
251struct GrantGate(AuthContext);
252
253impl<S: Send + Sync> FromRequestParts<S> for GrantGate {
254    type Rejection = AudienceGrantError;
255
256    async fn from_request_parts(parts: &mut Parts, state: &S) -> Result<Self, Self::Rejection> {
257        let AuthenticatedUser(caller) = AuthenticatedUser::from_request_parts(parts, state).await?;
258        let grants_state = parts
259            .extensions
260            .get::<AudienceGrantsState>()
261            .cloned()
262            // Reachable only if routing is misconfigured; the 503 body's wording doesn't
263            // literally describe this cause, but a 503 fail-closed beats a panic for a case that
264            // should never happen in a correctly wired router -- mirrors `MintGate`'s identical
265            // `.ok_or(...)`.
266            .ok_or(AudienceGrantError::NotConfigured)?;
267        if !caller.is_admin() && !grants_state.self_service_mint_enabled {
268            return Err(AudienceGrantError::Forbidden(
269                "self-service grant management is disabled".to_string(),
270            ));
271        }
272        Ok(GrantGate(caller))
273    }
274}

293/// Row shape shared by both branches of `insert_or_get`'s CTE (see its doc comment).
294#[derive(sqlx::FromRow)]
295struct UpsertedRow {
296    audience: String,
297    axis: String,
298    selector: String,
299    created_at: DateTime<Utc>,
300    created_by: String,
301    created: bool,
302}
303
304/// One round trip: a CTE that unions the just-inserted row with the pre-existing one, so there is
305/// no window between a failed insert and a re-`SELECT` for a concurrent `DELETE` to invalidate --
306/// unlike `ingestion_keys.rs::import_key`'s insert-then-re-`SELECT`, safe there only because that
307/// table never physically deletes rows.
308///
309/// This single statement can still return **zero rows**: Postgres data-modifying CTEs share one
310/// statement-level snapshot with the query around them, so when two callers race to create the
311/// same new `(audience, axis, selector)`, the loser's `ins` branch resolves to "do nothing" (its
312/// `INSERT ... ON CONFLICT` finds the winner's row already committed) while its plain-`SELECT`
313/// branch still runs against the snapshot taken before the winner committed -- neither branch
314/// sees the row. The caller retries the exact same statement once more (now that the winner's
315/// insert has definitely committed, the loser's re-`SELECT` branch will see it); a second
316/// zero-row result is treated as an internal error rather than looping further.
317const UPSERT_GRANT_SQL: &str = "
318    WITH ins AS (
319        INSERT INTO audience_grants (audience, axis, selector, created_at, created_by)
320        VALUES ($1, $2, $3, now(), $4)
321        ON CONFLICT (audience, axis, selector) DO NOTHING
322        RETURNING audience, axis, selector, created_at, created_by
323    )
324    SELECT audience, axis, selector, created_at, created_by, true AS created FROM ins
325    UNION ALL
326    SELECT audience, axis, selector, created_at, created_by, false AS created
327    FROM audience_grants
328    WHERE audience = $1 AND axis = $2 AND selector = $3
329      AND NOT EXISTS (SELECT 1 FROM ins)";
330
331async fn insert_or_get(
332    tx: &mut sqlx::Transaction<'_, sqlx::Postgres>,
333    audience: &str,
334    axis: &str,
335    selector: &str,
336    created_by: &str,
337) -> Result<UpsertedRow, AudienceGrantError> {
338    for _ in 0..2 {
339        let row = sqlx::query_as::<_, UpsertedRow>(UPSERT_GRANT_SQL)
340            .bind(audience)
341            .bind(axis)
342            .bind(selector)
343            .bind(created_by)
344            .fetch_optional(&mut **tx)
345            .await?;
346        if let Some(row) = row {
347            return Ok(row);
348        }
349    }
350    Err(AudienceGrantError::Internal(format!(
351        "audience_grants upsert for ({audience:?}, {axis:?}, {selector:?}) returned no row \
352         after a retry"
353    )))
354}

356/// Whether `caller` holds `(audience, axis)` via an identity selector -- `SELECT
357/// EXISTS(... selector = ANY($3))` with the leading `"*"` stripped from `caller_selectors`.
358/// A `*` row on the pair makes it publicly readable/mintable, but must not let
359/// every authenticated user plant durable `user:`/`group:` rows on that pair, which would outlive
360/// the `*` grant once an admin deletes it, silently re-widening access the admin thought they'd
361/// just closed.
362async fn caller_holds_pair(
363    pool: &PgPool,
364    audience: &str,
365    axis: &str,
366    caller: &AuthContext,
367) -> Result<bool, AudienceGrantError> {
368    let identity_selectors: Vec<String> = caller_selectors(caller)
369        .into_iter()
370        .filter(|s| s != "*")
371        .collect();
372    let held: bool = sqlx::query_scalar(
373        "SELECT EXISTS(SELECT 1 FROM audience_grants WHERE audience = $1 AND axis = $2 \
374         AND selector = ANY($3))",
375    )
376    .bind(audience)
377    .bind(axis)
378    .bind(&identity_selectors)
379    .fetch_one(pool)
380    .await?;
381    Ok(held)
382}

384/// `POST {base_path}/api/audience-grants` -- creates (or reports the pre-existing) grant row.
385/// `201` when this call created it, `200` when it already existed.
386///
387/// Gated by [`GrantGate`] (the knob check) plus, for a non-admin caller, four further checks
388/// that need the parsed body and so live here rather than in the gate:
389///
390/// 1. `selector` must be `user:…`/`group:…` -- `*` is refused with 403, since a caller who can
391///    read an audience must not be able to open it to every authenticated principal.
392/// 2. `(axis, selector)` must not be the caller's own `mint`/`user:<email>` claim marker -- that
393///    row is what `try_claim_and_mint` (`ingestion_keys.rs`) writes for a real claim, and letting
394///    Share plant a byte-identical one would make it permanently undeletable by its own creator
395///    and silently burn a `max_claims_per_caller` slot.
396/// 3. The caller must hold `(audience, axis)` via an identity selector ([`caller_holds_pair`]).
397///    Delegation is per axis: a `read` grant lets you share `read`, a `mint` grant lets you share
398///    `mint`, and neither confers the other.
399/// 4. The caller must be under `max_grants_per_caller` distinct rows already created
400///    (`created_by = <caller>`, counted across every pair, excluding the caller's own
401///    `user:<email>` identity-selector rows -- see the check's own comment) -- otherwise a
402///    caller who holds one grant could plant unlimited `group:<arbitrary-id>` rows on that pair.
403///
404/// An admin bypasses all four checks entirely, exactly as before.
405async fn create_grant(
406    Extension(state): Extension<AudienceGrantsState>,
407    GrantGate(caller): GrantGate,
408    Json(body): Json<CreateGrantRequest>,
409) -> Result<(StatusCode, Json<GrantResponse>), AudienceGrantError> {
410    let pool = require_pool(&state)?;
411    validate_audience(&body.audience)?;
412    validate_axis(&body.axis)?;
413    validate_selector(&body.selector)?;
414    let created_by = caller_identity(&caller);
415
416    if !caller.is_admin() {
417        if body.selector == "*" {
418            return Err(AudienceGrantError::Forbidden(
419                "non-admin callers may not grant '*' (every authenticated principal) access"
420                    .to_string(),
421            ));
422        }
423        // Refuse a caller planting their own `mint`/`user:<email>` row via Share: it is
424        // byte-identical to the claim marker `try_claim_and_mint` (`ingestion_keys.rs`) writes,
425        // which makes it permanently undeletable by its own creator (`delete_grant`'s own-claim
426        // guard above) and lets it silently consume one of the caller's `max_claims_per_caller`
427        // slots without ever claiming anything. Route them to the mint endpoint instead.
428        if let Some(email) = &caller.email
429            && body.axis == "mint"
430            && body.selector == format!("user:{email}")
431        {
432            return Err(AudienceGrantError::Forbidden(
433                "you cannot create your own mint claim marker via self-service; claim the \
434                 audience through the mint route instead"
435                    .to_string(),
436            ));
437        }
438        if !caller_holds_pair(&pool, &body.audience, &body.axis, &caller).await? {
439            return Err(AudienceGrantError::Forbidden(format!(
440                "you have no {} grant on {} to share",
441                body.axis, body.audience
442            )));
443        }
444        // Per-caller bound (mirrors `IngestionKeysState::max_keys_per_caller` on the mint side
445        // of this same knob): without it, a caller who holds one grant on a pair could plant
446        // unlimited `group:<arbitrary-id>` rows on it -- the PK only blocks exact duplicates.
447        // Best-effort under concurrency, not a hard ceiling -- same caveat as the mint bounds.
448        //
449        // Excludes the caller's own `user:<email>` identity-selector rows: `try_claim_and_mint`
450        // (`ingestion_keys.rs`) writes two such rows (`mint` and `read`) per lazy claim, also
451        // with `created_by = <caller>`, and those rows represent claiming, not sharing -- this
452        // bound exists to cap the latter (per the comment above: unlimited `group:<arbitrary-id>`
453        // rows), not to make claiming and sharing compete for the same budget. Mirrors
454        // `delete_grant`'s own has-email/no-email split for the same `user:<email>` predicate.
455        let grant_count: i64 = if let Some(email) = &caller.email {
456            let own_selector = format!("user:{email}");
457            sqlx::query_scalar(
458                "SELECT COUNT(*) FROM audience_grants WHERE created_by = $1 AND selector <> $2",
459            )
460            .bind(&created_by)
461            .bind(&own_selector)
462            .fetch_one(&pool)
463            .await?
464        } else {
465            sqlx::query_scalar("SELECT COUNT(*) FROM audience_grants WHERE created_by = $1")
466                .bind(&created_by)
467                .fetch_one(&pool)
468                .await?
469        };
470        if grant_count >= state.max_grants_per_caller {
471            return Err(AudienceGrantError::Forbidden(format!(
472                "you have created the maximum number of grants ({})",
473                state.max_grants_per_caller
474            )));
475        }
476    }
477
478    // Checked only now, after authorization succeeds (admin unconditionally, non-admin via the
479    // `caller_holds_pair` check above) -- an existence check run before authorization would let
480    // any authenticated non-admin caller probe for group names by selector alone, even on a pair
481    // they hold no grant on (`GET /api/groups` is `AdminUser`-only).
482    //
483    // The existence check and the insert run in one transaction, with the group's `groups` row
484    // locked first: `delete_group` (`groups.rs`) takes this same `FOR UPDATE` lock on its own row
485    // before scanning for `group:<name>` referrers (see its own comment), so a concurrent delete
486    // of the group and this insert serialize on that row instead of racing -- without that, a
487    // `group:<name>` grant inserted between the plain check and the delete would survive, leaving
488    // a dangling `audience_grants` row that silently re-activates if the name is later recreated.
489    let mut tx = pool.begin().await.map_err(|e| {
490        AudienceGrantError::Internal(format!("starting create-grant transaction: {e:#}"))
491    })?;
492
493    if let Some(group_name) = body.selector.strip_prefix("group:") {
494        let group_exists = sqlx::query("SELECT 1 FROM groups WHERE name = $1 FOR UPDATE")
495            .bind(group_name)
496            .fetch_optional(&mut *tx)
497            .await?
498            .is_some();
499        if !group_exists {
500            return Err(AudienceGrantError::GroupNotFound(format!(
501                "group '{group_name}' not found"
502            )));
503        }
504    }
505
506    let row = insert_or_get(
507        &mut tx,
508        &body.audience,
509        &body.axis,
510        &body.selector,
511        &created_by,
512    )
513    .await?;
514
515    tx.commit().await.map_err(|e| {
516        AudienceGrantError::Internal(format!("committing create-grant transaction: {e:#}"))
517    })?;
518
519    let status = if row.created {
520        StatusCode::CREATED
521    } else {
522        StatusCode::OK
523    };
524    info!(
525        "audience grant audience={} axis={} selector={} created={} created_by={}",
526        row.audience, row.axis, row.selector, row.created, row.created_by
527    );
528    Ok((
529        status,
530        Json(GrantResponse {
531            audience: row.audience,
532            axis: row.axis,
533            selector: row.selector,
534            created_at: row.created_at,
535            created_by: row.created_by,
536        }),
537    ))
538}

540/// Row shape shared by `GET .../visible` and (via `micromegas-query`) the SQL table function's
541/// column order -- kept as its own type since `GET .../visible` is a distinct route with its own

543#[derive(Serialize, sqlx::FromRow)]

559/// `DELETE {base_path}/api/audience-grants?audience=&axis=&selector=` -- natural key passed as
560/// query parameters, not path segments: `valid_selector` places no charset restriction on a
561/// `group:<id>` selector (a hierarchical IdP group name can contain `/`), so encoding it as a raw
562/// path segment the way every other route's `Uuid` id does would be unsafe here. `204` / `404`.
563///
564/// Gated by [`GrantGate`] (the knob check). An admin deletes any row unconditionally, including
565/// their own -- admin access never depends on grant rows, so this is not a restriction on them.
566/// A non-admin may delete only their own direct `user:<email>` row ("remove my access" -- never
567/// offered for `group:`/`*` rows, since those would affect other principals) or a row they
568/// themselves created (the revoke-a-share counterpart of `create_grant`) -- **except**
569/// their own `mint`/`user:<email>` row, which "remove my access" no longer covers. That row is
570/// exactly the shape `try_claim_and_mint`'s (`ingestion_keys.rs`) self-service claim marker takes,
571/// and its own per-caller claim count is read straight from `audience_grants` on that predicate;
572/// letting a non-admin delete it would let them claim, delete, and re-claim to dodge
573/// `max_claims_per_caller` for free. Their `read`-axis own-row removal is unaffected, and so is
574/// removing any row -- `mint` axis included -- that they merely created for someone else rather
575/// than for themselves. If the natural key names no row at all, `404`; if it names a row but no
576/// condition matches, `403` -- but only when the caller holds a grant on `(audience, axis)` at all
577/// ([`caller_holds_pair`]); otherwise `404`, so a non-admin can't use the 403-vs-404 split to
578/// probe for the existence of a grant on a pair they can't otherwise see via
579/// `/visible`/`list_audience_grants()`.
580async fn delete_grant(
581    Extension(state): Extension<AudienceGrantsState>,
582    GrantGate(caller): GrantGate,
583    Query(query): Query<DeleteGrantQuery>,
584) -> Result<StatusCode, AudienceGrantError> {
585    let pool = require_pool(&state)?;
586    validate_axis(&query.axis)?;
587
588    let deleted_by = caller_identity(&caller);
589    let rows_affected = if caller.is_admin() {
590        sqlx::query(
591            "DELETE FROM audience_grants WHERE audience = $1 AND axis = $2 AND selector = $3",
592        )
593        .bind(&query.audience)
594        .bind(&query.axis)
595        .bind(&query.selector)
596        .execute(&pool)
597        .await?
598        .rows_affected()
599    } else if let Some(email) = &caller.email {
600        let own_selector = format!("user:{email}");
601        // The trailing `AND NOT ($2 = 'mint' AND selector = $4)` carves the caller's own
602        // `mint`/`user:<email>` claim-marker row out of *both* preceding arms, not just the
603        // `selector = $4` one -- a self-claim also sets `created_by` to the same caller, so
604        // leaving the `created_by = $5` arm unguarded would still let it through. See this
605        // function's own doc comment for why that row specifically must stay undeletable by its
606        // own subject.
607        sqlx::query(
608            "DELETE FROM audience_grants WHERE audience = $1 AND axis = $2 AND selector = $3 \
609             AND (selector = $4 OR created_by = $5) \
610             AND NOT ($2 = 'mint' AND selector = $4)",
611        )
612        .bind(&query.audience)
613        .bind(&query.axis)
614        .bind(&query.selector)
615        .bind(&own_selector)
616        .bind(&deleted_by)
617        .execute(&pool)
618        .await?
619        .rows_affected()
620    } else {
621        // No email at all -- the own-row arm can never match (there is no `user:<email>`
622        // selector to be), so it's dropped from the predicate entirely rather than bound to a
623        // sentinel that could never occur naturally but would still read as a real comparison.
624        sqlx::query(
625            "DELETE FROM audience_grants WHERE audience = $1 AND axis = $2 AND selector = $3 \
626             AND created_by = $4",
627        )
628        .bind(&query.audience)
629        .bind(&query.axis)
630        .bind(&query.selector)
631        .bind(&deleted_by)
632        .execute(&pool)
633        .await?
634        .rows_affected()
635    };
636
637    if rows_affected == 0 {
638        if caller.is_admin() {
639            return Err(AudienceGrantError::NotFound);
640        }
641        // Distinguish "no such row" (404) from "exists, but not yours" (403) with a follow-up
642        // read -- the delete above already told us this caller's predicate didn't match. This
643        // must not become an unscoped existence oracle: a non-admin who holds no grant on
644        // `(audience, axis)` at all gets 404 regardless of whether the row actually exists,
645        // exactly as they'd learn nothing from `/visible` or `list_audience_grants()` either.
646        // `caller_holds_pair` is the same "what pairs can this caller act on" check
647        // `create_grant`'s hold check runs, not the wider read-visibility query.
648        if !caller_holds_pair(&pool, &query.audience, &query.axis, &caller).await? {
649            return Err(AudienceGrantError::NotFound);
650        }
651        let exists: bool = sqlx::query_scalar(
652            "SELECT EXISTS(SELECT 1 FROM audience_grants WHERE audience = $1 AND axis = $2 \
653             AND selector = $3)",
654        )
655        .bind(&query.audience)
656        .bind(&query.axis)
657        .bind(&query.selector)
658        .fetch_one(&pool)
659        .await?;
660        let is_own_mint_claim_marker = query.axis == "mint"
661            && caller
662                .email
663                .as_deref()
664                .is_some_and(|email| query.selector == format!("user:{email}"));
665        return Err(if !exists {
666            AudienceGrantError::NotFound
667        } else if is_own_mint_claim_marker {
668            AudienceGrantError::Forbidden(
669                "your own mint claim marker cannot be removed via self-service; ask an admin"
670                    .to_string(),
671            )
672        } else {
673            AudienceGrantError::Forbidden(
674                "this grant is neither your own direct access nor one you created".to_string(),
675            )
676        });
677    }
678
679    info!(
680        "deleted audience grant audience={} axis={} selector={} deleted_by={deleted_by}",
681        query.audience, query.axis, query.selector
682    );
683    Ok(StatusCode::NO_CONTENT)
684}

686/// `GET {base_path}/api/audience-grants/visible` -- the unpaginated read backing the Audience
687/// Access page's own list. [`AuthenticatedUser`]-gated, no admin
688/// requirement -- built the same way `/my-audiences` already is: no pagination, reading
689/// `AudienceGrantsState.pool` directly.
690///
691/// **Visibility, by caller:**
692/// - **Admin**: every row.
693/// - **Non-admin, self-service knob on**: every grant on each `(audience, axis)` pair the caller
694///   holds a matching grant on -- the same held-pair `EXISTS` query `list_audience_grants()`
695///   (the SQL table function) runs, bound to the caller's identity selectors with the leading
696///   `"*"` stripped from `caller_selectors` (the same [`caller_holds_pair`] convention): binding
697///   the unfiltered list would match every pair carrying a `*` grant row, leaking every sibling
698///   row on it to a caller who holds nothing there directly.
699/// - **Non-admin, self-service knob off**: narrows to the caller's own rows only
700///   (`selector = ANY(...)`, `"*"` stripped the same way) -- never a sibling's
701///   `selector`/`created_by`. This route (unlike the table function) *can* read the knob, since
702///   it lives in this same crate, and a default knob-off deployment must not hand a browsing
703///   non-admin the wider disclosure the held-pair query carries (the same reasoning
704///   `/my-audiences` already applies).
705///
706/// This function is what makes the two read paths deliberately asymmetric:
707/// `list_audience_grants()` cannot check this knob at all (a different crate/service), so it
708/// always runs the wider held-pair query for a non-admin -- an accepted, intentional exception,
709/// not a bug.
710async fn visible_grants(
711    Extension(state): Extension<AudienceGrantsState>,
712    AuthenticatedUser(caller): AuthenticatedUser,
713) -> Result<Json<Vec<VisibleGrantRow>>, AudienceGrantError> {
714    let pool = require_pool(&state)?;
715    let rows = if caller.is_admin() {
716        sqlx::query_as::<_, VisibleGrantRow>(
717            "SELECT audience, axis, selector, created_at, created_by
718             FROM audience_grants
719             ORDER BY audience, axis, selector",
720        )
721        .fetch_all(&pool)
722        .await?
723    } else if state.self_service_mint_enabled {
724        // `"*"` stripped before binding -- the same `caller_holds_pair` convention. Left in, it
725        // would match every pair carrying a `*` grant row and leak every sibling row on it to a
726        // caller who holds nothing there directly.
727        let identity_selectors: Vec<String> = caller_selectors(&caller)
728            .into_iter()
729            .filter(|s| s != "*")
730            .collect();
731        sqlx::query_as::<_, VisibleGrantRow>(
732            "SELECT g.audience, g.axis, g.selector, g.created_at, g.created_by
733             FROM audience_grants g
734             WHERE EXISTS (
735               SELECT 1 FROM audience_grants h
736               WHERE h.audience = g.audience AND h.axis = g.axis AND h.selector = ANY($1)
737             )
738             ORDER BY g.audience, g.axis, g.selector",
739        )
740        .bind(&identity_selectors)
741        .fetch_all(&pool)
742        .await?
743    } else {
744        // `"*"` stripped before binding -- otherwise this would return every `*`-selector row in
745        // the entire store rather than just the caller's own `user:`/`group:` rows.
746        let identity_selectors: Vec<String> = caller_selectors(&caller)
747            .into_iter()
748            .filter(|s| s != "*")
749            .collect();
750        sqlx::query_as::<_, VisibleGrantRow>(
751            "SELECT audience, axis, selector, created_at, created_by
752             FROM audience_grants
753             WHERE selector = ANY($1)
754             ORDER BY audience, axis, selector",
755        )
756        .bind(&identity_selectors)
757        .fetch_all(&pool)
758        .await?
759    };
760    Ok(Json(rows))
761}

763/// Derives the caller-scoped namespace prefix used to *suggest* a fresh audience name -- the web
764/// app's Mint dialog composes it live before commit, and `micromegas-setup-telemetry`'s CLI uses
765/// it only to render a concrete `--claim <prefix><name>` suggestion in its zero-match error; the
766/// CLI's own `--claim` claims the name it is given verbatim, never prefixing it itself.
767/// `pub`, not module-private, and pure/sync -- no DB, no `AuthContext` needed beyond the plain
768/// `Option<String>` email -- so the whole sanitization is unit-testable directly, the same reason
769/// `ingestion_keys::resolve_audience` is `pub`.
770///
771/// Takes the local part of `email` (everything before the first `@`), lowercases it, replaces
772/// every character outside `[a-z0-9_-]` with `-`, collapses any run of `-` to a single `-`, trims
773/// leading/trailing `-`, and appends one more `-` as the separator. `None` when `email` is `None`
774/// (a client-credentials service-account caller -- the same condition the lazy claim path itself
775/// gates on) or when sanitizing leaves an empty string.
776///
777/// Deliberately not injective (`alice.smith@x` and `alice-smith@x` both yield `alice-smith-`,
778/// and two different domains with the same local part collide too) -- harmless, since
779/// authorization is still the exact `user:<email>` selector the lazy claim writes, indifferent to
780/// the prefix; a collision just means the second caller's claim attempt hits the ordinary
781/// "audience already exists" denial.
782pub fn mint_prefix_for(email: &Option<String>) -> Option<String> {
783    let email = email.as_deref()?;
784    let local = email.split('@').next().unwrap_or("");
785    let mut sanitized = String::with_capacity(local.len() + 1);
786    let mut last_was_dash = false;
787    for ch in local.chars() {
788        let lower = ch.to_ascii_lowercase();
789        let mapped = if lower.is_ascii_alphanumeric() || lower == '_' || lower == '-' {
790            lower
791        } else {
792            '-'
793        };
794        if mapped == '-' {
795            if last_was_dash {
796                continue;
797            }
798            last_was_dash = true;
799        } else {
800            last_was_dash = false;
801        }
802        sanitized.push(mapped);
803    }
804    let trimmed = sanitized.trim_matches('-');
805    if trimmed.is_empty() {
806        None
807    } else {
808        Some(format!("{trimmed}-"))
809    }
810}

813struct MyAudiencesResponse {
814    is_admin: bool,
815    audiences: Vec<String>,
816    mint_prefix: Option<String>,
817    email: Option<String>,
818    /// `"{audience}:{axis}"` for every `(audience, axis)` pair `caller` holds a grant on via an
819    /// identity selector -- i.e. the pairs [`caller_holds_pair`] would return `true` for. Lets
820    /// the Audience Access page tell "a pair I hold" apart from "a pair I can merely see" (e.g.
821    /// visible only via a `*` row, or a `group:` row the caller isn't actually a member of --
822    /// the client has no group-membership info of its own to make that call). Always empty for
823    /// an admin: `isAdmin` already grants Share everywhere on the client, and every pair is a
824    /// held pair for an admin's own writes anyway.
825    held_pairs: Vec<String>,
826    /// The caller's resolved, transitive local-group membership -- straight off
827    /// `AuthContext.memberships`, no query. Lets the CLI and the Audience Access page show why a
828    /// caller holds a `group:` grant.
829    groups: Vec<String>,
830}

832/// `GET {base_path}/api/audience-grants/my-audiences` -- audiences `caller` may mint into today,
833/// per the DB store's current rows (no cache -- this reads `pool` directly), plus the caller's
834/// own `is_admin` flag, `mint_prefix`, and `email`.
835///
836/// Caller-scoped, so [`AuthenticatedUser`] (any authenticated caller), not admin-gated: this can
837/// never reveal another principal's selector, only whether *this* caller's own email/groups
838/// match one, plus facts about the caller's own identity. `is_admin`, `mint_prefix`, and `email`
839/// all ride on this response because there is no other route reachable from a CLI caller
840/// (authenticated purely with a Bearer header) that exposes any of them -- `/auth/me` reads its
841/// ID token only from the browser's `id_token` cookie, with no `Authorization: Bearer` fallback.
842///
843/// Gated on the same off-by-default `self_service_mint_enabled` knob `MintGate`/`GrantGate`
844/// enforce, for the same reason: this is new non-admin surface too, and must not widen on
845/// upgrade regardless of the knob. An admin caller is exempt, matching `MintGate`'s own
846/// `!caller.is_admin()` condition.
847async fn my_audiences(
848    Extension(state): Extension<AudienceGrantsState>,
849    AuthenticatedUser(caller): AuthenticatedUser,
850) -> Result<Json<MyAudiencesResponse>, AudienceGrantError> {
851    if !caller.is_admin() && !state.self_service_mint_enabled {
852        return Err(AudienceGrantError::Forbidden(
853            "self-service minting is disabled".to_string(),
854        ));
855    }
856    let pool = require_pool(&state)?;
857    // Push the selector test into SQL rather than pulling every `mint` row into Rust and
858    // filtering with `selector_matches`: `*` plus the caller's own `user:`/`group:` selectors
859    // are exactly the selectors `selector_matches` would accept, and binding them as an array
860    // lets Postgres do the filtering instead of materializing the whole (monotonically-growing)
861    // mint axis on every call. `caller_selectors` is the shared builder for this exact list.
862    let selectors = caller_selectors(&caller);
863    let rows: Vec<(String,)> = sqlx::query_as(
864        "SELECT DISTINCT audience FROM audience_grants WHERE axis = 'mint' AND selector = ANY($1)",
865    )
866    .bind(&selectors)
867    .fetch_all(&pool)
868    .await?;
869    let mut audiences: Vec<String> = rows.into_iter().map(|(audience,)| audience).collect();
870    audiences.sort();
871    let mint_prefix = mint_prefix_for(&caller.email);
872    let email = caller.email.clone();
873
874    // Ground truth for `canShareRow` on the client: the caller's own distinct held
875    // `(audience, axis)` pairs, by the exact
876    // same rule `caller_holds_pair` checks -- `*` filtered out of `caller_selectors`, matching
877    // that write-hold-check convention (a `*` row must not let a non-admin claim they "hold"
878    // every pair). An admin needs none of this on the client, so skip the query entirely.
879    let held_pairs = if caller.is_admin() {
880        Vec::new()
881    } else {
882        let identity_selectors: Vec<String> = caller_selectors(&caller)
883            .into_iter()
884            .filter(|s| s != "*")
885            .collect();
886        let rows: Vec<(String, String)> = sqlx::query_as(
887            "SELECT DISTINCT audience, axis FROM audience_grants WHERE selector = ANY($1)",
888        )
889        .bind(&identity_selectors)
890        .fetch_all(&pool)
891        .await?;
892        rows.into_iter()
893            .map(|(audience, axis)| format!("{audience}:{axis}"))
894            .collect()
895    };
896
897    let groups = caller.memberships.to_vec();
898
899    Ok(Json(MyAudiencesResponse {
900        is_admin: caller.is_admin(),
901        audiences,
902        mint_prefix,
903        email,
904        held_pairs,
905        groups,
906    }))
907}

909/// Routes only -- [`AudienceGrantsState`] is layered separately in
910/// `web_server.rs::build_protected_routes`, the same way `analytics_keys_state`/
911/// `ingestion_keys_state` are.
912///
913/// The collection path (`POST`/`DELETE {base_path}/api/audience-grants`) does not answer `GET`
914/// -- the Audience Access page's own list lives at `GET {base_path}/api/audience-grants/visible`
915/// instead, and ad-hoc auditing goes through `micromegas-query`/`list_audience_grants()`.
916pub fn audience_grants_router(base_path: &str) -> Router {
917    Router::new()
918        .route(
919            &format!("{base_path}/api/audience-grants"),
920            post(create_grant).delete(delete_grant),
921        )
922        .route(
923            &format!("{base_path}/api/audience-grants/visible"),
924            get(visible_grants),
925        )
926        .route(
927            &format!("{base_path}/api/audience-grants/my-audiences"),
928            get(my_audiences),
929        )
