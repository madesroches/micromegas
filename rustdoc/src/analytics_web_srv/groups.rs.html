1//! Group admin API for `analytics-web-srv`.
2//!
3//! Mirrors `analytics_keys.rs`/`audience_grants.rs`'s shape (`GroupsState { pool: Option<PgPool> }`,
4//! an `IntoResponse` error enum, admin-gated) over the `groups`/`group_members` tables
5//! (migration v10, `rust/ingestion/src/sql_migration.rs`). The two read routes stay
6//! [`AdminUser`]-gated directly; the four mutation routes go through `GroupAdminGate`, a thin
7//! wrapper around the same `caller.is_admin` check that additionally emits a control-plane audit
8//! record on rejection -- delegating group ownership later means widening that one gate, not
9//! this module's routing or handler shapes.
10

56/// Errors this API returns.
57///
58/// No `Forbidden` variant: the admin gate is the [`AdminUser`] extractor, whose rejection
59/// renders as `AdminRequired`'s own 403 body before any handler here starts running.
60pub enum GroupsError {
61    BadRequest(String),
62    NotFound,
63    /// `add_member`'s `group:<name>` member names a nested group that does not exist. Distinct
64    /// from `NotFound` (the *target* group missing) so the client can tell the two apart instead
65    /// of both rendering as the same generic "group not found". The message names the missing
66    /// nested group.
67    NestedGroupNotFound(String),
68    /// `add_member`'s `key:<id>` member names no live analytics API key.
69    KeyNotFound(String),
70    /// `409` -- the group already exists, is `admins` (undeletable), is still referenced, would
71    /// create a cycle, or would leave `admins` unreachable by any principal. The message names
72    /// which.
73    Conflict(String),
74    Database(sqlx::Error),
75    NotConfigured,
76    /// A non-DB internal error.
77    Internal(String),
78}
79
80impl IntoResponse for GroupsError {
81    fn into_response(self) -> Response {
82        match self {
83            GroupsError::BadRequest(msg) => (
84                StatusCode::BAD_REQUEST,
85                Json(ErrorResponse::new("BAD_REQUEST", msg)),
86            )
87                .into_response(),
88            GroupsError::NotFound => (
89                StatusCode::NOT_FOUND,
90                Json(ErrorResponse::new("NOT_FOUND", "group not found")),
91            )
92                .into_response(),
93            GroupsError::NestedGroupNotFound(msg) | GroupsError::KeyNotFound(msg) => (
94                StatusCode::NOT_FOUND,
95                Json(ErrorResponse::new("NOT_FOUND", msg)),
96            )
97                .into_response(),
98            GroupsError::Conflict(msg) => (
99                StatusCode::CONFLICT,
100                Json(ErrorResponse::new("CONFLICT", msg)),
101            )
102                .into_response(),
103            GroupsError::Database(err) => {
104                error!("groups: database error: {err}");
105                (
106                    StatusCode::INTERNAL_SERVER_ERROR,
107                    Json(ErrorResponse::new(
108                        "DATABASE_ERROR",
109                        "internal database error",
110                    )),
111                )
112                    .into_response()
113            }
114            GroupsError::NotConfigured => (
115                StatusCode::SERVICE_UNAVAILABLE,
116                Json(ErrorResponse::new(
117                    "NOT_CONFIGURED",
118                    "group store not configured: set MICROMEGAS_SQL_CONNECTION_STRING",
119                )),
120            )
121                .into_response(),
122            GroupsError::Internal(msg) => {
123                error!("groups: internal error: {msg}");
124                (
125                    StatusCode::INTERNAL_SERVER_ERROR,
126                    Json(ErrorResponse::new("INTERNAL_ERROR", "internal error")),
127                )
128                    .into_response()
129            }
130        }
131    }
132}

140impl AuditOutcome for GroupsError {
141    fn audit_outcome(&self) -> (&'static str, Option<String>) {
142        match self {
143            GroupsError::BadRequest(msg) => ("denied", Some(msg.clone())),
144            GroupsError::NotFound => ("denied", Some("group not found".to_string())),
145            GroupsError::NestedGroupNotFound(msg) => ("denied", Some(msg.clone())),
146            GroupsError::KeyNotFound(msg) => ("denied", Some(msg.clone())),
147            GroupsError::Conflict(msg) => ("denied", Some(msg.clone())),
148            // The raw `sqlx::Error` text must not reach the record -- it can carry
149            // SQL/connection detail -- so this uses the same fixed string `IntoResponse` already
150            // returns to the client instead. `Internal` gets the same treatment: it is built from
151            // `sqlx::Error` on several audited code paths and would otherwise leak the same detail.
152            GroupsError::Database(_) => ("error", Some("internal database error".to_string())),
153            GroupsError::NotConfigured => (
154                "error",
155                Some(
156                    "group store not configured: set MICROMEGAS_SQL_CONNECTION_STRING".to_string(),
157                ),
158            ),
159            GroupsError::Internal(_) => ("error", Some("internal error".to_string())),
160        }
161    }
162}

168fn validate_group_name(name: &str) -> Result<(), GroupsError> {
169    if !is_valid_group_name(name) {
170        return Err(GroupsError::BadRequest(format!(
171            "invalid group name {name:?}: must match [A-Za-z0-9_-]{{1,255}}"
172        )));
173    }
174    Ok(())
175}

186/// Derives the mutation-audit action from the request's method and route template
187/// (`axum::extract::MatchedPath`, read out of `parts.extensions` -- not `parts.uri.path()`,
188/// which would misclassify a group literally named `members`, a name `is_valid_group_name`
189/// allows). Every group mutation route matches one of the four pairs below, so an unmatched
190/// pair -- including a missing `MatchedPath` extension -- is unreachable through the router;
191/// mapped to `create_group` rather than adding an error path for that case, mirroring
192/// `GrantGate`'s identical fallback.
193fn group_gate_action(parts: &Parts) -> &'static str {
194    let matched_path = parts
195        .extensions
196        .get::<MatchedPath>()
197        .map(MatchedPath::as_str);
198    match (&parts.method, matched_path) {
199        (&Method::DELETE, Some(path)) if path.ends_with("/members") => action::REMOVE_MEMBER,
200        (&Method::POST, Some(path)) if path.ends_with("/members") => action::ADD_MEMBER,
201        (&Method::DELETE, Some(_)) => action::DELETE_GROUP,
202        _ => action::CREATE_GROUP,
203    }
204}

206/// `FromRequestParts` extractor for the four group-mutation routes: delegates to [`AdminUser`],
207/// and on rejection emits a `"denied"` mutation-audit record before returning `AdminRequired`
208/// unchanged, so the HTTP response stays byte-identical to today's. Deliberately not folded into
209/// `AdminUser` itself: that extractor gates every admin route in this crate (data sources,
210/// screens, folders, analytics keys, ingestion keys), and auditing all of them under the
211/// control-plane target is not what this record is for.
212struct GroupAdminGate(ValidatedUser);
213
214impl<S: Send + Sync> FromRequestParts<S> for GroupAdminGate {
215    type Rejection = AdminRequired;
216
217    async fn from_request_parts(parts: &mut Parts, state: &S) -> Result<Self, Self::Rejection> {
218        let action = group_gate_action(parts);
219        let ClientIp(client_ip) = ClientIp::from_request_parts(parts, state)
220            .await
221            .expect("ClientIp is infallible");
222        match AdminUser::from_request_parts(parts, state).await {
223            Ok(AdminUser(user)) => Ok(GroupAdminGate(user)),
224            Err(rejection) => {
225                let (actor, is_admin) = match parts.extensions.get::<ValidatedUser>() {
226                    Some(user) => (caller_identity(user), user.is_admin),
227                    None => ("unauthenticated".to_string(), false),
228                };
229                MutationAudit::new(action, actor, is_admin, client_ip)
230                    .emit_gate_outcome("denied", "Admin access required");
231                Err(rejection)
232            }
233        }
234    }
235}

246/// `GET {base_path}/api/groups` -- every group with its member count.
247async fn list_groups(
248    Extension(state): Extension<GroupsState>,
249    AdminUser(_user): AdminUser,
250) -> Result<Json<Vec<GroupSummary>>, GroupsError> {
251    let pool = require_pool(&state)?;
252    let rows = sqlx::query_as::<_, GroupSummary>(
253        "SELECT g.name, g.description, g.created_at, g.created_by,
254                COUNT(m.member) AS member_count
255         FROM groups g
256         LEFT JOIN group_members m ON m.group_name = g.name
257         GROUP BY g.name, g.description, g.created_at, g.created_by
258         ORDER BY g.name",
259    )
260    .fetch_all(&pool)
261    .await?;
262    Ok(Json(rows))
263}

291async fn create_group_inner(
292    state: GroupsState,
293    user: ValidatedUser,
294    body: CreateGroupRequest,
295) -> Result<(StatusCode, Json<GroupSummary>), GroupsError> {
296    let pool = require_pool(&state)?;
297    validate_group_name(&body.name)?;
298    let created_by = caller_identity(&user);
299
300    let row = sqlx::query_as::<_, GroupSummary>(
301        "INSERT INTO groups (name, description, created_at, created_by)
302         VALUES ($1, $2, now(), $3)
303         ON CONFLICT (name) DO NOTHING
304         RETURNING name, description, 0::bigint AS member_count, created_at, created_by",
305    )
306    .bind(&body.name)
307    .bind(&body.description)
308    .bind(&created_by)
309    .fetch_optional(&pool)
310    .await?

312
313    Ok((StatusCode::CREATED, Json(row)))

337async fn delete_group_inner(state: GroupsState, name: String) -> Result<StatusCode, GroupsError> {
338    let pool = require_pool(&state)?;
339    if name == ADMINS_GROUP {
340        return Err(GroupsError::Conflict(
341            "the admins group cannot be deleted".to_string(),
342        ));
343    }
344
345    // The referrer checks and the delete run in one transaction, with the `groups` row for
346    // `name` locked first: `add_member`'s nested-group check takes the same `FOR UPDATE` lock on
347    // that row before inserting a `group:<name>` member row (see its own comment), so a
348    // concurrent nest-in and this delete serialize on this row instead of racing -- without that,
349    // a `group:<name>` reference inserted between the plain check and the delete would survive,
350    // and silently re-activate if the name is later recreated.
351    let mut tx = pool
352        .begin()
353        .await
354        .map_err(|e| GroupsError::Internal(format!("starting delete transaction: {e:#}")))?;
355    sqlx::query("SELECT 1 FROM groups WHERE name = $1 FOR UPDATE")
356        .bind(&name)
357        .fetch_optional(&mut *tx)
358        .await?;
359
360    let nesting_selector = format!("group:{name}");
361    let nested_into: Vec<String> = sqlx::query_scalar(
362        "SELECT group_name FROM group_members WHERE member = $1 ORDER BY group_name",
363    )
364    .bind(&nesting_selector)
365    .fetch_all(&mut *tx)
366    .await?;
367    let granted_to: Vec<String> = sqlx::query_scalar(
368        "SELECT audience FROM audience_grants WHERE selector = $1 ORDER BY audience",
369    )
370    .bind(&nesting_selector)
371    .fetch_all(&mut *tx)
372    .await?;
373    if !nested_into.is_empty() || !granted_to.is_empty() {
374        let mut referrers = Vec::new();
375        if !nested_into.is_empty() {
376            referrers.push(format!("nested into: {}", nested_into.join(", ")));
377        }
378        if !granted_to.is_empty() {
379            referrers.push(format!("granted audiences: {}", granted_to.join(", ")));
380        }
381        return Err(GroupsError::Conflict(format!(
382            "group {name:?} is still referenced ({})",
383            referrers.join("; ")
384        )));
385    }
386
387    let rows_affected = sqlx::query("DELETE FROM groups WHERE name = $1")
388        .bind(&name)
389        .execute(&mut *tx)
390        .await?
391        .rows_affected();
392    if rows_affected == 0 {
393        return Err(GroupsError::NotFound);
394    }
395    tx.commit()
396        .await
397        .map_err(|e| GroupsError::Internal(format!("committing delete transaction: {e:#}")))?;
398    Ok(StatusCode::NO_CONTENT)
399}

412/// `GET {base_path}/api/groups/{name}/members`.
413async fn list_members(
414    Extension(state): Extension<GroupsState>,
415    AdminUser(_user): AdminUser,
416    Path(name): Path<String>,
417) -> Result<Json<Vec<MemberRow>>, GroupsError> {
418    let pool = require_pool(&state)?;
419    let exists: bool = sqlx::query_scalar("SELECT EXISTS(SELECT 1 FROM groups WHERE name = $1)")
420        .bind(&name)
421        .fetch_one(&pool)

425    }

429         FROM group_members m
430         LEFT JOIN analytics_api_keys k ON m.member = 'key:' || k.key_id::text
431         WHERE m.group_name = $1 ORDER BY m.member",
432    )
433    .bind(&name)
434    .fetch_all(&pool)
435    .await?;
436    Ok(Json(rows))
437}

444/// `POST {base_path}/api/groups/{name}/members` -- validates `valid_selector` + 255 bytes; for
445/// `group:X`, `404` if `X` does not exist and `409` if `nesting_would_cycle`; for `key:K`, `400`
446/// into `admins` and `404` unless `K` is a live analytics key; `201` created /
447/// `200` already existed, via the same CTE `insert_or_get` UPSERT pattern `audience_grants.rs`
448/// uses (see that module's `insert_or_get` doc comment).
449async fn add_member(
450    Extension(state): Extension<GroupsState>,
451    GroupAdminGate(user): GroupAdminGate,
452    client_ip: ClientIp,
453    Path(name): Path<String>,
454    Json(mut body): Json<AddMemberRequest>,
455) -> Result<(StatusCode, Json<MemberRow>), GroupsError> {
456    body.member = canonicalize_key_selector(&body.member);
457    let audit = MutationAudit::new(
458        action::ADD_MEMBER,
459        caller_identity(&user),
460        user.is_admin,
461        client_ip.0,
462    )
463    .group(&name)
464    .member(&body.member);
465    let result = add_member_inner(state, user, name, body).await;
466    let audit = match &result {
467        Ok((status, _)) => audit.created(*status == StatusCode::CREATED),
468        Err(_) => audit,
469    };
470    audit.emit(&result);
471    result
472}
473
474async fn add_member_inner(
475    state: GroupsState,
476    user: ValidatedUser,
477    name: String,
478    body: AddMemberRequest,
479) -> Result<(StatusCode, Json<MemberRow>), GroupsError> {
480    let pool = require_pool(&state)?;
481    if !valid_selector(&body.member) {
482        return Err(GroupsError::BadRequest(format!(
483            "invalid member {:?}: must be '*', 'user:<id>', 'group:<id>', or 'key:<key_id>'",
484            body.member
485        )));
486    }
487    if name == ADMINS_GROUP && body.member.starts_with("key:") {
488        return Err(GroupsError::BadRequest(
489            "API keys cannot be members of admins".to_string(),
490        ));
491    }
492    if body.member.len() > MAX_MEMBER_BYTES {
493        return Err(GroupsError::BadRequest(format!(
494            "member must be at most {MAX_MEMBER_BYTES} bytes"
495        )));
496    }
497
498    // The nested-group existence check and the insert run in one transaction, with the `groups`
499    // row(s) involved locked first: `delete_group` takes this same `FOR UPDATE` lock on its own
500    // row before scanning for `group:<name>` referrers (see its own comment), so a concurrent
501    // delete of the nested group and this insert serialize on that row instead of racing --
502    // without that, a `group:<nested>` member row could commit after `delete_group`'s referrer
503    // scan already found none, leaving `group_members` pointing at a group name that no longer
504    // exists (its FK is on `group_name`, not `member`, so nothing else rejects the orphan).
505    //
506    // For a `group:<nested>` member, *both* the target's and the nested group's rows are locked
507    // `FOR UPDATE` up front, in a fixed (lexicographic) order -- not just the nested row. The
508    // insert's FK on `group_name` takes an implicit `FOR KEY SHARE` lock on the target's row,
509    // which conflicts with `FOR UPDATE`; locking only the nested row here left
510    // `add_member(A, "group:B")` (holds `FOR UPDATE` on B, wants `KEY SHARE` on A) able to
511    // deadlock against a concurrent `add_member(B, "group:A")` (holds `FOR UPDATE` on A, wants
512    // `KEY SHARE` on B). Every caller taking these two locks in the same order removes that
513    // cycle.
514    let mut tx = pool
515        .begin()
516        .await
517        .map_err(|e| GroupsError::Internal(format!("starting add-member transaction: {e:#}")))?;
518
519    let nested = body.member.strip_prefix("group:");
520    let mut lock_names: Vec<&str> = vec![&name];
521    if let Some(nested) = nested {
522        lock_names.push(nested);
523    }
524    lock_names.sort_unstable();
525    lock_names.dedup();
526    for lock_name in &lock_names {
527        sqlx::query("SELECT 1 FROM groups WHERE name = $1 FOR UPDATE")
528            .bind(lock_name)
529            .fetch_optional(&mut *tx)
530            .await?;
531    }
532
533    let group_exists: bool =
534        sqlx::query_scalar("SELECT EXISTS(SELECT 1 FROM groups WHERE name = $1)")
535            .bind(&name)
536            .fetch_one(&mut *tx)
537            .await?;
538    if !group_exists {
539        return Err(GroupsError::NotFound);
540    }
541
542    let key_name = if body.member.starts_with("key:") {
543        match ensure_live_key(&mut tx, &body.member).await? {
544            Some(key_name) => Some(key_name),
545            None => {
546                return Err(GroupsError::KeyNotFound(key_not_found_message(
547                    &body.member,
548                )));
549            }
550        }
551    } else {
552        None
553    };
554
555    if let Some(nested) = nested {
556        let nested_exists: bool =
557            sqlx::query_scalar("SELECT EXISTS(SELECT 1 FROM groups WHERE name = $1)")
558                .bind(nested)
559                .fetch_one(&mut *tx)
560                .await?;
561        if !nested_exists {
562            return Err(GroupsError::NestedGroupNotFound(format!(
563                "group {nested:?} not found"
564            )));
565        }
566        // Queried fresh (never the TTL snapshot) and read through this same transaction's
567        // connection -- not a second `pool.begin()` via `GroupsLoader::fetch`, which would need a
568        // second connection from this same (2-connection) pool while this one is still held,
569        // deadlocking two concurrent nested-group adds against each other. Mirrors
570        // `remove_member`'s own in-transaction graph read below. No need for `remove_member`'s
571        // `admins`-lockout advisory lock here: this only ever adds a member, which can only add
572        // reachability to a target, never remove it, so it cannot defeat that guard.
573        let group_names: Vec<String> = sqlx::query_scalar("SELECT name FROM groups")
574            .fetch_all(&mut *tx)
575            .await?;
576        let member_rows: Vec<(String, String)> =
577            sqlx::query_as("SELECT group_name, member FROM group_members")
578                .fetch_all(&mut *tx)
579                .await?;
580        let parsed = GroupGraph::from_rows(group_names, member_rows);
581        report_skipped("group store", &parsed.skipped);
582        let graph = parsed.value;
583        if graph.nesting_would_cycle(&name, nested) {
584            return Err(GroupsError::Conflict(format!(
585                "adding {:?} as a member of {name:?} would create a cycle",
586                body.member
587            )));
588        }
589    }
590
591    let created_by = caller_identity(&user);
592    for _ in 0..2 {
593        let row = sqlx::query_as::<_, UpsertedMemberRow>(
594            "WITH ins AS (
595                INSERT INTO group_members (group_name, member, created_at, created_by)
596                VALUES ($1, $2, now(), $3)
597                ON CONFLICT (group_name, member) DO NOTHING
598                RETURNING group_name, member, created_at, created_by
599             )
600             SELECT group_name, member, created_at, created_by, true AS created FROM ins
601             UNION ALL
602             SELECT group_name, member, created_at, created_by, false AS created
603             FROM group_members
604             WHERE group_name = $1 AND member = $2
605               AND NOT EXISTS (SELECT 1 FROM ins)",
606        )
607        .bind(&name)
608        .bind(&body.member)
609        .bind(&created_by)
610        .fetch_optional(&mut *tx)
611        .await?;
612        if let Some(row) = row {
613            tx.commit().await.map_err(|e| {
614                GroupsError::Internal(format!("committing add-member transaction: {e:#}"))
615            })?;
616            let status = if row.created {
617                StatusCode::CREATED
618            } else {
619                StatusCode::OK
620            };
621            return Ok((
622                status,
623                Json(MemberRow {
624                    group_name: row.group_name,
625                    member: row.member,
626                    created_at: row.created_at,
627                    created_by: row.created_by,
628                    key_name,
629                    key_revoked: false,
630                }),
631            ));
632        }
633    }
634    Err(GroupsError::Internal(format!(
635        "group_members upsert for ({name:?}, {:?}) returned no row after a retry",
636        body.member
637    )))
638}

640/// Row shape shared by both branches of `add_member`'s CTE -- mirrors
641/// `audience_grants.rs::UpsertedRow`'s reasoning (a single statement that can return zero rows on
642/// a create/create race; the caller retries once, then treats a second zero-row result as
643/// internal).
644#[derive(sqlx::FromRow)]
645struct UpsertedMemberRow {
646    group_name: String,
647    member: String,
648    created_at: DateTime<Utc>,
649    created_by: String,
650    created: bool,
651}
652
653#[derive(Deserialize)]
654struct RemoveMemberQuery {
655    member: String,
656}
657
658/// `DELETE {base_path}/api/groups/{name}/members?member=` -- `204`; `404` unknown (a `key:`
659/// member is not checked for existence, so a row left by a revoked key can still be removed); `409` when
660/// removing this member would leave [`ADMINS_GROUP`] unreachable by any principal, direct or
661/// nested (the only lockout protection: losing that reachability would leave admin reachable only
662/// through `psql`).
663async fn remove_member(
664    Extension(state): Extension<GroupsState>,
665    GroupAdminGate(user): GroupAdminGate,
666    client_ip: ClientIp,
667    Path(name): Path<String>,
668    Query(mut query): Query<RemoveMemberQuery>,
669) -> Result<StatusCode, GroupsError> {
670    query.member = canonicalize_key_selector(&query.member);
671    let audit = MutationAudit::new(
672        action::REMOVE_MEMBER,
673        caller_identity(&user),
674        user.is_admin,
675        client_ip.0,
676    )
677    .group(&name)
678    .member(&query.member);
679    let result = remove_member_inner(state, name, query).await;
680    audit.emit(&result);
681    result
682}
683
684async fn remove_member_inner(
685    state: GroupsState,
686    name: String,
687    query: RemoveMemberQuery,
688) -> Result<StatusCode, GroupsError> {
689    let pool = require_pool(&state)?;
690
691    // The lockout guard and the delete run in one transaction. The guard is a whole-graph
692    // reachability question (`any_principal_reaches(ADMINS_GROUP)`), so a lock on just `name`'s
693    // rows isn't enough to serialize it: two concurrent removals from *different* groups that
694    // both feed into `admins` (e.g. `admins = [group:eng, group:ops]`) would take disjoint
695    // per-group locks, each build its post-removal graph from the other's uncommitted state,
696    // both see `admins` still reachable, and both commit -- leaving `admins` reachable by
697    // nobody. A transaction-scoped advisory lock on a fixed key, taken before the graph read,
698    // closes that: every removal that could affect the guard serializes on this one lock,
699    // regardless of which group it targets. `pg_advisory_xact_lock` auto-releases at
700    // commit/rollback.
701    let mut tx = pool
702        .begin()
703        .await
704        .map_err(|e| GroupsError::Internal(format!("starting remove-member transaction: {e:#}")))?;
705
706    sqlx::query("SELECT pg_advisory_xact_lock(hashtext('micromegas_groups_admins_lockout'))")
707        .execute(&mut *tx)
708        .await?;
709
710    // Also lock `name`'s own rows: unrelated to the cross-group race above (the advisory lock
711    // already covers that), but still serializes this removal against `add_member`'s nested-group
712    // insert into the same group name, matching `add_member`/`delete_group`'s own row-locking
713    // pattern.
714    sqlx::query("SELECT 1 FROM group_members WHERE group_name = $1 FOR UPDATE")
715        .bind(&name)
716        .fetch_all(&mut *tx)
717        .await?;
718
719    // Resolve the group graph as it would look immediately after this removal, read fresh inside
720    // this transaction (never the TTL-cached `DbGroupsSource` snapshot `AuthContext` uses for
721    // request-time admin checks) -- the same freshly-queried-graph approach `add_member`'s cycle
722    // check uses. Refuse the removal if no principal -- no `*` and no `user:` entry, direct or via
723    // nesting -- would still reach `admins` afterward. This subsumes the old
724    // `name == ADMINS_GROUP`-only special case: emptying a group nested into `admins` (not
725    // `admins` itself, e.g. `admins`'s only member is `group:eng-leads`) strands `admins` just as
726    // surely as emptying `admins` directly.
727    let group_names: Vec<String> = sqlx::query_scalar("SELECT name FROM groups")
728        .fetch_all(&mut *tx)
729        .await?;
730    let member_rows: Vec<(String, String)> =
731        sqlx::query_as("SELECT group_name, member FROM group_members")
732            .fetch_all(&mut *tx)
733            .await?;
734    let post_removal_members = member_rows
735        .into_iter()
736        .filter(|(group_name, member)| !(*group_name == name && *member == query.member));
737    let parsed = GroupGraph::from_rows(group_names, post_removal_members);
738    report_skipped("group store", &parsed.skipped);
739    let graph = parsed.value;
740    if !graph.any_principal_reaches(ADMINS_GROUP) {
741        return Err(GroupsError::Conflict(
742            "removing this member would leave admins reachable only through direct database \
743             access"
744                .to_string(),
745        ));
746    }
747
748    let rows_affected =
749        sqlx::query("DELETE FROM group_members WHERE group_name = $1 AND member = $2")
750            .bind(&name)
751            .bind(&query.member)
752            .execute(&mut *tx)
753            .await?
754            .rows_affected();
755    if rows_affected == 0 {
756        return Err(GroupsError::NotFound);
757    }
758    tx.commit().await.map_err(|e| {
759        GroupsError::Internal(format!("committing remove-member transaction: {e:#}"))
760    })?;
761    Ok(StatusCode::NO_CONTENT)
762}

764/// Routes only -- [`GroupsState`] is layered separately in `web_server.rs::build_protected_routes`.
765pub fn groups_router(base_path: &str) -> Router {
766    Router::new()
767        .route(
768            &format!("{base_path}/api/groups"),
769            get(list_groups).post(create_group),
770        )
771        .route(
772            &format!("{base_path}/api/groups/{{name}}"),
773            axum::routing::delete(delete_group),
774        )
775        .route(
776            &format!("{base_path}/api/groups/{{name}}/members"),
777            get(list_members).post(add_member).delete(remove_member),
778        )
779}
