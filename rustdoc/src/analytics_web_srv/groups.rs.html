1//! Group admin API for `analytics-web-srv`.
2//!
3//! Mirrors `analytics_keys.rs`/`audience_grants.rs`'s shape (`GroupsState { pool: Option<PgPool> }`,
4//! an `IntoResponse` error enum, `AdminUser`-gated) over the `groups`/`group_members` tables
5//! (migration v10, `rust/ingestion/src/sql_migration.rs`). Every write goes through the
6//! [`AdminUser`] extractor's gate, which is `caller.is_admin` today; delegating group ownership
7//! later means widening that one gate, not this module's routing or handler shapes.
8

22/// `valid_selector` places no charset/length bound on a `group:<id>` member (mirrors
23/// `audience_grants.rs`'s identical constant/reasoning) -- the `member` column is `VARCHAR(255)`.
24const MAX_MEMBER_BYTES: usize = 255;

50/// Errors this API returns.
51///
52/// No `Forbidden` variant: the admin gate is the [`AdminUser`] extractor, whose rejection
53/// renders as `AdminRequired`'s own 403 body before any handler here starts running.
54pub enum GroupsError {
55    BadRequest(String),
56    NotFound,
57    /// `add_member`'s `group:<name>` member names a nested group that does not exist. Distinct
58    /// from `NotFound` (the *target* group missing) so the client can tell the two apart instead
59    /// of both rendering as the same generic "group not found". The message names the missing
60    /// nested group.
61    NestedGroupNotFound(String),
62    /// `409` -- the group already exists, is `admins` (undeletable), is still referenced, would
63    /// create a cycle, or would leave `admins` unreachable by any principal. The message names
64    /// which.
65    Conflict(String),
66    Database(sqlx::Error),
67    NotConfigured,
68    /// A non-DB internal error -- e.g. `GroupGraph::from_rows` rejecting a snapshot read
69    /// in-transaction.
70    Internal(String),
71}
72
73impl IntoResponse for GroupsError {
74    fn into_response(self) -> Response {
75        match self {
76            GroupsError::BadRequest(msg) => (
77                StatusCode::BAD_REQUEST,
78                Json(ErrorResponse::new("BAD_REQUEST", msg)),
79            )
80                .into_response(),
81            GroupsError::NotFound => (
82                StatusCode::NOT_FOUND,
83                Json(ErrorResponse::new("NOT_FOUND", "group not found")),
84            )
85                .into_response(),
86            GroupsError::NestedGroupNotFound(msg) => (
87                StatusCode::NOT_FOUND,
88                Json(ErrorResponse::new("NOT_FOUND", msg)),
89            )
90                .into_response(),
91            GroupsError::Conflict(msg) => (
92                StatusCode::CONFLICT,
93                Json(ErrorResponse::new("CONFLICT", msg)),
94            )
95                .into_response(),
96            GroupsError::Database(err) => {
97                error!("groups: database error: {err}");
98                (
99                    StatusCode::INTERNAL_SERVER_ERROR,
100                    Json(ErrorResponse::new(
101                        "DATABASE_ERROR",
102                        "internal database error",
103                    )),
104                )
105                    .into_response()
106            }
107            GroupsError::NotConfigured => (
108                StatusCode::SERVICE_UNAVAILABLE,
109                Json(ErrorResponse::new(
110                    "NOT_CONFIGURED",
111                    "group store not configured: set MICROMEGAS_SQL_CONNECTION_STRING",
112                )),
113            )
114                .into_response(),
115            GroupsError::Internal(msg) => {
116                error!("groups: internal error: {msg}");
117                (
118                    StatusCode::INTERNAL_SERVER_ERROR,
119                    Json(ErrorResponse::new("INTERNAL_ERROR", "internal error")),
120                )
121                    .into_response()
122            }
123        }
124    }
125}

137fn validate_group_name(name: &str) -> Result<(), GroupsError> {
138    if !is_valid_group_name(name) {
139        return Err(GroupsError::BadRequest(format!(
140            "invalid group name {name:?}: must match [A-Za-z0-9_-]{{1,255}}"
141        )));
142    }
143    Ok(())
144}

152        .unwrap_or_else(|| caller.subject.clone())

155#[derive(Serialize, sqlx::FromRow)]
156struct GroupSummary {
157    name: String,
158    description: Option<String>,
159    member_count: i64,
160    created_at: DateTime<Utc>,
161    created_by: String,
162}

164/// `GET {base_path}/api/groups` -- every group with its member count.
165async fn list_groups(
166    Extension(state): Extension<GroupsState>,
167    AdminUser(_user): AdminUser,
168) -> Result<Json<Vec<GroupSummary>>, GroupsError> {
169    let pool = require_pool(&state)?;
170    let rows = sqlx::query_as::<_, GroupSummary>(
171        "SELECT g.name, g.description, g.created_at, g.created_by,
172                COUNT(m.member) AS member_count
173         FROM groups g
174         LEFT JOIN group_members m ON m.group_name = g.name
175         GROUP BY g.name, g.description, g.created_at, g.created_by
176         ORDER BY g.name",
177    )
178    .fetch_all(&pool)
179    .await?;
180    Ok(Json(rows))
181}

189/// `POST {base_path}/api/groups` -- creates a new, empty group. `400` on a malformed name, `409`
190/// if it already exists.
191async fn create_group(
192    Extension(state): Extension<GroupsState>,
193    AdminUser(user): AdminUser,
194    Json(body): Json<CreateGroupRequest>,
195) -> Result<(StatusCode, Json<GroupSummary>), GroupsError> {
196    let pool = require_pool(&state)?;
197    validate_group_name(&body.name)?;
198    let created_by = caller_identity(&user);
199
200    let row = sqlx::query_as::<_, GroupSummary>(
201        "INSERT INTO groups (name, description, created_at, created_by)
202         VALUES ($1, $2, now(), $3)
203         ON CONFLICT (name) DO NOTHING
204         RETURNING name, description, 0::bigint AS member_count, created_at, created_by",
205    )
206    .bind(&body.name)
207    .bind(&body.description)
208    .bind(&created_by)
209    .fetch_optional(&pool)
210    .await?
211    .ok_or_else(|| GroupsError::Conflict(format!("group {:?} already exists", body.name)))?;
212
213    info!("created group name={} created_by={created_by}", row.name);
214    Ok((StatusCode::CREATED, Json(row)))
215}

217/// `DELETE {base_path}/api/groups/{name}` -- `204`; `409` for `admins`; `409` while referenced by
218/// any `group_members.member = 'group:<name>'` or `audience_grants.selector = 'group:<name>'`
219/// row (the response lists the referrers).
220async fn delete_group(
221    Extension(state): Extension<GroupsState>,
222    AdminUser(_user): AdminUser,
223    Path(name): Path<String>,
224) -> Result<StatusCode, GroupsError> {
225    let pool = require_pool(&state)?;
226    if name == ADMINS_GROUP {
227        return Err(GroupsError::Conflict(
228            "the admins group cannot be deleted".to_string(),
229        ));
230    }
231
232    // The referrer checks and the delete run in one transaction, with the `groups` row for
233    // `name` locked first: `add_member`'s nested-group check takes the same `FOR UPDATE` lock on
234    // that row before inserting a `group:<name>` member row (see its own comment), so a
235    // concurrent nest-in and this delete serialize on this row instead of racing -- without that,
236    // a `group:<name>` reference inserted between the plain check and the delete would survive,
237    // and silently re-activate if the name is later recreated.
238    let mut tx = pool
239        .begin()
240        .await
241        .map_err(|e| GroupsError::Internal(format!("starting delete transaction: {e:#}")))?;
242    sqlx::query("SELECT 1 FROM groups WHERE name = $1 FOR UPDATE")
243        .bind(&name)
244        .fetch_optional(&mut *tx)
245        .await?;
246
247    let nesting_selector = format!("group:{name}");
248    let nested_into: Vec<String> = sqlx::query_scalar(
249        "SELECT group_name FROM group_members WHERE member = $1 ORDER BY group_name",
250    )
251    .bind(&nesting_selector)
252    .fetch_all(&mut *tx)
253    .await?;
254    let granted_to: Vec<String> = sqlx::query_scalar(
255        "SELECT audience FROM audience_grants WHERE selector = $1 ORDER BY audience",
256    )
257    .bind(&nesting_selector)
258    .fetch_all(&mut *tx)
259    .await?;
260    if !nested_into.is_empty() || !granted_to.is_empty() {
261        let mut referrers = Vec::new();
262        if !nested_into.is_empty() {
263            referrers.push(format!("nested into: {}", nested_into.join(", ")));
264        }
265        if !granted_to.is_empty() {
266            referrers.push(format!("granted audiences: {}", granted_to.join(", ")));
267        }
268        return Err(GroupsError::Conflict(format!(
269            "group {name:?} is still referenced ({})",
270            referrers.join("; ")
271        )));
272    }
273
274    let rows_affected = sqlx::query("DELETE FROM groups WHERE name = $1")
275        .bind(&name)
276        .execute(&mut *tx)
277        .await?
278        .rows_affected();
279    if rows_affected == 0 {
280        return Err(GroupsError::NotFound);
281    }
282    tx.commit()
283        .await
284        .map_err(|e| GroupsError::Internal(format!("committing delete transaction: {e:#}")))?;
285    info!("deleted group name={name}");
286    Ok(StatusCode::NO_CONTENT)
287}

326/// `POST {base_path}/api/groups/{name}/members` -- validates `valid_selector` + 255 bytes; for
327/// `group:X`, `404` if `X` does not exist and `409` if `nesting_would_cycle`; `201` created /
328/// `200` already existed, via the same CTE `insert_or_get` UPSERT pattern `audience_grants.rs`
329/// uses (see that module's `insert_or_get` doc comment).
330async fn add_member(
331    Extension(state): Extension<GroupsState>,
332    AdminUser(user): AdminUser,
333    Path(name): Path<String>,
334    Json(body): Json<AddMemberRequest>,
335) -> Result<(StatusCode, Json<MemberRow>), GroupsError> {
336    let pool = require_pool(&state)?;
337    if !valid_selector(&body.member) {
338        return Err(GroupsError::BadRequest(format!(
339            "invalid member {:?}: must be '*', 'user:<id>', or 'group:<id>'",
340            body.member
341        )));
342    }
343    if body.member.len() > MAX_MEMBER_BYTES {
344        return Err(GroupsError::BadRequest(format!(
345            "member must be at most {MAX_MEMBER_BYTES} bytes"
346        )));
347    }
348
349    // The nested-group existence check and the insert run in one transaction, with the `groups`
350    // row(s) involved locked first: `delete_group` takes this same `FOR UPDATE` lock on its own
351    // row before scanning for `group:<name>` referrers (see its own comment), so a concurrent
352    // delete of the nested group and this insert serialize on that row instead of racing --
353    // without that, a `group:<nested>` member row could commit after `delete_group`'s referrer
354    // scan already found none, leaving `group_members` pointing at a group name that no longer
355    // exists (its FK is on `group_name`, not `member`, so nothing else rejects the orphan).
356    //
357    // For a `group:<nested>` member, *both* the target's and the nested group's rows are locked
358    // `FOR UPDATE` up front, in a fixed (lexicographic) order -- not just the nested row. The
359    // insert's FK on `group_name` takes an implicit `FOR KEY SHARE` lock on the target's row,
360    // which conflicts with `FOR UPDATE`; locking only the nested row here left
361    // `add_member(A, "group:B")` (holds `FOR UPDATE` on B, wants `KEY SHARE` on A) able to
362    // deadlock against a concurrent `add_member(B, "group:A")` (holds `FOR UPDATE` on A, wants
363    // `KEY SHARE` on B). Every caller taking these two locks in the same order removes that
364    // cycle.
365    let mut tx = pool
366        .begin()
367        .await
368        .map_err(|e| GroupsError::Internal(format!("starting add-member transaction: {e:#}")))?;
369
370    let nested = body.member.strip_prefix("group:");
371    let mut lock_names: Vec<&str> = vec![&name];
372    if let Some(nested) = nested {
373        lock_names.push(nested);
374    }
375    lock_names.sort_unstable();
376    lock_names.dedup();
377    for lock_name in &lock_names {
378        sqlx::query("SELECT 1 FROM groups WHERE name = $1 FOR UPDATE")
379            .bind(lock_name)
380            .fetch_optional(&mut *tx)
381            .await?;
382    }
383
384    let group_exists: bool =
385        sqlx::query_scalar("SELECT EXISTS(SELECT 1 FROM groups WHERE name = $1)")
386            .bind(&name)
387            .fetch_one(&mut *tx)
388            .await?;
389    if !group_exists {
390        return Err(GroupsError::NotFound);
391    }
392
393    if let Some(nested) = nested {
394        let nested_exists: bool =
395            sqlx::query_scalar("SELECT EXISTS(SELECT 1 FROM groups WHERE name = $1)")
396                .bind(nested)
397                .fetch_one(&mut *tx)
398                .await?;
399        if !nested_exists {
400            return Err(GroupsError::NestedGroupNotFound(format!(
401                "group {nested:?} not found"
402            )));
403        }
404        // Queried fresh (never the TTL snapshot) and read through this same transaction's
405        // connection -- not a second `pool.begin()` via `GroupsLoader::fetch`, which would need a
406        // second connection from this same (2-connection) pool while this one is still held,
407        // deadlocking two concurrent nested-group adds against each other. Mirrors
408        // `remove_member`'s own in-transaction graph read below. No need for `remove_member`'s
409        // `admins`-lockout advisory lock here: this only ever adds a member, which can only add
410        // reachability to a target, never remove it, so it cannot defeat that guard.
411        let group_names: Vec<String> = sqlx::query_scalar("SELECT name FROM groups")
412            .fetch_all(&mut *tx)
413            .await?;
414        let member_rows: Vec<(String, String)> =
415            sqlx::query_as("SELECT group_name, member FROM group_members")
416                .fetch_all(&mut *tx)
417                .await?;
418        let graph = GroupGraph::from_rows(group_names, member_rows)
419            .map_err(|e| GroupsError::Internal(format!("querying the group graph: {e:#}")))?;
420        if graph.nesting_would_cycle(&name, nested) {
421            return Err(GroupsError::Conflict(format!(
422                "adding {:?} as a member of {name:?} would create a cycle",
423                body.member
424            )));
425        }
426    }
427
428    let created_by = caller_identity(&user);
429    for _ in 0..2 {
430        let row = sqlx::query_as::<_, UpsertedMemberRow>(
431            "WITH ins AS (
432                INSERT INTO group_members (group_name, member, created_at, created_by)
433                VALUES ($1, $2, now(), $3)
434                ON CONFLICT (group_name, member) DO NOTHING
435                RETURNING group_name, member, created_at, created_by
436             )
437             SELECT group_name, member, created_at, created_by, true AS created FROM ins
438             UNION ALL
439             SELECT group_name, member, created_at, created_by, false AS created
440             FROM group_members
441             WHERE group_name = $1 AND member = $2
442               AND NOT EXISTS (SELECT 1 FROM ins)",
443        )
444        .bind(&name)
445        .bind(&body.member)
446        .bind(&created_by)
447        .fetch_optional(&mut *tx)
448        .await?;
449        if let Some(row) = row {
450            tx.commit().await.map_err(|e| {
451                GroupsError::Internal(format!("committing add-member transaction: {e:#}"))
452            })?;
453            info!(
454                "group member added group={} member={} created={} created_by={created_by}",
455                row.group_name, row.member, row.created
456            );
457            let status = if row.created {
458                StatusCode::CREATED
459            } else {
460                StatusCode::OK
461            };
462            return Ok((
463                status,
464                Json(MemberRow {
465                    group_name: row.group_name,
466                    member: row.member,
467                    created_at: row.created_at,
468                    created_by: row.created_by,
469                }),
470            ));
471        }
472    }
473    Err(GroupsError::Internal(format!(
474        "group_members upsert for ({name:?}, {:?}) returned no row after a retry",
475        body.member
476    )))
477}

492#[derive(Deserialize)]
493struct RemoveMemberQuery {
494    member: String,
495}
496
497/// `DELETE {base_path}/api/groups/{name}/members?member=` -- `204`; `404` unknown; `409` when
498/// removing this member would leave [`ADMINS_GROUP`] unreachable by any principal, direct or
499/// nested (the only lockout protection: losing that reachability would leave admin reachable only
500/// through `psql`).
501async fn remove_member(
502    Extension(state): Extension<GroupsState>,
503    AdminUser(_user): AdminUser,
504    Path(name): Path<String>,
505    Query(query): Query<RemoveMemberQuery>,
506) -> Result<StatusCode, GroupsError> {
507    let pool = require_pool(&state)?;
508
509    // The lockout guard and the delete run in one transaction. The guard is a whole-graph
510    // reachability question (`any_principal_reaches(ADMINS_GROUP)`), so a lock on just `name`'s
511    // rows isn't enough to serialize it: two concurrent removals from *different* groups that
512    // both feed into `admins` (e.g. `admins = [group:eng, group:ops]`) would take disjoint
513    // per-group locks, each build its post-removal graph from the other's uncommitted state,
514    // both see `admins` still reachable, and both commit -- leaving `admins` reachable by
515    // nobody. A transaction-scoped advisory lock on a fixed key, taken before the graph read,
516    // closes that: every removal that could affect the guard serializes on this one lock,
517    // regardless of which group it targets. `pg_advisory_xact_lock` auto-releases at
518    // commit/rollback.
519    let mut tx = pool
520        .begin()
521        .await
522        .map_err(|e| GroupsError::Internal(format!("starting remove-member transaction: {e:#}")))?;
523
524    sqlx::query("SELECT pg_advisory_xact_lock(hashtext('micromegas_groups_admins_lockout'))")
525        .execute(&mut *tx)
526        .await?;
527
528    // Also lock `name`'s own rows: unrelated to the cross-group race above (the advisory lock
529    // already covers that), but still serializes this removal against `add_member`'s nested-group
530    // insert into the same group name, matching `add_member`/`delete_group`'s own row-locking
531    // pattern.
532    sqlx::query("SELECT 1 FROM group_members WHERE group_name = $1 FOR UPDATE")
533        .bind(&name)
534        .fetch_all(&mut *tx)
535        .await?;
536
537    // Resolve the group graph as it would look immediately after this removal, read fresh inside
538    // this transaction (never the TTL-cached `DbGroupsSource` snapshot `AuthContext` uses for
539    // request-time admin checks) -- the same freshly-queried-graph approach `add_member`'s cycle
540    // check uses. Refuse the removal if no principal -- no `*` and no `user:` entry, direct or via
541    // nesting -- would still reach `admins` afterward. This subsumes the old
542    // `name == ADMINS_GROUP`-only special case: emptying a group nested into `admins` (not
543    // `admins` itself, e.g. `admins`'s only member is `group:eng-leads`) strands `admins` just as
544    // surely as emptying `admins` directly.
545    let group_names: Vec<String> = sqlx::query_scalar("SELECT name FROM groups")
546        .fetch_all(&mut *tx)
547        .await?;
548    let member_rows: Vec<(String, String)> =
549        sqlx::query_as("SELECT group_name, member FROM group_members")
550            .fetch_all(&mut *tx)
551            .await?;
552    let post_removal_members = member_rows
553        .into_iter()
554        .filter(|(group_name, member)| !(*group_name == name && *member == query.member));
555    let graph = GroupGraph::from_rows(group_names, post_removal_members).map_err(|e| {
556        GroupsError::Internal(format!("building the post-removal group graph: {e:#}"))
557    })?;
558    if !graph.any_principal_reaches(ADMINS_GROUP) {
559        return Err(GroupsError::Conflict(
560            "removing this member would leave admins reachable only through direct database \
561             access"
562                .to_string(),
563        ));
564    }
565
566    let rows_affected =
567        sqlx::query("DELETE FROM group_members WHERE group_name = $1 AND member = $2")
568            .bind(&name)
569            .bind(&query.member)
570            .execute(&mut *tx)
571            .await?
572            .rows_affected();
573    if rows_affected == 0 {
574        return Err(GroupsError::NotFound);
575    }
576    tx.commit().await.map_err(|e| {
577        GroupsError::Internal(format!("committing remove-member transaction: {e:#}"))
578    })?;
579    info!("group member removed group={name} member={}", query.member);
580    Ok(StatusCode::NO_CONTENT)
581}

583/// Routes only -- [`GroupsState`] is layered separately in `web_server.rs::build_protected_routes`.
584pub fn groups_router(base_path: &str) -> Router {
585    Router::new()
586        .route(
587            &format!("{base_path}/api/groups"),
588            get(list_groups).post(create_group),
589        )
590        .route(
591            &format!("{base_path}/api/groups/{{name}}"),
592            axum::routing::delete(delete_group),
593        )
594        .route(
595            &format!("{base_path}/api/groups/{{name}}/members"),
596            get(list_members).post(add_member).delete(remove_member),
