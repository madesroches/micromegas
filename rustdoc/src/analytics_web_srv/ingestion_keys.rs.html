1//! Ingestion key management API for `analytics-web-srv` (#1458).
2//!
3//! Modeled directly on `analytics_keys.rs`, targeting `ingestion_api_keys`
4//! instead of `analytics_api_keys`. Replaces the proxy that used to forward
5//! mint/list/revoke calls to ingestion's own (now removed)
6//! `/auth/api_keys*` routes: ingestion should only do ingestion, so
7//! `analytics-web-srv` writes directly to `ingestion_api_keys` via the same
8//! telemetry-DB pool it already opens for `analytics_api_keys` — both tables
9//! live in the same database behind `MICROMEGAS_SQL_CONNECTION_STRING`.
10//! Routes live under `{base_path}/api/ingestion-api-keys`, distinct from this
11//! service's own `/auth/*` routes (login/callback/refresh/logout/me) — a
12//! completely different concern (browser session lifecycle).
13//!
14//! This is also the attribution fix: every mint/revoke/import records the
15//! acting caller's own OIDC identity, never a shared service credential the
16//! way the removed proxy did. `list_keys`/`revoke_key`/`import_key` still do
17//! so via the [`AdminUser`] extractor; `mint_key` (AbAC Stage 6, #1374) now
18//! runs through [`MintGate`]/[`AuthenticatedUser`] instead, since minting is
19//! no longer purely admin-gated -- see that extractor's own doc comment.
20//!
21//! **Duplication, accepted.** This duplicates most of `analytics_keys.rs`'s
22//! validation/SQL/error shape — deliberately, per that module's own doc
23//! comment: sharing it would mean a generic abstraction over a handful of
24//! near-identical handlers differing only in which table they target, the
25//! same shape the codebase already declines to share between
26//! `data_sources.rs`/`screens.rs`/`folders.rs` today.
27

46/// Bytes, not chars: deliberately stricter than the `VARCHAR(255)` column,
47/// which bounds characters — same rule as `analytics_keys.rs`.
48const MAX_NAME_BYTES: usize = 255;
49const DEFAULT_LIMIT: i64 = 100;
50const MAX_LIMIT: i64 = 500;

52/// Holds the (possibly absent) telemetry-DB pool for the ingestion-key
53/// routes. `None` only when `MICROMEGAS_SQL_CONNECTION_STRING` is unset — the
54/// routes stay registered either way and return 503 per-request in that case,
55/// the same always-register-503-when-unconfigured shape `AnalyticsKeysState`
56/// uses. An unmigrated DB (missing the v5 migration's tables) is a separate
57/// failure mode: the pool is still `Some`, and a request fails with a 500 at
58/// query time instead.
59#[derive(Clone)]
60pub struct IngestionKeysState {
61    pub pool: Option<PgPool>,
62    /// Resolved once at startup from `{prefix}_DEFAULT_KEY_AUDIENCE`
63    /// (`micromegas::auth::policy::default_key_audience_from_env`, `web_server.rs`). `None`
64    /// when the knob is unset -- `mint` then requires an explicit `audience` (400 otherwise);
65    /// `import` falls back further, to `PUBLIC_AUDIENCE`. See [`resolve_audience`].
66    pub default_audience: Option<String>,
67    /// Off-by-default self-service mint gate (AbAC Stage 6, #1374). Resolved once at startup
68    /// from `MICROMEGAS_SELF_SERVICE_MINT` (`web_server.rs`), default `false`. Checked by
69    /// [`MintGate`] for every non-admin caller before `mint_key`'s body runs at all -- with the
70    /// knob off, a deployment that upgrades to this stage keeps today's admin-only mint
71    /// behavior unchanged.
72    pub self_service_mint_enabled: bool,
73    /// `MICROMEGAS_SELF_SERVICE_MAX_CLAIMS_PER_CALLER`, default 25 -- caps how many distinct
74    /// audiences one non-admin caller may claim via the lazy claim path
75    /// ([`try_claim_and_mint`]). Best-effort under concurrency, not a hard ceiling -- see that
76    /// function's own doc comment.
77    pub max_claims_per_caller: i64,
78    /// `MICROMEGAS_SELF_SERVICE_MAX_KEYS_PER_CALLER`, default 100 -- caps how many *live* keys
79    /// one non-admin caller may hold at once, checked in `mint_key` regardless of which path
80    /// mints the next one. Best-effort under concurrency, not a hard ceiling.
81    pub max_keys_per_caller: i64,
82}

91    message: String,
92}

98            message: message.into(),
99        }
100    }

103/// Errors this API returns.
104///
105/// `Forbidden`/`Unavailable`/`Unauthenticated`/`Conflict` all back the self-service mint path
106/// (AbAC Stage 6, #1374): `mint_key` is no longer purely [`AdminUser`]-gated, whose own rejection
107/// (`AdminRequired`) used to make a `Forbidden` variant here dead code -- it is now
108/// [`MintGate`]/[`AuthenticatedUser`]-gated, and its own denials (the off-by-default gate, a
109/// missing-grant/malformed-audience `MintPolicy` denial, a per-caller bound, and lock contention
110/// on a lazy claim) need their own status codes. `list_keys`/`revoke_key`/`import_key` stay
111/// `AdminUser`-gated and never construct any of the four.
112#[derive(Debug)]
113pub enum IngestionKeyError {
114    /// Request body/query failed validation.
115    BadRequest(String),
116    /// Unknown `key_id`.
117    NotFound,
118    /// A DB error.
119    Database(sqlx::Error),
120    /// `state.pool == None` — the telemetry-DB pool was never configured
121    /// (`MICROMEGAS_SQL_CONNECTION_STRING` unset).
122    NotConfigured,
123    /// Self-service mint denied: the `MICROMEGAS_SELF_SERVICE_MINT` gate is off for a non-admin
124    /// caller, `MintPolicy::resolve_audience` denied the request (no matching grant, or a
125    /// malformed audience from an admin), the audience is not eligible for a lazy claim, or a
126    /// per-caller bound (`max_claims_per_caller`/`max_keys_per_caller`) was reached.
127    Forbidden(String),
128    /// The audience-grant point query behind `resolve_audience`'s policy call (or the row parse
129    /// immediately after it) failed -- a DB outage, not a denial, so it must not be
130    /// misattributed as "you have no grant." Distinct from `NotConfigured`, whose message is
131    /// specifically about an unset `MICROMEGAS_SQL_CONNECTION_STRING` and would mislead here.
132    Unavailable(String),
133    /// [`AuthenticatedUser`]/[`MintGate`] found no `AuthContext` extension in the request --
134    /// normally unreachable once routing is wired correctly; see `AuthenticatedUser`'s own doc
135    /// comment for the (fail-closed) case this covers.
136    Unauthenticated(String),
137    /// A lazy claim ([`try_claim_and_mint`]) lost the per-audience advisory-lock race to a
138    /// concurrent claimant. Transient lock contention, not a denial -- the caller (in particular
139    /// `micromegas-setup-telemetry`) should retry, not treat this as "you may not do this."
140    Conflict(String),
141}
142
143impl IntoResponse for IngestionKeyError {
144    fn into_response(self) -> Response {
145        match self {
146            IngestionKeyError::BadRequest(msg) => (
147                StatusCode::BAD_REQUEST,
148                Json(ErrorResponse::new("BAD_REQUEST", msg)),
149            )
150                .into_response(),
151            IngestionKeyError::NotFound => (
152                StatusCode::NOT_FOUND,
153                Json(ErrorResponse::new("NOT_FOUND", "key not found")),
154            )
155                .into_response(),
156            IngestionKeyError::Database(err) => {
157                error!("ingestion_keys: database error: {err}");
158                (
159                    StatusCode::INTERNAL_SERVER_ERROR,
160                    Json(ErrorResponse::new(
161                        "DATABASE_ERROR",
162                        "internal database error",
163                    )),
164                )
165                    .into_response()
166            }
167            IngestionKeyError::NotConfigured => (
168                StatusCode::SERVICE_UNAVAILABLE,
169                Json(ErrorResponse::new(
170                    "NOT_CONFIGURED",
171                    "ingestion key store not configured: set MICROMEGAS_SQL_CONNECTION_STRING",
172                )),
173            )
174                .into_response(),
175            IngestionKeyError::Forbidden(msg) => (
176                StatusCode::FORBIDDEN,
177                Json(ErrorResponse::new("FORBIDDEN", msg)),
178            )
179                .into_response(),
180            IngestionKeyError::Unavailable(msg) => (
181                StatusCode::SERVICE_UNAVAILABLE,
182                Json(ErrorResponse::new("UNAVAILABLE", msg)),
183            )
184                .into_response(),
185            IngestionKeyError::Unauthenticated(msg) => (
186                StatusCode::UNAUTHORIZED,
187                Json(ErrorResponse::new("UNAUTHENTICATED", msg)),
188            )
189                .into_response(),
190            IngestionKeyError::Conflict(msg) => (
191                StatusCode::CONFLICT,
192                Json(ErrorResponse::new("CLAIM_CONTENDED", msg)),
193            )
194                .into_response(),
195        }
196    }
197}

218fn validate_name(name: &str) -> Result<(), IngestionKeyError> {
219    if name.is_empty() {
220        return Err(IngestionKeyError::BadRequest(
221            "name must not be empty".to_string(),
222        ));
223    }
224    if name.len() > MAX_NAME_BYTES {
225        return Err(IngestionKeyError::BadRequest(format!(
226            "name must be at most {MAX_NAME_BYTES} bytes"
227        )));
228    }
229    Ok(())
230}

232/// Resolves the audience to stamp on a mint/import `INSERT`'s `NOT NULL` column
233/// (`tasks/1372_audience_on_keys_plan.md` §5-§6). `pub`, not module-private, and sync with no
234/// pool access, so the whole resolution matrix is unit-testable without a database.
235///
236/// `requested`: a missing field or an empty string counts as absent (the empty string is not a
237/// name -- it fails [`is_valid_audience`] either way); anything else is taken **verbatim**, no
238/// case folding. `fallback`: `None` for `mint` (an unresolved mint is a `BadRequest`, never a
239/// silent `public`), `Some(PUBLIC_AUDIENCE)` for `import` (continuity with the v6 backfill).
240///
241/// Resolution order: `requested` → `state.default_audience` → `fallback`; the first
242/// non-absent value is validated with [`is_valid_audience`] and returned. `BadRequest` when
243/// nothing resolves at all.
244///
245/// This is format/defaulting validation only -- it runs before `mint_key`'s
246/// `MintPolicy::resolve_audience` authorization decision (AbAC Stage 6, #1374, Design §4), and is
247/// unaware of grants or claims.
248pub fn resolve_audience(
249    state: &IngestionKeysState,
250    requested: Option<&str>,
251    fallback: Option<&str>,
252) -> Result<String, IngestionKeyError> {
253    let requested = requested.filter(|s| !s.is_empty());
254    let default_audience = state.default_audience.as_deref();
255    let candidate = requested.or(default_audience).or(fallback);
256    match candidate {
257        Some(aud) if is_valid_audience(aud) => Ok(aud.to_string()),
258        Some(aud) => Err(IngestionKeyError::BadRequest(format!(
259            "invalid audience {aud:?}: must match [A-Za-z0-9_-]{{1,255}}"
260        ))),
261        None => Err(IngestionKeyError::BadRequest(
262            "no audience given and MICROMEGAS_DEFAULT_KEY_AUDIENCE is not set".to_string(),
263        )),
264    }
265}

285/// `FromRequestParts` extractor for `mint_key` specifically -- yields the caller's `AuthContext`
286/// after enforcing the off-by-default self-service gate (AbAC Stage 6, #1374, Design §3). Because
287/// this is a `FromRequestParts` impl, axum runs it -- like `AuthenticatedUser` itself -- before
288/// `Json<MintRequest>` ever parses the request body: a knob-off non-admin caller is rejected
289/// before the body is ever touched, mirroring `AdminUser`'s own ordering guarantee.
290struct MintGate(AuthContext);
291
292impl<S: Send + Sync> FromRequestParts<S> for MintGate {
293    type Rejection = IngestionKeyError;
294
295    async fn from_request_parts(parts: &mut Parts, state: &S) -> Result<Self, Self::Rejection> {
296        let AuthenticatedUser(caller) = AuthenticatedUser::from_request_parts(parts, state).await?;
297        // `.ok_or(...)`, not `.expect(...)`: every other extractor in this crate fails closed
298        // with a status code for a missing extension rather than panicking (`AdminUser`'s own
299        // `.ok_or(AdminRequired)` for the identical "extension should always be layered"
300        // situation) -- `MintGate` should not be the one exception.
301        let ingestion_state = parts
302            .extensions
303            .get::<IngestionKeysState>()
304            .cloned()
305            .ok_or(IngestionKeyError::NotConfigured)?; // reachable only if routing is
306        // misconfigured; the 503 body's wording (about `MICROMEGAS_SQL_CONNECTION_STRING`)
307        // doesn't literally describe this cause, but a 503 fail-closed beats a panic for a
308        // case that should never happen in a correctly wired router.
309        if !caller.is_admin && !ingestion_state.self_service_mint_enabled {
310            return Err(IngestionKeyError::Forbidden(
311                "self-service minting is disabled".to_string(),
312            ));
313        }
314        Ok(MintGate(caller))
315    }
316}

318/// `POST {base_path}/api/ingestion-api-keys` — mints a new `ingestion_api_keys` row.
319///
320/// Authorization is `MintGate` (the off-by-default self-service gate, AbAC Stage 6, #1374) plus
321/// `MintPolicy::resolve_audience` (a per-request point query against `audience_grants`, Design
322/// §4) -- no longer a flat `AdminUser` gate. Format/defaulting validation still runs first,
323/// through the untouched free `resolve_audience` function below, which is what keeps this
324/// route's pre-stage 400s unchanged.
325async fn mint_key(
326    Extension(state): Extension<IngestionKeysState>,
327    MintGate(caller): MintGate,
328    Json(body): Json<MintRequest>,
329) -> Result<(StatusCode, Json<MintResponse>), IngestionKeyError> {
330    let pool = require_pool(&state)?;
331    validate_name(&body.name)?;
332    // `fallback: None` -- a new credential must never silently default to `public`; with
333    // neither an explicit `audience` nor `MICROMEGAS_DEFAULT_KEY_AUDIENCE` configured, this
334    // is a `BadRequest`, not a fail-open publish grant. `?` here is what preserves the exact
335    // existing 400 bodies -- `MintPolicy::resolve_audience`'s own `requested: None` /
336    // malformed-audience arms are never reached from this route at all.
337    let candidate = resolve_audience(&state, body.audience.as_deref(), None)?;
338
339    // Key material, generated here -- before the policy call, not after -- so both the
340    // ordinary and the lazy-claim path share one value to insert.
341    let key = generate_key();
342    let hash = hash_key(&key);
343    let key_id = Uuid::new_v4();
344    let created_at = Utc::now();
345
346    // Per-caller bound (Design §4a): caps how many *live* keys one non-admin may hold, regardless
347    // of which path below mints the next one. Admins are exempt -- this bounds self-service, not
348    // administration. Best-effort, not a hard ceiling, under concurrency: this `SELECT COUNT(*)`
349    // runs on `pool` outside any transaction, so N concurrent mint requests from the same caller
350    // all read the same pre-insert count and can all pass -- the cap is exact only for
351    // sequential use from one caller.
352    if !caller.is_admin {
353        let caller_id = caller.email.as_deref().unwrap_or(&caller.subject);
354        let key_count: i64 = sqlx::query_scalar(
355            "SELECT COUNT(*) FROM ingestion_api_keys WHERE created_by = $1 AND revoked_at IS NULL",
356        )
357        .bind(caller_id)
358        .fetch_one(&pool)
359        .await?;
360        if key_count >= state.max_keys_per_caller {
361            return Err(IngestionKeyError::Forbidden(format!(
362                "caller already holds {key_count} live self-service keys (limit {})",
363                state.max_keys_per_caller
364            )));
365        }
366    }
367
368    // Mint authorization is a point query, not a cached snapshot (Design §3): no
369    // `DbAudienceGrantsSource` is attached for mint at all. `audience` is the leading column of
370    // `audience_grants`'s `PRIMARY KEY (audience, axis, selector)`, so this is an index-only
371    // scan.
372    let mint_selectors: Vec<String> = sqlx::query_scalar(
373        "SELECT selector FROM audience_grants WHERE audience = $1 AND axis = 'mint'",
374    )
375    .bind(&candidate)
376    .fetch_all(&pool)
377    .await
378    // A failed query is a DB outage, not a denial -- must not be misattributed as "you have no
379    // grant," so it is mapped to `Unavailable` explicitly. Logs the real `sqlx::Error`
380    // server-side and renders a fixed, generic client message -- the raw error text (connection
381    // strings, table/column names) must not reach the client.
382    .map_err(|e| {
383        error!("ingestion_keys: audience grant point query failed: {e}");
384        IngestionKeyError::Unavailable("audience grant store unavailable".to_string())
385    })?;
386
387    let grants = AudienceGrants::from_rows(
388        mint_selectors
389            .into_iter()
390            .map(|selector| (candidate.clone(), GrantAxis::Mint, selector)),
391    )
392    // Every row read back out of `audience_grants` already passed its own `CHECK` constraints on
393    // write, so this arm is unreachable in practice; kept as a real status code rather than
394    // `.unwrap()`, for the same fail-closed reason `MintGate` gives its own `.ok_or(...)`.
395    .map_err(|e| {
396        error!("ingestion_keys: audience grant row parse failed: {e}");
397        IngestionKeyError::Unavailable("audience grant store unavailable".to_string())
398    })?;
399
400    // `store: None` (the `new` default) -- this stage never attaches a `DbAudienceGrantsSource`
401    // to a mint policy (Design §3).
402    let policy = AudienceMintPolicy::new(grants);
403
404    let audience = match policy.resolve_audience(&caller, Some(&candidate)).await {
405        Ok(aud) => aud,
406        // Malformed-audience arm; `candidate` is already valid-format (via `resolve_audience`
407        // above), so unreachable in practice.
408        Err(e) if caller.is_admin => return Err(IngestionKeyError::Forbidden(e.to_string())),
409        Err(_) => {
410            // Non-admin, no matching `mint` grant for `candidate` among the rows the point query
411            // above just read -- try the lazy claim (Design §4a) only when the caller explicitly
412            // named this audience (not merely `state.default_audience`), and has an email to
413            // claim with.
414            let explicit = body.audience.as_deref().filter(|s| !s.is_empty()).is_some();
415            match (explicit, caller.email.as_deref()) {
416                (true, Some(_email)) => {
417                    // Commits its own grant + key rows and returns the finished response
418                    // directly -- `mint_key` never reaches the ordinary `INSERT` below for this
419                    // path.
420                    return try_claim_and_mint(
421                        &pool, &state, &candidate, &caller, &body, key, key_id, &hash, created_at,
422                    )
423                    .await
424                    .map(|resp| (StatusCode::CREATED, Json(resp)));
425                }
426                _ => {
427                    return Err(IngestionKeyError::Forbidden(format!(
428                        "audience {candidate:?} is not in the caller's mintable set"
429                    )));
430                }
431            }
432        }
433    };
434
435    // Ordinary (non-claim) path: single INSERT, as today, using the `key`/`hash`/`key_id`/
436    // `created_at` generated above.
437    let created_by = caller
438        .email
439        .clone()
440        .unwrap_or_else(|| caller.subject.clone());
441
442    // Table name is a literal, never derived from caller input: no route in
443    // this module ever writes to `analytics_api_keys`.
444    sqlx::query(
445        "INSERT INTO ingestion_api_keys (key_id, key_hash, name, created_at, created_by, audience)
446         VALUES ($1, $2, $3, $4, $5, $6)",
447    )
448    .bind(key_id)
449    .bind(&hash[..])
450    .bind(&body.name)
451    .bind(created_at)
452    .bind(&created_by)
453    .bind(&audience)
454    .execute(&pool)
455    .await?;
456
457    info!(
458        "minted ingestion api key key_id={key_id} name={} created_by={created_by} audience={audience}",
459        body.name
460    );
461
462    Ok((
463        StatusCode::CREATED,
464        Json(MintResponse {
465            key_id,
466            name: body.name,
467            created_at,
468            audience,
469            key,
470        }),
471    ))
472}

474/// The lazy audience claim (AbAC Stage 6, #1374, Design §4a). Reached only from `mint_key`, only
475/// for a non-admin caller who explicitly named `audience`, has a known `caller.email`, and whose
476/// `MintPolicy::resolve_audience` call was just denied because `audience` carried no matching
477/// `mint` grant.
478///
479/// One transaction, on the same pool `mint_key` already has: takes a non-blocking, per-audience
480/// Postgres advisory lock to serialize concurrent claims for the *same* audience name (across
481/// processes, since advisory locks are server-instance-wide), then checks both `audience_grants`
482/// and `ingestion_api_keys` for any existing row naming this audience -- a genuinely fresh
483/// audience has neither. On a fresh audience: rejects the two reserved names (`public` and
484/// `state.default_audience`), enforces the per-caller claim bound
485/// (`max_claims_per_caller`), and writes `user:<email>` grant rows on **both** the `mint` and
486/// `read` axes (so the caller who just claimed the audience can read back what their own new key
487/// uploads) plus the `ingestion_api_keys` row itself, all in the same transaction.
488#[allow(clippy::too_many_arguments)]
489async fn try_claim_and_mint(
490    pool: &PgPool,
491    state: &IngestionKeysState,
492    audience: &str,
493    caller: &AuthContext,
494    body: &MintRequest,
495    key: String,
496    key_id: Uuid,
497    hash: &[u8],
498    created_at: DateTime<Utc>,
499) -> Result<MintResponse, IngestionKeyError> {
500    let mut tx = pool.begin().await?;
501
502    // `pg_try_advisory_xact_lock` (non-blocking), not the blocking `pg_advisory_xact_lock`: this
503    // transaction runs on a 2-connection pool shared with every other admin route in this crate,
504    // so a blocking acquire could exhaust it under concurrent contention. The `_try_` form
505    // returns immediately instead; Postgres advisory locks are server-instance-wide, so this
506    // still correctly serializes two concurrent claims for the *same* audience name across
507    // different processes. `_xact_lock` (not the session-level `pg_advisory_lock`) releases
508    // automatically at COMMIT/ROLLBACK, so a claim that errors out never leaks a held lock.
509    let locked: bool =
510        sqlx::query_scalar("SELECT pg_try_advisory_xact_lock(hashtextextended($1, 0))")
511            .bind(audience)
512            .fetch_one(&mut *tx)
513            .await?;
514    if !locked {
515        tx.rollback().await?;
516        // `Conflict` (409, `CLAIM_CONTENDED`), not `Forbidden` (403): this is transient lock
517        // contention, not a denial -- the caller should retry, not treat this as "you may not do
518        // this."
519        return Err(IngestionKeyError::Conflict(format!(
520            "audience {audience:?} is being claimed by another request -- retry"
521        )));
522    }
523
524    let caller_email = caller
525        .email
526        .as_deref()
527        .expect("mint_key only calls try_claim_and_mint when caller.email is Some");
528    let selector = format!("user:{caller_email}");
529    // Same 255-byte limit `audience_grants.rs`'s admin grant-write route enforces: the
530    // `audience_grants.selector` column is `VARCHAR(255)`, and an RFC-max email can push
531    // `"user:" + email` past that, which would otherwise 500 at the INSERT below instead of
532    // failing cleanly.
533    if selector.len() > 255 {
534        tx.rollback().await?;
535        return Err(IngestionKeyError::Forbidden(
536            "caller email is too long to form a valid grant selector".to_string(),
537        ));
538    }
539
540    // "Does this audience already have an owner?" -- true for any grant row (any axis/selector)
541    // *or* any existing `ingestion_api_keys` row: `audience_grants` alone would miss an audience
542    // an admin minted straight through `AudienceMintPolicy`'s `is_admin` arm before any grant
543    // ever existed for it.
544    let exists: bool = sqlx::query_scalar(
545        "SELECT EXISTS(SELECT 1 FROM audience_grants WHERE audience = $1)
546            OR EXISTS(SELECT 1 FROM ingestion_api_keys WHERE audience = $1)",
547    )
548    .bind(audience)
549    .fetch_one(&mut *tx)
550    .await?;
551
552    if exists {
553        // No recheck needed: reaching this function at all already means `resolve_audience` ran
554        // a fresh, uncached point query against exactly this audience's `mint` selectors moments
555        // earlier in this same request and found no match for the caller.
556        tx.rollback().await?;
557        return Err(IngestionKeyError::Forbidden(format!(
558            "audience {audience:?} already exists and the caller has no grant for it"
559        )));
560    } else {
561        // Genuinely fresh audience -- no grant row and no key row for it at all, so this is a
562        // real claim attempt. Reject the two reserved names here, not before the transaction
563        // opened: a caller who already holds a genuine `mint` grant on either name never reaches
564        // this branch at all -- `resolve_audience`'s point query would already have found that
565        // grant and approved the mint directly.
566        if audience == PUBLIC_AUDIENCE || Some(audience) == state.default_audience.as_deref() {
567            tx.rollback().await?;
568            return Err(IngestionKeyError::Forbidden(format!(
569                "audience {audience:?} cannot be claimed"
570            )));
571        }
572        // Genuinely fresh audience: this is a real claim, so it counts against the per-caller
573        // bound. Best-effort under concurrency: exact against another claim for *this same*
574        // audience (serialized by the lock above), but not against concurrent claims by the same
575        // caller for other, distinct fresh audience names, which take different locks.
576        let claim_count: i64 = sqlx::query_scalar(
577            "SELECT COUNT(DISTINCT audience) FROM audience_grants
578             WHERE axis = 'mint' AND selector = $1 AND created_by = $2",
579        )
580        .bind(&selector)
581        .bind(caller_email)
582        .fetch_one(&mut *tx)
583        .await?;
584        if claim_count >= state.max_claims_per_caller {
585            tx.rollback().await?;
586            return Err(IngestionKeyError::Forbidden(format!(
587                "caller has already claimed {claim_count} audiences (limit {})",
588                state.max_claims_per_caller
589            )));
590        }
591        for axis in ["mint", "read"] {
592            sqlx::query(
593                "INSERT INTO audience_grants (audience, axis, selector, created_at, created_by)
594                 VALUES ($1, $2, $3, now(), $4)",
595            )
596            .bind(audience)
597            .bind(axis)
598            .bind(&selector)
599            .bind(caller_email)
600            .execute(&mut *tx)
601            .await?;
602        }
603    }
604
605    sqlx::query(
606        "INSERT INTO ingestion_api_keys (key_id, key_hash, name, created_at, created_by, audience)
607         VALUES ($1, $2, $3, $4, $5, $6)",
608    )
609    .bind(key_id)
610    .bind(hash)
611    .bind(&body.name)
612    .bind(created_at)
613    .bind(caller_email)
614    .bind(audience)
615    .execute(&mut *tx)
616    .await?;
617
618    tx.commit().await?;
619
620    // `mint_key`'s own mint audit line is never reached for this path -- it returns from here
621    // directly -- so log both the mint and the claim here instead. The `exists`-true branch above
622    // always returns early, so every call that reaches this point took the `else`
623    // (genuinely-fresh) branch and wrote both new grant rows -- the claim line is unconditional.
624    info!(
625        "minted ingestion api key key_id={key_id} name={} created_by={caller_email} audience={audience}",
626        body.name
627    );
628    info!(
629        "claimed audience via lazy self-service mint audience={audience} selector={selector} created_by={caller_email} axes=mint,read"
630    );
631
632    Ok(MintResponse {
633        key_id,
634        name: body.name.clone(),
635        created_at,
636        audience: audience.to_string(),
637        key,
638    })
639}

641#[derive(Deserialize)]
642struct ListQuery {
643    limit: Option<i64>,
644    offset: Option<i64>,
645    include_revoked: Option<bool>,
646}
647
648#[derive(Serialize, sqlx::FromRow)]
649struct KeyListEntry {
650    key_id: Uuid,
651    name: String,
652    created_at: DateTime<Utc>,
653    created_by: String,
654    last_used_at: Option<DateTime<Utc>>,
655    revoked_at: Option<DateTime<Utc>>,
656    revoked_by: Option<String>,
657    audience: String,
658}
659
660/// `GET {base_path}/api/ingestion-api-keys?limit=&offset=&include_revoked=` —
661/// lists `ingestion_api_keys` rows, newest first. Never `key_hash`, never the
662/// key.
663async fn list_keys(
664    Extension(state): Extension<IngestionKeysState>,
665    AdminUser(_user): AdminUser,
666    Query(query): Query<ListQuery>,
667) -> Result<Json<Vec<KeyListEntry>>, IngestionKeyError> {
668    let pool = require_pool(&state)?;
669
670    let limit = query.limit.unwrap_or(DEFAULT_LIMIT);
671    if limit <= 0 {
672        return Err(IngestionKeyError::BadRequest(
673            "limit must be > 0".to_string(),
674        ));
675    }
676    // A read endpoint, so capping is safer than erroring: silently clamp
677    // rather than reject values above MAX_LIMIT.
678    let limit = limit.min(MAX_LIMIT);
679    let offset = query.offset.unwrap_or(0);
680    if offset < 0 {
681        return Err(IngestionKeyError::BadRequest(
682            "offset must be >= 0".to_string(),
683        ));
684    }
685    let include_revoked = query.include_revoked.unwrap_or(true);
686
687    let rows = if include_revoked {
688        sqlx::query_as::<_, KeyListEntry>(
689            "SELECT key_id, name, created_at, created_by, last_used_at, revoked_at, revoked_by, audience
690             FROM ingestion_api_keys
691             ORDER BY created_at DESC
692             LIMIT $1 OFFSET $2",
693        )
694        .bind(limit)
695        .bind(offset)
696        .fetch_all(&pool)
697        .await?
698    } else {
699        sqlx::query_as::<_, KeyListEntry>(
700            "SELECT key_id, name, created_at, created_by, last_used_at, revoked_at, revoked_by, audience
701             FROM ingestion_api_keys
702             WHERE revoked_at IS NULL
703             ORDER BY created_at DESC
704             LIMIT $1 OFFSET $2",
705        )
706        .bind(limit)
707        .bind(offset)
708        .fetch_all(&pool)
709        .await?
710    };
711
712    Ok(Json(rows))
713}

715#[derive(Serialize)]
716struct RevokeResponse {
717    revoked_at: DateTime<Utc>,
718}
719
720/// `DELETE {base_path}/api/ingestion-api-keys/{key_id}` — idempotent in one
721/// statement, preserving the original revocation time on a repeat call.
722///
723/// No `effective_within_seconds` field, unlike the removed ingestion-hosted
724/// `revoke_key`: that field threaded the *validating* provider's
725/// `cache_ttl_secs`, but nothing in `analytics-web-srv` runs a
726/// `DbApiKeyAuthProvider` — there is no running cache TTL here to report. The
727/// revocation latency is still bounded by whichever ingestion/flight-sql
728/// process's cache TTL is validating the key, documented in the runbook
729/// rather than echoed by this response.
730async fn revoke_key(
731    Extension(state): Extension<IngestionKeysState>,
732    AdminUser(user): AdminUser,
733    Path(key_id): Path<Uuid>,
734) -> Result<Json<RevokeResponse>, IngestionKeyError> {
735    let pool = require_pool(&state)?;
736    let revoked_by = user.email.clone().unwrap_or_else(|| user.subject.clone());
737
738    let row = sqlx::query(
739        "UPDATE ingestion_api_keys
740         SET revoked_at = COALESCE(revoked_at, now()),
741             revoked_by = COALESCE(revoked_by, $2)
742         WHERE key_id = $1
743         RETURNING revoked_at",
744    )
745    .bind(key_id)
746    .bind(&revoked_by)
747    .fetch_optional(&pool)
748    .await?;
749
750    match row {
751        Some(row) => {
752            let revoked_at: DateTime<Utc> = row.try_get("revoked_at")?;
753            info!("revoked ingestion api key key_id={key_id} revoked_by={revoked_by}");
754            Ok(Json(RevokeResponse { revoked_at }))
755        }
756        None => Err(IngestionKeyError::NotFound),
757    }
758}

760#[derive(Deserialize)]
761#[serde(deny_unknown_fields)]
762struct ImportRequest {
763    name: String,
764    key: String,
765    audience: Option<String>,
766}
767
768#[derive(Serialize)]
769struct ImportResponse {
770    key_id: Uuid,
771    name: String,
772    created_at: DateTime<Utc>,
773    created_by: String,
774    /// `null` unless the already-present row (on the `imported: false` path)
775    /// was itself revoked.
776    revoked_at: Option<DateTime<Utc>>,
777    /// `true` on a fresh insert; `false` when `key_hash` already existed.
778    imported: bool,
779    /// The audience the row actually carries. On the already-present
780    /// (`imported: false`) path this is the **existing** row's audience, never the
781    /// request's -- the binding is immutable, so an import never rewrites it.
782    audience: String,
783}
784
785/// Row shape shared by both branches of `import_key`'s `INSERT ... ON
786/// CONFLICT` / fallback `SELECT`.
787#[derive(sqlx::FromRow)]
788struct ImportedRow {
789    key_id: Uuid,
790    name: String,
791    created_at: DateTime<Utc>,
792    created_by: String,
793    revoked_at: Option<DateTime<Utc>>,
794    audience: String,
795}
796
797/// `POST {base_path}/api/ingestion-api-keys/import` — a route the removed
798/// proxy never had (the CLI called ingestion's own import route directly
799/// instead); now required since the CLI's `--table ingestion` path always
800/// targets `analytics-web-srv` and ingestion no longer has an import route of
801/// its own to fall back on.
802///
803/// Hashes and stores a caller-supplied key string verbatim, rather than
804/// generating a fresh one. `created_by` is the importing caller's own OIDC
805/// identity, never the literal string `"import"`.
806///
807/// No format validation on `key` beyond non-empty: `hash_key` covers the whole
808/// string regardless of shape, which is what lets an operator-chosen legacy
809/// key of any format import cleanly.
810async fn import_key(
811    Extension(state): Extension<IngestionKeysState>,
812    AdminUser(user): AdminUser,
813    Json(body): Json<ImportRequest>,
814) -> Result<(StatusCode, Json<ImportResponse>), IngestionKeyError> {
815    let pool = require_pool(&state)?;
816    validate_name(&body.name)?;
817    if body.key.is_empty() {
818        return Err(IngestionKeyError::BadRequest(
819            "key must not be empty".to_string(),
820        ));
821    }
822    // `fallback: Some(PUBLIC_AUDIENCE)` -- continuity with the v6 backfill: a legacy key's
823    // already-ingested history was just stamped `public`, so an import with no explicit
824    // audience and no knob keeps the new rows under the same audience rather than a 400.
825    let audience = resolve_audience(&state, body.audience.as_deref(), Some(PUBLIC_AUDIENCE))?;
826
827    let hash = hash_key(&body.key);
828    let key_id = Uuid::new_v4();
829    let created_at = Utc::now();
830    let created_by = user.email.clone().unwrap_or_else(|| user.subject.clone());
831
832    let inserted = sqlx::query_as::<_, ImportedRow>(
833        "INSERT INTO ingestion_api_keys (key_id, key_hash, name, created_at, created_by, audience)
834         VALUES ($1, $2, $3, $4, $5, $6)
835         ON CONFLICT (key_hash) DO NOTHING
836         RETURNING key_id, name, created_at, created_by, revoked_at, audience",
837    )
838    .bind(key_id)
839    .bind(&hash[..])
840    .bind(&body.name)
841    .bind(created_at)
842    .bind(&created_by)
843    .bind(&audience)
844    .fetch_optional(&pool)
845    .await?;
846
847    let (row, imported, status) = match inserted {
848        Some(row) => (row, true, StatusCode::CREATED),
849        None => {
850            // The hash already exists: report the existing row (including
851            // whether it's revoked, and its actual, immutable audience) instead
852            // of the freshly-generated values above, which never made it into
853            // the table.
854            let row = sqlx::query_as::<_, ImportedRow>(
855                "SELECT key_id, name, created_at, created_by, revoked_at, audience
856                 FROM ingestion_api_keys
857                 WHERE key_hash = $1",
858            )
859            .bind(&hash[..])
860            .fetch_one(&pool)
861            .await?;
862            (row, false, StatusCode::OK)
863        }
864    };
865
866    info!(
867        "imported ingestion api key key_id={} name={} created_by={} imported={imported} audience={}",
868        row.key_id, row.name, row.created_by, row.audience
869    );
870
871    Ok((
872        status,
873        Json(ImportResponse {
874            key_id: row.key_id,
875            name: row.name,
876            created_at: row.created_at,
877            created_by: row.created_by,
878            revoked_at: row.revoked_at,
879            imported,
880            audience: row.audience,
881        }),
882    ))
883}

885/// Routes only — [`IngestionKeysState`] is layered separately in
886/// `web_server.rs::build_protected_routes`, the same way `app_db_pool`/
887/// `maps_state`/`analytics_keys_state` are.
888pub fn ingestion_keys_router(base_path: &str) -> Router {
889    Router::new()
890        .route(
891            &format!("{base_path}/api/ingestion-api-keys"),
892            post(mint_key).get(list_keys),
893        )
894        .route(
895            &format!("{base_path}/api/ingestion-api-keys/{{key_id}}"),
896            delete(revoke_key),
897        )
898        .route(
899            &format!("{base_path}/api/ingestion-api-keys/import"),
900            post(import_key),
901        )
