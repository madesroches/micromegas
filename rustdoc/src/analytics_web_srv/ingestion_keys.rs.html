1//! Ingestion key management API for `analytics-web-srv`.
2//!
3//! Modeled directly on `analytics_keys.rs`, targeting `ingestion_api_keys`
4//! instead of `analytics_api_keys`. Ingestion should only do ingestion, so
5//! `analytics-web-srv` writes directly to `ingestion_api_keys` via the same
6//! telemetry-DB pool it already opens for `analytics_api_keys` — both tables
7//! live in the same database behind `MICROMEGAS_SQL_CONNECTION_STRING`.
8//! Routes live under `{base_path}/api/ingestion-api-keys`, distinct from this
9//! service's own `/auth/*` routes (login/callback/refresh/logout/me) — a
10//! completely different concern (browser session lifecycle).
11//!
12//! Every mint/revoke/import records the acting caller's own OIDC identity,
13//! never a shared service credential. `list_keys`/`revoke_key`/`import_key`
14//! are gated via the [`AdminUser`] extractor; `mint_key` runs through
15//! [`MintGate`]/[`AuthenticatedUser`] instead, since minting is not purely
16//! admin-gated -- see that extractor's own doc comment.
17//!
18//! **Duplication, accepted.** This duplicates most of `analytics_keys.rs`'s
19//! validation/SQL/error shape — deliberately, per that module's own doc
20//! comment: sharing it would mean a generic abstraction over a handful of
21//! near-identical handlers differing only in which table they target, the
22//! same shape the codebase already declines to share between
23//! `data_sources.rs`/`screens.rs`/`folders.rs` today.
24

49/// Holds the (possibly absent) telemetry-DB pool for the ingestion-key
50/// routes. `None` only when `MICROMEGAS_SQL_CONNECTION_STRING` is unset — the
51/// routes stay registered either way and return 503 per-request in that case,
52/// the same always-register-503-when-unconfigured shape `AnalyticsKeysState`
53/// uses. An unmigrated DB (missing the v5 migration's tables) is a separate
54/// failure mode: the pool is still `Some`, and a request fails with a 500 at
55/// query time instead.
56#[derive(Clone)]
57pub struct IngestionKeysState {
58    pub pool: Option<PgPool>,
59    /// The deployment default audience, resolved once at startup from `{prefix}_DEFAULT_AUDIENCE`
60    /// (`micromegas::auth::policy::default_audience_from_env`, `web_server.rs`), `public` when
61    /// unset. Always present: both `mint` and `import` fall back to it when a request names no
62    /// audience. See [`resolve_audience`].
63    pub default_audience: String,
64    /// Off-by-default self-service mint gate. Resolved once at startup
65    /// from `MICROMEGAS_SELF_SERVICE_MINT` (`web_server.rs`), default `false`. Checked by
66    /// [`MintGate`] for every non-admin caller before `mint_key`'s body runs at all -- with the
67    /// knob off, mint stays admin-only.
68    pub self_service_mint_enabled: bool,
69    /// `MICROMEGAS_SELF_SERVICE_MAX_CLAIMS_PER_CALLER`, default 25 -- caps how many distinct
70    /// audiences one non-admin caller may claim via the lazy claim path
71    /// ([`try_claim_and_mint`]). Best-effort under concurrency, not a hard ceiling -- see that
72    /// function's own doc comment.
73    pub max_claims_per_caller: i64,
74    /// `MICROMEGAS_SELF_SERVICE_MAX_KEYS_PER_CALLER`, default 100 -- caps how many *live* keys
75    /// one non-admin caller may hold at once, checked in `mint_key` regardless of which path
76    /// mints the next one. Best-effort under concurrency, not a hard ceiling.
77    pub max_keys_per_caller: i64,
78}

99/// Errors this API returns.
100///
101/// `Forbidden`/`Unavailable`/`Unauthenticated`/`Conflict` all back the self-service mint path:
102/// `mint_key` is [`MintGate`]/[`AuthenticatedUser`]-gated, not [`AdminUser`]-gated, and its own
103/// denials (the off-by-default gate, a missing-grant/malformed-audience `MintPolicy` denial, a
104/// per-caller bound, and lock contention on a lazy claim) need their own status codes.
105/// `list_keys`/`revoke_key`/`import_key` stay `AdminUser`-gated and never construct any of the
106/// four.
107#[derive(Debug)]
108pub enum IngestionKeyError {
109    /// Request body/query failed validation.
110    BadRequest(String),
111    /// Unknown `key_id`.
112    NotFound,
113    /// A DB error.
114    Database(sqlx::Error),
115    /// `state.pool == None` — the telemetry-DB pool was never configured
116    /// (`MICROMEGAS_SQL_CONNECTION_STRING` unset).
117    NotConfigured,
118    /// Self-service mint denied: the `MICROMEGAS_SELF_SERVICE_MINT` gate is off for a non-admin
119    /// caller, `MintPolicy::resolve_audience` denied the request (no matching grant, or a
120    /// malformed audience from an admin), the audience is not eligible for a lazy claim, or a
121    /// per-caller bound (`max_claims_per_caller`/`max_keys_per_caller`) was reached.
122    Forbidden(String),
123    /// The audience-grant point query behind `resolve_audience`'s policy call (or the row parse
124    /// immediately after it) failed -- a DB outage, not a denial, so it must not be
125    /// misattributed as "you have no grant." Distinct from `NotConfigured`, whose message is
126    /// specifically about an unset `MICROMEGAS_SQL_CONNECTION_STRING` and would mislead here.
127    Unavailable(String),
128    /// [`AuthenticatedUser`]/[`MintGate`] found no `AuthContext` extension in the request --
129    /// normally unreachable once routing is wired correctly; see `AuthenticatedUser`'s own doc
130    /// comment for the (fail-closed) case this covers.
131    Unauthenticated(String),
132    /// A lazy claim ([`try_claim_and_mint`]) lost the per-audience advisory-lock race to a
133    /// concurrent claimant. Transient lock contention, not a denial -- the caller (in particular
134    /// `micromegas-setup-telemetry`) should retry, not treat this as "you may not do this."
135    Conflict(String),
136}
137
138impl IntoResponse for IngestionKeyError {
139    fn into_response(self) -> Response {
140        match self {
141            IngestionKeyError::BadRequest(msg) => (
142                StatusCode::BAD_REQUEST,
143                Json(ErrorResponse::new("BAD_REQUEST", msg)),
144            )
145                .into_response(),
146            IngestionKeyError::NotFound => (
147                StatusCode::NOT_FOUND,
148                Json(ErrorResponse::new("NOT_FOUND", "key not found")),
149            )
150                .into_response(),
151            IngestionKeyError::Database(err) => {
152                error!("ingestion_keys: database error: {err}");
153                (
154                    StatusCode::INTERNAL_SERVER_ERROR,
155                    Json(ErrorResponse::new(
156                        "DATABASE_ERROR",
157                        "internal database error",
158                    )),
159                )
160                    .into_response()
161            }
162            IngestionKeyError::NotConfigured => (
163                StatusCode::SERVICE_UNAVAILABLE,
164                Json(ErrorResponse::new(
165                    "NOT_CONFIGURED",
166                    "ingestion key store not configured: set MICROMEGAS_SQL_CONNECTION_STRING",
167                )),
168            )
169                .into_response(),
170            IngestionKeyError::Forbidden(msg) => (
171                StatusCode::FORBIDDEN,
172                Json(ErrorResponse::new("FORBIDDEN", msg)),
173            )
174                .into_response(),
175            IngestionKeyError::Unavailable(msg) => (
176                StatusCode::SERVICE_UNAVAILABLE,
177                Json(ErrorResponse::new("UNAVAILABLE", msg)),
178            )
179                .into_response(),
180            IngestionKeyError::Unauthenticated(msg) => (
181                StatusCode::UNAUTHORIZED,
182                Json(ErrorResponse::new("UNAUTHENTICATED", msg)),
183            )
184                .into_response(),
185            IngestionKeyError::Conflict(msg) => (
186                StatusCode::CONFLICT,
187                Json(ErrorResponse::new("CLAIM_CONTENDED", msg)),
188            )
189                .into_response(),
190        }
191    }
192}

209fn require_pool(state: &IngestionKeysState) -> Result<PgPool, IngestionKeyError> {

213fn validate_name(name: &str) -> Result<(), IngestionKeyError> {
214    if name.is_empty() {
215        return Err(IngestionKeyError::BadRequest(
216            "name must not be empty".to_string(),
217        ));
218    }
219    if name.len() > MAX_NAME_BYTES {
220        return Err(IngestionKeyError::BadRequest(format!(
221            "name must be at most {MAX_NAME_BYTES} bytes"
222        )));
223    }
224    Ok(())
225}

227/// Resolves the audience to stamp on a mint/import `INSERT`'s `NOT NULL` column. `pub`, not
228/// module-private, and sync with no pool access, so the whole resolution matrix is
229/// unit-testable without a database.
230///
231/// `requested`: a missing field or an empty string counts as absent (the empty string is not a
232/// name -- it fails [`is_valid_audience`] either way); anything else is taken **verbatim**, no
233/// case folding.
234///
235/// Resolution order: `requested` → `state.default_audience`, which is itself never absent
236/// (`MICROMEGAS_DEFAULT_AUDIENCE`, `public` when unset), so this always resolves *something* and
237/// the only failure left is a malformed explicit request.
238///
239/// This is format/defaulting validation only -- it runs before `mint_key`'s
240/// `MintPolicy::resolve_audience` authorization decision, and is
241/// unaware of grants or claims. Defaulting is not a grant: the policy call is still what decides
242/// whether this caller may mint for the audience that came out of here.
243pub fn resolve_audience(
244    state: &IngestionKeysState,
245    requested: Option<&str>,
246) -> Result<String, IngestionKeyError> {
247    let candidate = requested
248        .filter(|s| !s.is_empty())
249        .unwrap_or(&state.default_audience);
250    if is_valid_audience(candidate) {
251        Ok(candidate.to_string())
252    } else {
253        Err(IngestionKeyError::BadRequest(format!(
254            "invalid audience {candidate:?}: must match [A-Za-z0-9_-]{{1,255}}"
255        )))
256    }
257}

262    name: String,
263    audience: Option<String>,
264}

266#[derive(Serialize)]
267struct MintResponse {
268    key_id: Uuid,
269    name: String,
270    created_at: DateTime<Utc>,
271    audience: String,
272    /// The cleartext key, returned exactly once. Never logged, never
273    /// retrievable afterwards.
274    key: String,
275    /// `true` only when this call actually created `audience`'s first grant rows -- i.e. an
276    /// admin caller minting into a brand-new audience, server-side-claiming it exactly as
277    /// `try_claim_and_mint` already does for a non-admin's lazy claim.
278    /// `false` on every other path. Appended last -- additive JSON, existing
279    /// consumers unaffected.
280    claimed: bool,
281}

283/// `FromRequestParts` extractor for `mint_key` specifically -- yields the caller's `AuthContext`
284/// after enforcing the off-by-default self-service gate. Because
285/// this is a `FromRequestParts` impl, axum runs it -- like `AuthenticatedUser` itself -- before
286/// `Json<MintRequest>` ever parses the request body: a knob-off non-admin caller is rejected
287/// before the body is ever touched, mirroring `AdminUser`'s own ordering guarantee.
288struct MintGate(AuthContext);
289
290impl<S: Send + Sync> FromRequestParts<S> for MintGate {
291    type Rejection = IngestionKeyError;
292
293    async fn from_request_parts(parts: &mut Parts, state: &S) -> Result<Self, Self::Rejection> {
294        let AuthenticatedUser(caller) = AuthenticatedUser::from_request_parts(parts, state).await?;
295        // `.ok_or(...)`, not `.expect(...)`: every other extractor in this crate fails closed
296        // with a status code for a missing extension rather than panicking (`AdminUser`'s own
297        // `.ok_or(AdminRequired)` for the identical "extension should always be layered"
298        // situation) -- `MintGate` should not be the one exception.
299        let ingestion_state = parts
300            .extensions
301            .get::<IngestionKeysState>()
302            .cloned()
303            .ok_or(IngestionKeyError::NotConfigured)?; // reachable only if routing is
304        // misconfigured; the 503 body's wording (about `MICROMEGAS_SQL_CONNECTION_STRING`)
305        // doesn't literally describe this cause, but a 503 fail-closed beats a panic for a
306        // case that should never happen in a correctly wired router.
307        if !caller.is_admin() && !ingestion_state.self_service_mint_enabled {
308            return Err(IngestionKeyError::Forbidden(
309                "self-service minting is disabled".to_string(),
310            ));
311        }
312        Ok(MintGate(caller))
313    }
314}

316/// `POST {base_path}/api/ingestion-api-keys` — mints a new `ingestion_api_keys` row.
317///
318/// Authorization is `MintGate` (the off-by-default self-service gate) plus
319/// `MintPolicy::resolve_audience` (a per-request point query against `audience_grants`)
320/// -- not a flat `AdminUser` gate. Format/defaulting validation still runs first,
321/// through the free `resolve_audience` function below, so existing 400s stay unchanged.
322async fn mint_key(
323    Extension(state): Extension<IngestionKeysState>,
324    MintGate(caller): MintGate,
325    Json(body): Json<MintRequest>,
326) -> Result<(StatusCode, Json<MintResponse>), IngestionKeyError> {
327    let pool = require_pool(&state)?;
328    validate_name(&body.name)?;
329    // A request naming no audience mints for the deployment default (`MICROMEGAS_DEFAULT_AUDIENCE`,
330    // `public` when unset) -- the same value data written by a credential with no bound audience is
331    // stamped with. What still gates the mint is `MintPolicy::resolve_audience` below, which asks
332    // whether *this caller* may mint for that audience; the default only decides which audience
333    // gets asked about. `?` here still rejects a malformed explicit `audience` with the existing
334    // 400, so `MintPolicy::resolve_audience`'s own malformed arm stays unreachable from this route.
335    let candidate = resolve_audience(&state, body.audience.as_deref())?;
336
337    // Key material, generated here -- before the policy call, not after -- so both the
338    // ordinary and the lazy-claim path share one value to insert.
339    let key = generate_key();
340    let hash = hash_key(&key);
341    let key_id = Uuid::new_v4();
342    let created_at = Utc::now();
343
344    // Per-caller bound: caps how many *live* keys one non-admin may hold, regardless
345    // of which path below mints the next one. Admins are exempt -- this bounds self-service, not
346    // administration. Best-effort, not a hard ceiling, under concurrency: this `SELECT COUNT(*)`
347    // runs on `pool` outside any transaction, so N concurrent mint requests from the same caller
348    // all read the same pre-insert count and can all pass -- the cap is exact only for
349    // sequential use from one caller.
350    if !caller.is_admin() {
351        let caller_id = caller.email.as_deref().unwrap_or(&caller.subject);
352        let key_count: i64 = sqlx::query_scalar(
353            "SELECT COUNT(*) FROM ingestion_api_keys WHERE created_by = $1 AND revoked_at IS NULL",
354        )
355        .bind(caller_id)
356        .fetch_one(&pool)
357        .await?;
358        if key_count >= state.max_keys_per_caller {
359            return Err(IngestionKeyError::Forbidden(format!(
360                "caller already holds {key_count} live self-service keys (limit {})",
361                state.max_keys_per_caller
362            )));
363        }
364    }
365
366    // Mint authorization is a point query, not a cached snapshot: no
367    // `DbAudienceGrantsSource` is attached for mint at all. `audience` is the leading column of
368    // `audience_grants`'s `PRIMARY KEY (audience, axis, selector)`, so this is an index-only
369    // scan.
370    let mint_selectors: Vec<String> = sqlx::query_scalar(
371        "SELECT selector FROM audience_grants WHERE audience = $1 AND axis = 'mint'",
372    )
373    .bind(&candidate)
374    .fetch_all(&pool)
375    .await
376    // A failed query is a DB outage, not a denial -- must not be misattributed as "you have no
377    // grant," so it is mapped to `Unavailable` explicitly. Logs the real `sqlx::Error`
378    // server-side and renders a fixed, generic client message -- the raw error text (connection
379    // strings, table/column names) must not reach the client.
380    .map_err(|e| {
381        error!("ingestion_keys: audience grant point query failed: {e}");
382        IngestionKeyError::Unavailable("audience grant store unavailable".to_string())
383    })?;
384
385    let grants = AudienceGrants::from_rows(
386        mint_selectors
387            .into_iter()
388            .map(|selector| (candidate.clone(), GrantAxis::Mint, selector)),
389    )
390    // Every row read back out of `audience_grants` already passed its own `CHECK` constraints on
391    // write, so this arm is unreachable in practice; kept as a real status code rather than
392    // `.unwrap()`, for the same fail-closed reason `MintGate` gives its own `.ok_or(...)`.
393    .map_err(|e| {
394        error!("ingestion_keys: audience grant row parse failed: {e}");
395        IngestionKeyError::Unavailable("audience grant store unavailable".to_string())
396    })?;
397
398    // `store: None` (the `new` default) -- a mint policy here never attaches a
399    // `DbAudienceGrantsSource`.
400    let policy = AudienceMintPolicy::new(grants);
401
402    let audience = match policy.resolve_audience(&caller, Some(&candidate)).await {
403        Ok(aud) => {
404            // Admin server-side claim: `AudienceMintPolicy`'s
405            // admin arm always resolves `Ok`, so this route never reaches the non-admin lazy
406            // claim's `Err` arm below for an admin caller -- decided here, as a second, separate
407            // query, whether `aud` looks unclaimed (no grant row, no key row on any axis) before
408            // falling through to the ordinary insert. Skipped outright for a reserved name
409            // (`public`, `state.default_audience` -- never claimable) or when the admin has no
410            // email (an admin with no email can't be granted a `user:` row).
411            if caller.is_admin() {
412                let reserved = aud == PUBLIC_AUDIENCE || aud.as_str() == state.default_audience;
413                if !reserved && caller.email.is_some() {
414                    let already_owned: bool = sqlx::query_scalar(
415                        "SELECT EXISTS(SELECT 1 FROM audience_grants WHERE audience = $1)
416                            OR EXISTS(SELECT 1 FROM ingestion_api_keys WHERE audience = $1)",
417                    )
418                    .bind(&aud)
419                    .fetch_one(&pool)
420                    .await?;
421                    if !already_owned {
422                        // Commits its own grant + key rows and returns the finished response
423                        // directly, same as the non-admin lazy claim below.
424                        return try_claim_and_mint(
425                            &pool, &state, &aud, &caller, &body, key, key_id, &hash, created_at,
426                        )
427                        .await
428                        .map(|resp| (StatusCode::CREATED, Json(resp)));
429                    }
430                }
431            }
432            aud
433        }
434        // Malformed-audience arm; `candidate` is already valid-format (via `resolve_audience`
435        // above), so unreachable in practice.
436        Err(e) if caller.is_admin() => return Err(IngestionKeyError::Forbidden(e.to_string())),
437        Err(_) => {
438            // Non-admin, no matching `mint` grant for `candidate` among the rows the point query
439            // above just read -- try the lazy claim only when the caller explicitly
440            // named this audience (not merely `state.default_audience`), and has an email to
441            // claim with.
442            let explicit = body.audience.as_deref().filter(|s| !s.is_empty()).is_some();
443            match (explicit, caller.email.as_deref()) {
444                (true, Some(_email)) => {
445                    // Commits its own grant + key rows and returns the finished response
446                    // directly -- `mint_key` never reaches the ordinary `INSERT` below for this
447                    // path.
448                    return try_claim_and_mint(
449                        &pool, &state, &candidate, &caller, &body, key, key_id, &hash, created_at,
450                    )
451                    .await
452                    .map(|resp| (StatusCode::CREATED, Json(resp)));
453                }
454                _ => {
455                    return Err(IngestionKeyError::Forbidden(format!(
456                        "audience {candidate:?} is not in the caller's mintable set"
457                    )));
458                }
459            }
460        }
461    };
462
463    // Ordinary (non-claim) path: single INSERT, as today, using the `key`/`hash`/`key_id`/
464    // `created_at` generated above. `claimed: false` -- this path never writes a grant row (an
465    // admin minting into an *existing* audience, or any non-admin mint that didn't take the
466    // lazy-claim branch above).
467    let created_by = caller
468        .email
469        .clone()
470        .unwrap_or_else(|| caller.subject.clone());
471    let response = insert_key(
472        &pool,
473        &audience,
474        &created_by,
475        &body,
476        key,
477        key_id,
478        &hash,
479        created_at,
480        false,
481    )
482    .await?;
483
484    Ok((StatusCode::CREATED, Json(response)))
485}

487/// Plain, non-transactional `INSERT INTO ingestion_api_keys` -- what `mint_key`'s own ordinary
488/// (non-claim) path calls directly, and what every "the in-lock recheck disagreed with the
489/// pre-check" branch inside [`try_claim_and_mint`] falls through to for an admin caller instead
490/// of erroring (the claim is best-effort for an admin, never a
491/// mint failure). Table name is a literal, never derived from caller input: no route in this
492/// module ever writes to `analytics_api_keys`.
493#[allow(clippy::too_many_arguments)]
494async fn insert_key(
495    pool: &PgPool,
496    audience: &str,
497    created_by: &str,
498    body: &MintRequest,
499    key: String,
500    key_id: Uuid,
501    hash: &[u8],
502    created_at: DateTime<Utc>,
503    claimed: bool,
504) -> Result<MintResponse, IngestionKeyError> {
505    sqlx::query(
506        "INSERT INTO ingestion_api_keys (key_id, key_hash, name, created_at, created_by, audience)
507         VALUES ($1, $2, $3, $4, $5, $6)",
508    )
509    .bind(key_id)
510    .bind(hash)
511    .bind(&body.name)
512    .bind(created_at)
513    .bind(created_by)
514    .bind(audience)
515    .execute(pool)
516    .await?;
517
518    info!(
519        "minted ingestion api key key_id={key_id} name={} created_by={created_by} audience={audience}",
520        body.name
521    );
522
523    Ok(MintResponse {
524        key_id,
525        name: body.name.clone(),
526        created_at,
527        audience: audience.to_string(),
528        key,
529        claimed,
530    })
531}

533/// Counts the audiences a caller has already claimed, for `try_claim_and_mint`'s
534/// `max_claims_per_caller` check: `$1` binds the `user:<email>` selector, `$2` the caller email.
535///
536/// Counted from `audience_grants`, not `ingestion_api_keys`: counting distinct audiences in
537/// `ingestion_api_keys` over-counts, because that table can't tell a self-service claim apart
538/// from an ordinary mint into an audience the caller already held a `mint` grant on for some
539/// other reason (an admin grant, a `group:` share). A non-admin who legitimately holds
540/// pre-existing `mint` grants on many audiences and mints keys into each would be refused on
541/// their next genuinely fresh claim despite never having claimed anything. `audience_grants`
542/// with this exact predicate -- `axis = 'mint' AND selector = 'user:<email>' AND created_by =
543/// <email>` -- matches only the row shape a lazy claim itself writes, so it counts claims and
544/// nothing else.
545///
546/// This predicate alone would be resettable (delete the caller's own `mint`/`user:<email>` row
547/// via `delete_grant`'s "remove my access" own-row arm, then re-claim), so `delete_grant`
548/// (`audience_grants.rs`) refuses that specific deletion for a non-admin -- see its own comment.
549/// That closes the reset hole without reintroducing the over-count.
550///
551/// `pub` so the shape a live database would otherwise be needed to observe can be asserted from
552/// the test crate.
553pub const CLAIM_COUNT_SQL: &str = "SELECT COUNT(DISTINCT audience) FROM audience_grants \
554     WHERE axis = 'mint' AND selector = $1 AND created_by = $2";
555
556/// The lazy audience claim. Reached only from `mint_key`, only
557/// for a non-admin caller who explicitly named `audience`, has a known `caller.email`, and whose
558/// `MintPolicy::resolve_audience` call was just denied because `audience` carried no matching
559/// `mint` grant.
560///
561/// One transaction, on the same pool `mint_key` already has: takes a non-blocking, per-audience
562/// Postgres advisory lock to serialize concurrent claims for the *same* audience name (across
563/// processes, since advisory locks are server-instance-wide), then checks both `audience_grants`
564/// and `ingestion_api_keys` for any existing row naming this audience -- a genuinely fresh
565/// audience has neither. On a fresh audience: rejects the two reserved names (`public` and
566/// `state.default_audience`), enforces the per-caller claim bound
567/// (`max_claims_per_caller`, skipped for an admin), and writes `user:<email>` grant rows on
568/// **both** the `mint` and `read` axes (so the caller who just claimed the audience can read
569/// back what their own new key uploads) plus the `ingestion_api_keys` row itself, all in the
570/// same transaction.
571///
572/// **Admin mode never turns a claim attempt into a mint failure.**
573/// Called for an admin (`mint_key`'s own pre-check already found the audience looking
574/// unclaimed moments earlier, outside any lock), the in-lock recheck can still disagree -- lock
575/// contention, the row now existing, or the selector exceeding 255 bytes -- and for a non-admin
576/// each of those is a hard failure. For an admin, all three instead fall through to
577/// [`insert_key`]'s plain, non-transactional insert: the claim is best-effort, `claimed` comes
578/// back `false`, and the admin still gets their key.
579#[allow(clippy::too_many_arguments)]
580async fn try_claim_and_mint(
581    pool: &PgPool,
582    state: &IngestionKeysState,
583    audience: &str,
584    caller: &AuthContext,
585    body: &MintRequest,
586    key: String,
587    key_id: Uuid,
588    hash: &[u8],
589    created_at: DateTime<Utc>,
590) -> Result<MintResponse, IngestionKeyError> {
591    // Both the admin and non-admin call sites only ever call this function when
592    // `caller.email` is `Some` (mint_key's own pre-checks).
593    let caller_email = caller
594        .email
595        .as_deref()
596        .expect("mint_key only calls try_claim_and_mint when caller.email is Some");
597
598    let mut tx = pool.begin().await?;
599
600    // `pg_try_advisory_xact_lock` (non-blocking), not the blocking `pg_advisory_xact_lock`: this
601    // transaction runs on a 2-connection pool shared with every other admin route in this crate,
602    // so a blocking acquire could exhaust it under concurrent contention. The `_try_` form
603    // returns immediately instead; Postgres advisory locks are server-instance-wide, so this
604    // still correctly serializes two concurrent claims for the *same* audience name across
605    // different processes. `_xact_lock` (not the session-level `pg_advisory_lock`) releases
606    // automatically at COMMIT/ROLLBACK, so a claim that errors out never leaks a held lock.
607    let locked: bool =
608        sqlx::query_scalar("SELECT pg_try_advisory_xact_lock(hashtextextended($1, 0))")
609            .bind(audience)
610            .fetch_one(&mut *tx)
611            .await?;
612    if !locked {
613        tx.rollback().await?;
614        if caller.is_admin() {
615            return insert_key(
616                pool,
617                audience,
618                caller_email,
619                body,
620                key,
621                key_id,
622                hash,
623                created_at,
624                false,
625            )
626            .await;
627        }
628        // `Conflict` (409, `CLAIM_CONTENDED`), not `Forbidden` (403): this is transient lock
629        // contention, not a denial -- the caller should retry, not treat this as "you may not do
630        // this."
631        return Err(IngestionKeyError::Conflict(format!(
632            "audience {audience:?} is being claimed by another request -- retry"
633        )));
634    }
635
636    let selector = format!("user:{caller_email}");
637    // Same 255-byte limit `audience_grants.rs`'s grant-write routes enforce: the
638    // `audience_grants.selector` column is `VARCHAR(255)`, and an RFC-max email can push
639    // `"user:" + email` past that, which would otherwise 500 at the INSERT below instead of
640    // failing cleanly.
641    if selector.len() > 255 {
642        tx.rollback().await?;
643        if caller.is_admin() {
644            return insert_key(
645                pool,
646                audience,
647                caller_email,
648                body,
649                key,
650                key_id,
651                hash,
652                created_at,
653                false,
654            )
655            .await;
656        }
657        return Err(IngestionKeyError::Forbidden(
658            "caller email is too long to form a valid grant selector".to_string(),
659        ));
660    }
661
662    // "Does this audience already have an owner?" -- true for any grant row (any axis/selector)
663    // *or* any existing `ingestion_api_keys` row: `audience_grants` alone would miss an audience
664    // an admin minted straight through `AudienceMintPolicy`'s `is_admin` arm before any grant
665    // ever existed for it.
666    let exists: bool = sqlx::query_scalar(
667        "SELECT EXISTS(SELECT 1 FROM audience_grants WHERE audience = $1)
668            OR EXISTS(SELECT 1 FROM ingestion_api_keys WHERE audience = $1)",
669    )
670    .bind(audience)
671    .fetch_one(&mut *tx)
672    .await?;
673
674    if exists {
675        // No recheck needed for a non-admin: reaching this function at all already means
676        // `resolve_audience` ran a fresh, uncached point query against exactly this audience's
677        // `mint` selectors moments earlier in this same request and found no match for the
678        // caller. For an admin, `mint_key`'s own pre-check ran the identical `EXISTS` query
679        // moments earlier and found it `false` -- a concurrent claim by someone else can still
680        // have landed in between, which is exactly the race this in-lock recheck exists to
681        // catch.
682        tx.rollback().await?;
683        if caller.is_admin() {
684            return insert_key(
685                pool,
686                audience,
687                caller_email,
688                body,
689                key,
690                key_id,
691                hash,
692                created_at,
693                false,
694            )
695            .await;
696        }
697        return Err(IngestionKeyError::Forbidden(format!(
698            "audience {audience:?} already exists and the caller has no grant for it"
699        )));
700    } else {
701        // Genuinely fresh audience -- no grant row and no key row for it at all, so this is a
702        // real claim attempt. Reject the two reserved names here, not before the transaction
703        // opened: a caller who already holds a genuine `mint` grant on either name never reaches
704        // this branch at all -- `resolve_audience`'s point query would already have found that
705        // grant and approved the mint directly. (`mint_key`'s own admin pre-check already skips
706        // calling this function at all for a reserved name, so this arm is unreachable for an
707        // admin in practice; the `caller.is_admin()` fallback stays for defense in depth.)
708        if audience == PUBLIC_AUDIENCE || audience == state.default_audience {
709            tx.rollback().await?;
710            if caller.is_admin() {
711                return insert_key(
712                    pool,
713                    audience,
714                    caller_email,
715                    body,
716                    key,
717                    key_id,
718                    hash,
719                    created_at,
720                    false,
721                )
722                .await;
723            }
724            return Err(IngestionKeyError::Forbidden(format!(
725                "audience {audience:?} cannot be claimed"
726            )));
727        }
728        // Genuinely fresh audience: this is a real claim, so it counts against the per-caller
729        // bound -- skipped for an admin, same as the plain path's `max_keys_per_caller` bound.
730        // Best-effort under concurrency: exact against another claim for *this same* audience
731        // (serialized by the lock above), but not against concurrent claims by the same caller
732        // for other, distinct fresh audience names, which take different locks.
733        if !caller.is_admin() {
734            let claim_count: i64 = sqlx::query_scalar(CLAIM_COUNT_SQL)
735                .bind(&selector)
736                .bind(caller_email)
737                .fetch_one(&mut *tx)
738                .await?;
739            if claim_count >= state.max_claims_per_caller {
740                tx.rollback().await?;
741                return Err(IngestionKeyError::Forbidden(format!(
742                    "caller has already claimed {claim_count} audiences (limit {})",
743                    state.max_claims_per_caller
744                )));
745            }
746        }
747        for axis in ["mint", "read"] {
748            sqlx::query(
749                "INSERT INTO audience_grants (audience, axis, selector, created_at, created_by)
750                 VALUES ($1, $2, $3, now(), $4)",
751            )
752            .bind(audience)
753            .bind(axis)
754            .bind(&selector)
755            .bind(caller_email)
756            .execute(&mut *tx)
757            .await?;
758        }
759    }
760
761    sqlx::query(
762        "INSERT INTO ingestion_api_keys (key_id, key_hash, name, created_at, created_by, audience)
763         VALUES ($1, $2, $3, $4, $5, $6)",
764    )
765    .bind(key_id)
766    .bind(hash)
767    .bind(&body.name)
768    .bind(created_at)
769    .bind(caller_email)
770    .bind(audience)
771    .execute(&mut *tx)
772    .await?;
773
774    tx.commit().await?;
775
776    // `mint_key`'s own mint audit line is never reached for this path -- it returns from here
777    // directly -- so log both the mint and the claim here instead. The `exists`-true branch above
778    // always returns early, so every call that reaches this point took the `else`
779    // (genuinely-fresh) branch and wrote both new grant rows -- the claim line is unconditional.
780    info!(
781        "minted ingestion api key key_id={key_id} name={} created_by={caller_email} audience={audience}",
782        body.name
783    );
784    info!(
785        "claimed audience via self-service mint audience={audience} selector={selector} created_by={caller_email} axes=mint,read"
786    );
787
788    Ok(MintResponse {
789        key_id,
790        name: body.name.clone(),
791        created_at,
792        audience: audience.to_string(),
793        key,
794        claimed: true,
795    })
796}

805#[derive(Serialize, sqlx::FromRow)]
806struct KeyListEntry {
807    key_id: Uuid,
808    name: String,
809    created_at: DateTime<Utc>,
810    created_by: String,
811    last_used_at: Option<DateTime<Utc>>,
812    revoked_at: Option<DateTime<Utc>>,
813    revoked_by: Option<String>,
814    audience: String,
815}

817/// `GET {base_path}/api/ingestion-api-keys?limit=&offset=&include_revoked=` —
818/// lists `ingestion_api_keys` rows, newest first. Never `key_hash`, never the
819/// key.
820async fn list_keys(
821    Extension(state): Extension<IngestionKeysState>,
822    AdminUser(_user): AdminUser,
823    Query(query): Query<ListQuery>,
824) -> Result<Json<Vec<KeyListEntry>>, IngestionKeyError> {
825    let pool = require_pool(&state)?;
826
827    let limit = query.limit.unwrap_or(DEFAULT_LIMIT);
828    if limit <= 0 {
829        return Err(IngestionKeyError::BadRequest(
830            "limit must be > 0".to_string(),
831        ));
832    }
833    // A read endpoint, so capping is safer than erroring: silently clamp
834    // rather than reject values above MAX_LIMIT.
835    let limit = limit.min(MAX_LIMIT);
836    let offset = query.offset.unwrap_or(0);
837    if offset < 0 {
838        return Err(IngestionKeyError::BadRequest(
839            "offset must be >= 0".to_string(),
840        ));
841    }
842    let include_revoked = query.include_revoked.unwrap_or(true);
843
844    let rows = if include_revoked {
845        sqlx::query_as::<_, KeyListEntry>(
846            "SELECT key_id, name, created_at, created_by, last_used_at, revoked_at, revoked_by, audience
847             FROM ingestion_api_keys
848             ORDER BY created_at DESC
849             LIMIT $1 OFFSET $2",
850        )
851        .bind(limit)
852        .bind(offset)
853        .fetch_all(&pool)
854        .await?
855    } else {
856        sqlx::query_as::<_, KeyListEntry>(
857            "SELECT key_id, name, created_at, created_by, last_used_at, revoked_at, revoked_by, audience
858             FROM ingestion_api_keys
859             WHERE revoked_at IS NULL
860             ORDER BY created_at DESC
861             LIMIT $1 OFFSET $2",
862        )
863        .bind(limit)
864        .bind(offset)
865        .fetch_all(&pool)
866        .await?
867    };
868
869    Ok(Json(rows))
870}

873struct RevokeResponse {
874    revoked_at: DateTime<Utc>,
875}
876
877/// `DELETE {base_path}/api/ingestion-api-keys/{key_id}` — idempotent in one
878/// statement, preserving the original revocation time on a repeat call.
879///
880/// No `effective_within_seconds` field: that field threads the *validating* provider's
881/// `cache_ttl_secs`, but nothing in `analytics-web-srv` runs a
882/// `DbApiKeyAuthProvider` — there is no running cache TTL here to report. The
883/// revocation latency is still bounded by whichever ingestion/flight-sql
884/// process's cache TTL is validating the key, documented in the runbook
885/// rather than echoed by this response.
886async fn revoke_key(
887    Extension(state): Extension<IngestionKeysState>,
888    AdminUser(user): AdminUser,
889    Path(key_id): Path<Uuid>,
890) -> Result<Json<RevokeResponse>, IngestionKeyError> {
891    let pool = require_pool(&state)?;
892    let revoked_by = user.email.clone().unwrap_or_else(|| user.subject.clone());
893
894    let row = sqlx::query(
895        "UPDATE ingestion_api_keys
896         SET revoked_at = COALESCE(revoked_at, now()),
897             revoked_by = COALESCE(revoked_by, $2)
898         WHERE key_id = $1
899         RETURNING revoked_at",
900    )
901    .bind(key_id)
902    .bind(&revoked_by)
903    .fetch_optional(&pool)
904    .await?;
905
906    match row {
907        Some(row) => {
908            let revoked_at: DateTime<Utc> = row.try_get("revoked_at")?;
909            info!("revoked ingestion api key key_id={key_id} revoked_by={revoked_by}");
910            Ok(Json(RevokeResponse { revoked_at }))
911        }
912        None => Err(IngestionKeyError::NotFound),
913    }
914}

924#[derive(Serialize)]
925struct ImportResponse {
926    key_id: Uuid,
927    name: String,
928    created_at: DateTime<Utc>,
929    created_by: String,
930    /// `null` unless the already-present row (on the `imported: false` path)
931    /// was itself revoked.
932    revoked_at: Option<DateTime<Utc>>,
933    /// `true` on a fresh insert; `false` when `key_hash` already existed.
934    imported: bool,
935    /// The audience the row actually carries. On the already-present
936    /// (`imported: false`) path this is the **existing** row's audience, never the
937    /// request's -- the binding is immutable, so an import never rewrites it.
938    audience: String,
939}

941/// Row shape shared by both branches of `import_key`'s `INSERT ... ON
942/// CONFLICT` / fallback `SELECT`.
943#[derive(sqlx::FromRow)]
944struct ImportedRow {
945    key_id: Uuid,
946    name: String,
947    created_at: DateTime<Utc>,
948    created_by: String,
949    revoked_at: Option<DateTime<Utc>>,
950    audience: String,
951}

953/// `POST {base_path}/api/ingestion-api-keys/import` -- required because the CLI's
954/// `--table ingestion` path always targets `analytics-web-srv`, which is the only place
955/// that can mint or import ingestion keys.
956///
957/// Hashes and stores a caller-supplied key string verbatim, rather than
958/// generating a fresh one. `created_by` is the importing caller's own OIDC
959/// identity, never the literal string `"import"`.
960///
961/// No format validation on `key` beyond non-empty: `hash_key` covers the whole
962/// string regardless of shape, which is what lets an operator-chosen legacy
963/// key of any format import cleanly.
964async fn import_key(
965    Extension(state): Extension<IngestionKeysState>,
966    AdminUser(user): AdminUser,
967    Json(body): Json<ImportRequest>,
968) -> Result<(StatusCode, Json<ImportResponse>), IngestionKeyError> {
969    let pool = require_pool(&state)?;
970    validate_name(&body.name)?;
971    if body.key.is_empty() {
972        return Err(IngestionKeyError::BadRequest(
973            "key must not be empty".to_string(),
974        ));
975    }
976    // Falls back to the deployment default (`public` unless configured otherwise via
977    // `MICROMEGAS_DEFAULT_AUDIENCE`) -- matching how already-ingested legacy data is stamped.
978    let audience = resolve_audience(&state, body.audience.as_deref())?;
979
980    let hash = hash_key(&body.key);
981    let key_id = Uuid::new_v4();
982    let created_at = Utc::now();
983    let created_by = user.email.clone().unwrap_or_else(|| user.subject.clone());
984
985    let inserted = sqlx::query_as::<_, ImportedRow>(
986        "INSERT INTO ingestion_api_keys (key_id, key_hash, name, created_at, created_by, audience)
987         VALUES ($1, $2, $3, $4, $5, $6)
988         ON CONFLICT (key_hash) DO NOTHING
989         RETURNING key_id, name, created_at, created_by, revoked_at, audience",
990    )
991    .bind(key_id)
992    .bind(&hash[..])
993    .bind(&body.name)
994    .bind(created_at)
995    .bind(&created_by)
996    .bind(&audience)
997    .fetch_optional(&pool)
998    .await?;
999
1000    let (row, imported, status) = match inserted {
1001        Some(row) => (row, true, StatusCode::CREATED),
1002        None => {
1003            // The hash already exists: report the existing row (including
1004            // whether it's revoked, and its actual, immutable audience) instead
1005            // of the freshly-generated values above, which never made it into
1006            // the table.
1007            let row = sqlx::query_as::<_, ImportedRow>(
1008                "SELECT key_id, name, created_at, created_by, revoked_at, audience
1009                 FROM ingestion_api_keys
1010                 WHERE key_hash = $1",
1011            )
1012            .bind(&hash[..])
1013            .fetch_one(&pool)
1014            .await?;
1015            (row, false, StatusCode::OK)
1016        }
1017    };
1018
1019    info!(
1020        "imported ingestion api key key_id={} name={} created_by={} imported={imported} audience={}",
1021        row.key_id, row.name, row.created_by, row.audience
1022    );
1023
1024    Ok((
1025        status,
1026        Json(ImportResponse {
1027            key_id: row.key_id,
1028            name: row.name,
1029            created_at: row.created_at,
1030            created_by: row.created_by,
1031            revoked_at: row.revoked_at,
1032            imported,
1033            audience: row.audience,
1034        }),
1035    ))
1036}

1041pub fn ingestion_keys_router(base_path: &str) -> Router {
