1//! Settings admin API for `analytics-web-srv`.
2//!
3//! Admin-only REST routes over the `settings` table (migration v12). The registry of known keys
4//! and the single validation rule live in `micromegas::settings`; this module only reads and
5//! writes rows and audits every change. The read route is [`AdminUser`]-gated directly; the two
6//! mutation routes go through `SettingsAdminGate`, which additionally emits a control-plane audit
7//! record on rejection, the same shape as `groups.rs`.
8

23/// Holds the (possibly absent) telemetry-DB pool for the settings routes. `None` only when
24/// `MICROMEGAS_SQL_CONNECTION_STRING` is unset -- the routes stay registered either way and
25/// return 503 per-request in that case, the same shape `GroupsState` uses.
26#[derive(Clone)]
27pub struct SettingsState {
28    pub pool: Option<PgPool>,
29}

32struct ErrorResponse {

34    message: String,

37impl ErrorResponse {

39        Self {
40            code: code.to_string(),
41            message: message.into(),
42        }
43    }

46/// Errors this API returns. No `Forbidden` variant: the admin gate is the [`AdminUser`]
47/// extractor, whose rejection renders as `AdminRequired`'s own 403 body.
48pub enum SettingsError {
49    BadRequest(String),
50    NotFound,
51    Database(sqlx::Error),
52    NotConfigured,
53    /// A stored value that does not parse as JSON.
54    Internal(String),
55}
56
57impl IntoResponse for SettingsError {
58    fn into_response(self) -> Response {
59        match self {
60            SettingsError::BadRequest(msg) => (
61                StatusCode::BAD_REQUEST,
62                Json(ErrorResponse::new("BAD_REQUEST", msg)),
63            )
64                .into_response(),
65            SettingsError::NotFound => (
66                StatusCode::NOT_FOUND,
67                Json(ErrorResponse::new("NOT_FOUND", "setting not found")),
68            )
69                .into_response(),
70            SettingsError::Database(err) => {
71                error!("settings: database error: {err}");
72                (
73                    StatusCode::INTERNAL_SERVER_ERROR,
74                    Json(ErrorResponse::new(
75                        "DATABASE_ERROR",
76                        "internal database error",
77                    )),
78                )
79                    .into_response()
80            }
81            SettingsError::NotConfigured => (
82                StatusCode::SERVICE_UNAVAILABLE,
83                Json(ErrorResponse::new(
84                    "NOT_CONFIGURED",
85                    "settings store not configured: set MICROMEGAS_SQL_CONNECTION_STRING",
86                )),
87            )
88                .into_response(),
89            SettingsError::Internal(msg) => {
90                error!("settings: internal error: {msg}");
91                (
92                    StatusCode::INTERNAL_SERVER_ERROR,
93                    Json(ErrorResponse::new("INTERNAL_ERROR", "internal error")),
94                )
95                    .into_response()
96            }
97        }
98    }
99}

101impl From<sqlx::Error> for SettingsError {
102    fn from(err: sqlx::Error) -> Self {
103        SettingsError::Database(err)
104    }
105}

107impl AuditOutcome for SettingsError {
108    fn audit_outcome(&self) -> (&'static str, Option<String>) {
109        match self {
110            SettingsError::BadRequest(msg) => ("denied", Some(msg.clone())),
111            SettingsError::NotFound => ("denied", Some("setting not found".to_string())),
112            // The raw `sqlx::Error` text can carry SQL/connection detail, so the record gets the
113            // same fixed string the client does.
114            SettingsError::Database(_) => ("error", Some("internal database error".to_string())),
115            SettingsError::NotConfigured => (
116                "error",
117                Some(
118                    "settings store not configured: set MICROMEGAS_SQL_CONNECTION_STRING"
119                        .to_string(),
120                ),
121            ),
122            SettingsError::Internal(_) => ("error", Some("internal error".to_string())),
123        }
124    }
125}

127fn require_pool(state: &SettingsState) -> Result<PgPool, SettingsError> {
128    state.pool.clone().ok_or(SettingsError::NotConfigured)

131/// `caller`'s own identity for `updated_by` -- the same resolution every admin route in this
132/// crate uses.
133fn caller_identity(caller: &ValidatedUser) -> String {
134    caller
135        .email
136        .clone()
137        .unwrap_or_else(|| caller.subject.clone())
138}

140fn settings_gate_action(parts: &Parts) -> &'static str {
141    if parts.method == Method::DELETE {
142        action::RESET_SETTING
143    } else {
144        action::SET_SETTING
145    }
146}

148/// `FromRequestParts` extractor for the two mutation routes: delegates to [`AdminUser`] and on
149/// rejection emits a `"denied"` mutation-audit record before returning `AdminRequired`
150/// unchanged.
151struct SettingsAdminGate(ValidatedUser);
152
153impl<S: Send + Sync> FromRequestParts<S> for SettingsAdminGate {
154    type Rejection = AdminRequired;
155
156    async fn from_request_parts(parts: &mut Parts, state: &S) -> Result<Self, Self::Rejection> {
157        let action = settings_gate_action(parts);
158        let ClientIp(client_ip) = ClientIp::from_request_parts(parts, state)
159            .await
160            .expect("ClientIp is infallible");
161        match AdminUser::from_request_parts(parts, state).await {
162            Ok(AdminUser(user)) => Ok(SettingsAdminGate(user)),
163            Err(rejection) => {
164                let (actor, is_admin) = match parts.extensions.get::<ValidatedUser>() {
165                    Some(user) => (caller_identity(user), user.is_admin),
166                    None => ("unauthenticated".to_string(), false),
167                };
168                MutationAudit::new(action, actor, is_admin, client_ip)
169                    .emit_gate_outcome("denied", "Admin access required");
170                Err(rejection)
171            }
172        }
173    }
174}

177pub struct StoredSetting {

179    pub value: serde_json::Value,
180    pub updated_at: DateTime<Utc>,
181    pub updated_by: String,

184/// One entry of the `GET /api/settings` response.
185#[derive(Debug, Serialize)]
186pub struct SettingEntry {
187    pub key: &'static str,
188    pub kind: &'static str,
189    pub env_var: Option<&'static str>,
190    pub default: serde_json::Value,
191    pub description: &'static str,
192    pub value: Option<serde_json::Value>,
193    pub updated_at: Option<DateTime<Utc>>,
194    pub updated_by: Option<String>,
195}
196
197/// Turns the registry plus the stored rows into response entries, one per registered setting.
198/// Stored rows for unregistered keys (written by a newer replica) are not listed.
199pub fn build_entries(specs: &[SettingSpec], rows: &[StoredSetting]) -> Vec<SettingEntry> {
200    specs
201        .iter()
202        .map(|spec| {
203            let stored = rows.iter().find(|r| r.key == spec.key);
204            SettingEntry {
205                key: spec.key,
206                kind: spec.kind.as_str(),
207                env_var: spec.env_var,
208                default: serde_json::from_str(spec.default)
209                    .expect("SETTINGS defaults are valid JSON"),
210                description: spec.description,
211                value: stored.map(|r| r.value.clone()),
212                updated_at: stored.map(|r| r.updated_at),
213                updated_by: stored.map(|r| r.updated_by.clone()),
214            }
215        })
216        .collect()
217}
218
219fn parse_stored(key: &str, raw: &str) -> Result<serde_json::Value, SettingsError> {
220    serde_json::from_str(raw)
221        .map_err(|e| SettingsError::Internal(format!("stored value for settings.{key}: {e}")))
222}
223
224/// Reads the table directly, not through the cache, so the page shows what was just written.
225async fn load_rows(pool: &PgPool) -> Result<Vec<StoredSetting>, SettingsError> {
226    let rows: Vec<(String, String, DateTime<Utc>, String)> =
227        sqlx::query_as("SELECT key, value::text, updated_at, updated_by FROM settings")
228            .fetch_all(pool)
229            .await?;
230    rows.into_iter()
231        .map(|(key, raw, updated_at, updated_by)| {
232            Ok(StoredSetting {
233                value: parse_stored(&key, &raw)?,
234                key,
235                updated_at,
236                updated_by,
237            })
238        })
239        .collect()
240}

242/// `GET {base_path}/api/settings` -- one entry per registered setting.
243async fn list_settings(
244    Extension(state): Extension<SettingsState>,
245    AdminUser(_user): AdminUser,
246) -> Result<Json<Vec<SettingEntry>>, SettingsError> {
247    let pool = require_pool(&state)?;
248    let rows = load_rows(&pool).await?;
249    Ok(Json(build_entries(SETTINGS, &rows)))
250}

257/// `PUT {base_path}/api/settings/{key}` -- `404` on an unregistered key, `400` on a value that
258/// fails the key's validation.
259async fn put_setting(
260    Extension(state): Extension<SettingsState>,
261    SettingsAdminGate(user): SettingsAdminGate,
262    client_ip: ClientIp,
263    Path(key): Path<String>,
264    Json(body): Json<PutSettingRequest>,
265) -> Result<Json<SettingEntry>, SettingsError> {
266    let audit = MutationAudit::new(
267        action::SET_SETTING,
268        caller_identity(&user),
269        user.is_admin,
270        client_ip.0,
271    )
272    .setting(&key)
273    .value(&body.value.to_string());
274    let result = put_setting_inner(state, user, &key, body.value).await;
275    let audit = match &result {
276        Ok((_, Some(previous))) => audit.previous_value(&previous.to_string()),
277        _ => audit,
278    };
279    audit.emit(&result);
280    result.map(|(entry, _)| Json(entry))
281}
282
283async fn put_setting_inner(
284    state: SettingsState,
285    user: ValidatedUser,
286    key: &str,
287    value: serde_json::Value,
288) -> Result<(SettingEntry, Option<serde_json::Value>), SettingsError> {
289    let spec = find_spec(key).ok_or(SettingsError::NotFound)?;
290    spec.validate(&value)
291        .map_err(|e| SettingsError::BadRequest(format!("{e:#}")))?;
292    let pool = require_pool(&state)?;
293    let updated_by = caller_identity(&user);
294
295    let mut tx = pool.begin().await?;
296    // FOR UPDATE locks nothing when the row doesn't exist yet, so concurrent first writes would
297    // both see no previous value.
298    sqlx::query("SELECT pg_advisory_xact_lock(hashtext($1))")
299        .bind(key)
300        .execute(&mut *tx)
301        .await?;
302    let previous_raw: Option<String> =
303        sqlx::query_scalar("SELECT value::text FROM settings WHERE key = $1 FOR UPDATE")
304            .bind(key)
305            .fetch_optional(&mut *tx)
306            .await?;
307    let previous = previous_raw
308        .map(|raw| parse_stored(key, &raw))
309        .transpose()?;
310    let (raw, updated_at, updated_by): (String, DateTime<Utc>, String) = sqlx::query_as(
311        "INSERT INTO settings (key, value, updated_at, updated_by)
312         VALUES ($1, $2::text::jsonb, now(), $3)
313         ON CONFLICT (key) DO UPDATE
314           SET value = EXCLUDED.value, updated_at = EXCLUDED.updated_at,
315               updated_by = EXCLUDED.updated_by
316         RETURNING value::text, updated_at, updated_by",
317    )
318    .bind(key)
319    .bind(value.to_string())
320    .bind(&updated_by)
321    .fetch_one(&mut *tx)
322    .await?;
323    tx.commit().await?;
324
325    let stored = StoredSetting {
326        key: key.to_string(),
327        value: parse_stored(key, &raw)?,
328        updated_at,
329        updated_by,
330    };
331    let entry = build_entries(std::slice::from_ref(spec), &[stored])
332        .pop()
333        .expect("one spec yields one entry");
334    Ok((entry, previous))
335}

337/// `DELETE {base_path}/api/settings/{key}` -- the key reverts to its env var or default. `204`;
338/// `404` on an unregistered key or one with no row.
339async fn delete_setting(
340    Extension(state): Extension<SettingsState>,
341    SettingsAdminGate(user): SettingsAdminGate,
342    client_ip: ClientIp,
343    Path(key): Path<String>,
344) -> Result<StatusCode, SettingsError> {
345    let audit = MutationAudit::new(
346        action::RESET_SETTING,
347        caller_identity(&user),
348        user.is_admin,
349        client_ip.0,
350    )
351    .setting(&key);
352    let result = delete_setting_inner(state, &key).await;
353    let audit = match &result {
354        Ok(previous) => audit.previous_value(&previous.to_string()),
355        Err(_) => audit,
356    };
357    audit.emit(&result);
358    result.map(|_| StatusCode::NO_CONTENT)
359}
360
361async fn delete_setting_inner(
362    state: SettingsState,
363    key: &str,
364) -> Result<serde_json::Value, SettingsError> {
365    find_spec(key).ok_or(SettingsError::NotFound)?;
366    let pool = require_pool(&state)?;
367    let raw: Option<String> =
368        sqlx::query_scalar("DELETE FROM settings WHERE key = $1 RETURNING value::text")
369            .bind(key)
370            .fetch_optional(&pool)
371            .await?;
372    let raw = raw.ok_or(SettingsError::NotFound)?;
373    parse_stored(key, &raw)
374}

376pub fn settings_router(base_path: &str) -> Router {
377    Router::new()
378        .route(&format!("{base_path}/api/settings"), get(list_settings))
379        .route(
380            &format!("{base_path}/api/settings/{{key}}"),
381            axum::routing::put(put_setting).delete(delete_setting),
382        )
