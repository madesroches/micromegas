19/// Error response for screen operations.
20#[derive(Debug, Serialize)]
21pub struct ErrorResponse {
22    code: String,
23    message: String,
24}
25
26impl ErrorResponse {
27    fn new(code: &str, message: &str) -> Self {
28        Self {
29            code: code.to_string(),
30            message: message.to_string(),
31        }
32    }
33}
34
35/// Unified error type for screen handlers.
36#[derive(Debug)]
37pub enum ScreenError {
38    NotFound(String),
39    BadRequest(ErrorResponse),
40    Database(sqlx::Error),
41}
42
43impl IntoResponse for ScreenError {
44    fn into_response(self) -> Response {
45        match self {
46            ScreenError::NotFound(name) => {
47                let body = ErrorResponse::new("NOT_FOUND", &format!("Screen '{name}' not found"));
48                (StatusCode::NOT_FOUND, Json(body)).into_response()
49            }
50            ScreenError::BadRequest(err) => (StatusCode::BAD_REQUEST, Json(err)).into_response(),
51            ScreenError::Database(err) => {
52                error!("Database error: {}", err);
53                let body = ErrorResponse::new("DATABASE_ERROR", "Internal database error");
54                (StatusCode::INTERNAL_SERVER_ERROR, Json(body)).into_response()
55            }
56        }
57    }
58}
59
60impl From<sqlx::Error> for ScreenError {
61    fn from(err: sqlx::Error) -> Self {
62        ScreenError::Database(err)
63    }
64}
65
66impl From<ValidationError> for ScreenError {
67    fn from(err: ValidationError) -> Self {
68        ScreenError::BadRequest(ErrorResponse::new(&err.code, &err.message))
69    }
70}
71
72type ScreenResult<T> = Result<T, ScreenError>;
73
74// Locking every ancestor prefix of a folder path (not just the exact
75// destination path, via `expand_path_prefixes(path, false)`) is what makes
76// `create_screen`/`update_screen` serialize against a concurrent
77// `update_folder`/`delete_folder` on any ancestor folder: renaming/deleting
78// "team" takes a lock on "team" alone, so a screen create/move into
79// "team/sub" must also lock "team" (in addition to "team/sub") to contend
80// on that same key — otherwise the two transactions target disjoint
81// advisory-lock keys and can race under READ COMMITTED. Root (empty path)
82// is excluded — it is never locked, matching `folder_exists` in folders.rs.
83

88/// List all available screen types.
89#[span_fn]
90pub async fn list_screen_types() -> Json<Vec<serde_json::Value>> {
91    let types: Vec<_> = ScreenType::all()
92        .into_iter()
93        .map(|t| {
94            let info = t.info();
95            serde_json::json!({
96                "name": info.name,
97                "display_name": info.display_name,
98                "icon": info.icon,
99                "description": info.description
100            })
101        })
102        .collect();
103
104    Json(types)
105}

107/// Get the default configuration for a screen type.
108#[span_fn]
109pub async fn get_default_config(
110    Path(type_name): Path<String>,
111) -> ScreenResult<Json<serde_json::Value>> {
112    let screen_type: ScreenType = type_name.parse().map_err(|_| {
113        ScreenError::BadRequest(ErrorResponse::new(
114            "INVALID_SCREEN_TYPE",
115            &format!("Invalid screen type: {type_name}"),
116        ))
117    })?;
118
119    Ok(Json(screen_type.default_config()))
120}

128/// List all screens.
129#[span_fn]
130pub async fn list_screens(Extension(pool): Extension<PgPool>) -> ScreenResult<Json<Vec<Screen>>> {
131    let query = format!("SELECT {SCREEN_COLUMNS} FROM screens ORDER BY name");
132    let screens = instrument_named!(
133        sqlx::query_as::<_, Screen>(&query).fetch_all(&pool),
134        "sql_select_screens"
135    )
136    .await?;
137
138    Ok(Json(screens))
139}

150            .bind(&name)
151            .fetch_optional(&pool),

155    .ok_or_else(|| ScreenError::NotFound(name))?;

160/// Create a new screen.
161#[span_fn]
162pub async fn create_screen(
163    Extension(pool): Extension<PgPool>,
164    Extension(user): Extension<ValidatedUser>,
165    Json(request): Json<CreateScreenRequest>,
166) -> ScreenResult<(StatusCode, Json<Screen>)> {
167    // Normalize and validate name
168    let name = normalize_name(&request.name);
169    validate_name(&name)?;
170
171    // Validate destination folder path
172    validate_folder_path(&request.folder_path)?;
173
174    // Validate screen type
175    let _screen_type: ScreenType = request.screen_type.parse().map_err(|_| {
176        ScreenError::BadRequest(ErrorResponse::new(
177            "INVALID_SCREEN_TYPE",
178            &format!("Invalid screen type: {}", request.screen_type),
179        ))
180    })?;
181
182    let mut tx = pool.begin().await?;
183
184    // Lock the destination folder path *and all of its ancestors* so a
185    // concurrent delete/rename of any ancestor folder (which locks only its
186    // own exact path in `folders.rs`) can't race past this create and leave
187    // the screen under a folder that was just renamed/deleted away. Locked
188    // shortest-prefix-first (root-to-leaf), matching `update_folder`'s
189    // sorted-order deadlock-avoidance convention.
190    for lock_path in expand_path_prefixes(&request.folder_path, false) {
191        instrument_named!(
192            sqlx::query("SELECT pg_advisory_xact_lock(hashtext($1))")
193                .bind(&lock_path)
194                .execute(&mut *tx),
195            "sql_folder_advisory_lock"
196        )
197        .await?;
198    }
199
200    // Check for duplicate name
201    let exists = instrument_named!(
202        sqlx::query_scalar::<_, bool>("SELECT EXISTS(SELECT 1 FROM screens WHERE name = $1)")
203            .bind(&name)
204            .fetch_one(&mut *tx),
205        "sql_select_screen_exists"
206    )
207    .await?;
208
209    if exists {
210        return Err(ScreenError::BadRequest(ErrorResponse::new(
211            "DUPLICATE_NAME",
212            &format!("Screen with name '{name}' already exists"),
213        )));
214    }
215
216    // Use email if available, otherwise fall back to subject
217    let user_id = user.email.as_deref().unwrap_or(&user.subject);
218
219    // Insert screen
220    let query = format!(
221        "INSERT INTO screens (name, screen_type, config, created_by, updated_by, managed_by, folder_path)
222         VALUES ($1, $2, $3, $4, $4, $5, $6)
223         RETURNING {SCREEN_COLUMNS}"
224    );
225    let screen = instrument_named!(
226        sqlx::query_as::<_, Screen>(&query)
227            .bind(&name)
228            .bind(&request.screen_type)
229            .bind(&request.config)
230            .bind(user_id)
231            .bind(&request.managed_by)
232            .bind(&request.folder_path)
233            .fetch_one(&mut *tx),
234        "sql_insert_screen"
235    )
236    .await?;
237
238    tx.commit().await?;
239
240    info!("Created screen: {} by {}", name, user_id);
241    Ok((StatusCode::CREATED, Json(screen)))
242}

244/// Update an existing screen.
245#[span_fn]
246pub async fn update_screen(
247    Extension(pool): Extension<PgPool>,
248    Extension(user): Extension<ValidatedUser>,
249    Path(name): Path<String>,
250    Json(request): Json<UpdateScreenRequest>,
251) -> ScreenResult<Json<Screen>> {
252    if let Some(ref folder_path) = request.folder_path {
253        validate_folder_path(folder_path)?;
254    }
255
256    // Use email if available, otherwise fall back to subject
257    let user_id = user.email.as_deref().unwrap_or(&user.subject);
258
259    let mut tx = pool.begin().await?;
260
261    // If the folder is changing, lock both the screen's current folder path
262    // and the destination folder path *and all of their ancestors* (sorted,
263    // to match `update_folder`'s deadlock-avoidance convention) so a
264    // concurrent delete/rename of any ancestor of either folder can't race
265    // past this move. Root (empty path) is never locked.
266    if let Some(ref new_folder_path) = request.folder_path {
267        let current_folder_path = instrument_named!(
268            sqlx::query_scalar::<_, String>("SELECT folder_path FROM screens WHERE name = $1")
269                .bind(&name)
270                .fetch_optional(&mut *tx),
271            "sql_select_screen_folder_path"
272        )
273        .await?
274        .ok_or_else(|| ScreenError::NotFound(name.clone()))?;
275
276        let mut lock_paths: Vec<String> = expand_path_prefixes(&current_folder_path, false);
277        lock_paths.extend(expand_path_prefixes(new_folder_path, false));
278        lock_paths.sort_unstable();
279        lock_paths.dedup();
280
281        for lock_path in lock_paths {
282            instrument_named!(
283                sqlx::query("SELECT pg_advisory_xact_lock(hashtext($1))")
284                    .bind(lock_path)
285                    .execute(&mut *tx),
286                "sql_folder_advisory_lock"
287            )
288            .await?;
289        }
290    }
291
292    let query = format!(
293        "UPDATE screens
294         SET config = COALESCE($1, config), updated_by = $2, updated_at = NOW(),
295             managed_by = COALESCE($4, managed_by), folder_path = COALESCE($5, folder_path)
296         WHERE name = $3
297         RETURNING {SCREEN_COLUMNS}"
298    );
299    let screen = instrument_named!(
300        sqlx::query_as::<_, Screen>(&query)
301            .bind(&request.config)
302            .bind(user_id)
303            .bind(&name)
304            .bind(&request.managed_by)
305            .bind(&request.folder_path)
306            .fetch_optional(&mut *tx),
307        "sql_update_screen"
308    )
309    .await?
310    .ok_or_else(|| ScreenError::NotFound(name.clone()))?;
311
312    tx.commit().await?;
313
314    info!("Updated screen: {} by {}", name, user_id);
315    Ok(Json(screen))
316}

318/// Delete a screen.
319#[span_fn]
320pub async fn delete_screen(
321    Extension(pool): Extension<PgPool>,
322    Path(name): Path<String>,
323) -> ScreenResult<StatusCode> {
324    let result = instrument_named!(
325        sqlx::query("DELETE FROM screens WHERE name = $1")
326            .bind(&name)
327            .execute(&pool),
328        "sql_delete_screen"
329    )
330    .await?;
331
332    if result.rows_affected() == 0 {
333        return Err(ScreenError::NotFound(name));
334    }
335
336    info!("Deleted screen: {}", name);
337    Ok(StatusCode::NO_CONTENT)
338}
