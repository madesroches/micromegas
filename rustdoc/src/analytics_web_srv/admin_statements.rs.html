1//! Classification of SQL statements that modify the lakehouse.
2//!
3//! The web server refuses these on the standard query endpoint so that content written by
4//! someone else (a notebook) cannot run them with a viewing admin's token.
5
6use micromegas::servers::view_ddl::parse_view_ddl;
7
8/// A statement that modifies the lakehouse.
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub enum AdminStatement {
11    /// View DDL (`CREATE [OR REPLACE] / DROP MATERIALIZED VIEW`). `parsed` is false when the text
12    /// looked like view DDL but could not be parsed.
13    ViewDdl { parsed: bool },
14    /// A call to a mutating admin function, by name.
15    Function(&'static str),
16}
17
18/// Admin functions that change lakehouse state. Read-only admin functions are deliberately absent.
19pub const MUTATING_ADMIN_FUNCTIONS: &[&str] = &[
20    "retire_partitions",
21    "retire_partition_by_metadata",
22    "retire_partition_by_file",
23    "materialize_partitions",
24    "regenerate_partitions",
25    "deny_queries",
26    "remove_query_denial",
27];
28
29/// Returns the lakehouse-modifying statement `sql` contains, if any.
30///
31/// DDL detection reuses the FlightSQL parser so both agree on what counts as DDL; a parse error
32/// is treated as DDL (fail closed). Functions use a lower-cased substring match: it can
33/// over-block (a string literal naming a function) but never under-blocks, whatever the
34/// comments, quoting or whitespace.
35pub fn classify_admin_statement(sql: &str) -> Option<AdminStatement> {
36    match parse_view_ddl(sql) {
37        Ok(None) => {}
38        Ok(Some(_)) => return Some(AdminStatement::ViewDdl { parsed: true }),
39        Err(_) => return Some(AdminStatement::ViewDdl { parsed: false }),
40    }
41    let sql_lower = sql.to_lowercase();
42    MUTATING_ADMIN_FUNCTIONS
43        .iter()
44        .find(|&func| sql_lower.contains(func))
45        .map(|func| AdminStatement::Function(func))
46}
47
48impl AdminStatement {
49    /// Message returned to the client when the standard endpoint refuses this statement.
50    pub fn refusal_message(&self) -> String {
51        const FIX: &str = "run it from an admin page, or, as an admin, enable admin mode on a notebook from the user menu";
52        match self {
53            AdminStatement::ViewDdl { parsed: true } => {
54                format!("View DDL modifies the lakehouse; {FIX}")
55            }
56            AdminStatement::ViewDdl { parsed: false } => format!(
57                "The statement could not be parsed as view DDL and is refused as if it modified the lakehouse; {FIX}"
58            ),
59            AdminStatement::Function(name) => {
60                format!("'{name}' modifies the lakehouse; {FIX}")
61            }
62        }
63    }
64}
