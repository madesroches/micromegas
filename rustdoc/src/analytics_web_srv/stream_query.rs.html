29/// Request for streaming SQL query
30#[derive(Debug, Serialize, Deserialize)]
31pub struct StreamQueryRequest {
32    pub sql: String,
33    #[serde(default)]
34    pub params: HashMap<String, String>,
35    pub begin: Option<DateTime<Utc>>,
36    pub end: Option<DateTime<Utc>>,
37    #[serde(default)]
38    pub data_source: String,
39    /// Originating notebook name, for query attribution. Omitted outside a notebook.
40    pub notebook: Option<String>,
41    /// Originating cell name within the notebook, for query attribution.
42    pub cell: Option<String>,
43}

45/// Error codes for stream query errors
46#[derive(Debug, Clone, Copy, Serialize)]
47#[serde(rename_all = "SCREAMING_SNAKE_CASE")]
48pub enum ErrorCode {
49    InvalidSql,
50    ConnectionFailed,
51    Internal,
52    Forbidden,
53    DataSourceNotFound,
54}

56/// Schema and batch frames: size-prefixed binary with optional row count
57#[derive(Serialize)]
58struct DataHeader {
59    #[serde(rename = "type")]
60    frame_type: &'static str,
61    size: usize,
62    #[serde(skip_serializing_if = "Option::is_none")]
63    rows: Option<usize>,
64}
65
66/// Done frame to indicate successful completion
67#[derive(Serialize)]
68struct DoneFrame {
69    #[serde(rename = "type")]
70    frame_type: &'static str,
71}

73/// Error frame for streaming errors
74#[derive(Serialize)]
75struct ErrorFrame {
76    #[serde(rename = "type")]
77    frame_type: &'static str,
78    code: ErrorCode,
79    message: String,
80}

82/// Serialize a value to a JSON line (with trailing newline)
83fn json_line<T: Serialize>(value: &T) -> Bytes {
84    let mut json = serde_json::to_string(value).expect("serialization failed");
85    json.push('\n');
86    Bytes::from(json)
87}

89/// List of destructive functions that should be blocked in web queries
90const BLOCKED_FUNCTIONS: &[&str] = &[
91    "retire_partitions",
92    "retire_partition_by_metadata",
93    "retire_partition_by_file",
94];
95
96/// Check if the SQL query contains any blocked destructive functions
97pub fn contains_blocked_function(sql: &str) -> Option<&'static str> {
98    let sql_lower = sql.to_lowercase();
99    BLOCKED_FUNCTIONS
100        .iter()
101        .find(|&func| sql_lower.contains(func))
102        .copied()
103}

105/// Maximum length for a client-supplied origin label (notebook/cell name) before
106/// it's rejected outright rather than forwarded into a gRPC header and a log line.
107const MAX_ORIGIN_LABEL_LEN: usize = 128;
108
109/// Returns `value` if it's a safe gRPC metadata value (printable ASCII, no control
110/// characters, bounded length) and non-empty after trimming, else `None` so the
111/// caller silently omits the field — same reject-whole-value strategy as the
112/// Python client's `_sanitize_override` (attribution.py) for the analogous
113/// MICROMEGAS_CLIENT_AGENT/MICROMEGAS_CLIENT_ENTRYPOINT overrides.
114pub fn sanitize_origin_label(value: &str) -> Option<String> {
115    let trimmed = value.trim();
116    if trimmed.is_empty() || trimmed.len() > MAX_ORIGIN_LABEL_LEN {
117        return None;
118    }
119    if !trimmed.chars().all(|c| (' '..='~').contains(&c)) {
120        return None;
121    }
122    Some(trimmed.to_string())
123}

125/// Substitute macro variables in SQL query
126pub fn substitute_macros(sql: &str, params: &HashMap<String, String>) -> String {
127    let mut result = sql.to_string();
128    for (key, value) in params {
129        // Escape single quotes in values to prevent SQL injection
130        let escaped_value = value.replace('\'', "''");
131        // Replace $key with the escaped value
132        result = result.replace(&format!("${key}"), &escaped_value);
133    }
134    result
135}

137/// Encode a schema to Arrow IPC format
138///
139/// The tracker should be the same instance used for subsequent batch encoding
140/// to ensure dictionary IDs are consistent between schema and batches.
141pub fn encode_schema(
142    schema: &Schema,
143    tracker: &mut arrow_ipc::writer::DictionaryTracker,
144    options: &IpcWriteOptions,
145) -> Result<Vec<u8>> {
146    let mut buffer = Vec::new();
147    let data_gen = IpcDataGenerator::default();
148
149    let encoded = data_gen.schema_to_bytes_with_dictionary_tracker(schema, tracker, options);
150    write_message(&mut buffer, encoded, options).context("writing schema message")?;
151    Ok(buffer)
152}
153
154/// Encode a RecordBatch to Arrow IPC format
155pub fn encode_batch(
156    batch: &datafusion::arrow::array::RecordBatch,
157    tracker: &mut arrow_ipc::writer::DictionaryTracker,
158    options: &IpcWriteOptions,
159) -> Result<Vec<u8>> {
160    let mut buffer = Vec::new();
161    let data_gen = IpcDataGenerator::default();
162    let mut compression = CompressionContext::default();
163
164    let (encoded_dicts, encoded_batch) = data_gen
165        .encode(batch, tracker, options, &mut compression)
166        .context("encoding batch")?;
167
168    // Write dictionary batches first (if any)
169    for dict in encoded_dicts {
170        write_message(&mut buffer, dict, options).context("writing dictionary message")?;
171    }
172
173    // Write the main batch
174    write_message(&mut buffer, encoded_batch, options).context("writing batch message")?;
175
176    Ok(buffer)
177}

179/// Streaming SQL query endpoint using Arrow IPC protocol
180///
181/// Returns a stream of JSON-framed Arrow IPC messages:
182/// - `{"type":"schema","size":N}\n` followed by N bytes of schema IPC
183/// - `{"type":"batch","size":N}\n` followed by N bytes of batch IPC
184/// - `{"type":"done"}\n` on success
185/// - `{"type":"error","code":"..","message":"..}\n` on error
186#[span_fn]
187pub async fn stream_query_handler(
188    Extension(auth_token): Extension<AuthToken>,
189    Extension(cache): Extension<DataSourceCache>,
190    Json(request): Json<StreamQueryRequest>,
191) -> impl IntoResponse {
192    let notebook = request.notebook.as_deref().and_then(sanitize_origin_label);
193    let cell = request.cell.as_deref().and_then(sanitize_origin_label);
194
195    info!(
196        "stream query sql={} params={:?} begin={:?} end={:?} data_source={} notebook={notebook:?} cell={cell:?}",
197        request.sql, request.params, request.begin, request.end, request.data_source
198    );
199
200    // Check for blocked functions first (before starting the stream)
201    if let Some(blocked_func) = contains_blocked_function(&request.sql) {
202        return (
203            StatusCode::FORBIDDEN,
204            Json(ErrorFrame {
205                frame_type: "error",
206                code: ErrorCode::Forbidden,
207                message: format!(
208                    "The function '{blocked_func}' is not allowed in web queries for security reasons",
209                ),
210            }),
211        )
212            .into_response();
213    }
214
215    // Resolve data source
216    if request.data_source.is_empty() {
217        return (
218            StatusCode::BAD_REQUEST,
219            Json(ErrorFrame {
220                frame_type: "error",
221                code: ErrorCode::DataSourceNotFound,
222                message: "No data source specified".to_string(),
223            }),
224        )
225            .into_response();
226    }
227
228    let data_source_config = match cache.resolve(&request.data_source).await {
229        Ok(Some(config)) => config,
230        Ok(None) => {
231            return (
232                StatusCode::NOT_FOUND,
233                Json(ErrorFrame {
234                    frame_type: "error",
235                    code: ErrorCode::DataSourceNotFound,
236                    message: format!("Data source '{}' not found", request.data_source),
237                }),
238            )
239                .into_response();
240        }
241        Err(e) => {
242            error!(
243                "Failed to resolve data source '{}': {e}",
244                request.data_source
245            );
246            return (
247                StatusCode::INTERNAL_SERVER_ERROR,
248                Json(ErrorFrame {
249                    frame_type: "error",
250                    code: ErrorCode::Internal,
251                    message: "Failed to resolve data source".to_string(),
252                }),
253            )
254                .into_response();
255        }
256    };
257
258    let flightsql_url = data_source_config.url;
259
260    // Substitute macros
261    let sql = substitute_macros(&request.sql, &request.params);
262
263    // Build time range if provided
264    let time_range = match (request.begin, request.end) {
265        (Some(begin), Some(end)) => Some(TimeRange::new(begin, end)),
266        _ => None,
267    };
268
269    let stream = stream! {
270        // Create FlightSQL client
271        let mut client_factory = BearerFlightSQLClientFactory::new_with_client_type(
272            flightsql_url,
273            auth_token.0,
274            "web".to_string(),
275        );
276        if let Some(notebook) = &notebook {
277            client_factory = client_factory.with_metadata("x-client-notebook", notebook.clone());
278        }
279        if let Some(cell) = &cell {
280            client_factory = client_factory.with_metadata("x-client-cell", cell.clone());
281        }
282        let mut client = match client_factory.make_client().await {
283            Ok(c) => c,
284            Err(e) => {
285                yield Ok::<_, std::io::Error>(json_line(&ErrorFrame {
286                    frame_type: "error",
287                    code: ErrorCode::ConnectionFailed,
288                    message: e.to_string(),
289                }));
290                return;
291            }
292        };
293
294        // Start streaming query
295        let mut batch_stream = match client.query_stream(sql, time_range).await {
296            Ok(s) => s,
297            Err(e) => {
298                yield Ok(json_line(&ErrorFrame {
299                    frame_type: "error",
300                    code: ErrorCode::InvalidSql,
301                    message: e.to_string(),
302                }));
303                return;
304            }
305        };
306
307        // FlightSQL limitation: The schema is embedded in the first FlightData message,
308        // not sent separately. The FlightRecordBatchStream only populates its schema field
309        // after we read the first message. We must read the first batch here, extract the
310        // schema, send it to the client, then process this batch along with the rest.
311        let first_batch = batch_stream.next().await;
312
313        // Get schema from the stream (now available after reading first message)
314        let schema = match batch_stream.schema() {
315            Some(s) => s.clone(),
316            None => {
317                yield Ok(json_line(&ErrorFrame {
318                    frame_type: "error",
319                    code: ErrorCode::Internal,
320                    message: "No schema in response".to_string(),
321                }));
322                return;
323            }
324        };
325
326        // Track dictionaries across schema and batches
327        // Must use same tracker for schema and batches to ensure dictionary IDs align
328        let mut dict_tracker = arrow_ipc::writer::DictionaryTracker::new(false);
329        let options = IpcWriteOptions::default()
330            .try_with_compression(Some(arrow_ipc::CompressionType::LZ4_FRAME))
331            .expect("enabling LZ4 compression");
332
333        // Encode and send schema
334        let schema_bytes = match encode_schema(&schema, &mut dict_tracker, &options) {
335            Ok(bytes) => bytes,
336            Err(e) => {
337                yield Ok(json_line(&ErrorFrame {
338                    frame_type: "error",
339                    code: ErrorCode::Internal,
340                    message: format!("{e:#}"),
341                }));
342                return;
343            }
344        };
345
346        yield Ok(json_line(&DataHeader {
347            frame_type: "schema",
348            size: schema_bytes.len(),
349            rows: None,
350        }));
351        yield Ok(Bytes::from(schema_bytes));
352
353        // Helper to encode and yield a batch
354        macro_rules! yield_batch {
355            ($batch:expr) => {
356                let batch = &$batch;
357                let num_rows = batch.num_rows();
358                let batch_bytes = match encode_batch(batch, &mut dict_tracker, &options) {
359                    Ok(bytes) => bytes,
360                    Err(e) => {
361                        yield Ok(json_line(&ErrorFrame {
362                            frame_type: "error",
363                            code: ErrorCode::Internal,
364                            message: format!("{e:#}"),
365                        }));
366                        return;
367                    }
368                };
369                yield Ok(json_line(&DataHeader {
370                    frame_type: "batch",
371                    size: batch_bytes.len(),
372                    rows: Some(num_rows),
373                }));
374                yield Ok(Bytes::from(batch_bytes));
375            };
376        }
377
378        // Process the first batch we read earlier (to get the schema)
379        if let Some(result) = first_batch {
380            match result {
381                Ok(batch) => {
382                    yield_batch!(batch);
383                }
384                Err(e) => {
385                    yield Ok(json_line(&ErrorFrame {
386                        frame_type: "error",
387                        code: ErrorCode::Internal,
388                        message: e.to_string(),
389                    }));
390                    return;
391                }
392            }
393        }
394
395        // Stream remaining batches
396        while let Some(result) = batch_stream.next().await {
397            match result {
398                Ok(batch) => {
399                    yield_batch!(batch);
400                }
401                Err(e) => {
402                    yield Ok(json_line(&ErrorFrame {
403                        frame_type: "error",
404                        code: ErrorCode::Internal,
405                        message: e.to_string(),
406                    }));
407                    return;
408                }
409            }
410        }
411
412        // Success
413        yield Ok(json_line(&DoneFrame { frame_type: "done" }));
414    };
415
416    (
417        [(
418            header::CONTENT_TYPE,
419            "application/x-micromegas-arrow-stream",
420        )],
421        Body::from_stream(stream),
422    )
423        .into_response()
424}
