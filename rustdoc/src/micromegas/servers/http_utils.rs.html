3/// Extracts the client IP address from HTTP headers and extensions.
4///
5/// This function checks sources in order of priority:
6/// 1. X-Forwarded-For (rightmost entry of the last header field line -- the address the nearest
7///    trusted proxy observed)
8/// 2. X-Real-IP (used by some proxies like nginx)
9/// 3. Socket address from extensions (direct connection)
10///
11/// The *rightmost* `X-Forwarded-For` entry is used, not the leftmost, because the AWS ALB every
12/// service is deployed behind *appends* the address it observed rather than overwriting the
13/// header (`routing.http.xff_header_processing.mode = append`, the ALB default). Every entry to
14/// the left of the last one is caller-supplied and therefore spoofable; the last entry is the
15/// ALB's own observation and cannot be forged by the caller *for requests that actually traversed
16/// the ALB* -- this holds even if the caller sends its own `X-Forwarded-For` as a *separate*
17/// header field line, since `HeaderMap::get` returns only the first such line and would surface a
18/// fully caller-chosen value; `get_all(...).iter().next_back()` is required to reach the ALB's
19/// line. This is correct for exactly one trusted proxy hop -- putting a second trusted proxy in front of the ALB
20/// would mean skipping one more entry from the right. For a request that reaches this service
21/// *without* going through the ALB (local dev, an in-cluster peer, or any other direct connection),
22/// there is no trusted proxy appending anything, so the rightmost entry is just as caller-chosen
23/// as every other entry and this guarantee does not apply -- the socket-address fallback (branch
24/// 3) is the only non-forgeable source in that case.
25///
26/// Every candidate is validated as a parseable `IpAddr` before being returned; a header value
27/// that fails to parse (e.g. malformed input or an attempted log-injection payload) is treated as
28/// absent and the function falls through to the next source in priority order instead of
29/// returning the raw text.
30///
31/// Returns "unknown" if no IP can be extracted.
32pub fn get_client_ip(headers: &http::HeaderMap, extensions: &http::Extensions) -> String {
33    // Check X-Forwarded-For header first (for load balancers/proxies).
34    // `get_all(...).iter().next_back()` -- not `get(...)`, which returns only the *first* field
35    // line -- picks the last field line, since the ALB appends its own observation as (or onto)
36    // the last line; within that line, the rightmost comma-separated entry is what the ALB
37    // itself observed. Everything else (earlier lines in full, and earlier entries within the
38    // last line) is caller-supplied and spoofable. `next_back()` (not `.last()`, which would
39    // needlessly walk the whole iterator) is equivalent here since `GetAll::iter()` is a
40    // `DoubleEndedIterator`.
41    if let Some(forwarded_for) = headers.get_all("x-forwarded-for").iter().next_back()
42        && let Ok(value) = forwarded_for.to_str()
43        && let Some(candidate) = value.rsplit(',').map(str::trim).find(|s| !s.is_empty())
44        && let Ok(client_ip) = candidate.parse::<std::net::IpAddr>()
45    {
46        return client_ip.to_string();
47    }
48
49    // Check X-Real-IP header (used by some proxies like nginx)
50    if let Some(real_ip) = headers.get("x-real-ip")
51        && let Ok(value) = real_ip.to_str()
52        && let Ok(client_ip) = value.trim().parse::<std::net::IpAddr>()
53    {
54        return client_ip.to_string();
55    }
56
57    // Fall back to socket address from extensions
58    // Axum provides ConnectInfo<SocketAddr>, Tonic provides SocketAddr directly
59    if let Some(connect_info) = extensions.get::<axum::extract::ConnectInfo<std::net::SocketAddr>>()
60    {
61        return connect_info.0.ip().to_string();
62    }
63
64    if let Some(remote_addr) = extensions.get::<std::net::SocketAddr>() {
65        return remote_addr.ip().to_string();
66    }
67
68    "unknown".to_string()
69}
