47/// Reads the caller's identity from the HTTP request the MCP service is handling.
48fn call_context(ctx: &RequestContext<RoleServer>) -> Option<CallContext> {
49    let parts = ctx.extensions.get::<http::request::Parts>()?;
50    let auth = parts.extensions.get::<AuthContext>()?.clone();
51    let header = |name: http::header::HeaderName| {
52        parts
53            .headers
54            .get(name)
55            .and_then(|v| v.to_str().ok())
56            .map(str::to_string)
57    };
58    Some(CallContext {
59        auth,
60        authorization: header(http::header::AUTHORIZATION).unwrap_or_default(),
61        client_agent: header(http::header::USER_AGENT).unwrap_or_else(|| "unknown".to_string()),
62        client_ip: get_client_ip(&parts.headers, &parts.extensions),
63    })
64}

66impl ServerHandler for McpServer {
67    fn get_info(&self) -> ServerConfig {
68        ServerConfig::new(
69            ServerCapabilities::builder()
70                .enable_tools()
71                .enable_resources()
72                .enable_skills()
73                .build(),
74        )
75        .with_instructions(INSTRUCTIONS)
76    }
77
78    async fn list_tools(
79        &self,
80        _request: Option<PaginatedRequestParams>,
81        context: RequestContext<RoleServer>,
82    ) -> Result<ListToolsResult, ErrorData> {
83        let call = call_context(&context).ok_or_else(|| {
84            ErrorData::internal_error("request carries no authenticated identity", None)
85        })?;
86        let tools = self
87            .inner
88            .providers
89            .iter()
90            .flat_map(|p| p.tools(&call))
91            .collect();
92        Ok(ListToolsResult::with_all_items(tools))
93    }
94
95    async fn call_tool(
96        &self,
97        request: CallToolRequestParams,
98        context: RequestContext<RoleServer>,
99    ) -> Result<CallToolResponse, ErrorData> {
100        let Some(call) = call_context(&context) else {
101            return Ok(CallToolResult::error(vec![ContentBlock::text(
102                "Unauthenticated: the request carries no validated identity",
103            )])
104            .into());
105        };
106        let arguments = request.arguments.unwrap_or_default();
107        for provider in &self.inner.providers {
108            if let Some(result) = provider
109                .call_tool(&call, &request.name, arguments.clone())
110                .await
111            {
112                return Ok(result.into());
113            }
114        }
115        Err(ErrorData::invalid_params(
116            format!("unknown tool '{}'", request.name),
117            None,
118        ))
119    }
120
121    async fn list_skills(
122        &self,
123        _request: Option<PaginatedRequestParams>,
124        _context: RequestContext<RoleServer>,
125    ) -> Result<ListSkillsResult, ErrorData> {
126        Ok(self.inner.skills.list_skills())
127    }
128
129    async fn get_skill(
130        &self,
131        request: GetSkillRequestParams,
132        _context: RequestContext<RoleServer>,
133    ) -> Result<GetSkillResult, ErrorData> {
134        self.inner.skills.get_skill(&request.uri).ok_or_else(|| {
135            ErrorData::invalid_params(format!("unknown skill {}", request.uri), None)
136        })
137    }
138
139    async fn list_resources(
140        &self,
141        _request: Option<PaginatedRequestParams>,
142        _context: RequestContext<RoleServer>,
143    ) -> Result<ListResourcesResult, ErrorData> {
144        Ok(ListResourcesResult::with_all_items(
145            self.inner.skills.resources(),
146        ))
147    }
148
149    async fn read_resource(
150        &self,
151        request: ReadResourceRequestParams,
152        _context: RequestContext<RoleServer>,
153    ) -> Result<ReadResourceResponse, ErrorData> {
154        self.inner
155            .skills
156            .read_resource(&request.uri)
157            .map(Into::into)
158            .ok_or_else(|| ErrorData::resource_not_found(request.uri, None))
159    }
