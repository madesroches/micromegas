115/// Classifies the root cause of a `DataFusionError` into the gRPC status code
116/// that best distinguishes "the caller's SQL/input was bad" from "the server
117/// broke": `Execution`/`Configuration` land under `InvalidArgument` since they
118/// signal bad input, while `ArrowError` stays `Internal` since it reflects a
119/// real server-side decode failure.
120pub fn classify_datafusion_error(err: &DataFusionError) -> tonic::Code {
121    use datafusion::error::DataFusionError as DFE;
122    match err.find_root() {
123        DFE::SQL(..)
124        | DFE::Plan(_)
125        | DFE::SchemaError(..)
126        | DFE::Execution(_)
127        | DFE::Configuration(_) => tonic::Code::InvalidArgument,
128        DFE::ResourcesExhausted(_) => tonic::Code::ResourceExhausted,
129        DFE::NotImplemented(_) => tonic::Code::Unimplemented,
130        _ => tonic::Code::Internal,
131    }
132}

134/// Maps a gRPC status code to the `QueryAuditRecord.error_class` bucket used
135/// both by the audit log and to gate the `query_failed`/
136/// `query_duration_with_error` metrics. This is the *derived* mapping used for
137/// every `DataFusionError`-rooted failure; the query-deny-list check stamps a fourth value,
138/// `"denied"`, directly via `QueryAuditState::fail_with_class` rather than through this
139/// function, since a denial's `ResourceExhausted` code would otherwise land in the
140/// `"resource"` bucket alongside a genuine resource-budget failure.
141pub fn error_class(code: tonic::Code) -> &'static str {
142    match code {
143        tonic::Code::InvalidArgument | tonic::Code::Unimplemented => "user",
144        tonic::Code::ResourceExhausted => "resource",
145        _ => "internal",
146    }
147}

149/// Cap on the physical-plan text appended to the server-side log line. The
150/// plan can render arbitrarily long `file_groups=` sections for wide
151/// scans, and this text is server-log-only, never returned to the client.
152pub const MAX_PLAN_CHARS: usize = 2000;
153
154/// Truncate `text` at a char boundary once it exceeds `MAX_PLAN_CHARS`, tagging
155/// the cut with a trailing marker. Driven by `char_indices()` (not byte
156/// length) so a multibyte plan text (e.g. a non-ASCII string literal from a
157/// query predicate, rendered via `ScalarValue`'s `Display` impl) can't panic
158/// on a byte-offset slice landing mid-character.
159fn truncate_plan_text(text: &str) -> String {
160    match text.char_indices().nth(MAX_PLAN_CHARS) {
161        None => text.to_string(),
162        Some((i, _)) => format!("{}... (truncated)", &text[..i]),
163    }
164}

166/// Builds the full server-side log line for a classified `DataFusionError`:
167/// `desc`, the error's own `Display` (its outer `Context` chain intact, with
168/// backtrace if enabled -- unlike `find_root()`, which is for classification
169/// only), the `query_id`, and -- when there's no diagnostic span to show the
170/// caller instead -- the truncated physical plan text. `pub` so
171/// tests can assert its content, including the truncation marker,
172/// without capturing log output.
173pub fn build_log_line(
174    desc: &str,
175    err: &DataFusionError,
176    query_id: &str,
177    plan: Option<&Arc<dyn ExecutionPlan>>,
178) -> String {
179    let mut full = format!("{desc}: {err} (query_id={query_id})");
180    if let Some(plan) = plan {
181        let plan_text = format!(
182            "{}",
183            datafusion::physical_plan::displayable(plan.as_ref()).indent(true)
184        );
185        full.push_str(&format!(
186            "\nphysical plan:\n{}",
187            truncate_plan_text(&plan_text)
188        ));
189    }
190    full
191}

233/// Classifies `err`, builds the client-facing `Status` -- `desc` + root-error
234/// text (backtrace stripped) + an optional diagnostic span/notes/helps +
235/// `query_id` -- and logs the full error server-side. The physical
236/// plan, when present and there's no diagnostic span to show instead, only
237/// ever reaches the server log, never the returned `Status`: `displayable`
238/// can render object-store partition paths from Micromegas view scans, which
239/// would leak internal lakehouse details to every caller hitting an
240/// execution-time error.
241pub fn client_error(
242    desc: &str,
243    err: DataFusionError,
244    query_id: &str,
245    plan: Option<&Arc<dyn ExecutionPlan>>,
246) -> Status {
247    let code = classify_datafusion_error(&err);
248    let mut msg = format!("{desc}: {}", err.find_root().strip_backtrace());
249    let mut has_span = false;
250    if let Some(diag) = err.diagnostic() {
251        if let Some(span) = diag.span {
252            has_span = true;
253            msg.push_str(&format!(
254                " (at line {}, column {})",
255                span.start.line, span.start.column
256            ));
257        }
258        for note in &diag.notes {
259            msg.push_str(&format!("\nnote: {}", note.message));
260        }
261        for help in &diag.helps {
262            msg.push_str(&format!("\nhelp: {}", help.message));
263        }
264    }
265    msg.push_str(&format!(" (query_id={query_id})"));
266    error_or_warn_log(
267        code,
268        desc,
269        &err,
270        query_id,
271        if has_span { None } else { plan },
272    );
273    Status::new(code, msg)
274}

276/// Recovers the `DataFusionError` from the per-batch `FlightError` re-wrap
277/// (`FlightError::ExternalError(Box::new(e))`, applied once the raw
278/// `DataFusionError` stream is wrapped for Arrow Flight) and classifies it via
279/// `client_error`. Every branch logs exactly once before returning, so this is
280/// a single, unconditional log point for the per-batch stream-error site
281/// regardless of which branch is taken. `pub` so step 9's external
282/// integration tests can exercise this recovery path directly.
283pub fn classify_flight_error(
284    err: FlightError,
285    query_id: &str,
286    plan: Option<&Arc<dyn ExecutionPlan>>,
287) -> Status {
288    match err {
289        FlightError::ExternalError(inner) => match inner.downcast::<DataFusionError>() {
290            Ok(df_err) => client_error("error building data stream", *df_err, query_id, plan),
291            Err(inner) => {
292                error!("error building data stream: {inner} (query_id={query_id})");
293                Status::internal(format!(
294                    "error building data stream: {inner} (query_id={query_id})"
295                ))
296            }
297        },
298        other => {
299            error!("error building data stream: {other} (query_id={query_id})");
300            Status::internal(format!(
301                "error building data stream: {other} (query_id={query_id})"
302            ))
303        }
304    }
305}

354impl QueryAuditState {
355    /// Aggregate the plan's DataFusion metrics (if a physical plan was
356    /// created before the failure/completion), assemble the audit record,
357    /// and emit it as a single JSON log line.
358    ///
359    /// Takes `&self` (rather than consuming) so it can be called from a
360    /// setup-error `map_err` closure while leaving the state available for
361    /// further updates/completion on the success path, and so `Drop` can
362    /// call it on an abandoned/cancelled stream without needing to
363    /// reconstruct anything.
364    fn emit(&self, status: &'static str, error: Option<String>, error_class: Option<&'static str>) {
365        let scan = match &self.plan {
366            Some(plan) => aggregate_scan_metrics(plan.as_ref()),
367            None => ScanMetrics {
368                output_rows: None,
369                bytes_scanned: 0,
370                spilled_bytes: 0,
371                spill_count: 0,
372            },
373        };
374        let peak_memory_bytes = self.pool.peak() as u64;
375        imetric!("query_peak_memory_bytes", "bytes", peak_memory_bytes);
376        let total_ms = self.request_start.elapsed().as_secs_f64() * 1000.0;
377        let record = QueryAuditRecord {
378            query_id: self.query_id.clone(),
379            client_ip: self.client_ip.clone(),
380            client: self.client.clone(),
381            agent: self.agent.clone(),
382            entrypoint: self.entrypoint.clone(),
383            session: self.session.clone(),
384            notebook: self.notebook.clone(),
385            cell: self.cell.clone(),
386            user: self.user.clone(),
387            email: self.email.clone(),
388            name: self.name.clone(),
389            service_account: self.service_account,
390            service_account_name: self.service_account_name.clone(),
391            sql: self.sql.clone(),
392            sql_hash: self.sql_hash.clone(),
393            range_begin: self.range_begin.clone(),
394            range_end: self.range_end.clone(),
395            limit: self.limit,
396            context_init_ms: self.context_init_ms,
397            planning_ms: self.planning_ms,
398            execution_ms: self.execution_ms,
399            setup_ms: self.setup_ms,
400            total_ms,
401            status,
402            error,
403            error_class,
404            output_rows: scan.output_rows,
405            bytes_scanned: scan.bytes_scanned,
406            peak_memory_bytes,
407            spilled_bytes: scan.spilled_bytes,
408            spill_count: scan.spill_count,
409        };
410        match serde_json::to_string(&record) {
411            Ok(json) => info!(target: "flightsql_query_audit", "{json}"),
412            Err(e) => warn!("failed to serialize query audit record: {e}"),
413        }
414    }
415
416    /// Emits an "error" audit record for an already-built client-facing `status`, with an
417    /// explicit `error_class` rather than one derived from the gRPC code -- used by the
418    /// query-deny-list check, whose `error_class` is `"denied"`, a value
419    /// `error_class(status.code())` (`ResourceExhausted` -> `"resource"`) would get wrong.
420    /// Returns `status` unchanged.
421    fn fail_with_class(&self, status: Status, class: &'static str) -> Status {
422        self.emit("error", Some(status.message().to_string()), Some(class));
423        status
424    }
425
426    /// Emits an "error" audit record for an already-built client-facing
427    /// `status` (message + `error_class` derived from its code) and returns it
428    /// unchanged -- the shared tail of every setup-phase `map_err` in
429    /// `execute_query`. A thin wrapper over [`Self::fail_with_class`] that derives the class
430    /// from the code.
431    fn fail(&self, status: Status) -> Status {
432        let class = error_class(status.code());
433        self.fail_with_class(status, class)
434    }
435}

456impl<S> Drop for CompletionTrackedStream<S> {
457    /// If the stream is dropped before yielding `None` or an `Err` (client
458    /// disconnect/cancel mid-drain), `poll_next` never ran its completion
459    /// arms, so `self.audit` is still `Some(...)`. Emit it here with a
460    /// terminal "incomplete" status so cancelled/abandoned queries are still
461    /// audited instead of silently vanishing. A no-op for streams that
462    /// completed or errored normally, since those arms already `take()` the
463    /// audit state.
464    fn drop(&mut self) {
465        if let Some(state) = self.audit.take() {
466            state.emit("incomplete", None, None);
467        }
468    }
469}
470
471impl<S> Stream for CompletionTrackedStream<S>
472where
473    S: Stream<Item = Result<arrow_flight::FlightData, Status>> + Unpin + Send,
474{
475    type Item = Result<arrow_flight::FlightData, Status>;
476
477    fn poll_next(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
478        match Pin::new(&mut self.inner).poll_next(cx) {
479            Poll::Ready(Some(result)) => {
480                // Every `Status` reaching this stream was already logged once by
481                // `classify_flight_error` (the sole `map_err` feeding it) when it
482                // built that `Status` -- this arm only reads the code to derive
483                // the audit/metric class, it never logs again.
484                if let Err(ref err) = result
485                    && !self.completed
486                {
487                    let class = error_class(err.code());
488                    let total_duration = now() - self.start_time;
489                    match class {
490                        "internal" => {
491                            imetric!("query_duration_with_error", "ticks", total_duration as u64);
492                            imetric!("query_failed", "count", 1);
493                        }
494                        "resource" => {
495                            imetric!("query_failed_resource", "count", 1);
496                        }
497                        _ => {
498                            imetric!("query_failed_user", "count", 1);
499                        }
500                    }
501                    self.completed = true;
502                    if let Some(state) = self.audit.take() {
503                        state.emit("error", Some(err.message().to_string()), Some(class));
504                    }
505                }
506                Poll::Ready(Some(result))
507            }
508            Poll::Ready(None) => {
509                // Stream completed successfully
510                if !self.completed {
511                    let total_duration = now() - self.start_time;
512                    imetric!("query_duration_total", "ticks", total_duration as u64);
513                    imetric!("query_completed_successfully", "count", 1);
514                    self.completed = true;
515                    if let Some(state) = self.audit.take() {
516                        state.emit("ok", None, None);
517                    }
518                }
519                Poll::Ready(None)
520            }
521            Poll::Pending => Poll::Pending,
522        }
523    }
524}

574    /// **Absent-extension convention.** When no `AuthContext` extension is present at all (no
575    /// auth provider configured, e.g. `--disable-auth`), the resolved scope is
576    /// [`ReadScope::All`]. This is the *only* permissive branch here: `AuthService::call`
577    /// (`rust/auth/src/tower.rs`) rejects the request before this inner service ever runs
578    /// whenever a provider *is* configured, so the extension is always present in that case --
579    /// the same safety argument `is_admin`'s absent-header-⇒-trusted convention already relies
580    /// on (`user_attribution.rs`).
581    ///
582    /// **Failure convention.** A [`ReadPolicy::resolve`] that returns `Err` is a hard failure and
583    /// must never become a scope -- not `ReadScope::Audiences(Arc::from([]))` (which would read
584    /// as a legitimate, audited fail-closed decision by the isolation-config layer) and not
585    /// `ReadScope::All` (a silent fail-open bypass). Mirrors `tower.rs`'s own discriminator:
586    /// `Status::unavailable` when the error downcasts to [`ProviderUnavailable`] (a
587    /// store/provider outage), `Status::permission_denied` otherwise.
588    ///
589    /// `is_admin` is read from `md` unchanged (`is_admin(md)`), equivalent to reading
590    /// `AuthContext.is_admin` off `ext` when the extension is present, since the header is
591    /// derived from the same `AuthContext` with client-supplied copies stripped
592    /// (`tower.rs:107-139`) -- and it is what preserves today's `--disable-auth`
593    /// absent-header-⇒-trusted convention when `ext` has none.
594    async fn caller_context(
595        &self,
596        ext: &http::Extensions,
597        md: &MetadataMap,
598    ) -> Result<CallerContext, Status> {
599        // `deny_queries` requires `Some` -- resolved from the
600        // same attribution `execute_query` already validates, so a query already carries this by
601        // the time it could call `deny_queries`. This is simply "the caller's resolved user id"
602        // (`user_attribution::validate_and_resolve_user_attribution_grpc`).
603        let identity = validate_and_resolve_user_attribution_grpc(md)
604            .ok()
605            .map(|attr| attr.user_id);
606        // Read once, up front, so both `read_scope`'s resolution below and `grant_selectors`
607        // can consult the same `AuthContext` -- the `Some(auth_ctx)` arm's
608        // binding otherwise goes out of scope before the `CallerContext` literal is built.
609        let auth_ctx = ext.get::<AuthContext>();
610        let read_scope = match auth_ctx {
611            Some(auth_ctx) => match self.read_policy.resolve(auth_ctx).await {
612                Ok(audiences) => ReadScope::Audiences(audiences.into_inner()),
613                Err(e) => {
614                    return Err(if e.downcast_ref::<ProviderUnavailable>().is_some() {
615                        Status::unavailable(format!("read policy unavailable: {e:#}"))
616                    } else {
617                        Status::permission_denied(format!("read scope denied: {e:#}"))
618                    });
619                }
620            },
621            None => ReadScope::All,
622        };
623        Ok(CallerContext {
624            read_scope,
625            is_admin: is_admin(md),
626            // Not permission-sensitive the way `read_scope` is (it's deployment config, not
627            // derived from the caller's identity), so it is copied verbatim on both branches
628            // above rather than participating in the absent-extension/`Err` distinction.
629            isolation_config: self.isolation_config.clone(),
630            identity,
631            // Empty when there's no `AuthContext` extension at all (no auth provider configured,
632            // e.g. `--disable-auth`) -- the same absent-extension convention `read_scope` follows
633            // above.
634            grant_selectors: auth_ctx.map(caller_selectors).unwrap_or_default().into(),
635        })
636    }

655    #[span_fn]
656    async fn execute_query(
657        &self,
658        ticket_stmt: TicketStatementQuery,
659        metadata: &MetadataMap,
660        extensions: &http::Extensions,
661        client_ip: &str,
662    ) -> Result<Response<FlightDataStream>, Status> {
663        // Minted first, before any fallible step, so it's Always available --
664        // for every client-facing `Status` built by `client_error`/
665        // `classify_flight_error` and for this query's audit record.
666        let query_id = Uuid::new_v4().to_string();
667        let begin_request = now();
668        let request_start = Instant::now();
669        let sql = std::str::from_utf8(&ticket_stmt.statement_handle)
670            .map_err(|e| status!("Unable to parse query", e))?;
671
672        let mut begin = metadata.get("query_range_begin");
673        if let Some(s) = &begin
674            && s.is_empty()
675        {
676            begin = None;
677        }
678        let mut end = metadata.get("query_range_end");
679        if let Some(s) = &end
680            && s.is_empty()
681        {
682            end = None;
683        }
684        let query_range = if begin.is_some() && end.is_some() {
685            let begin_datetime =
686                chrono::DateTime::parse_from_rfc3339(begin.unwrap().to_str().map_err(|e| {
687                    client_input_error!("Unable to convert query_range_begin to string", e)
688                })?)
689                .map_err(|e| {
690                    client_input_error!(
691                        "Unable to parse query_range_begin as a rfc3339 datetime",
692                        e
693                    )
694                })?;
695            let end_datetime =
696                chrono::DateTime::parse_from_rfc3339(end.unwrap().to_str().map_err(|e| {
697                    client_input_error!("Unable to convert query_range_end to string", e)
698                })?)
699                .map_err(|e| {
700                    client_input_error!("Unable to parse query_range_end as a rfc3339 datetime", e)
701                })?;
702            Some(TimeRange::new(begin_datetime.into(), end_datetime.into()))
703        } else {
704            None
705        };
706
707        // Validate and resolve user attribution
708        let attr = validate_and_resolve_user_attribution_grpc(metadata).map_err(|e| *e)?;
709
710        let client_type = metadata
711            .get("x-client-type")
712            .and_then(|v| v.to_str().ok())
713            .unwrap_or("unknown");
714        let client_agent = metadata
715            .get("x-client-agent")
716            .and_then(|v| v.to_str().ok())
717            .unwrap_or("unknown");
718        let client_entrypoint = metadata
719            .get("x-client-entrypoint")
720            .and_then(|v| v.to_str().ok())
721            .unwrap_or("unknown");
722        let client_session = Self::optional_metadata(metadata, "x-client-session");
723        let client_notebook = Self::optional_metadata(metadata, "x-client-notebook");
724        let client_cell = Self::optional_metadata(metadata, "x-client-cell");
725
726        let user_name_display = attr.user_name.as_deref().unwrap_or("");
727
728        // Log query with full attribution
729        if let Some(service_account_name) = &attr.service_account {
730            info!(
731                "execute_query range={query_range:?} sql={sql:?} limit={:?} user={} email={} name={user_name_display:?} service_account={service_account_name} client={client_type} agent={client_agent} entrypoint={client_entrypoint} notebook={client_notebook:?} cell={client_cell:?} client_ip={client_ip}",
732                metadata.get("limit"),
733                attr.user_id,
734                attr.user_email
735            );
736        } else {
737            info!(
738                "execute_query range={query_range:?} sql={sql:?} limit={:?} user={} email={} name={user_name_display:?} client={client_type} agent={client_agent} entrypoint={client_entrypoint} notebook={client_notebook:?} cell={client_cell:?} client_ip={client_ip}",
739                metadata.get("limit"),
740                attr.user_id,
741                attr.user_email
742            );
743        }
744
745        // The per-query memory-pool wrapper is infallible to construct, so it's built
746        // before `audit_state` and owned by it from the start -- every subsequent
747        // setup failure (session context, planning, limit, physical plan, stream
748        // construction) can then still emit an "error" audit record carrying whatever
749        // peak the pool had accrued by that point, instead of silently disappearing on
750        // an early `?` return.
751        let scoped_pool = Arc::new(ScopedMemoryPool::new(
752            self.lakehouse.runtime().memory_pool.clone(),
753        ));
754
755        // Attribution is resolved from here on, so build the audit state now
756        // (durations/limit/plan filled in as they become known) instead of
757        // only after the physical plan exists.
758        // Computed once per query, regardless of whether the deny list is in active use --
759        // it's what an operator pastes into `deny_queries` after finding an offender in the
760        // audit log.
761        let sql_hash = fingerprint_of(sql);
762
763        let mut audit_state = QueryAuditState {
764            query_id: query_id.clone(),
765            client_ip: client_ip.to_string(),
766            client: client_type.to_string(),
767            agent: client_agent.to_string(),
768            entrypoint: client_entrypoint.to_string(),
769            session: client_session,
770            notebook: client_notebook,
771            cell: client_cell,
772            user: attr.user_id.clone(),
773            email: attr.user_email.clone(),
774            name: attr.user_name.clone(),
775            service_account: attr.service_account.is_some(),
776            service_account_name: attr.service_account.clone(),
777            sql: sql.to_string(),
778            sql_hash: sql_hash.clone(),
779            range_begin: query_range.as_ref().map(|r| r.begin.to_rfc3339()),
780            range_end: query_range.as_ref().map(|r| r.end.to_rfc3339()),
781            limit: None,
782            context_init_ms: 0.0,
783            planning_ms: 0.0,
784            execution_ms: 0.0,
785            setup_ms: 0.0,
786            request_start,
787            plan: None,
788            pool: scoped_pool.clone(),
789        };
790
791        // The deny-list check, in front of everything expensive -- before the scoped
792        // runtime/session context/planning below. `check` returns `None` immediately on an
793        // empty snapshot, so this costs nothing while the deny list stands empty.
794        if let Some(rule) = self.lakehouse.query_denials().check(&QueryAttribution {
795            user_id: &attr.user_id,
796            email: &attr.user_email,
797            service_account: attr.service_account.as_deref(),
798            client: client_type,
799            agent: client_agent,
800            entrypoint: client_entrypoint,
801            session: audit_state.session.as_deref(),
802            notebook: audit_state.notebook.as_deref(),
803            cell: audit_state.cell.as_deref(),
804            client_ip,
805            sql,
806            sql_hash: &sql_hash,
807        }) && !skip_for_admin_recovery(sql, is_admin(metadata))
808        {
809            let status = Status::resource_exhausted(format!(
810                "query denied by rule {} (reason: {}); ask an admin to lift it with \
811                 remove_query_denial('{}'); query_id={query_id}",
812                rule.row.rule_id, rule.row.reason, rule.row.rule_id
813            ));
814            warn!(
815                "{}",
816                build_denial_warn_line(
817                    &rule.row.rule_id.to_string(),
818                    &rule.row.reason,
819                    &sql_hash,
820                    &attr.user_id,
821                    &attr.user_email,
822                    client_type,
823                    client_entrypoint,
824                    client_ip,
825                    &query_id,
826                )
827            );
828            imetric!("query_denied", "count", rule_tags(&rule.row.rule_id), 1_u64);
829            rule.record_hit();
830            return Err(audit_state.fail_with_class(status, "denied"));
831        }
832
833        // Caller is resolved here, ahead of the scoped runtime, because the view-DDL intercept
834        // below needs it and returns before either the scoped runtime or a session context is
835        // ever built for a `CREATE`/`DROP MATERIALIZED VIEW` statement.
836        //
837        // `context_init_ms` is measured starting here (not at `make_session_context`) because
838        // caller resolution can itself hit the read-policy store (e.g. a DB round trip to
839        // resolve audience grants), and that cost belongs in the same phase.
840        let session_begin = now();
841        let session_begin_instant = Instant::now();
842        let caller = self
843            .caller_context(extensions, metadata)
844            .await
845            .map_err(|status| audit_state.fail(status))?;
846
847        // View DDL is intercepted here, ahead of the scoped runtime/`make_session_context`: a
848        // `CREATE`/`DROP MATERIALIZED VIEW` statement is never planned as an ordinary query.
849        match parse_view_ddl(sql) {
850            Ok(Some(ddl)) => {
851                return self
852                    .execute_view_ddl(ddl, &caller, begin_request, audit_state)
853                    .await;
854            }
855            Ok(None) => {}
856            Err(e) => {
857                return Err(audit_state.fail(client_input_error!("error parsing view DDL", e)));
858            }
859        }
860
861        // Build a `RuntimeEnv`/`LakehouseContext` scoped to this query's memory-pool
862        // wrapper, so every session context created from it (including nested ones,
863        // e.g. Perfetto trace queries and JIT materialization) attributes its memory
864        // to this query alone instead of the process-shared pool.
865        let scoped_env = scoped_runtime(self.lakehouse.runtime(), scoped_pool.clone())
866            .map_err(|e| audit_state.fail(status!("error building scoped runtime", e)))?;
867        let lakehouse = self.lakehouse.with_runtime(scoped_env);
868
869        // Session context creation phase
870        let ctx = make_session_context(
871            lakehouse,
872            self.part_provider.clone(),
873            query_range,
874            self.view_registry.current(),
875            self.session_configurator.clone(),
876            caller,
877        )
878        .await
879        .map_err(|e| audit_state.fail(status!("error in make_session_context", e)))?;
880        let context_init_duration = now() - session_begin;
881        audit_state.context_init_ms = session_begin_instant.elapsed().as_secs_f64() * 1000.0;
882
883        // Query planning phase
884        let planning_begin = now();
885        let planning_begin_instant = Instant::now();
886        let mut df = ctx.sql(sql).await.map_err(|e| {
887            audit_state.fail(client_error("error building dataframe", e, &query_id, None))
888        })?;
889        let planning_duration = now() - planning_begin;
890        audit_state.planning_ms = planning_begin_instant.elapsed().as_secs_f64() * 1000.0;
891
892        if let Some(limit_str) = metadata.get("limit") {
893            let parsed_limit: usize = usize::from_str(limit_str.to_str().map_err(|e| {
894                audit_state.fail(client_input_error!("error converting limit to str", e))
895            })?)
896            .map_err(|e| audit_state.fail(client_input_error!("error parsing limit", e)))?;
897            audit_state.limit = Some(parsed_limit as u64);
898            df = df.limit(0, Some(parsed_limit)).map_err(|e| {
899                audit_state.fail(client_error(
900                    "error building dataframe with limit",
901                    e,
902                    &query_id,
903                    None,
904                ))
905            })?;
906        }
907
908        // Query execution phase: build the physical plan (kept for post-drain metrics)
909        // and run it via the free-function `execute_stream`, which is what
910        // `DataFrame::execute_stream` does internally, minus dropping the plan.
911        let execution_begin = now();
912        let execution_begin_instant = Instant::now();
913        let schema = Arc::new(df.schema().as_arrow().clone());
914        let task_ctx = Arc::new(df.task_ctx());
915        let plan = df.create_physical_plan().await.map_err(|e| {
916            audit_state.fail(client_error(
917                "error creating physical plan",
918                e,
919                &query_id,
920                None,
921            ))
922        })?;
923        audit_state.plan = Some(plan.clone());
924        // `plan`/`query_id` are cloned here because both are moved shortly:
925        // `plan` into `execute_stream` below, `audit_state` (which owns
926        // `query_id`'s sibling on the struct) into `CompletionTrackedStream::new`
927        // further down. The per-batch closure below outlives this function call
928        // (it's polled from the returned stream), so it needs its own owned
929        // clones rather than borrowing `plan`/`query_id` directly.
930        let plan_for_errors = plan.clone();
931        let query_id_for_stream = query_id.clone();
932        let stream = execute_stream(plan, task_ctx)
933            .map_err(|e| {
934                audit_state.fail(client_error(
935                    "error executing plan",
936                    e,
937                    &query_id_for_stream,
938                    Some(&plan_for_errors),
939                ))
940            })?
941            .map_err(|e| FlightError::ExternalError(Box::new(e)));
942        let builder = if Self::should_preserve_dictionary(metadata) {
943            FlightDataEncoderBuilder::new()
944                .with_schema(schema.clone())
945                .with_dictionary_handling(DictionaryHandling::Resend)
946        } else {
947            FlightDataEncoderBuilder::new().with_schema(schema.clone())
948        };
949        let flight_data_stream = builder.build(stream);
950        let execution_duration = now() - execution_begin;
951        audit_state.execution_ms = execution_begin_instant.elapsed().as_secs_f64() * 1000.0;
952
953        // Calculate total setup time and record detailed metrics
954        let total_setup_duration = now() - begin_request;
955        audit_state.setup_ms = request_start.elapsed().as_secs_f64() * 1000.0;
956
957        // Record detailed timing metrics
958        imetric!(
959            "context_init_duration",
960            "ticks",
961            context_init_duration as u64
962        );
963        imetric!("query_planning_duration", "ticks", planning_duration as u64);
964        imetric!(
965            "query_execution_duration",
966            "ticks",
967            execution_duration as u64
968        );
969        imetric!("query_setup_duration", "ticks", total_setup_duration as u64);
970
971        // Create instrumented stream that tracks completion. This closure owns
972        // `plan_for_errors`/`query_id_for_stream` (moved in) since it's polled
973        // from the returned stream, long after this function returns -- see the
974        // comment above `plan_for_errors`'s definition.
975        let instrumented_stream = flight_data_stream.map_err(move |e| {
976            classify_flight_error(e, &query_id_for_stream, Some(&plan_for_errors))
977        });
978        let completion_tracked_stream =
979            CompletionTrackedStream::new(instrumented_stream.boxed(), begin_request, audit_state);
980        Ok(Response::new(
981            Box::pin(completion_tracked_stream) as FlightDataStream
982        ))
983    }
984
985    /// Executes a parsed `CREATE [OR REPLACE] MATERIALIZED VIEW` / `DROP MATERIALIZED VIEW`
986    /// statement: admin-gates it, validates/upserts or deletes its row inside one transaction
987    /// (serialized cluster-wide by an advisory lock), refuses a mutation that would break a
988    /// dependent, retires partitions on `DROP`, reloads this replica's registry inline after
989    /// commit, and returns a one-row `(view_set_name, status)` result through the same
990    /// `CompletionTrackedStream` every other query uses.
991    async fn execute_view_ddl(
992        &self,
993        ddl: ViewDdl,
994        caller: &CallerContext,
995        begin_request: i64,
996        audit_state: QueryAuditState,
997    ) -> Result<Response<FlightDataStream>, Status> {
998        authorize_view_ddl(caller).map_err(|status| audit_state.fail(status))?;
999
1000        let pool = self.lakehouse.lake().db_pool.clone();
1001        let mut tx = pool
1002            .begin()
1003            .await
1004            .map_err(|e| audit_state.fail(status!("error starting view DDL transaction", e)))?;
1005        acquire_lock(&mut tx, VIEW_DDL_ADVISORY_LOCK_KEY)
1006            .await
1007            .map_err(|e| audit_state.fail(status!("error acquiring view DDL lock", e)))?;
1008        let pre_rows = list_tx(&mut tx)
1009            .await
1010            .map_err(|e| audit_state.fail(status!("error reading view definitions", e)))?;
1011
1012        let (view_set_name, status_text) = match ddl {
1013            ViewDdl::Create {
1014                name,
1015                or_replace,
1016                definition,
1017                sql,
1018            } => {
1019                let existing = pre_rows.iter().any(|r| r.view_set_name == name);
1020                if existing && !or_replace {
1021                    return Err(audit_state.fail(Status::already_exists(format!(
1022                        "materialized view '{name}' already exists; use CREATE OR REPLACE to \
1023                         redefine it"
1024                    ))));
1025                }
1026                // Validated against the factory the loader would build for this row: the
1027                // pre-mutation rows, this row's own name excluded (identical to the
1028                // post-mutation set for this single-row mutation) -- so a definition reading
1029                // another DDL view validates against the real thing, and a `CREATE OR REPLACE`
1030                // that raises its own update_group cannot validate against a since-superseded
1031                // copy of itself.
1032                let validation_rows: Vec<_> = pre_rows
1033                    .iter()
1034                    .filter(|r| r.view_set_name != name)
1035                    .cloned()
1036                    .collect();
1037                // A `Probe`: a row in `validation_rows` that no longer builds is a pre-existing
1038                // breakage the live reload already reports on every tick -- re-logging it here
1039                // would pin it on whoever happened to run the next DDL statement.
1040                let (factory, _failed) = self
1041                    .view_registry
1042                    .build_from_rows(&validation_rows, BuildPurpose::Probe)
1043                    .await
1044                    .map_err(|e| {
1045                        audit_state.fail(status!("error building validation factory", e))
1046                    })?;
1047                let view = build_sql_batch_view(
1048                    &definition,
1049                    self.lakehouse.runtime().clone(),
1050                    self.lakehouse.lake().clone(),
1051                    factory.clone(),
1052                    self.session_configurator.clone(),
1053                )
1054                .await
1055                .map_err(|e| {
1056                    audit_state.fail(client_input_error!("error building materialized view", e))
1057                })?;
1058                validate_view_definition(
1059                    &definition,
1060                    &view,
1061                    &factory,
1062                    self.lakehouse.runtime().clone(),
1063                    self.lakehouse.lake().clone(),
1064                    self.session_configurator.clone(),
1065                )
1066                .await
1067                .map_err(|e| {
1068                    audit_state.fail(client_input_error!("view definition failed validation", e))
1069                })?;
1070                let updated_by = caller.identity.as_deref().unwrap_or("unknown");
1071                upsert_tx(&mut tx, &definition, &sql, updated_by)
1072                    .await
1073                    .map_err(|e| audit_state.fail(status!("error upserting view definition", e)))?;
1074                (name, if existing { "replaced" } else { "created" })
1075            }
1076            ViewDdl::Drop { name, if_exists } => {
1077                let existing = pre_rows.iter().any(|r| r.view_set_name == name);
1078                if !existing {
1079                    if if_exists {
1080                        (name, "not_found")
1081                    } else {
1082                        return Err(audit_state.fail(Status::not_found(format!(
1083                            "materialized view '{name}' does not exist"
1084                        ))));
1085                    }
1086                } else {
1087                    let range = partition_insert_range(&mut tx, &name).await.map_err(|e| {
1088                        audit_state.fail(status!("error reading partition insert range", e))
1089                    })?;
1090                    delete_tx(&mut tx, &name).await.map_err(|e| {
1091                        audit_state.fail(status!("error deleting view definition", e))
1092                    })?;
1093                    if let Some((begin, end)) = range {
1094                        let null_logger = Arc::new(ResponseWriter::new(None));
1095                        retire_partitions(
1096                            &mut tx,
1097                            &name,
1098                            "global",
1099                            begin,
1100                            end,
1101                            RetireMatch::Containment,
1102                            &[],
1103                            null_logger,
1104                        )
1105                        .await
1106                        .map_err(|e| audit_state.fail(status!("error retiring partitions", e)))?;
1107                    }
1108                    (name, "dropped")
1109                }
1110            }
1111        };
1112
1113        // `DROP ... IF EXISTS` on an absent view writes nothing, so `post_rows` would be
1114        // byte-identical to `pre_rows`; skip the probe rather than replanning every stored
1115        // definition twice over for a statement that changed nothing, all while holding the
1116        // advisory lock and this open transaction.
1117        if status_text != "not_found" {
1118            let post_rows = list_tx(&mut tx)
1119                .await
1120                .map_err(|e| audit_state.fail(status!("error reading view definitions", e)))?;
1121            self.view_registry
1122                .check_dependents_survive(&pre_rows, &post_rows)
1123                .await
1124                .map_err(|e| {
1125                    // The inner error already says what would break and why; this prefix only has
1126                    // to mark where the refusal came from.
1127                    audit_state.fail(client_input_error!("view DDL refused", e))
1128                })?;
1129        }
1130
1131        tx.commit()
1132            .await
1133            .map_err(|e| audit_state.fail(status!("error committing view DDL transaction", e)))?;
1134
1135        // Run after the commit: `reload()` starts with a second, uncommitted-write-blind
1136        // connection, so calling it any earlier would have it read the pre-mutation rows and
1137        // swap in the old factory. `Err` is `warn!`-logged and discarded -- the row is already
1138        // committed, and the next periodic refresh will pick it up regardless.
1139        if let Err(e) = self.view_registry.reload().await {
1140            warn!("view_ddl: inline reload after DDL failed: {e:#}");
1141        }
1142
1143        let schema = Arc::new(Schema::new(vec![
1144            Field::new("view_set_name", DataType::Utf8, false),
1145            Field::new("status", DataType::Utf8, false),
1146        ]));
1147        let batch = RecordBatch::try_new(
1148            schema.clone(),
1149            vec![
1150                Arc::new(StringArray::from(vec![view_set_name])),
1151                Arc::new(StringArray::from(vec![status_text.to_string()])),
1152            ],
1153        )
1154        .map_err(|e| audit_state.fail(status!("error building view DDL result batch", e)))?;
1155        let stream = FlightDataEncoderBuilder::new()
1156            .with_schema(schema)
1157            .build(futures::stream::once(async { Ok(batch) }))
1158            .map_err(Status::from);
1159        let completion_tracked_stream =
1160            CompletionTrackedStream::new(stream.boxed(), begin_request, audit_state);
1161        Ok(Response::new(
1162            Box::pin(completion_tracked_stream) as FlightDataStream
1163        ))
1164    }
1165}

1553    async fn do_action_create_prepared_statement(
1554        &self,
1555        query: ActionCreatePreparedStatementRequest,
1556        request: Request<Action>,
1557    ) -> Result<ActionCreatePreparedStatementResult, Status> {
1558        info!("do_action_create_prepared_statement query={}", &query.query);
1559
1560        // DDL (`CREATE`/`DROP MATERIALIZED VIEW`) is never prepared: nothing in the repo
1561        // prepares DDL, and planning it here via `ctx.sql` would fail with an opaque DataFusion
1562        // error instead of this clear message.
1563        if parse_view_ddl(&query.query)
1564            .map_err(|e| client_input_error!("error parsing view DDL", e))?
1565            .is_some()
1566        {
1567            return Err(Status::invalid_argument(
1568                "CREATE/DROP MATERIALIZED VIEW must be executed directly, not prepared",
1569            ));
1570        }
1571
1572        // Prepared statements resolve the same CallerContext as the do_get execute path.
1573        let caller = self
1574            .caller_context(request.extensions(), request.metadata())
1575            .await?;
1576        let ctx = make_session_context(
1577            self.lakehouse.clone(),
1578            self.part_provider.clone(),
1579            None,
1580            self.view_registry.current(),
1581            self.session_configurator.clone(),
1582            caller,
1583        )
1584        .await
1585        .map_err(|e| status!("error in make_session_context", e))?;
1586
1587        // No `QueryAuditState`/audit record exists on this RPC (it only plans the
1588        // query, never executes it) -- this id exists solely so the client-facing
1589        // message and a server log line can be correlated with each other.
1590        let local_query_id = Uuid::new_v4().to_string();
1591        let df = ctx
1592            .sql(&query.query)
1593            .await
1594            .map_err(|e| client_error("error building dataframe", e, &local_query_id, None))?;
1595        let schema = df.schema().as_arrow();
1596        let mut schema_buffer = Vec::new();
1597        let mut writer = StreamWriter::try_new(&mut schema_buffer, schema)
1598            .map_err(|e| status!("error writing schema to in-memory buffer", e))?;
1599        writer
1600            .finish()
1601            .map_err(|e| status!("error closing arrow ipc stream writer", e))?;
1602        // here we could serialize the logical plan and return that as the prepared statement, but we would
1603        // need to register LogicalExtensionCodec for user-defined functions
1604        // instead, we are sending back the sql as we received it
1605        let result = ActionCreatePreparedStatementResult {
1606            prepared_statement_handle: query.query.into(),
1607            dataset_schema: schema_buffer.into(),
1608            parameter_schema: "".into(),
1609        };
1610        Ok(result)
1611    }
