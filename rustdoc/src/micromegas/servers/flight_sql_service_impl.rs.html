98/// Classifies the root cause of a `DataFusionError` into the gRPC status code
99/// that best distinguishes "the caller's SQL/input was bad" from "the server
100/// broke" -- see the FlightSQL error-classification plan (issue #1435) for the
101/// full rationale, including why `Execution`/`Configuration` land under
102/// `InvalidArgument` and why `ArrowError` stays `Internal`.
103pub fn classify_datafusion_error(err: &DataFusionError) -> tonic::Code {
104    use datafusion::error::DataFusionError as DFE;
105    match err.find_root() {
106        DFE::SQL(..)
107        | DFE::Plan(_)
108        | DFE::SchemaError(..)
109        | DFE::Execution(_)
110        | DFE::Configuration(_) => tonic::Code::InvalidArgument,
111        DFE::ResourcesExhausted(_) => tonic::Code::ResourceExhausted,
112        DFE::NotImplemented(_) => tonic::Code::Unimplemented,
113        _ => tonic::Code::Internal,
114    }
115}

117/// Maps a gRPC status code to the `QueryAuditRecord.error_class` bucket used
118/// both by the audit log and to gate the `query_failed`/
119/// `query_duration_with_error` metrics (see Design §4). This is the *derived* mapping used for
120/// every `DataFusionError`-rooted failure; the query-deny-list check (§5,
121/// `tasks/query_deny_list_plan.md`) stamps a fourth value, `"denied"`, directly via
122/// `QueryAuditState::fail_with_class` rather than through this function, since a denial's
123/// `ResourceExhausted` code would otherwise land in the `"resource"` bucket alongside a genuine
124/// resource-budget failure.
125pub fn error_class(code: tonic::Code) -> &'static str {
126    match code {
127        tonic::Code::InvalidArgument | tonic::Code::Unimplemented => "user",
128        tonic::Code::ResourceExhausted => "resource",
129        _ => "internal",
130    }
131}

133/// Cap on the physical-plan text appended to the server-side log line (Design
134/// §2). The plan can render arbitrarily long `file_groups=` sections for wide
135/// scans, and this text is server-log-only, never returned to the client.
136pub const MAX_PLAN_CHARS: usize = 2000;
137
138/// Truncate `text` at a char boundary once it exceeds `MAX_PLAN_CHARS`, tagging
139/// the cut with a trailing marker. Driven by `char_indices()` (not byte
140/// length) so a multibyte plan text (e.g. a non-ASCII string literal from a
141/// query predicate, rendered via `ScalarValue`'s `Display` impl) can't panic
142/// on a byte-offset slice landing mid-character.
143fn truncate_plan_text(text: &str) -> String {
144    match text.char_indices().nth(MAX_PLAN_CHARS) {
145        None => text.to_string(),
146        Some((i, _)) => format!("{}... (truncated)", &text[..i]),
147    }
148}

150/// Builds the full server-side log line for a classified `DataFusionError`:
151/// `desc`, the error's own `Display` (its outer `Context` chain intact, with
152/// backtrace if enabled -- unlike `find_root()`, which is for classification
153/// only), the `query_id`, and -- when there's no diagnostic span to show the
154/// caller instead (Design §2) -- the truncated physical plan text. `pub` so
155/// step 9's tests can assert its content, including the truncation marker,
156/// without capturing log output.
157pub fn build_log_line(
158    desc: &str,
159    err: &DataFusionError,
160    query_id: &str,
161    plan: Option<&Arc<dyn ExecutionPlan>>,
162) -> String {
163    let mut full = format!("{desc}: {err} (query_id={query_id})");
164    if let Some(plan) = plan {
165        let plan_text = format!(
166            "{}",
167            datafusion::physical_plan::displayable(plan.as_ref()).indent(true)
168        );
169        full.push_str(&format!(
170            "\nphysical plan:\n{}",
171            truncate_plan_text(&plan_text)
172        ));
173    }
174    full
175}

200/// Logs `build_log_line`'s output at `error!` for `error_class == "internal"`,
201/// `warn!` otherwise -- the single log point for every `DataFusionError`
202/// reaching `client_error`/`classify_flight_error`.
203fn error_or_warn_log(
204    code: tonic::Code,
205    desc: &str,
206    err: &DataFusionError,
207    query_id: &str,
208    plan: Option<&Arc<dyn ExecutionPlan>>,
209) {

213        _ => warn!("{full}"),
214    }

217/// Classifies `err`, builds the client-facing `Status` -- `desc` + root-error
218/// text (backtrace stripped) + an optional diagnostic span/notes/helps +
219/// `query_id` -- and logs the full error server-side (Design §2). The physical
220/// plan, when present and there's no diagnostic span to show instead, only
221/// ever reaches the server log, never the returned `Status`: `displayable`
222/// can render object-store partition paths from Micromegas view scans, which
223/// would leak internal lakehouse details to every caller hitting an
224/// execution-time error.
225pub fn client_error(
226    desc: &str,
227    err: DataFusionError,
228    query_id: &str,
229    plan: Option<&Arc<dyn ExecutionPlan>>,
230) -> Status {
231    let code = classify_datafusion_error(&err);
232    let mut msg = format!("{desc}: {}", err.find_root().strip_backtrace());
233    let mut has_span = false;
234    if let Some(diag) = err.diagnostic() {
235        if let Some(span) = diag.span {
236            has_span = true;
237            msg.push_str(&format!(
238                " (at line {}, column {})",
239                span.start.line, span.start.column
240            ));
241        }
242        for note in &diag.notes {
243            msg.push_str(&format!("\nnote: {}", note.message));
244        }
245        for help in &diag.helps {
246            msg.push_str(&format!("\nhelp: {}", help.message));
247        }
248    }
249    msg.push_str(&format!(" (query_id={query_id})"));
250    error_or_warn_log(
251        code,
252        desc,
253        &err,
254        query_id,
255        if has_span { None } else { plan },
256    );
257    Status::new(code, msg)
258}

260/// Recovers the `DataFusionError` from the per-batch `FlightError` re-wrap
261/// (`FlightError::ExternalError(Box::new(e))`, applied once the raw
262/// `DataFusionError` stream is wrapped for Arrow Flight) and classifies it via
263/// `client_error`. Every branch logs exactly once before returning, so this is
264/// a single, unconditional log point for the per-batch stream-error site
265/// regardless of which branch is taken. `pub` so step 9's external
266/// integration tests can exercise this recovery path directly.
267pub fn classify_flight_error(
268    err: FlightError,
269    query_id: &str,
270    plan: Option<&Arc<dyn ExecutionPlan>>,
271) -> Status {
272    match err {
273        FlightError::ExternalError(inner) => match inner.downcast::<DataFusionError>() {
274            Ok(df_err) => client_error("error building data stream", *df_err, query_id, plan),
275            Err(inner) => {
276                error!("error building data stream: {inner} (query_id={query_id})");
277                Status::internal(format!(
278                    "error building data stream: {inner} (query_id={query_id})"
279                ))
280            }
281        },
282        other => {
283            error!("error building data stream: {other} (query_id={query_id})");
284            Status::internal(format!(
285                "error building data stream: {other} (query_id={query_id})"
286            ))
287        }
288    }
289}

291/// Attribution, per-stage timing, and (once created) the physical plan for
292/// one query. Built as soon as attribution is resolved, updated as each
293/// setup stage completes, and emitted exactly once as a [`QueryAuditRecord`]
294/// under the `flightsql_query_audit` log target — either on an early setup
295/// failure, at stream completion/error, or (if the stream is dropped mid-drain)
296/// from `CompletionTrackedStream`'s `Drop` impl.
297struct QueryAuditState {
298    /// Minted as the very first statement of `execute_query`, before any
299    /// fallible step -- included in every client-facing `Status` built by
300    /// `client_error`/`classify_flight_error` and in this query's audit
301    /// record, so a failure's server-log line and its audit record can be
302    /// correlated by grepping this id.
303    query_id: String,
304    client_ip: String,
305    client: String,
306    agent: String,
307    entrypoint: String,
308    session: Option<String>,
309    notebook: Option<String>,
310    cell: Option<String>,
311    user: String,
312    email: String,
313    name: Option<String>,
314    service_account: bool,
315    service_account_name: Option<String>,
316    sql: String,
317    /// Normalized SQL fingerprint (`tasks/query_deny_list_plan.md` §2), computed once per query
318    /// regardless of whether the deny list is in active use.
319    sql_hash: String,
320    range_begin: Option<String>,
321    range_end: Option<String>,
322    limit: Option<u64>,
323    context_init_ms: f64,
324    planning_ms: f64,
325    execution_ms: f64,
326    setup_ms: f64,
327    request_start: Instant,
328    /// `None` until the physical plan is created; set as soon as it is, and
329    /// still `None` for a record emitted on a setup failure that happens
330    /// before that point.
331    plan: Option<Arc<dyn ExecutionPlan>>,
332    /// This query's per-query memory-pool wrapper; owned from construction so
333    /// `emit()` can read its peak on every terminal path, including setup
334    /// failures.
335    pool: Arc<ScopedMemoryPool>,
336}

338impl QueryAuditState {
339    /// Aggregate the plan's DataFusion metrics (if a physical plan was
340    /// created before the failure/completion), assemble the audit record,
341    /// and emit it as a single JSON log line.
342    ///
343    /// Takes `&self` (rather than consuming) so it can be called from a
344    /// setup-error `map_err` closure while leaving the state available for
345    /// further updates/completion on the success path, and so `Drop` can
346    /// call it on an abandoned/cancelled stream without needing to
347    /// reconstruct anything.
348    fn emit(&self, status: &'static str, error: Option<String>, error_class: Option<&'static str>) {
349        let scan = match &self.plan {
350            Some(plan) => aggregate_scan_metrics(plan.as_ref()),
351            None => ScanMetrics {
352                output_rows: None,
353                bytes_scanned: 0,
354                spilled_bytes: 0,
355                spill_count: 0,
356            },
357        };
358        let peak_memory_bytes = self.pool.peak() as u64;
359        imetric!("query_peak_memory_bytes", "bytes", peak_memory_bytes);
360        let total_ms = self.request_start.elapsed().as_secs_f64() * 1000.0;
361        let record = QueryAuditRecord {
362            query_id: self.query_id.clone(),
363            client_ip: self.client_ip.clone(),
364            client: self.client.clone(),
365            agent: self.agent.clone(),
366            entrypoint: self.entrypoint.clone(),
367            session: self.session.clone(),
368            notebook: self.notebook.clone(),
369            cell: self.cell.clone(),
370            user: self.user.clone(),
371            email: self.email.clone(),
372            name: self.name.clone(),
373            service_account: self.service_account,
374            service_account_name: self.service_account_name.clone(),
375            sql: self.sql.clone(),
376            sql_hash: self.sql_hash.clone(),
377            range_begin: self.range_begin.clone(),
378            range_end: self.range_end.clone(),
379            limit: self.limit,
380            context_init_ms: self.context_init_ms,
381            planning_ms: self.planning_ms,
382            execution_ms: self.execution_ms,
383            setup_ms: self.setup_ms,
384            total_ms,
385            status,
386            error,
387            error_class,
388            output_rows: scan.output_rows,
389            bytes_scanned: scan.bytes_scanned,
390            peak_memory_bytes,
391            spilled_bytes: scan.spilled_bytes,
392            spill_count: scan.spill_count,
393        };
394        match serde_json::to_string(&record) {
395            Ok(json) => info!(target: "flightsql_query_audit", "{json}"),
396            Err(e) => warn!("failed to serialize query audit record: {e}"),
397        }
398    }
399
400    /// Emits an "error" audit record for an already-built client-facing `status`, with an
401    /// explicit `error_class` rather than one derived from the gRPC code -- used by the
402    /// query-deny-list check (Design §5), whose `error_class` is `"denied"`, a value
403    /// `error_class(status.code())` (`ResourceExhausted` -> `"resource"`) would get wrong.
404    /// Returns `status` unchanged.
405    fn fail_with_class(&self, status: Status, class: &'static str) -> Status {
406        self.emit("error", Some(status.message().to_string()), Some(class));
407        status
408    }
409
410    /// Emits an "error" audit record for an already-built client-facing
411    /// `status` (message + `error_class` derived from its code) and returns it
412    /// unchanged -- the shared tail of every setup-phase `map_err` in
413    /// `execute_query`. A thin wrapper over [`Self::fail_with_class`] that derives the class
414    /// from the code.
415    fn fail(&self, status: Status) -> Status {
416        let class = error_class(status.code());
417        self.fail_with_class(status, class)
418    }
419}

421/// Stream wrapper that tracks when the stream is fully consumed
422struct CompletionTrackedStream<S> {
423    inner: S,
424    start_time: i64,
425    completed: bool,
426    audit: Option<QueryAuditState>,
427}

429impl<S> CompletionTrackedStream<S> {

432            inner,
433            start_time,
434            completed: false,
435            audit: Some(audit),
436        }
437    }
438}

440impl<S> Drop for CompletionTrackedStream<S> {
441    /// If the stream is dropped before yielding `None` or an `Err` (client
442    /// disconnect/cancel mid-drain), `poll_next` never ran its completion
443    /// arms, so `self.audit` is still `Some(...)`. Emit it here with a
444    /// terminal "incomplete" status so cancelled/abandoned queries are still
445    /// audited instead of silently vanishing. A no-op for streams that
446    /// completed or errored normally, since those arms already `take()` the
447    /// audit state.
448    fn drop(&mut self) {
449        if let Some(state) = self.audit.take() {
450            state.emit("incomplete", None, None);
451        }
452    }
453}
454
455impl<S> Stream for CompletionTrackedStream<S>
456where
457    S: Stream<Item = Result<arrow_flight::FlightData, Status>> + Unpin + Send,
458{
459    type Item = Result<arrow_flight::FlightData, Status>;
460
461    fn poll_next(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
462        match Pin::new(&mut self.inner).poll_next(cx) {
463            Poll::Ready(Some(result)) => {
464                // Every `Status` reaching this stream was already logged once by
465                // `classify_flight_error` (the sole `map_err` feeding it) when it
466                // built that `Status` -- this arm only reads the code to derive
467                // the audit/metric class, it never logs again (Design §4).
468                if let Err(ref err) = result
469                    && !self.completed
470                {
471                    let class = error_class(err.code());
472                    let total_duration = now() - self.start_time;
473                    match class {
474                        "internal" => {
475                            imetric!("query_duration_with_error", "ticks", total_duration as u64);
476                            imetric!("query_failed", "count", 1);
477                        }
478                        "resource" => {
479                            imetric!("query_failed_resource", "count", 1);
480                        }
481                        _ => {
482                            imetric!("query_failed_user", "count", 1);
483                        }
484                    }
485                    self.completed = true;
486                    if let Some(state) = self.audit.take() {
487                        state.emit("error", Some(err.message().to_string()), Some(class));
488                    }
489                }
490                Poll::Ready(Some(result))
491            }
492            Poll::Ready(None) => {
493                // Stream completed successfully
494                if !self.completed {
495                    let total_duration = now() - self.start_time;
496                    imetric!("query_duration_total", "ticks", total_duration as u64);
497                    imetric!("query_completed_successfully", "count", 1);
498                    self.completed = true;
499                    if let Some(state) = self.audit.take() {
500                        state.emit("ok", None, None);
501                    }
502                }
503                Poll::Ready(None)
504            }
505            Poll::Pending => Poll::Pending,
506        }
507    }
508}

567    /// (`rust/auth/src/tower.rs`) rejects the request before this inner service ever runs
568    /// whenever a provider *is* configured, so the extension is always present in that case --
569    /// the same safety argument `is_admin`'s absent-header-⇒-trusted convention already relies
570    /// on (`user_attribution.rs`).
571    ///
572    /// **Failure convention.** A [`ReadPolicy::resolve`] that returns `Err` is a hard failure and
573    /// must never become a scope -- not `ReadScope::Audiences(Arc::from([]))` (which would read
574    /// as a legitimate, audited fail-closed decision to Stage 2/3) and not `ReadScope::All` (a
575    /// silent fail-open bypass). Mirrors `tower.rs`'s own discriminator: `Status::unavailable`
576    /// when the error downcasts to [`ProviderUnavailable`] (a store/provider outage),
577    /// `Status::permission_denied` otherwise.
578    ///
579    /// `is_admin` is read from `md` unchanged (`is_admin(md)`), equivalent to reading
580    /// `AuthContext.is_admin` off `ext` when the extension is present, since the header is
581    /// derived from the same `AuthContext` with client-supplied copies stripped
582    /// (`tower.rs:107-139`) -- and it is what preserves today's `--disable-auth`
583    /// absent-header-⇒-trusted convention when `ext` has none.
584    async fn caller_context(
585        &self,
586        ext: &http::Extensions,
587        md: &MetadataMap,
588    ) -> Result<CallerContext, Status> {
589        // `deny_queries` requires `Some` (query_deny_list_plan.md §7/§8) -- resolved from the
590        // same attribution `execute_query` already validates, so a query already carries this by
591        // the time it could call `deny_queries`. This is simply "the caller's resolved user id"
592        // (`user_attribution::validate_and_resolve_user_attribution_grpc`).
593        let identity = validate_and_resolve_user_attribution_grpc(md)
594            .ok()
595            .map(|attr| attr.user_id);
596        let read_scope = match ext.get::<AuthContext>() {
597            Some(auth_ctx) => match self.read_policy.resolve(auth_ctx).await {
598                Ok(audiences) => ReadScope::Audiences(audiences.into_inner()),
599                Err(e) => {
600                    return Err(if e.downcast_ref::<ProviderUnavailable>().is_some() {
601                        Status::unavailable(format!("read policy unavailable: {e:#}"))
602                    } else {
603                        Status::permission_denied(format!("read scope denied: {e:#}"))
604                    });
605                }
606            },
607            None => ReadScope::All,
608        };
609        Ok(CallerContext {
610            read_scope,
611            is_admin: is_admin(md),
612            // Not permission-sensitive the way `read_scope` is (it's deployment config, not
613            // derived from the caller's identity), so it is copied verbatim on both branches
614            // above rather than participating in the absent-extension/`Err` distinction.
615            isolation_config: self.isolation_config.clone(),
616            // Deployment-wide, derived once at startup (`flight_sql_server.rs`) -- same
617            // treatment as `isolation_config` above.
618            admin_principal_possible: self.admin_principal_possible,
619            identity,
620        })
621    }

640    #[span_fn]
641    async fn execute_query(
642        &self,
643        ticket_stmt: TicketStatementQuery,
644        metadata: &MetadataMap,
645        extensions: &http::Extensions,
646        client_ip: &str,
647    ) -> Result<Response<FlightDataStream>, Status> {
648        // Minted first, before any fallible step, so it's Always available --
649        // for every client-facing `Status` built by `client_error`/
650        // `classify_flight_error` and for this query's audit record (Design §3).
651        let query_id = Uuid::new_v4().to_string();
652        let begin_request = now();
653        let request_start = Instant::now();
654        let sql = std::str::from_utf8(&ticket_stmt.statement_handle)
655            .map_err(|e| status!("Unable to parse query", e))?;
656
657        let mut begin = metadata.get("query_range_begin");
658        if let Some(s) = &begin
659            && s.is_empty()
660        {
661            begin = None;
662        }
663        let mut end = metadata.get("query_range_end");
664        if let Some(s) = &end
665            && s.is_empty()
666        {
667            end = None;
668        }
669        let query_range = if begin.is_some() && end.is_some() {
670            let begin_datetime =
671                chrono::DateTime::parse_from_rfc3339(begin.unwrap().to_str().map_err(|e| {
672                    client_input_error!("Unable to convert query_range_begin to string", e)
673                })?)
674                .map_err(|e| {
675                    client_input_error!(
676                        "Unable to parse query_range_begin as a rfc3339 datetime",
677                        e
678                    )
679                })?;
680            let end_datetime =
681                chrono::DateTime::parse_from_rfc3339(end.unwrap().to_str().map_err(|e| {
682                    client_input_error!("Unable to convert query_range_end to string", e)
683                })?)
684                .map_err(|e| {
685                    client_input_error!("Unable to parse query_range_end as a rfc3339 datetime", e)
686                })?;
687            Some(TimeRange::new(begin_datetime.into(), end_datetime.into()))
688        } else {
689            None
690        };
691
692        // Validate and resolve user attribution
693        let attr = validate_and_resolve_user_attribution_grpc(metadata).map_err(|e| *e)?;
694
695        let client_type = metadata
696            .get("x-client-type")
697            .and_then(|v| v.to_str().ok())
698            .unwrap_or("unknown");
699        let client_agent = metadata
700            .get("x-client-agent")
701            .and_then(|v| v.to_str().ok())
702            .unwrap_or("unknown");
703        let client_entrypoint = metadata
704            .get("x-client-entrypoint")
705            .and_then(|v| v.to_str().ok())
706            .unwrap_or("unknown");
707        let client_session = Self::optional_metadata(metadata, "x-client-session");
708        let client_notebook = Self::optional_metadata(metadata, "x-client-notebook");
709        let client_cell = Self::optional_metadata(metadata, "x-client-cell");
710
711        let user_name_display = attr.user_name.as_deref().unwrap_or("");
712
713        // Log query with full attribution
714        if let Some(service_account_name) = &attr.service_account {
715            info!(
716                "execute_query range={query_range:?} sql={sql:?} limit={:?} user={} email={} name={user_name_display:?} service_account={service_account_name} client={client_type} agent={client_agent} entrypoint={client_entrypoint} notebook={client_notebook:?} cell={client_cell:?} client_ip={client_ip}",
717                metadata.get("limit"),
718                attr.user_id,
719                attr.user_email
720            );
721        } else {
722            info!(
723                "execute_query range={query_range:?} sql={sql:?} limit={:?} user={} email={} name={user_name_display:?} client={client_type} agent={client_agent} entrypoint={client_entrypoint} notebook={client_notebook:?} cell={client_cell:?} client_ip={client_ip}",
724                metadata.get("limit"),
725                attr.user_id,
726                attr.user_email
727            );
728        }
729
730        // The per-query memory-pool wrapper is infallible to construct, so it's built
731        // before `audit_state` and owned by it from the start -- every subsequent
732        // setup failure (session context, planning, limit, physical plan, stream
733        // construction) can then still emit an "error" audit record carrying whatever
734        // peak the pool had accrued by that point, instead of silently disappearing on
735        // an early `?` return.
736        let scoped_pool = Arc::new(ScopedMemoryPool::new(
737            self.lakehouse.runtime().memory_pool.clone(),
738        ));
739
740        // Attribution is resolved from here on, so build the audit state now
741        // (durations/limit/plan filled in as they become known) instead of
742        // only after the physical plan exists.
743        // Computed once per query, regardless of whether the deny list is in active use --
744        // it's what an operator pastes into `deny_queries` after finding an offender in the
745        // audit log (Design §2/§7).
746        let sql_hash = fingerprint_of(sql);
747
748        let mut audit_state = QueryAuditState {
749            query_id: query_id.clone(),
750            client_ip: client_ip.to_string(),
751            client: client_type.to_string(),
752            agent: client_agent.to_string(),
753            entrypoint: client_entrypoint.to_string(),
754            session: client_session,
755            notebook: client_notebook,
756            cell: client_cell,
757            user: attr.user_id.clone(),
758            email: attr.user_email.clone(),
759            name: attr.user_name.clone(),
760            service_account: attr.service_account.is_some(),
761            service_account_name: attr.service_account.clone(),
762            sql: sql.to_string(),
763            sql_hash: sql_hash.clone(),
764            range_begin: query_range.as_ref().map(|r| r.begin.to_rfc3339()),
765            range_end: query_range.as_ref().map(|r| r.end.to_rfc3339()),
766            limit: None,
767            context_init_ms: 0.0,
768            planning_ms: 0.0,
769            execution_ms: 0.0,
770            setup_ms: 0.0,
771            request_start,
772            plan: None,
773            pool: scoped_pool.clone(),
774        };
775
776        // Design §5: the deny-list check, in front of everything expensive -- before the scoped
777        // runtime/session context/planning below. `check` returns `None` immediately on an
778        // empty snapshot, so this costs nothing while the deny list stands empty.
779        if let Some(rule) = self.lakehouse.query_denials().check(&QueryAttribution {
780            user_id: &attr.user_id,
781            email: &attr.user_email,
782            service_account: attr.service_account.as_deref(),
783            client: client_type,
784            agent: client_agent,
785            entrypoint: client_entrypoint,
786            session: audit_state.session.as_deref(),
787            notebook: audit_state.notebook.as_deref(),
788            cell: audit_state.cell.as_deref(),
789            client_ip,
790            sql,
791            sql_hash: &sql_hash,
792        }) && !skip_for_admin_recovery(sql, is_admin(metadata), self.admin_principal_possible)
793        {
794            let status = Status::resource_exhausted(format!(
795                "query denied by rule {} (reason: {}); ask an admin to lift it with \
796                 remove_query_denial('{}'); query_id={query_id}",
797                rule.row.rule_id, rule.row.reason, rule.row.rule_id
798            ));
799            warn!(
800                "{}",
801                build_denial_warn_line(
802                    &rule.row.rule_id.to_string(),
803                    &rule.row.reason,
804                    &sql_hash,
805                    &attr.user_id,
806                    &attr.user_email,
807                    client_type,
808                    client_entrypoint,
809                    client_ip,
810                    &query_id,
811                )
812            );
813            imetric!("query_denied", "count", rule_tags(&rule.row.rule_id), 1_u64);
814            rule.record_hit();
815            return Err(audit_state.fail_with_class(status, "denied"));
816        }
817
818        // Build a `RuntimeEnv`/`LakehouseContext` scoped to this query's memory-pool
819        // wrapper, so every session context created from it (including nested ones,
820        // e.g. Perfetto trace queries and JIT materialization) attributes its memory
821        // to this query alone instead of the process-shared pool.
822        let scoped_env = scoped_runtime(self.lakehouse.runtime(), scoped_pool.clone())
823            .map_err(|e| audit_state.fail(status!("error building scoped runtime", e)))?;
824        let lakehouse = self.lakehouse.with_runtime(scoped_env);
825
826        // Session context creation phase
827        let session_begin = now();
828        let session_begin_instant = Instant::now();
829        let caller = self
830            .caller_context(extensions, metadata)
831            .await
832            .map_err(|status| audit_state.fail(status))?;
833        let ctx = make_session_context(
834            lakehouse,
835            self.part_provider.clone(),
836            query_range,
837            self.view_factory.clone(),
838            self.session_configurator.clone(),
839            caller,
840        )
841        .await
842        .map_err(|e| audit_state.fail(status!("error in make_session_context", e)))?;
843        let context_init_duration = now() - session_begin;
844        audit_state.context_init_ms = session_begin_instant.elapsed().as_secs_f64() * 1000.0;
845
846        // Query planning phase
847        let planning_begin = now();
848        let planning_begin_instant = Instant::now();
849        let mut df = ctx.sql(sql).await.map_err(|e| {
850            audit_state.fail(client_error("error building dataframe", e, &query_id, None))
851        })?;
852        let planning_duration = now() - planning_begin;
853        audit_state.planning_ms = planning_begin_instant.elapsed().as_secs_f64() * 1000.0;
854
855        if let Some(limit_str) = metadata.get("limit") {
856            let parsed_limit: usize = usize::from_str(limit_str.to_str().map_err(|e| {
857                audit_state.fail(client_input_error!("error converting limit to str", e))
858            })?)
859            .map_err(|e| audit_state.fail(client_input_error!("error parsing limit", e)))?;
860            audit_state.limit = Some(parsed_limit as u64);
861            df = df.limit(0, Some(parsed_limit)).map_err(|e| {
862                audit_state.fail(client_error(
863                    "error building dataframe with limit",
864                    e,
865                    &query_id,
866                    None,
867                ))
868            })?;
869        }
870
871        // Query execution phase: build the physical plan (kept for post-drain metrics)
872        // and run it via the free-function `execute_stream`, which is what
873        // `DataFrame::execute_stream` does internally, minus dropping the plan.
874        let execution_begin = now();
875        let execution_begin_instant = Instant::now();
876        let schema = Arc::new(df.schema().as_arrow().clone());
877        let task_ctx = Arc::new(df.task_ctx());
878        let plan = df.create_physical_plan().await.map_err(|e| {
879            audit_state.fail(client_error(
880                "error creating physical plan",
881                e,
882                &query_id,
883                None,
884            ))
885        })?;
886        audit_state.plan = Some(plan.clone());
887        // `plan`/`query_id` are cloned here because both are moved shortly:
888        // `plan` into `execute_stream` below, `audit_state` (which owns
889        // `query_id`'s sibling on the struct) into `CompletionTrackedStream::new`
890        // further down. The per-batch closure below outlives this function call
891        // (it's polled from the returned stream), so it needs its own owned
892        // clones rather than borrowing `plan`/`query_id` directly (Design §2).
893        let plan_for_errors = plan.clone();
894        let query_id_for_stream = query_id.clone();
895        let stream = execute_stream(plan, task_ctx)
896            .map_err(|e| {
897                audit_state.fail(client_error(
898                    "error executing plan",
899                    e,
900                    &query_id_for_stream,
901                    Some(&plan_for_errors),
902                ))
903            })?
904            .map_err(|e| FlightError::ExternalError(Box::new(e)));
905        let builder = if Self::should_preserve_dictionary(metadata) {
906            FlightDataEncoderBuilder::new()
907                .with_schema(schema.clone())
908                .with_dictionary_handling(DictionaryHandling::Resend)
909        } else {
910            FlightDataEncoderBuilder::new().with_schema(schema.clone())
911        };
912        let flight_data_stream = builder.build(stream);
913        let execution_duration = now() - execution_begin;
914        audit_state.execution_ms = execution_begin_instant.elapsed().as_secs_f64() * 1000.0;
915
916        // Calculate total setup time and record detailed metrics
917        let total_setup_duration = now() - begin_request;
918        audit_state.setup_ms = request_start.elapsed().as_secs_f64() * 1000.0;
919
920        // Record detailed timing metrics
921        imetric!(
922            "context_init_duration",
923            "ticks",
924            context_init_duration as u64
925        );
926        imetric!("query_planning_duration", "ticks", planning_duration as u64);
927        imetric!(
928            "query_execution_duration",
929            "ticks",
930            execution_duration as u64
931        );
932        imetric!("query_setup_duration", "ticks", total_setup_duration as u64);
933
934        // Create instrumented stream that tracks completion. This closure owns
935        // `plan_for_errors`/`query_id_for_stream` (moved in) since it's polled
936        // from the returned stream, long after this function returns -- see the
937        // comment above `plan_for_errors`'s definition.
938        let instrumented_stream = flight_data_stream.map_err(move |e| {
939            classify_flight_error(e, &query_id_for_stream, Some(&plan_for_errors))
940        });
941        let completion_tracked_stream =
942            CompletionTrackedStream::new(instrumented_stream.boxed(), begin_request, audit_state);
943        Ok(Response::new(
944            Box::pin(completion_tracked_stream) as FlightDataStream
945        ))
946    }
947}

1169    async fn do_get_tables(
1170        &self,
1171        query: CommandGetTables,
1172        _request: Request<Ticket>,
1173    ) -> Result<Response<<Self as FlightService>::DoGetStream>, Status> {
1174        let begin_request = now();
1175        info!("do_get_tables {query:?}");
1176        let mut builder = query.into_builder();
1177        for view in self.view_factory.get_global_views() {
1178            let catalog_name = "";
1179            let schema_name = "";
1180            builder
1181                .append(
1182                    catalog_name,
1183                    schema_name,
1184                    &*view.get_view_set_name(),
1185                    "table",
1186                    &view.get_file_schema(),
1187                )
1188                .map_err(Status::from)?;
1189        }
1190        let schema = builder.schema();
1191        let batch = builder.build();
1192        let stream = FlightDataEncoderBuilder::new()
1193            .with_schema(schema)
1194            .build(futures::stream::once(async { batch }))
1195            .map_err(Status::from);
1196        let duration = now() - begin_request;
1197        imetric!("request_duration", "ticks", duration as u64);
1198        Ok(Response::new(Box::pin(stream)))
1199    }

1274    /// `bulk_ingest` (routed here from `CommandStatementIngest`) is a replication/administrative
1275    /// API, not an ordinary analytics write path: it writes row properties -- including
1276    /// `micromegas.audience` on `processes` rows -- verbatim, with none of the server-side
1277    /// stamping or reserved-namespace stripping the HTTP ingestion paths apply. Gating this RPC
1278    /// on `is_admin` is what keeps that verbatim write safe: an admin-run replication tool
1279    /// re-ingesting rows already stamped at their origin lake is exactly the case the docs
1280    /// describe (`mkdocs/docs/query-guide/python-api.md`), while an ordinary authenticated
1281    /// analytics client must not be able to set `micromegas.audience` directly (#1373, AbAC
1282    /// Stage 5).
1283    #[span_fn]
1284    async fn do_put_statement_ingest(
1285        &self,
1286        command: CommandStatementIngest,
1287        request: Request<PeekableFlightDataStream>,
1288    ) -> Result<i64, Status> {
1289        let table_name = command.table;
1290        info!("do_put_statement_ingest table_name={table_name}");
1291        if !is_admin(request.metadata()) {
1292            return Err(Status::permission_denied(
1293                "bulk_ingest requires admin privileges",
1294            ));
1295        }
1296        let stream = FlightRecordBatchStream::new_from_flight_data(
1297            request.into_inner().map_err(|e| e.into()),
1298        );
1299        bulk_ingest(self.lakehouse.lake().clone(), &table_name, stream)
1300            .await
1301            .map_err(|e| {
1302                let msg = format!("error ingesting into {table_name}: {e:?}");
1303                error!("{msg}");
1304                status!(msg, e)
1305            })
1306    }

1332    #[span_fn]
1333    async fn do_action_create_prepared_statement(
1334        &self,
1335        query: ActionCreatePreparedStatementRequest,
1336        request: Request<Action>,
1337    ) -> Result<ActionCreatePreparedStatementResult, Status> {
1338        info!("do_action_create_prepared_statement query={}", &query.query);
1339
1340        // Closes hole #1 (#1369): prepared statements now resolve the same CallerContext as the
1341        // do_get execute path, instead of reading only is_admin and no identity at all.
1342        let caller = self
1343            .caller_context(request.extensions(), request.metadata())
1344            .await?;
1345        let ctx = make_session_context(
1346            self.lakehouse.clone(),
1347            self.part_provider.clone(),
1348            None,
1349            self.view_factory.clone(),
1350            self.session_configurator.clone(),
1351            caller,
1352        )
1353        .await
1354        .map_err(|e| status!("error in make_session_context", e))?;
1355
1356        // No `QueryAuditState`/audit record exists on this RPC (it only plans the
1357        // query, never executes it) -- this id exists solely so the client-facing
1358        // message and a server log line can be correlated with each other.
1359        let local_query_id = Uuid::new_v4().to_string();
1360        let df = ctx
1361            .sql(&query.query)
1362            .await
1363            .map_err(|e| client_error("error building dataframe", e, &local_query_id, None))?;
1364        let schema = df.schema().as_arrow();
1365        let mut schema_buffer = Vec::new();
1366        let mut writer = StreamWriter::try_new(&mut schema_buffer, schema)
1367            .map_err(|e| status!("error writing schema to in-memory buffer", e))?;
1368        writer
1369            .finish()
1370            .map_err(|e| status!("error closing arrow ipc stream writer", e))?;
1371        // here we could serialize the logical plan and return that as the prepared statement, but we would
1372        // need to register LogicalExtensionCodec for user-defined functions
1373        // instead, we are sending back the sql as we received it
1374        let result = ActionCreatePreparedStatementResult {
1375            prepared_statement_handle: query.query.into(),
1376            dataset_schema: schema_buffer.into(),
1377            parameter_schema: "".into(),
1378        };
1379        Ok(result)
1380    }
