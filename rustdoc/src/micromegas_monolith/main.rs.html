1//! Single-process deployment of all Micromegas roles.
2//!
3//! Runs ingestion, FlightSQL, maintenance, and web in one tokio runtime
4//! sharing a single data-lake connection, cache, and SIGTERM fanout.
5//!
6//! Env variables (in addition to role-specific vars):
7//!  - `MICROMEGAS_SQL_CONNECTION_STRING`      — shared data-lake Postgres
8//!  - `MICROMEGAS_OBJECT_STORE_URI`           — shared object store
9//!  - `MICROMEGAS_APP_SQL_CONNECTION_STRING`  — web-app Postgres
10//!  - `MICROMEGAS_WEB_CORS_ORIGIN`            — required by web role
11//!  - `MICROMEGAS_BASE_PATH`                  — required by web role (e.g. `/`)
12//!  - `MICROMEGAS_MONOLITH_ROLES`             — comma-separated role list (default `all`)
13//!  - `MICROMEGAS_OBJECT_CACHE_URL`           — optional external object-cache server; the monolith always runs its own in-process L1 `RangeCache`, and additionally routes reads and write-time warming through this L2 server if set (with the API key below)
14//!  - `MICROMEGAS_OBJECT_CACHE_API_KEY`       — optional; required alongside the URL above, else the cache is bypassed
15

45// ---------------------------------------------------------------------------
46// Role parsing
47// ---------------------------------------------------------------------------
48
49const ROLE_INGESTION: &str = "ingestion";
50const ROLE_FLIGHTSQL: &str = "flightsql";
51const ROLE_MAINTENANCE: &str = "maintenance";
52const ROLE_WEB: &str = "web";
53const ALL_ROLES: &[&str] = &[ROLE_INGESTION, ROLE_FLIGHTSQL, ROLE_MAINTENANCE, ROLE_WEB];
54
55struct Roles {
56    ingestion: bool,
57    flightsql: bool,
58    maintenance: bool,
59    web: bool,
60}
61
62impl Roles {
63    fn parse(spec: &str) -> Result<Self> {
64        let normalized = spec.trim().to_lowercase();
65        let set: HashSet<&str> = if normalized == "all" {
66            ALL_ROLES.iter().copied().collect()
67        } else {
68            normalized
69                .split(',')
70                .map(|s| s.trim())
71                .filter(|s| !s.is_empty())
72                .collect::<HashSet<_>>()
73                .into_iter()
74                .map(|s| {
75                    if ALL_ROLES.contains(&s) {
76                        Ok(s)
77                    } else {
78                        anyhow::bail!("Unknown role '{s}'. Valid roles: {}", ALL_ROLES.join(", "))
79                    }
80                })
81                .collect::<Result<HashSet<_>>>()?
82        };
83
84        if set.is_empty() {
85            anyhow::bail!("At least one role must be specified");
86        }
87
88        Ok(Self {
89            ingestion: set.contains(ROLE_INGESTION),
90            flightsql: set.contains(ROLE_FLIGHTSQL),
91            maintenance: set.contains(ROLE_MAINTENANCE),
92            web: set.contains(ROLE_WEB),
93        })
94    }
95
96    fn needs_lakehouse(&self) -> bool {
97        self.ingestion || self.flightsql || self.maintenance
98    }
99}
100
101impl fmt::Display for Roles {
102    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
103        let mut active = vec![];
104        if self.ingestion {
105            active.push(ROLE_INGESTION);
106        }
107        if self.flightsql {
108            active.push(ROLE_FLIGHTSQL);
109        }
110        if self.maintenance {
111            active.push(ROLE_MAINTENANCE);
112        }
113        if self.web {
114            active.push(ROLE_WEB);
115        }
116        write!(f, "{}", active.join(", "))
117    }
118}

144    /// Frontend build directory (web role)

146    frontend_dir: String,
147
148    /// Disable authentication for all roles (development only)
149    #[clap(long)]
150    disable_auth: bool,
151
152    /// Disable authentication for the ingestion role only (useful when web uses OIDC)
153    #[clap(long)]

172#[micromegas_main(interop_max_level = "info", max_level_override = "debug")]
173async fn main() -> Result<()> {
174    let args = Cli::parse();
175    let roles = Roles::parse(&args.roles)?;
176    let grace = args.common.grace();
177
178    info!("Starting micromegas-monolith with roles: {roles}");
179
180    // Build shared data lake for lake-backed roles
181    let lakehouse: Option<Arc<LakehouseContext>> = if roles.needs_lakehouse() {
182        let cfg = DataLakeConfig::from_env()?;
183        info!("Connecting to data lake (migrate_db + migrate_lakehouse)");
184        let lake = connect_to_remote_data_lake(&cfg.sql_connection_string, &cfg.object_store_uri)
185            .await
186            .with_context(|| "connecting to data lake")?;
187        let lh = LakehouseContext::from_connection(Arc::new(lake))
188            .await
189            .with_context(|| "building LakehouseContext")?;
190        Some(lh)
191    } else {
192        None
193    };
194
195    // Taken while `lakehouse` is still borrowed, before the role join_set spawns
196    // below — each dedicated key-store pool is built from this shared lake pool's
197    // connect options, not a clone of the lake pool itself (see
198    // `db_api_key::dedicated_key_store_pool`).
199    let lake_pool = lakehouse.as_ref().map(|lh| lh.lake().db_pool.clone());
200
201    // Role-scoped auth providers (ingestion = machine-to-machine, analytics = human/tooling)
202    let ingestion_auth = if roles.ingestion && !args.disable_auth && !args.disable_ingestion_auth {
203        let pool = lake_pool
204            .clone()
205            .expect("lakehouse must be Some when ingestion role is enabled");
206        let key_store_pool = dedicated_key_store_pool(&pool);
207        match ProviderBuilder::new("MICROMEGAS_INGESTION")
208            .with_db_key_store(key_store_pool, ApiKeyTable::Ingestion)
209            .build()
210            .await?
211        {
212            Some(p) => Some(p),
213            None => {
214                anyhow::bail!(
215                    "Ingestion auth required but no providers configured. \
216                     Set MICROMEGAS_INGESTION_API_KEYS, MICROMEGAS_API_KEYS, populate the \
217                     ingestion_api_keys DB table, or --disable-auth"
218                );
219            }
220        }
221    } else {
222        None
223    };
224
225    let analytics_auth = if roles.flightsql && !args.disable_auth {
226        let pool = lake_pool
227            .clone()
228            .expect("lakehouse must be Some when flightsql role is enabled");
229        let key_store_pool = dedicated_key_store_pool(&pool);
230        match ProviderBuilder::new("MICROMEGAS_ANALYTICS")
231            .with_db_key_store(key_store_pool, ApiKeyTable::Analytics)
232            .build()
233            .await?
234        {
235            Some(p) => Some(p),
236            None => {
237                anyhow::bail!(
238                    "Analytics auth required but no providers configured. \
239                     Set MICROMEGAS_ANALYTICS_OIDC_CONFIG, MICROMEGAS_OIDC_CONFIG, populate the \
240                     analytics_api_keys DB table, or --disable-auth. \
241                     To also disable analytics auth, pass `--disable-auth` instead of `--disable-ingestion-auth`."
242                );
243            }
244        }
245    } else {
246        None
247    };
248
249    // Resolved alongside `analytics_auth` (#1369, AbAC Stage 1 step 12; grant map rewrite
250    // #1372, Stage 4): unset `MICROMEGAS_ANALYTICS_AUDIENCE_GRANTS`/`MICROMEGAS_AUDIENCE_GRANTS`
251    // -> an empty grant map -> a real caller's resolved scope is just `{public}`, filtered by
252    // `OwnershipRewrite` (#1370, AbAC Stage 2) -- `MICROMEGAS_UNSTAMPED_AUDIENCE` defaults to
253    // `public` (`IsolationConfig::from_env`), so legacy, never-stamped data stays visible unless
254    // an operator opts back into fail-closed by setting it to an empty string.
255    let analytics_read_policy = if roles.flightsql && !args.disable_auth {
256        // One shared snapshot cache for this process (#1489, AbAC Stage 6a), built from its own
257        // dedicated pool via the same `dedicated_key_store_pool` convention `analytics_auth`
258        // above already uses. Resolved under the same `MICROMEGAS_ANALYTICS` prefix
259        // `AudienceReadPolicy::from_env` beside it uses, so the cache-TTL knob follows the same
260        // `{prefix}_` fallback every other knob at this wiring site does.
261        let pool = lake_pool
262            .clone()
263            .expect("lakehouse must be Some when flightsql role is enabled");
264        let audience_grants_pool = dedicated_key_store_pool(&pool);
265        let audience_grants_config =
266            DbAudienceGrantsConfig::from_env_with_prefix("MICROMEGAS_ANALYTICS");
267        let audience_grants_store = Arc::new(DbAudienceGrantsSource::new(
268            audience_grants_pool,
269            audience_grants_config,
270        ));
271        Some(Arc::new(
272            AudienceReadPolicy::from_env("MICROMEGAS_ANALYTICS")?
273                .with_store(Some(audience_grants_store)),
274        ) as Arc<dyn ReadPolicy>)
275    } else {
276        None
277    };
278
279    // Resolved alongside `analytics_read_policy` (#1370, AbAC Stage 2): `OwnershipRewrite`'s two
280    // data-isolation deployment knobs, parsed in `micromegas-analytics` (the crate that consumes
281    // them) rather than `micromegas-auth`. The mutating-function registration gate (#1371, AbAC
282    // Stage 3) is derived from the auth provider at server startup instead, not a knob here.
283    let analytics_isolation_config = if roles.flightsql && !args.disable_auth {
284        Some(Arc::new(IsolationConfig::from_env("MICROMEGAS_ANALYTICS")?))
285    } else {
286        None
287    };
288
289    // Resolve the analytics admin var once (web and FlightSQL share the same source)
290    let analytics_admin_var = if std::env::var("MICROMEGAS_ANALYTICS_ADMINS").is_ok() {
291        "MICROMEGAS_ANALYTICS_ADMINS".to_string()
292    } else {
293        "MICROMEGAS_ADMINS".to_string()
294    };
295
296    // One SIGTERM drives all roles
297    let fanout = ShutdownFanout::new(wait_for_sigterm());
298
299    let mut join_set: JoinSet<Result<()>> = JoinSet::new();
300
301    // ── Ingestion ──────────────────────────────────────────────────────────
302    if roles.ingestion {
303        let lake = lakehouse
304            .as_ref()
305            .expect("lakehouse must be Some when ingestion role is enabled")
306            .lake()
307            .as_ref()
308            .clone();
309        let shutdown = fanout.subscribe();
310        let listen_addr = args.listen_endpoint_http;
311        let grace_c = grace;
312        let auth = ingestion_auth;
313        join_set.spawn(
314            async move { serve_ingestion(listen_addr, lake, auth, shutdown, grace_c).await },
315        );
316    }
317
318    // ── FlightSQL ──────────────────────────────────────────────────────────
319    if roles.flightsql {
320        let lh = lakehouse
321            .as_ref()
322            .expect("lakehouse must be Some when flightsql role is enabled")
323            .clone();
324        let shutdown = fanout.subscribe();
325        let grace_c = grace;
326        let auth = analytics_auth;
327        let read_policy = analytics_read_policy;
328        let isolation_config = analytics_isolation_config;
329        join_set.spawn(async move {
330            let mut builder = FlightSqlServer::builder()
331                .with_lakehouse(lh)
332                .with_shutdown(shutdown)
333                .with_shutdown_grace(grace_c);
334            if let Some(provider) = auth {
335                builder = builder.with_auth_provider(provider);
336            }
337            if let Some(policy) = read_policy {
338                builder = builder.with_read_policy(policy);
339            }
340            if let Some(config) = isolation_config {
341                builder = builder.with_isolation_config(config);
342            }
343            builder.build_and_serve().await
344        });
345    }
346
347    // ── Maintenance ────────────────────────────────────────────────────────
348    if roles.maintenance {
349        let lh = lakehouse
350            .as_ref()
351            .expect("lakehouse must be Some when maintenance role is enabled")
352            .clone();
353        let shutdown = fanout.subscribe();
354        let grace_c = grace;
355        let retention_days = args.retention_days;
356        join_set.spawn(async move {
357            let view_factory =
358                default_view_factory(lh.runtime().clone(), lh.lake().clone()).await?;
359            let views_to_update = get_global_views_with_update_group(&view_factory);
360            daemon(lh, views_to_update, retention_days, shutdown, grace_c).await
361        });
362    }
363
364    // ── Web ────────────────────────────────────────────────────────────────
365    if roles.web {
366        let web_config = WebServerConfig::from_cli_and_env(WebCliArgs {
367            port: args.port,
368            frontend_dir: args.frontend_dir.clone(),
369            disable_auth: args.disable_auth,
370            admin_var_name: analytics_admin_var,
371        })?;
372
373        // Auto-seed the local FlightSQL data source when web + flightsql are both enabled
374        if roles.flightsql
375            && !args.no_seed_data_source
376            && let Err(e) = seed_local_data_source(&web_config.app_db_string).await
377        {
378            warn!("Failed to seed local FlightSQL data source: {e}");
379        }
380
381        let shutdown = fanout.subscribe();
382        let grace_c = grace;
383
384        join_set.spawn(async move { run_web_server(web_config, shutdown, grace_c).await });
385    }
386
387    // Wait for all roles; fail-fast on first error
388    while let Some(result) = join_set.join_next().await {
389        match result {
390            Ok(Ok(())) => {}
391            Ok(Err(e)) => {
392                error!("Role failed: {e:#}");
393                join_set.abort_all();
394                return Err(e);
395            }
396            Err(join_err) => {
397                error!("Role task panicked: {join_err}");
398                join_set.abort_all();
399                anyhow::bail!("Role task panicked: {join_err}");
400            }
401        }
402    }
403
404    info!("All roles shut down cleanly");
405    Ok(())
406}

412/// Idempotent first-run seed: insert a "local" FlightSQL data source pointing
413/// at the in-process loopback listener when the app DB has no data sources.
414async fn seed_local_data_source(app_db_string: &str) -> Result<()> {
415    let pool = sqlx::PgPool::connect(app_db_string)
416        .await
417        .with_context(|| "seed_local_data_source: connecting to app DB")?;
418    // Ensure the schema exists before querying (idempotent)
419    app_db::execute_migration(pool.clone()).await?;
420
421    let count = sqlx::query_scalar::<_, i64>("SELECT COUNT(*) FROM data_sources")
422        .fetch_one(&pool)
423        .await?;
424
425    if count == 0 {
426        sqlx::query(
427            "INSERT INTO data_sources (name, config, is_default, created_by, updated_by) \
428             VALUES ('local', $1, TRUE, 'monolith', 'monolith')",
429        )
430        .bind(serde_json::json!({"url": "http://127.0.0.1:50051"}))
431        .execute(&pool)
432        .await?;
433        info!("Seeded local FlightSQL data source pointing at http://127.0.0.1:50051");
434    }
435    Ok(())
