1//! Label resolution for the object cache's dimensioned metrics (`prefix`, `class`, `reason`).
2//!
3//! Every dimension used here is a small, closed label set so cardinality stays bounded, and
4//! every value is a compile-time (or leaked-once-at-startup) `&'static str`, per the
5//! tagged-metric contract documented on `micromegas_tracing::property_set::PropertySet`
6//! ("the user is expected to manage the cardinality"). Call sites attach the labels with
7//! `key = static label`; this module only keeps the taxonomy of label strings and resolves a
8//! key to its `prefix` label.
9
10use std::sync::Arc;
11
12/// Demand-vs-prefetch `class` dimension values. Kept as plain string
13/// constants here (rather than a method on `range_cache`'s private
14/// `Priority` enum) so the taxonomy of label strings lives in one place.
15pub const CLASS_DEMAND: &str = "demand";
16pub const CLASS_PREFETCH: &str = "prefetch";
17
18/// Fallback `prefix` label for a key that matches none of the server's
19/// configured allowed prefixes (or when no prefixes were configured at all,
20/// e.g. a `RangeCache` built without `RangeCache::with_prefix_labels`).
21pub const PREFIX_OTHER: &str = "other";
22
23/// Longest-prefix match of `key` against `labels`, using the same
24/// equal-or-`/`-boundary admission rule as `validation::validate_key` (so a
25/// label `"blobs"` matches `"blobs"` and `"blobs/x"` but not
26/// `"blobs-secret"`). Returns the index of the longest matching label, or
27/// `None` if none match -- the caller falls back to `PREFIX_OTHER`.
28pub fn longest_prefix_match(labels: &[&'static str], key: &str) -> Option<usize> {
29    labels
30        .iter()
31        .enumerate()
32        .filter(|(_, p)| {
33            key.starts_with(*p) && (key.len() == p.len() || key.as_bytes()[p.len()] == b'/')
34        })
35        .max_by_key(|(_, p)| p.len())
36        .map(|(i, _)| i)
37}
38
39/// RAM-tier eviction `reason` dimension values (`foyer::Event` mapped to a
40/// stable label string by `foyer_backend::reason_str`).
41pub const REASON_EVICT: &str = "evict";
42pub const REASON_REPLACE: &str = "replace";
43pub const REASON_REMOVE: &str = "remove";
44pub const REASON_CLEAR: &str = "clear";
45
46/// Resolves a key to its `prefix` label, shared (via `Arc`) between the RAM eviction listener and
47/// `FoyerBackend::get`, so the key-to-prefix matching rule (`longest_prefix_match`) is not
48/// duplicated between the two call sites.
49pub struct EvictionTagTable {
50    labels: Arc<[&'static str]>,
51}
52
53impl EvictionTagTable {
54    pub fn new(labels: Arc<[&'static str]>) -> Self {
55        Self { labels }
56    }
57
58    /// The `prefix` label `key` falls under, resolved by longest-prefix match against `labels`
59    /// (`PREFIX_OTHER` on no match).
60    pub fn classify(&self, key: &str) -> &'static str {
61        match longest_prefix_match(&self.labels, key) {
62            Some(i) => self.labels[i],
63            None => PREFIX_OTHER,
64        }
65    }
66}
