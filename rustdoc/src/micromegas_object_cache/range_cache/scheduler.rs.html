12/// Relative urgency of an origin fetch. Lower is more urgent.
13#[derive(Copy, Clone, Debug, PartialEq, Eq)]
14#[repr(u8)]
15pub(super) enum Priority {
16    Demand = 0,
17    Prefetch = 1,
18}
19
20impl Priority {
21    fn from_u8(v: u8) -> Self {
22        if v == Priority::Demand as u8 {
23            Priority::Demand
24        } else {
25            Priority::Prefetch
26        }
27    }
28
29    /// The `class` metric-tag value for this priority (see `metric_tags`).
30    pub(super) fn class_label(self) -> &'static str {
31        match self {
32            Priority::Demand => metric_tags::CLASS_DEMAND,
33            Priority::Prefetch => metric_tags::CLASS_PREFETCH,
34        }
35    }
36}
37
38/// The `class` tag for one coalesced run: `Demand` if any of its entries is
39/// currently demand priority, else `Prefetch`. Mirrors `acquire_run_permit`'s
40/// own `all_prefetch` check, evaluated once the permit has been acquired so a
41/// promotion that raced the wait is reflected in the tag.
42pub(super) fn effective_priority(entries: &[Arc<InFlight>]) -> Priority {
43    if entries.iter().any(|e| e.priority() == Priority::Demand) {
44        Priority::Demand
45    } else {
46        Priority::Prefetch
47    }
48}

50/// The set of entries submitted in one prefetch call. Used only when
51/// `promote_whole_batch` is enabled: a demand joiner into any sibling entry
52/// promotes every other entry in the batch, not just the one it joined.
53///
54/// Siblings are tracked by `Weak<InFlight>`, captured once each entry is
55/// created or joined, rather than by key string: a key can be removed from
56/// `FetchScheduler::inflight` and later reused by an unrelated fetch (e.g.
57/// after eviction), and promoting by key alone would then spuriously promote
58/// that new, logically unrelated `InFlight`.
59pub(super) struct BatchState {
60    entries: StdMutex<Vec<Weak<InFlight>>>,
61}
62
63impl BatchState {
64    pub(super) fn new(capacity: usize) -> Self {
65        Self {
66            entries: StdMutex::new(Vec::with_capacity(capacity)),
67        }
68    }
69}

73/// One outstanding origin fetch (a single block or a `size()` head), shared
74/// across every concurrent caller asking for the same key.
75pub(super) struct InFlight {
76    priority: AtomicU8,
77    promote: Notify,
78    result: watch::Sender<Option<FetchResult>>,
79    batch: Option<Arc<BatchState>>,
80    /// Set by the first `fulfill()` call. Lets a `FulfillGuard`'s panic-path
81    /// fallback tell which entries in a partially-completed run already got
82    /// a real result, so it never clobbers one with a synthesized error.
83    fulfilled: std::sync::atomic::AtomicBool,
84}
85
86impl InFlight {
87    fn new(priority: Priority, batch: Option<Arc<BatchState>>) -> Self {
88        let (result, _rx) = watch::channel(None);
89        Self {
90            priority: AtomicU8::new(priority as u8),
91            promote: Notify::new(),
92            result,
93            batch,
94            fulfilled: std::sync::atomic::AtomicBool::new(false),
95        }
96    }
97
98    fn priority(&self) -> Priority {
99        Priority::from_u8(self.priority.load(Ordering::Acquire))
100    }
101
102    fn promote_to_demand(&self) {
103        self.priority
104            .store(Priority::Demand as u8, Ordering::Release);
105        self.promote.notify_one();
106    }
107
108    /// Deliver `result` to every joiner. Idempotent: only the first call
109    /// actually sends — later calls (notably a `FulfillGuard`'s panic-path
110    /// fallback racing a fetch that in fact completed normally) are no-ops,
111    /// so a real result is never overwritten by a synthesized error.
112    pub(super) fn fulfill(&self, result: FetchResult) {
113        if self.fulfilled.swap(true, Ordering::AcqRel) {
114            return;
115        }
116        // `send_replace`, not `send`: `send` drops the value without storing
117        // it when the channel currently has zero receivers, and joiners
118        // subscribe lazily inside `join()`. A fetch task that completes
119        // before any joiner's `subscribe()` would lose the result and hang
120        // every later joiner forever. `send_replace` stores the value
121        // unconditionally.
122        self.result.send_replace(Some(result));
123    }
124
125    /// Wait for the fetch to complete, returning immediately if it already
126    /// has. Safe against the "subscribe after send" race: a fresh `watch`
127    /// receiver's initial value counts as unseen until `borrow_and_update`
128    /// checks it, so a result sent before this call is still observed here.
129    pub(super) async fn join(&self) -> FetchResult {
130        let mut rx = self.result.subscribe();
131        loop {
132            if let Some(r) = rx.borrow_and_update().clone() {
133                return r;
134            }
135            if rx.changed().await.is_err() {
136                return Err(Arc::new(anyhow!(
137                    "in-flight fetch entry dropped without a result"
138                )));
139            }
140        }
141    }
142}

149/// Owns the in-flight single-flight map and the priority-aware origin-fetch
150/// budget.
151pub(super) struct FetchScheduler {
152    inflight: StdMutex<HashMap<String, Arc<InFlight>>>,
153    /// Bounds total concurrent origin GETs (blocks + size heads don't count
154    /// against this; only run GETs acquire it).
155    shared_permits: Arc<Semaphore>,
156    /// Total capacity of `shared_permits`, stored alongside it since
157    /// `tokio::sync::Semaphore` has no capacity accessor. Used by
158    /// `fetch_budget_stats` for the saturation sampler.
159    shared_total: usize,
160    /// Prefetch runs must additionally hold one of these; sized to
161    /// `total - demand_reserved` so demand always finds a free shared permit.
162    prefetch_permits: Arc<Semaphore>,
163    /// Total capacity of `prefetch_permits`, for the same reason as
164    /// `shared_total`.
165    prefetch_total: usize,
166    promote_whole_batch: bool,
167    /// Count of detached fetch tasks (`spawn_run_fetch` runs + `size()` HEADs)
168    /// currently in flight, tracked by `FetchTaskGuard`. Used by graceful
169    /// shutdown to wait for in-flight origin GETs instead of letting the
170    /// tokio runtime drop them.
171    outstanding_tasks: AtomicUsize,
172    /// Notified when `outstanding_tasks` drops to zero. See `wait_drained`.
173    drained: Notify,
174}
175
176impl FetchScheduler {
177    pub(super) fn new(total: usize, demand_reserved: usize, promote_whole_batch: bool) -> Self {
178        assert!(total > 0, "fetch concurrency total must be > 0");
179        // Strictly less: `demand_reserved == total` would leave the prefetch
180        // semaphore with zero permits, hanging every prefetch run forever.
181        assert!(
182            demand_reserved < total,
183            "demand_reserved ({demand_reserved}) must be < total ({total})"
184        );
185        let prefetch_total = total - demand_reserved;
186        Self {
187            inflight: StdMutex::new(HashMap::new()),
188            shared_permits: Arc::new(Semaphore::new(total)),
189            shared_total: total,
190            prefetch_permits: Arc::new(Semaphore::new(prefetch_total)),
191            prefetch_total,
192            promote_whole_batch,
193            outstanding_tasks: AtomicUsize::new(0),
194            drained: Notify::new(),
195        }
196    }
197
198    /// Register one detached fetch task as outstanding. Call *before*
199    /// `tokio::spawn`, synchronously, so there is no window where a
200    /// queued-but-not-yet-polled task is invisible to `wait_drained()`.
201    pub(super) fn track_task(scheduler: &Arc<FetchScheduler>) -> FetchTaskGuard {
202        scheduler.outstanding_tasks.fetch_add(1, Ordering::AcqRel);
203        FetchTaskGuard(scheduler.clone())
204    }
205
206    /// Number of detached fetch tasks currently in flight to origin.
207    pub(super) fn outstanding_tasks(&self) -> usize {
208        self.outstanding_tasks.load(Ordering::Acquire)
209    }
210
211    /// Resolves once no detached fetch task is outstanding. Race-free by
212    /// construction: `Notified::enable`'s docs guarantee that a
213    /// `notify_waiters()` call is observed by a `Notified` future as long as
214    /// that call happens after the `Notified` was created, regardless of
215    /// whether `enable`/`poll` has run yet -- so creating the `Notified`
216    /// *before* re-checking the count (rather than after) means a
217    /// `notify_waiters()` racing the check in between is never missed. This
218    /// is a different guarantee than `any_entry_promoted`'s below, which
219    /// relies on `notify_one`'s stored-permit semantics; `notify_waiters`
220    /// stores no permit, so the ordering above -- not a permit -- is what
221    /// makes this race-free.
222    pub(super) async fn wait_drained(&self) {
223        loop {
224            if self.outstanding_tasks() == 0 {
225                return;
226            }
227            let notified = self.drained.notified();
228            if self.outstanding_tasks() == 0 {
229                return;
230            }
231            notified.await;
232        }
233    }
234
235    /// `(shared_available, shared_total, prefetch_available, prefetch_total)`
236    /// -- the fetch-permit budget's current occupancy, for the saturation
237    /// sampler (`object-cache-srv/src/saturation_monitor.rs`).
238    pub(super) fn fetch_budget_stats(&self) -> (usize, usize, usize, usize) {
239        (
240            self.shared_permits.available_permits(),
241            self.shared_total,
242            self.prefetch_permits.available_permits(),
243            self.prefetch_total,
244        )
245    }
246
247    /// Number of keys (blocks or `size()` heads) currently in flight to
248    /// origin.
249    pub(super) fn inflight_len(&self) -> usize {
250        self.inflight.lock().expect("inflight lock").len()
251    }
252
253    /// Look up `key` in the in-flight map: become the owner if absent, or a
254    /// joiner if present. A demand joiner into a prefetch-priority entry
255    /// promotes it (and, if `promote_whole_batch`, its batch siblings) to
256    /// demand so it competes for reserved capacity instead of sitting behind
257    /// other prefetch work.
258    ///
259    /// The entry is registered in `batch` (owner or joiner alike) while the
260    /// inflight lock is still held, i.e. before any concurrent demand joiner
261    /// can find the entry and run `promote_batch_siblings` over a
262    /// partially-populated list.
263    pub(super) fn own_or_join(
264        &self,
265        key: String,
266        prio: Priority,
267        batch: Option<Arc<BatchState>>,
268    ) -> Ownership {
269        let mut promote_batch: Option<(Arc<BatchState>, Arc<InFlight>)> = None;
270        let ownership = {
271            let mut map = self.inflight.lock().expect("inflight lock");
272            if let Some(existing) = map.get(&key) {
273                if prio == Priority::Demand && existing.priority() == Priority::Prefetch {
274                    existing.promote_to_demand();
275                    if self.promote_whole_batch
276                        && let Some(bs) = existing.batch.clone()
277                    {
278                        promote_batch = Some((bs, existing.clone()));
279                    }
280                }
281                if let Some(bs) = &batch {
282                    bs.entries
283                        .lock()
284                        .expect("batch lock")
285                        .push(Arc::downgrade(existing));
286                }
287                Ownership::Joiner(existing.clone())
288            } else {
289                let entry = Arc::new(InFlight::new(prio, batch.clone()));
290                if let Some(bs) = &batch {
291                    bs.entries
292                        .lock()
293                        .expect("batch lock")
294                        .push(Arc::downgrade(&entry));
295                }
296                map.insert(key.clone(), entry.clone());
297                Ownership::Owner(entry)
298            }
299        };
300        if let Some((bs, skip)) = promote_batch {
301            self.promote_batch_siblings(&bs, &skip);
302        }
303        ownership
304    }
305
306    /// Promote every sibling in `batch` other than `skip` to demand priority.
307    /// Siblings are resolved through `Weak<InFlight>` references captured at
308    /// batch-membership time, not by re-looking up their key in `inflight`:
309    /// the key may since have been removed and reused by an unrelated fetch,
310    /// and identity (not the key string) is what must match here.
311    fn promote_batch_siblings(&self, batch: &BatchState, skip: &Arc<InFlight>) {
312        let entries = batch.entries.lock().expect("batch lock");
313        for weak in entries.iter() {
314            if let Some(entry) = weak.upgrade()
315                && !Arc::ptr_eq(&entry, skip)
316            {
317                entry.promote_to_demand();
318            }
319        }
320    }
321
322    pub(super) fn remove_entry(&self, key: &str) {
323        self.inflight.lock().expect("inflight lock").remove(key);
324    }
325}

327/// Scope guard held by the task that owns one or more in-flight entries
328/// (either a single `size()` head or the members of one coalesced block
329/// run). If the owning task exits normally, it calls `disarm()` after
330/// fulfilling every entry and removing it from `FetchScheduler::inflight`.
331///
332/// If instead the task panics (e.g. `Bytes::slice` on a short origin read)
333/// tokio catches the unwind at the task boundary and silently drops the
334/// result, but this guard's `Drop` still runs during that unwind: it
335/// fulfills any not-yet-fulfilled entry with an error (the `fulfilled` flag
336/// on `InFlight` makes this a no-op for entries that already got a real
337/// result) and removes every entry from the map. Without this, a panicking
338/// owner would leave `fulfill()` never called, hanging every joiner
339/// (including the owner itself) forever and leaking the entry permanently.
340pub(super) struct FulfillGuard {
341    scheduler: Arc<FetchScheduler>,
342    entries: Vec<(String, Arc<InFlight>)>,
343    armed: bool,
344}
345
346impl FulfillGuard {
347    pub(super) fn new(
348        scheduler: Arc<FetchScheduler>,
349        entries: Vec<(String, Arc<InFlight>)>,
350    ) -> Self {
351        Self {
352            scheduler,
353            entries,
354            armed: true,
355        }
356    }
357
358    /// Call once the normal completion path has fulfilled and removed every
359    /// entry, so the guard's `Drop` becomes a no-op.
360    pub(super) fn disarm(mut self) {
361        self.armed = false;
362    }
363}
364
365impl Drop for FulfillGuard {
366    fn drop(&mut self) {
367        if !self.armed {
368            return;
369        }
370        let n = self.entries.len();
371        // `std::thread::panicking()` is `true` only while the current thread
372        // is unwinding from a real panic; it is `false` when a task future is
373        // simply dropped without being polled to completion (e.g. the tokio
374        // runtime shutting down), so it's exactly the signal needed to tell
375        // the two cases apart.
376        let panicking = std::thread::panicking();
377        if panicking {
378            warn!("fetch task panicked; fulfilling {n} in-flight entries with an error");
379        } else {
380            warn!(
381                "cache service shutting down; abandoning {n} in-flight fetch entries \
382                 (joiners see a synthesized error and must refetch after restart)"
383            );
384        }
385        let msg = if panicking {
386            "fetch task panicked before producing a result"
387        } else {
388            "fetch task was abandoned (cache service shutting down)"
389        };
390        for (key, entry) in &self.entries {
391            entry.fulfill(Err(Arc::new(anyhow!(msg))));
392            self.scheduler.remove_entry(key);
393        }
394    }
395}

397/// Scope guard tracking the lifetime of one detached fetch task, held for
398/// its whole body. Distinct from `FulfillGuard`, which tracks *entry
399/// fulfillment*, not task lifetime -- the two guards serve different
400/// purposes and both must survive a panicking OR shutdown-dropped task.
401/// Because this is a plain local variable inside the spawned async block,
402/// its `Drop` runs whether the task finishes normally, panics, or is dropped
403/// without being polled to completion, so `outstanding_tasks` stays accurate
404/// in all three cases without any special-casing.
405pub(super) struct FetchTaskGuard(Arc<FetchScheduler>);
406
407impl Drop for FetchTaskGuard {
408    fn drop(&mut self) {
409        if self.0.outstanding_tasks.fetch_sub(1, Ordering::AcqRel) == 1 {
410            self.0.drained.notify_waiters();
411        }
412    }
413}

415/// Held for the duration of one origin GET; dropping it frees the slot(s) for
416/// the next waiter. Fields are never read, only held for their `Drop` effect.
417pub(super) struct RunPermit {
418    _shared: OwnedSemaphorePermit,
419    _prefetch: Option<OwnedSemaphorePermit>,
420}

422/// Resolves when any of `entries`' priority has been promoted since this call
423/// started. Constructing the `Notified` future(s) up front (before the caller
424/// re-checks priority) is what makes this race-free: tokio guarantees a
425/// `notify_one()` that happens after a `Notified` is created is never missed,
426/// even if that `Notified` is later dropped without completing.
427async fn any_entry_promoted(entries: &[Arc<InFlight>]) {
428    match entries {
429        [] => std::future::pending::<()>().await,
430        [only] => only.promote.notified().await,
431        many => {
432            let futs: Vec<_> = many
433                .iter()
434                .map(|e| Box::pin(e.promote.notified()))
435                .collect();
436            futures::future::select_all(futs).await;
437        }
438    }
439}

441/// Acquire the permit(s) needed to run one coalesced GET covering `entries`,
442/// honoring promotion: the run's effective priority is the most urgent
443/// (minimum) of its entries', re-checked every time a promotion wakes this
444/// loop so a promotion mid-wait drops the prefetch-class requirement.
445pub(super) async fn acquire_run_permit(
446    scheduler: &FetchScheduler,
447    entries: &[Arc<InFlight>],
448) -> RunPermit {
449    loop {
450        let all_prefetch = entries.iter().all(|e| e.priority() == Priority::Prefetch);
451        if !all_prefetch {
452            let shared = scheduler
453                .shared_permits
454                .clone()
455                .acquire_owned()
456                .await
457                .expect("shared_permits semaphore is never closed");
458            return RunPermit {
459                _shared: shared,
460                _prefetch: None,
461            };
462        }
463
464        tokio::select! {
465            prefetch = scheduler.prefetch_permits.clone().acquire_owned() => {
466                let prefetch = prefetch.expect("prefetch_permits semaphore is never closed");
467                tokio::select! {
468                    shared = scheduler.shared_permits.clone().acquire_owned() => {
469                        let shared = shared.expect("shared_permits semaphore is never closed");
470                        return RunPermit { _shared: shared, _prefetch: Some(prefetch) };
471                    }
472                    _ = any_entry_promoted(entries) => {
473                        drop(prefetch);
474                        continue;
475                    }
476                }
477            }
478            _ = any_entry_promoted(entries) => {
479                continue;
480            }
481        }
482    }
483}

485/// Reconstruct an owned error from a shared in-flight failure. `anyhow::Error`
486/// is not `Clone`, so a joiner cannot move it out of the `Arc`; and it must
487/// not be stringified, or a missing key would stop downcasting to
488/// `object_store::Error::NotFound` and would surface as a 500 instead of a
489/// 404 (see `validation::is_not_found`).
490pub(super) fn reconstruct_shared_error(shared: &Arc<anyhow::Error>) -> anyhow::Error {
491    if let Some(object_store::Error::NotFound { path, source }) =
492        shared.downcast_ref::<object_store::Error>()
493    {
494        let rebuilt = object_store::Error::NotFound {
495            path: path.clone(),
496            source: Box::new(std::io::Error::new(
497                std::io::ErrorKind::NotFound,
498                source.to_string(),
499            )),
500        };
501        return anyhow::Error::from(rebuilt);
502    }
503    // `{shared:?}` keeps the full context chain (and backtrace, if captured)
504    // so joiners' 500 log lines are as informative as the owner's.
505    anyhow!("{shared:?}")
506}
