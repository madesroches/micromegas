1//! Metric sampling and emission. Modeled on
2//! `object-cache-srv/src/saturation_monitor.rs`: pure emit functions driven
3//! by a loop in `main.rs`, so tests can drive one round without the loop.
4//!
5//! `imetric!`/`fmetric!` require literal metric names (per-call-site static
6//! metadata), so each metric is an explicit call; dynamic strings appear
7//! only as property values, interned to `'static` (bounded cardinality:
8//! command names, latency events, db indices, one instance name).
9use anyhow::{Context, Result};
10use micromegas::tracing::intern_string::intern_string;
11use micromegas::tracing::prelude::*;
12use micromegas::tracing::property_set::{Property, PropertySet};
13
14use crate::cli::MetricsPreset;
15use crate::info_parser::ParsedInfo;

17/// Properties attached to every metric: `instance` plus user-supplied pairs.
18pub fn build_base_properties(
19    target_name: &str,
20    extra: &[(String, String)],
21) -> &'static PropertySet {
22    let mut props = vec![Property::new("instance", intern_string(target_name))];
23    for (key, value) in extra {
24        props.push(Property::new(intern_string(key), intern_string(value)));
25    }
26    PropertySet::find_or_create(props)
27}
28
29/// Base properties + one extra tag (e.g. `command=get`, `db=0`).
30fn with_extra(base: &'static PropertySet, name: &'static str, value: &str) -> &'static PropertySet {
31    let mut props = base.get_properties().to_vec();
32    props.push(Property::new(name, intern_string(value)));
33    PropertySet::find_or_create(props)
34}

36/// One full sampling round. `now_unix_secs` is passed in for testability
37/// (used to derive the RDB last-save age).
38pub async fn sample_once(
39    conn: &mut redis::aio::ConnectionManager,
40    preset: MetricsPreset,
41    props: &'static PropertySet,
42    now_unix_secs: u64,
43) -> Result<()> {
44    // `INFO all` also returns the Commandstats section needed by `full`;
45    // plain `INFO` (default sections) is enough below that and cheaper.
46    let raw: String = if preset >= MetricsPreset::Full {
47        redis::cmd("INFO").arg("all").query_async(conn).await
48    } else {
49        redis::cmd("INFO").query_async(conn).await
50    }
51    .context("INFO")?;
52    let info = ParsedInfo::parse(&raw);
53    emit_info_metrics(&info, props, now_unix_secs);
54
55    if preset >= MetricsPreset::Extended {
56        let slowlog_len: u64 = redis::cmd("SLOWLOG")
57            .arg("LEN")
58            .query_async(conn)
59            .await
60            .context("SLOWLOG LEN")?;
61        imetric!("redis_slowlog_length", "count", props, slowlog_len);
62    }
63
64    if preset >= MetricsPreset::Full {
65        emit_command_stats(&info, props);
66        // Rows of (event, unix timestamp, latest ms, max ms).
67        let latency: Vec<(String, u64, u64, u64)> = redis::cmd("LATENCY")
68            .arg("LATEST")
69            .query_async(conn)
70            .await
71            .context("LATENCY LATEST")?;
72        for (event, _timestamp, latest_ms, max_ms) in &latency {
73            let tagged = with_extra(props, "event", event);
74            imetric!("redis_latency_latest_ms", "ms", tagged, *latest_ms);
75            imetric!("redis_latency_max_ms", "ms", tagged, *max_ms);
76        }
77    }
78    Ok(())
79}

81/// Core metrics from one INFO response. Absent or unparseable fields are
82/// skipped (Redis versions differ), never fatal.
83pub fn emit_info_metrics(info: &ParsedInfo, props: &'static PropertySet, now_unix_secs: u64) {
84    if let Some(v) = info.get_u64("uptime_in_seconds") {
85        imetric!("redis_uptime_seconds", "seconds", props, v);
86    }
87    if let Some(v) = info.get_u64("connected_clients") {
88        imetric!("redis_connected_clients", "count", props, v);
89    }
90    if let Some(v) = info.get_u64("blocked_clients") {
91        imetric!("redis_blocked_clients", "count", props, v);
92    }
93    if let Some(v) = info.get_u64("used_memory") {
94        imetric!("redis_used_memory_bytes", "bytes", props, v);
95    }
96    if let Some(v) = info.get_u64("used_memory_rss") {
97        imetric!("redis_used_memory_rss_bytes", "bytes", props, v);
98    }
99    if let Some(v) = info.get_u64("maxmemory") {
100        imetric!("redis_maxmemory_bytes", "bytes", props, v);
101    }
102    if let Some(v) = info.get_f64("mem_fragmentation_ratio") {
103        fmetric!("redis_mem_fragmentation_ratio", "ratio", props, v);
104    }
105    if let Some(v) = info.get_u64("instantaneous_ops_per_sec") {
106        imetric!("redis_ops_per_sec", "ops_per_sec", props, v);
107    }
108    if let Some(v) = info.get_u64("total_commands_processed") {
109        imetric!("redis_total_commands_processed", "count", props, v);
110    }
111    if let Some(v) = info.get_u64("total_net_input_bytes") {
112        imetric!("redis_total_net_input_bytes", "bytes", props, v);
113    }
114    if let Some(v) = info.get_u64("total_net_output_bytes") {
115        imetric!("redis_total_net_output_bytes", "bytes", props, v);
116    }
117    if let Some(v) = info.get_u64("keyspace_hits") {
118        imetric!("redis_keyspace_hits", "count", props, v);
119    }
120    if let Some(v) = info.get_u64("keyspace_misses") {
121        imetric!("redis_keyspace_misses", "count", props, v);
122    }
123    if let Some(v) = info.get_u64("expired_keys") {
124        imetric!("redis_expired_keys", "count", props, v);
125    }
126    if let Some(v) = info.get_u64("evicted_keys") {
127        imetric!("redis_evicted_keys", "count", props, v);
128    }
129    if let Some(v) = info.get_u64("connected_slaves") {
130        imetric!("redis_connected_replicas", "count", props, v);
131    }
132    if let Some(v) = info.get_u64("master_repl_offset") {
133        imetric!("redis_master_repl_offset", "count", props, v);
134    }
135    if info.get_str("role") == Some("slave") {
136        if let Some(v) = info.get_u64("slave_repl_offset") {
137            imetric!("redis_replica_repl_offset", "count", props, v);
138        }
139        if let Some(v) = info.get_u64("master_last_io_seconds_ago") {
140            imetric!("redis_master_last_io_seconds_ago", "seconds", props, v);
141        }
142        if let Some(status) = info.get_str("master_link_status") {
143            let up = u64::from(status == "up");
144            imetric!("redis_master_link_up", "count", props, up);
145        }
146    }
147    if let Some(v) = info.get_u64("rdb_changes_since_last_save") {
148        imetric!("redis_rdb_changes_since_last_save", "count", props, v);
149    }
150    if let Some(t) = info.get_u64("rdb_last_save_time") {
151        // Mixes the server's clock (rdb_last_save_time) with the exporter's
152        // clock (now_unix_secs): clock skew between the two distorts this
153        // age; saturating_sub only guards against underflow, not skew.
154        imetric!(
155            "redis_rdb_last_save_age_seconds",
156            "seconds",
157            props,
158            now_unix_secs.saturating_sub(t)
159        );
160    }
161    if info.get_u64("aof_enabled") == Some(1)
162        && let Some(v) = info.get_u64("aof_current_size")
163    {
164        imetric!("redis_aof_current_size_bytes", "bytes", props, v);
165    }
166    for entry in info.keyspace() {
167        let tagged = with_extra(props, "db", &entry.db.to_string());
168        imetric!("redis_db_keys", "count", tagged, entry.keys);
169        imetric!("redis_db_expires", "count", tagged, entry.expires);
170    }
171}

173/// Per-command metrics from the Commandstats section (present in `INFO all`).
174pub fn emit_command_stats(info: &ParsedInfo, props: &'static PropertySet) {
175    for stat in info.command_stats() {
176        let tagged = with_extra(props, "command", &stat.name);
177        imetric!("redis_command_calls", "count", tagged, stat.calls);
178        imetric!("redis_command_usec", "us", tagged, stat.usec);
179        fmetric!(
180            "redis_command_usec_per_call",
181            "us",
182            tagged,
183            stat.usec_per_call
184        );
185    }
