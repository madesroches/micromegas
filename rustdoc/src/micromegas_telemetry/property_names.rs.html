1//! Property-key constants shared by producers and the ingestion stack.
2//!
3//! Kept apart from the `property` module, which is gated behind the `server` feature because its
4//! `Property` type carries sqlx impls: client crates need these names to honor the reserved
5//! namespace before they send anything, and must not pull a database driver in to do it.
6
7/// Reserved, server-written property namespace. A client-supplied property whose key starts
8/// with this prefix is dropped at ingestion (AbAC Stage 5, #1373).
9pub const RESERVED_PROPERTY_PREFIX: &str = "micromegas.";
10
11/// The audience a process's data belongs to -- written server-side from the authenticated
12/// ingestion credential, read by `OwnershipRewrite`.
13pub const PROPERTY_AUDIENCE: &str = "micromegas.audience";
